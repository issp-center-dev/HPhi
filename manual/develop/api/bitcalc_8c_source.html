58int GetSplitBit(
59 const int Nsite,
60 long unsigned int *irght,
61 long unsigned int *ilft,
62 long unsigned int *ihfbit
63){
64 if(Nsite<1){
65 fprintf(stderr, "%s", cErrSiteNumber);
66 return -1;
67 }

143int GetSplitBitForGeneralSpin(
144 const int Nsite,
145 long unsigned int *ihfbit,
146 const long int *SiteToBit
147){
148 int isite=0;
149 long int isqrtMaxDim=1;
150 long int tmpbit=1;
151
152 if(Nsite<1){
153 fprintf(stderr, "%s", cErrSiteNumber);
154 return -1;
155 }
156
157 for(isite=1; isite<=Nsite ; isite++){
158 isqrtMaxDim *= SiteToBit[isite-1];
159 }
160 isqrtMaxDim =(long int)sqrt(isqrtMaxDim);
161
162 for(isite=1; isite<=Nsite ; isite++){
163 tmpbit *= SiteToBit[isite-1];
164 if(tmpbit >= isqrtMaxDim) break;
165 }
166 *ihfbit=tmpbit;
167 return 0;
168}

342void SgnBit_old(
343 const long unsigned int org_bit,
344 int *sgn
345)
346{
347 long unsigned int bit;
348
349 bit = org_bit^(org_bit>>1);
350 bit = (bit^(bit>>2) ) & 0x11111111;
351 bit = bit*0x11111111;
352 *sgn = 1-2*((bit>>28) & 1); // sgn = pm 1

521unsigned long int snoob(unsigned long int x){
522 unsigned long int smallest, ripple, ones;
523 smallest = x &(-x);
524 ripple = x+ smallest;
525 ones = x ^ ripple;
526 ones = (ones>>2)/smallest;
527 return ripple|ones;
528}
