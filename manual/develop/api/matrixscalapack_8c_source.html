173void GetEigenVector(long int i, long int m, double complex *Z, int *descZ, double complex *vec) {
174 double complex alpha;

200int diag_scalapack_cmp(long int xNsize, double complex **A,
201 double complex *r, double complex *Z, int *descZ) {
202 const int i_one=1, i_zero=0;
203 const long int i_negone=-1;
204 const double zero=0.0, one=1.0;
205 long int m, n, mb, nb;
206 int nprow, npcol;
207 int myrow, mycol, info, lld;
208 long int mp, nq;
209 int ictxt;
210 complex double *A_distr, *work, *rwork;
211 double *W;
212 int descA_distr[9];
213 int rank, size, iam, nprocs;
214 long int lwork, lrwork;
215 int dims[2]={0,0};
216 long int i, j, ip, jp;
217 m=n=xNsize;
218
219 MPI_Comm_rank(MPI_COMM_WORLD, &rank);
220 MPI_Comm_size(MPI_COMM_WORLD, &size);
221 MPI_Dims_create(size,2,dims);
222 nprow=dims[0]; npcol=dims[1];
223
224 blacs_pinfo_(&iam, &nprocs);
225 blacs_get_((int *)&i_negone, &i_zero, &ictxt);
226 blacs_gridinit_(&ictxt, "R", &nprow, &npcol);
227 blacs_gridinfo_(&ictxt, &nprow, &npcol, &myrow, &mycol);
228
229 mb = GetBlockSize(m, size);
230 nb = GetBlockSize(n, size);
231
232 mp = numroc_(&m, &mb, &myrow, &i_zero, &nprow);
233 nq = numroc_(&n, &nb, &mycol, &i_zero, &npcol);
234 W = malloc(n*sizeof(double));
235 A_distr = malloc(mp*nq*sizeof(complex double));
236
237 lld = (mp>0) ? mp : 1;
238 descinit_(descA_distr, &m, &n, &mb, &nb, &i_zero, &i_zero, &ictxt, &lld, &info);
239 descinit_(descZ, &m, &n, &mb, &nb, &i_zero, &i_zero, &ictxt, &lld, &info);
240
241 for(i=0; i<m; i++){
242 for(j=0; j<n; j++){
243 DivMat(i, j, A[i][j], A_distr, descA_distr);
244 }
245 }
246
247 double complex wkopt, rwkopt;
248 pzheev_("V", "U", &n, A_distr, &i_one, &i_one, descA_distr, W, Z, &i_one, &i_one, descZ, &wkopt, &i_negone, &rwkopt, &i_negone, &info);
249
250 lwork = (long int)wkopt;
251 lrwork = (long int)rwkopt;
252 work = malloc(lwork*sizeof(complex double));
253 rwork = malloc(lrwork*sizeof(complex double));
254
255
256 pzheev_("V", "U", &n, A_distr, &i_one, &i_one, descA_distr, W, Z, &i_one, &i_one, descZ, work, &lwork, rwork, &lrwork, &info);
257
258 if(rank == 0){
259 for(i=0; i<n; i++){
260 r[i] = W[i];
261 }
262 }
263
264 free(A_distr);
265 free(work);
266 free(rwork);
267 free(W);
268
269 use_scalapack = 1;
270
271 return 0;
272}

285static void InitEigenVectorGatherContext(long int xNsize) {
286 int i_negone = -1, i_zero = 0;
287 int imap[1] = {0};
288 int ld = 1, np_gather = 1;
289 int myrow_g, mycol_g, nprow_g, npcol_g;
290
291 blacs_get_(&i_negone, &i_zero, &ictxt_gather);
292 blacs_gridmap_(&ictxt_gather, imap, &ld, &np_gather, &np_gather);
293
294 /* Fully initialize the descriptor on ALL ranks (some implementations
295 inspect fields other than CTXT_), then mark non-participants. */
296 desc_gather[0] = 1; /* DTYPE_: dense */
297 desc_gather[1] = ictxt_gather; /* CTXT_ */
298 desc_gather[2] = (int)xNsize; /* M_ */
299 desc_gather[3] = 1; /* N_ */
300 desc_gather[4] = (int)xNsize; /* MB_ */
301 desc_gather[5] = 1; /* NB_ */
302 desc_gather[6] = 0; /* RSRC_ */
303 desc_gather[7] = 0; /* CSRC_ */
304 desc_gather[8] = (int)xNsize; /* LLD_ */
305
306 blacs_gridinfo_(&ictxt_gather, &nprow_g, &npcol_g, &myrow_g, &mycol_g);
307 if (myrow_g < 0) {
308 desc_gather[1] = -1; /* not in the destination grid */
309 }
310}

328int GetEigenVectorBlock(long int idx, long int xNsize,
329 double complex *Z, int *descZ,
330 double complex *vec) {
331 const long int i_one = 1;
332 long int icol = idx + 1;
333 long int m = xNsize, n = 1;
334
335 if (ictxt_gather == -100) {
336 InitEigenVectorGatherContext(xNsize);
337 }
338 /* Last argument: a context containing the union of both grids
339 = the all-rank 2D context of Z. */
340 pzgemr2d_(&m, &n, Z, (long int *)&i_one, &icol, descZ,
341 vec, (long int *)&i_one, (long int *)&i_one, desc_gather,
342 &descZ[1]);
343 return 0;
344}

367void FreeDistributedEigenvectors(double complex **Z, int *descZ, int *used) {
368 int ictxt;
369 if (Z == NULL || descZ == NULL || used == NULL || !*used) return;
370 FreeEigenVectorGatherContext();
371 free(*Z);
372 *Z = NULL;
373 ictxt = descZ[1];
374 if (ictxt >= 0) blacs_gridexit_(&ictxt);
375 descZ[1] = -1;
376 *used = 0;

400int RedistPanelToBlockCyclic(long int xNsize, long int jbegin,
401 long int ncols_panel, long int panel_ld,
402 double complex *panel,
403 double complex *A_distr, int *descA_2d) {
404 int i_negone = -1, i_zero_i = 0, info;
405 int ictxt_1d, nprow_1, npcol_1, myrow_1, mycol_1;
406 int desc1d[9];
407 int lld;
408 long int NC, mb1, nb1;
409 const long int i_one = 1;
410 int size;
411
412 MPI_Comm_size(MPI_COMM_WORLD, &size);
413 NC = (xNsize + size - 1) / size;
414
415 blacs_get_(&i_negone, &i_zero_i, &ictxt_1d);
416 nprow_1 = 1; npcol_1 = size;
417 blacs_gridinit_(&ictxt_1d, "R", &nprow_1, &npcol_1);
418 blacs_gridinfo_(&ictxt_1d, &nprow_1, &npcol_1, &myrow_1, &mycol_1);
419
420 /* The 1x P 'R' grid must map mycol to this rank 1:1, matching
421 xsetmem's panel ownership (jb = myrank*NC + 1). A mismatch would
422 silently scramble columns, so check at runtime (survives NDEBUG)
423 and synchronize the verdict so no rank enters the collective
424 pzgemr2d_ alone (same pattern as SyncError in matrixlapack_elpa.c).
425 Also verify that the caller-supplied ownership parameters
426 (jbegin, ncols_panel) agree with the ownership derived internally
427 from mycol_1/NC/xNsize -- this is what jbegin/ncols_panel are for,
428 rather than being unused (void)-discarded. */
429 {
430 int ok = (mycol_1 == myrank) ? 0 : -1, gok;
431 long int nc_expect = (mycol_1 < (int)((xNsize + NC - 1) / NC))
432 ? (((long int)mycol_1 + 1) * NC <= xNsize
433 ? NC : xNsize - (long int)mycol_1 * NC)
434 : 0;
435 long int jb_expect = (long int)mycol_1 * NC + 1;
436 if (ok != 0) {
437 fprintf(stdout,
438 " Error: BLACS 1D grid column (%d) does not match MPI rank (%d):\n"
439 " panel ownership is inconsistent; aborting redistribution.\n",
440 mycol_1, myrank);
441 }
442 if (ncols_panel != nc_expect ||
443 (ncols_panel > 0 && jbegin != jb_expect)) {
444 ok = -1;
445 fprintf(stdout,
446 " Error: caller-supplied panel ownership (jbegin=%ld, ncols=%ld) does\n"
447 " not match the internally derived ownership (jbegin=%ld, ncols=%ld)\n"
448 " for rank %d: panel ownership is inconsistent; aborting redistribution.\n",
449 jbegin, ncols_panel, jb_expect, nc_expect, myrank);
450 }
451 MPI_Allreduce(&ok, &gok, 1, MPI_INT, MPI_MIN, MPI_COMM_WORLD);
452 if (gok != 0) {
453 blacs_gridexit_(&ictxt_1d);
454 return -1;
455 }
456 }
457
458 lld = (panel_ld > 0) ? (int)panel_ld : 1;
459 mb1 = xNsize;
460 nb1 = NC;
461 descinit_(desc1d, &xNsize, &xNsize, &mb1, &nb1, &i_zero_i, &i_zero_i,
462 &ictxt_1d, &lld, &info);
463 /* Synchronize the descinit_ verdict so no rank enters the collective
464 pzgemr2d_ with an invalid descriptor (same pattern as
465 RedistBlockCyclicToStatePanel below). */
466 {
467 int ok = (info == 0) ? 0 : -1, gok;
468 if (ok != 0) {
469 fprintf(stdout,
470 " Error: descinit_ failed (info=%d) for the generation-panel descriptor\n"
471 " on rank %d; aborting redistribution.\n", info, myrank);
472 }
473 MPI_Allreduce(&ok, &gok, 1, MPI_INT, MPI_MIN, MPI_COMM_WORLD);
474 if (gok != 0) {
475 blacs_gridexit_(&ictxt_1d);
476 return -1;
477 }
478 }
479
480 pzgemr2d_(&xNsize, &xNsize,
481 panel, (long int *)&i_one, (long int *)&i_one, desc1d,
482 A_distr, (long int *)&i_one, (long int *)&i_one, descA_2d,
483 &descA_2d[1]);
484
485 blacs_gridexit_(&ictxt_1d);
486 return 0;
487}

558 (ncols > 0 && jbegin != jb_expect)) {
