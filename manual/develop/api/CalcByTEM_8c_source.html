297void MakeTEDTransfer(struct BindStruct *X, const int timeidx) {
298 int i,j;
299 //Clear values
300 for(i=0; i<X->Def.NTETransferMax ;i++) {
301 for(j =0; j<4; j++) {
302 X->Def.EDGeneralTransfer[i + X->Def.EDNTransfer][j] = 0;
303 }
304 X->Def.EDParaGeneralTransfer[i+X->Def.EDNTransfer]=0.0;
305 }
306
307 //Input values
308 for(i=0; i<X->Def.NTETransfer[timeidx] ;i++){
309 for(j =0; j<4; j++) {
310 X->Def.EDGeneralTransfer[i + X->Def.EDNTransfer][j] = X->Def.TETransfer[timeidx][i][j];
311 }
312 X->Def.EDParaGeneralTransfer[i+X->Def.EDNTransfer]=X->Def.ParaTETransfer[timeidx][i];
313 }
314 X->Def.EDNTransfer += X->Def.NTETransfer[timeidx];
315}

320void MakeTEDInterAll(struct BindStruct *X, const int timeidx) {
321 int i, j;
322 //Clear values
323 for (i = 0; i < X->Def.NTEInterAllMax; i++) {
324 for (j = 0; j < 8; j++) {
325 X->Def.InterAll_OffDiagonal[i + X->Def.NInterAll_OffDiagonal][j] = 0;
326 }
327 X->Def.ParaInterAll_OffDiagonal[i + X->Def.NInterAll_OffDiagonal] = 0.0;
328 }
329
330 //Input values
331 for (i = 0; i < X->Def.NTEInterAllOffDiagonal[timeidx]; i++) {
332 for (j = 0; j < 8; j++) {
333 X->Def.InterAll_OffDiagonal[i + X->Def.NInterAll_OffDiagonal][j] = X->Def.TEInterAllOffDiagonal[timeidx][i][j];
334 }
335 X->Def.ParaInterAll_OffDiagonal[i + X->Def.NInterAll_OffDiagonal] = X->Def.ParaTEInterAllOffDiagonal[timeidx][i];
336 }
337 X->Def.NInterAll_OffDiagonal += X->Def.NTEInterAllOffDiagonal[timeidx];
338}
