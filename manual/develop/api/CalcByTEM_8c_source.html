72static int record_sector_te(const struct BindStruct *X, const struct SymmetryCheckpointInfo *source,
73 const struct SymmetryTEHamiltonian *view)
74{
75 char name[D_FileNameMax];
76 FILE *fp = NULL;
77 int error = 0;
78 if (myrank == 0) {
79 int n = snprintf(name, sizeof(name), "%s%ssymmetry_sector.dat",
80 X->Def.iOutputDataHead ? X->Def.CDataFileHead : "",
81 X->Def.iOutputDataHead ? "_" : "");
82 error = n < 0 || (size_t)n + strlen(cParentOutputFolder) >= sizeof(name);
83 if (!error) error = childfopenMPI(name, "a", &fp) != 0;
84 if (!error) {
85 fprintf(fp, "te_hamiltonian=%s\nte_steps=%u\nexpand_coef=%d\n"
86 "source_method=%" PRIu64 "\nsource_state=%" PRIu64 "\nsource_step=%" PRIu64
87 "\nsource_time=%.17g\nsource_hamiltonian_digest=%016" PRIx64 "\n",
88 view ? "time_dependent" : "static", X->Def.Lanczos_max, X->Def.Param.ExpandCoef, source->source_method,
89 source->state_index, source->step, source->time, source->hamiltonian_digest);
90 for (unsigned int i = 0; i < X->Def.Lanczos_max; ++i) {
91 fprintf(fp, "te_time_%u=%.17g\n", i, SymmetryTETime(&X->Def, i));
92 if (view) fprintf(fp, "te_hamiltonian_%u=%016" PRIx64 "\n", i, view->digests[i]);
93 }
94 if (view) fprintf(fp, "te_integrator=right_endpoint_taylor\nte_plan_update=rebuild_with_fixed_basis\n");
95 if (ferror(fp)) error = 1;
96 if (fclose(fp) != 0) error = 1;
97 }
98 }
99 if (SumMPI_i(error) != 0) {
100 fprintf(stdoutMPI, "Error: failed to record symmetry TE schedule.\n");
101 return -1;
102 }
103 return 0;
104}

183
184 int invalid = X->Bind.Def.Param.ExpandCoef < 1 ||
185 X->Bind.Def.Param.ExpandCoef == INT_MAX || ExpecInterval <= 0 ||
186 (X->Bind.Def.iOutputEigenVec && X->Bind.Def.Param.OutputInterval <= 0);

190 }

192 fprintf(stdoutMPI, "Error: TE requires finite initial time and nonnegative time step.\n");

194 }
195 if (!(view && X->Bind.Def.NLaser) && X->Bind.Def.NTETimeSteps < X->Bind.Def.Lanczos_max){
196 fprintf(stdoutMPI, "Error: NTETimeSteps must be larger than Lanczos_max.\n");
197 return -1;
198 }
199 if (X->Bind.Def.NLaser == 0) {
200 invalid = X->Bind.Def.TETime == NULL;
201 for (unsigned int i = 0; !invalid && i < X->Bind.Def.Lanczos_max; ++i) {
202 double time = X->Bind.Def.TETime[i];
203 invalid = !isfinite(time) ||
204 (i > 0 && (time < X->Bind.Def.TETime[i-1] ||
205 !isfinite(time-X->Bind.Def.TETime[i-1])));
206 }
207 if (SumMPI_i(invalid) != 0) {
208 fprintf(stdoutMPI, "Error: TE time grid must be finite and nondecreasing.\n");
209 return -1;
210 }
211 }
212 if (view && ValidateSymmetryTESchedule(&(X->Bind), view) != 0) return -1;
213 step_spin = ExpecInterval;
214 X->Bind.Def.St = 0;
215 fprintf(stdoutMPI, "%s", cLogTEM_Start);
216 if (X->Bind.Def.iInputEigenVec == FALSE) {
217 fprintf(stderr, "Error: A file of Inputvector is not inputted.\n");
218 return -1;
219 } else {
220 //input v1
221 fprintf(stdoutMPI, "%s","An Initial Vector is inputted.\n");
222 TimeKeeper(&(X->Bind), cFileNameTimeKeep, c_InputEigenVectorStart, "a");
223 invalid = GetFileNameByKW(KWSpectrumVec, &defname) != 0 || defname == NULL;
224 int length = invalid ? -1 : snprintf(sdt, sizeof(sdt), "%s_rank_%d.dat", defname, myrank);
225 if (SumMPI_i(length < 0 || (size_t)length + strlen(cParentOutputFolder) >= sizeof(sdt)) != 0) {
226 fprintf(stdoutMPI, "Error: missing or overlong TE SpectrumVec filename.\n");
227 return -1;
228 }
229 if (X->Bind.Def.iFlgSymmetryBasis) {
230 if (ReadSymmetryCheckpoint(&(X->Bind), sdt, v1, &source) != 0) return -1;
231 if (record_sector_te(&(X->Bind), &source, view) != 0) return -1;
232 } else {
233 invalid = childfopenALL(sdt, "rb", &fp) != 0;
234 if (!invalid) {
235 invalid = fread(&step_initial, sizeof(step_initial), 1, fp) != 1 ||
236 fread(&i_max, sizeof(i_max), 1, fp) != 1 ||
237 i_max < 0 || (unsigned long)i_max != X->Bind.Check.idim_max;
238 if (!invalid) invalid = fread(v1, sizeof(*v1), X->Bind.Check.idim_max + 1, fp)
239 != X->Bind.Check.idim_max + 1;
240 if (!invalid && (fgetc(fp) != EOF || ferror(fp))) invalid = 1;
241 if (fclose(fp) != 0) invalid = 1;
242 fp = NULL;
243 }
244 if (SumMPI_i(invalid) != 0) {
245 fprintf(stdoutMPI, "Error: missing, truncated or incompatible TE Inputvector file.\n");
246 return -1;
247 }
248 double norm = 0;
249 for (unsigned long i = 1; i <= X->Bind.Check.idim_max; ++i) {
250 double re = creal(v1[i]), im = cimag(v1[i]);
251 if (!isfinite(re) || !isfinite(im)) invalid = 1;
252 norm += re*re + im*im;
253 }
254 norm = SumMPI_d(norm);
255 if (SumMPI_i(invalid || !isfinite(norm) || norm <= 0) != 0) {
256 fprintf(stdoutMPI, "Error: TE Inputvector has zero or non-finite global norm.\n");
257 return -1;
258 }
259 if (X->Bind.Def.iReStart == RESTART_NOT || X->Bind.Def.iReStart == RESTART_OUT)
260 step_initial = 0;
261 int root_step = BcastMPI_i(0, step_initial);
262 if (SumMPI_i(step_initial < 0 || (unsigned int)step_initial >= X->Bind.Def.Lanczos_max ||
263 step_initial != root_step) != 0) {
264 fprintf(stdoutMPI, "Error: TE restart step must be consistent across ranks and within [0, Lanczos_max).\n");
265 return -1;
266 }
267 }
268 }
269
270 if(X->Bind.Def.iOutputDataHead==1){
271 sprintf(sdt_phys, "%s_%s", X->Bind.Def.CDataFileHead, cFileNameSS);
272 }else{
273 sprintf(sdt_phys, "%s", cFileNameSS);
274 }
275 if (childfopenMPI(sdt_phys, "w", &fp) != 0) {
276 return -1;
277 }
278 fprintf(fp, "%s",cLogSS);
279 fclose(fp);
280
281 if(X->Bind.Def.iOutputDataHead==1){
282 sprintf(sdt_norm, "%s_%s", X->Bind.Def.CDataFileHead, cFileNameNorm);
283 }else{
284 sprintf(sdt_norm, "%s", cFileNameNorm);
285 }
286 if (childfopenMPI(sdt_norm, "w", &fp) != 0) {
287 return -1;
288 }
289 fprintf(fp, "%s",cLogNorm);
290 fclose(fp);
291
292 if(X->Bind.Def.iOutputDataHead==1){
293 sprintf(sdt_flct, "%s_%s", X->Bind.Def.CDataFileHead, cFileNameFlct);
294 }else{
295 sprintf(sdt_flct, "%s", cFileNameFlct);
296 }
297 if (childfopenMPI(sdt_flct, "w", &fp) != 0) {
298 return -1;
299 }
300 fprintf(fp, "%s",cLogFlct);
301 fclose(fp);
302
303 if (X->Bind.Def.iReStart == RESTART_NOT || X->Bind.Def.iReStart == RESTART_OUT) {
304 if (GreenOutputInitializeAggregateFiles(&(X->Bind)) != 0) {
305 return -1;
306 }
307 }
308
309 int iInterAllOffDiagonal_org = X->Bind.Def.NInterAll_OffDiagonal;
310 int iTransfer_org = X->Bind.Def.EDNTransfer;
311 if (step_initial > 0) {
312 /* Raw TE files store the next step. Restore H|psi> at the preceding
313 * time point: MultiplyForTEM consumes both psi (v1) and H psi (v0).
314 * Without this, a restarted run silently loses its first Taylor term. */
315 step_i = X->Bind.Def.istep = step_initial - 1;
316 if (X->Bind.Def.NLaser > 0) {
317 double previous_time = Time + (step_initial - 1) * dt;
318 Time += step_initial * dt;
319 if (!isfinite(previous_time) || !isfinite(Time)) return -1;
320 TransferWithPeierls(&(X->Bind), previous_time);
321 } else if (X->Bind.Def.NTETransferMax > 0) {
322 MakeTEDTransfer(&(X->Bind), step_initial - 1);
323 } else if (X->Bind.Def.NTEInterAllMax > 0) {
324 MakeTEDInterAll(&(X->Bind), step_initial - 1);
325 }
326 for (unsigned long i = 1; i <= X->Bind.Check.idim_max; ++i) v0[i] = v1[i];
327 if (expec_energy_flct(&(X->Bind)) != 0) return -1;
328 }
329 int progress_interval = X->Bind.Def.Lanczos_max / 10;
330 if (progress_interval < 1) progress_interval = 1; /* avoid % 0 when Lanczos_max < 10 */
331 for (step_i = step_initial; step_i < X->Bind.Def.Lanczos_max; step_i++) {
332 X->Bind.Def.istep = step_i;
333
334 //Reset total number of interactions (changed in MakeTED***function.)
335 X->Bind.Def.EDNTransfer = iTransfer_org;
336 X->Bind.Def.NInterAll_OffDiagonal = iInterAllOffDiagonal_org;
337
338 if (step_i % progress_interval == 0) {
339 fprintf(stdoutMPI, cLogTEStep, step_i, X->Bind.Def.Lanczos_max);
340 }
341
342 if (view) {
343 Time = SymmetryTETime(&view->base, step_i);
344 dt = step_i ? Time - SymmetryTETime(&view->base, step_i - 1) : 0;
345 if (SelectSymmetryTEHamiltonian(&(X->Bind), view, step_i, Time) != 0 ||
346 RebuildSymmetryTEPlan(&(X->Bind)) != 0) return -1;
347 X->Bind.Def.Param.TimeSlice = dt;
348 /* All Taylor powers use the SAME current H, including the first one. */
349 for (unsigned long i = 1; i <= X->Bind.Check.idim_max; ++i) v0[i] = v1[i];
350 if (expec_energy_flct(&(X->Bind)) != 0) return -1;
351 }
352 else if(X->Bind.Def.NLaser !=0) {
353 TransferWithPeierls(&(X->Bind), Time);
354 }
355 else {
356 // common procedure
357 Time = X->Bind.Def.TETime[step_i];
358 if (step_i == 0) dt = 0.0;
359 else {
360 dt = X->Bind.Def.TETime[step_i] - X->Bind.Def.TETime[step_i - 1];
361 }
362 X->Bind.Def.Param.TimeSlice = dt;
363
364 // Set interactions
365 if(X->Bind.Def.NTETransferMax != 0 && X->Bind.Def.NTEInterAllMax!=0){
366 fprintf(stdoutMPI,
367 "Error: Time Evolution mode does not support TEOneBody and TETwoBody interactions at the same time. \n");
368 return -1;
369 }
370 else if (X->Bind.Def.NTETransferMax > 0) { //One-Body type
371 MakeTEDTransfer(&(X->Bind), step_i);
372 }else if (X->Bind.Def.NTEInterAllMax > 0) { //Two-Body type
373 MakeTEDInterAll(&(X->Bind), step_i);
374 }
375 //[e] Yoshimi
376 }
377
378 if(step_i == step_initial){
379 TimeKeeperWithStep(&(X->Bind), cFileNameTEStep, cTEStep, "w", step_i);
380 }
381 else {
382 TimeKeeperWithStep(&(X->Bind), cFileNameTEStep, cTEStep, "a", step_i);
383 }
384 if (MultiplyForTEM(&(X->Bind)) != 0) return -1;
385 //Add Diagonal Parts
386 //Multiply Diagonal
387 if (expec_energy_flct(&(X->Bind)) != 0) return -1;
388
389 if(!view && X->Bind.Def.NLaser >0 ) Time+=dt;
390 if (childfopenMPI(sdt_phys, "a", &fp) != 0) {
391 return -1;
392 }
393 fprintf(fp, "%.16lf %.16lf %.16lf %.16lf %.16lf %d\n", Time, X->Bind.Phys.energy, X->Bind.Phys.var,
394 X->Bind.Phys.doublon, X->Bind.Phys.num, step_i);
395 fclose(fp);
396
397 if (childfopenMPI(sdt_norm, "a", &fp) != 0) {
398 return -1;
399 }
400 fprintf(fp, "%.16lf %.16lf %d\n", Time, global_norm, step_i);
401 fclose(fp);
402
403 if (childfopenMPI(sdt_flct, "a", &fp) != 0) {
404 return -1;
405 }
406 fprintf(fp, "%.16lf %.16lf %.16lf %.16lf %.16lf %.16lf %.16lf %d\n", Time,X->Bind.Phys.num,X->Bind.Phys.num2, X->Bind.Phys.doublon,X->Bind.Phys.doublon2, X->Bind.Phys.Sz,X->Bind.Phys.Sz2,step_i);
407 fclose(fp);
408
409
410
411 if (step_i % step_spin == 0) {
412 if (expec_cisajs(&(X->Bind), v1) != 0) {
413 return -1;
414 }
415 if (expec_cisajscktaltdc(&(X->Bind), v1) != 0) {
416 return -1;
417 }
418 if (expec_nbodyg(&(X->Bind), v1) != 0) {
419 return -1;
420 }
421 if (expec_anomalousg(&(X->Bind), v1) != 0) {
422 return -1;
423 }
424 }
425 if (X->Bind.Def.iOutputEigenVec == TRUE) {
426 if (step_i % X->Bind.Def.Param.OutputInterval == 0) {
427 if (X->Bind.Def.iFlgSymmetryBasis) {
428 if (write_sector_te_vector(&(X->Bind), step_i, Time, source.state_index, 0) != 0) return -1;
429 } else if (write_raw_te_vector(&(X->Bind), step_i, step_i + 1) != 0) return -1;
430 }
431 }
432 }
433 if (X->Bind.Def.iOutputEigenVec == TRUE) {
434 if (X->Bind.Def.iFlgSymmetryBasis) {
435 if (write_sector_te_vector(&(X->Bind), step_i - 1, Time, source.state_index, 1) != 0) return -1;
436 } else if (write_raw_te_vector(&(X->Bind), rand_i, step_i) != 0) return -1;
437 }
438
439 fprintf(stdoutMPI, "%s",cLogTEM_End);
440 return 0;
441}

443int CalcByTEM(const int ExpecInterval, struct EDMainCalStruct *X)
444{
445 if (!SymmetryTEIsDynamic(&X->Bind.Def)) return calc_by_tem(ExpecInterval, X, NULL);
446 struct SymmetryTEHamiltonian view = {0};

457void MakeTEDTransfer(struct BindStruct *X, const int timeidx) {
458 int i,j;
459 //Clear values
460 for(i=0; i<X->Def.NTETransferMax ;i++) {
461 for(j =0; j<4; j++) {
462 X->Def.EDGeneralTransfer[i + X->Def.EDNTransfer][j] = 0;
463 }
464 X->Def.EDParaGeneralTransfer[i+X->Def.EDNTransfer]=0.0;
465 }
466
467 //Input values
468 for(i=0; i<X->Def.NTETransfer[timeidx] ;i++){
469 for(j =0; j<4; j++) {
470 X->Def.EDGeneralTransfer[i + X->Def.EDNTransfer][j] = X->Def.TETransfer[timeidx][i][j];
471 }
472 X->Def.EDParaGeneralTransfer[i+X->Def.EDNTransfer]=X->Def.ParaTETransfer[timeidx][i];
473 }
474 X->Def.EDNTransfer += X->Def.NTETransfer[timeidx];
475}

480void MakeTEDInterAll(struct BindStruct *X, const int timeidx) {
481 int i, j;
482 //Clear values
483 for (i = 0; i < X->Def.NTEInterAllMax; i++) {
484 for (j = 0; j < 8; j++) {
485 X->Def.InterAll_OffDiagonal[i + X->Def.NInterAll_OffDiagonal][j] = 0;
486 }
487 X->Def.ParaInterAll_OffDiagonal[i + X->Def.NInterAll_OffDiagonal] = 0.0;

489
490 //Input values
491 for (i = 0; i < X->Def.NTEInterAllOffDiagonal[timeidx]; i++) {
492 for (j = 0; j < 8; j++) {
493 X->Def.InterAll_OffDiagonal[i + X->Def.NInterAll_OffDiagonal][j] = X->Def.TEInterAllOffDiagonal[timeidx][i][j];
494 }
495 X->Def.ParaInterAll_OffDiagonal[i + X->Def.NInterAll_OffDiagonal] = X->Def.ParaTEInterAllOffDiagonal[timeidx][i];
496 }
497 X->Def.NInterAll_OffDiagonal += X->Def.NTEInterAllOffDiagonal[timeidx];
498}
