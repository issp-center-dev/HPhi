115int mltplySpinlessFermion(struct BindStruct *X, double complex *tmp_v0, double complex *tmp_v1) {
116 long unsigned int i, j = 0;
117 long unsigned int isite1, isite2;
118
119 double complex dam_pr;
120 double complex tmp_trans;
121
122 int ihermite = 0;
123 int idx = 0;
124 int isGC = (X->Def.iCalcModel == SpinlessFermionGC);
125 dam_pr = 0.0;
126
127 //Transfer
128 // Disable batching for TimeEvolution with step-dependent transfers, since
129 // the cached batched groups (initialized once on the first call) become
130 // stale when MakeTEDTransfer updates the transfer list each step. The
131 // Hubbard path applies the same guard (mltplyHubbard.c).
132 int use_batching = MPIBatchingEnabled() &&
133 !(X->Def.iCalcType == TimeEvolution &&
134 X->Def.NTETransferMax > 0);
135 // Time the whole transfer block; the matching StopTimer calls are at the
136 // end of the function and run unconditionally for both the batched and
137 // the per-term branches below.
138 StartTimer(600);
139 StartTimer(610);
140 if (X->Large.mode == M_MLTPLY && use_batching) {
141 for (i = 0; i < X->Def.EDNTransfer; i += 2) {
142 if (X->Def.EDGeneralTransfer[i][0] + 1 > X->Def.Nsite &&
143 X->Def.EDGeneralTransfer[i][2] + 1 > X->Def.Nsite) {
144 StartTimer(611);
145 if (isGC) {
146 child_GC_general_hopp_SpinlessFermion_MPIdouble(i, X, tmp_v0, tmp_v1);
147 } else {
148 child_general_hopp_Spinless_MPIdouble(i, X, tmp_v0, tmp_v1);
149 }
150 StopTimer(611);
151 } else {
152 continue;
153 }
154 }
155
156#ifdef MPI
157 // Use batched MPI communication for SpinlessFermionGC
158 if (isGC) {
159 StartTimer(612);
160 // Initialize batched transfers on first call
161 if (!batched_transfers_GC_initialized) {
162 if (InitializeMPIBatchedTransfers_SpinlessFermionGC(X, &batched_transfers_GC) != 0) {
163 fprintf(stderr, "Error: Failed to initialize batched MPI transfers\n");
164 return -1;
165 }
166 batched_transfers_GC_initialized = 1;
167 }
168
169 // Process all groups with batched communication
170 for (int g = 0; g < batched_transfers_GC.num_groups; g++) {
171 dam_pr = X_child_GC_general_hopp_SpinlessFermion_MPIsingle_batched(
172 &batched_transfers_GC.groups[g], X, tmp_v0, tmp_v1);
173 X->Large.prdct += dam_pr;
174 }
175 StopTimer(612);
176 } else {
177 // Canonical mode: use batched MPI processing
178 StartTimer(612);
179 if (!batched_transfers_canonical_initialized) {
180 if (InitializeMPIBatchedTransfers_SpinlessFermion(X, &batched_transfers_canonical) != 0) {
181 fprintf(stderr, "Error: Failed to initialize batched MPI transfers (canonical)\n");
182 return -1;
183 }
184 batched_transfers_canonical_initialized = 1;
185 }
186
187 for (int g = 0; g < batched_transfers_canonical.num_groups; g++) {
188 dam_pr = X_child_general_hopp_Spinless_MPIsingle_batched(
189 &batched_transfers_canonical.groups[g], X, tmp_v0, tmp_v1);
190 X->Large.prdct += dam_pr;
191 }
192 StopTimer(612);
193 }
194#else
195 // Non-MPI: skip inter-process transfers
196 (void)i;
197#endif
198
199 for (i = 0; i < X->Def.EDNTransfer; i += 2) {
200 if (X->Def.EDGeneralTransfer[i][0] + 1 > X->Def.Nsite ||
201 X->Def.EDGeneralTransfer[i][2] + 1 > X->Def.Nsite) {
202 continue;
203 } else {
204 StartTimer(613);
205 isite1 = X->Def.EDGeneralTransfer[i][0] + 1;
206 isite2 = X->Def.EDGeneralTransfer[i][2] + 1;
207 if (child_general_hopp_GetInfo_Spinless(X, isite1, isite2) != 0) {
208 return -1;
209 }
210 tmp_trans = -X->Def.EDParaGeneralTransfer[i];
211 dam_pr = child_general_hopp_Spinless(tmp_v0, tmp_v1, X, tmp_trans);
212 X->Large.prdct += dam_pr;
213 StopTimer(613);
214 }
215 }
216 } else {
217 for (i = 0; i < X->Def.EDNTransfer; i += 2) {
218 if (X->Def.EDGeneralTransfer[i][0] + 1 > X->Def.Nsite &&
219 X->Def.EDGeneralTransfer[i][2] + 1 > X->Def.Nsite) {
220 StartTimer(611);
221 if (isGC) {
222 child_GC_general_hopp_SpinlessFermion_MPIdouble(i, X, tmp_v0, tmp_v1);
223 } else {
224 child_general_hopp_Spinless_MPIdouble(i, X, tmp_v0, tmp_v1);
225 }
226 StopTimer(611);
227 } else if (X->Def.EDGeneralTransfer[i][2] + 1 > X->Def.Nsite) {
228 StartTimer(612);
229 if (isGC) {
230 child_GC_general_hopp_SpinlessFermion_MPIsingle(i, X, tmp_v0, tmp_v1);
231 } else {
232 child_general_hopp_Spinless_MPIsingle(i, X, tmp_v0, tmp_v1);
233 }
234 StopTimer(612);
235 } else if (X->Def.EDGeneralTransfer[i][0] + 1 > X->Def.Nsite) {
236 StartTimer(612);
237 if (isGC) {
238 child_GC_general_hopp_SpinlessFermion_MPIsingle(i + 1, X, tmp_v0, tmp_v1);
239 } else {
240 child_general_hopp_Spinless_MPIsingle(i + 1, X, tmp_v0, tmp_v1);
241 }
242 StopTimer(612);
243 } else {
244 StartTimer(613);
245 isite1 = X->Def.EDGeneralTransfer[i][0] + 1;
246 isite2 = X->Def.EDGeneralTransfer[i][2] + 1;
247 if (child_general_hopp_GetInfo_Spinless(X, isite1, isite2) != 0) {
248 return -1;
249 }
250 tmp_trans = -X->Def.EDParaGeneralTransfer[i];
251 dam_pr = child_general_hopp_Spinless(tmp_v0, tmp_v1, X, tmp_trans);
252 X->Large.prdct += dam_pr;
253 StopTimer(613);
254 }
255 }
256 }
257 StopTimer(610);
258 StopTimer(600);
259
260 if (X->Def.NNBodyInterAll_OffDiagonal > 0) {
261 if (isGC) {
262 if (MultiplyNBodyInterAllSpinlessGC(X, tmp_v0, tmp_v1) != 0) {
263 return -1;
264 }
265 }
266 else {
267 if (MultiplyNBodyInterAllSpinless(X, tmp_v0, tmp_v1) != 0) {
268 return -1;
269 }
270 }
271 }
272
273 return 0;
274}

306double complex child_general_hopp_Spinless
307 (
308 double complex *tmp_v0,
309 double complex *tmp_v1,
310 struct BindStruct *X,
311 double complex trans
312 ) {
313
314 long unsigned int j, isite1, isite2, Asum, Adiff;
315 long unsigned int i_max = X->Large.i_max;
316
317 isite1 = X->Large.is1_spin;
318 isite2 = X->Large.is2_spin;
319 Asum = X->Large.isA_spin;
320 Adiff = X->Large.A_spin;
321
322 double complex dam_pr = 0;
323 double complex X_dam_pr = 0;
324 double complex tmp_trans;
325#pragma omp parallel for default(none) reduction(+:dam_pr) firstprivate(i_max, X, Asum, Adiff, isite1, isite2, trans) private(j, X_dam_pr, tmp_trans) shared(tmp_v0, tmp_v1)
326 for (j = 1; j <= i_max; j++) {
327 tmp_trans = trans;
328 X_dam_pr = CisAjt_Hermite(j, tmp_v0, tmp_v1, X, isite1, isite2, Asum, Adiff, &tmp_trans);
329 dam_pr += X_dam_pr * tmp_trans;
330 }
331 return dam_pr;
332}

450double complex CisAjt_Hermite(
451 long unsigned int j,
452 double complex *tmp_v0,
453 double complex *tmp_v1,
454 struct BindStruct *X,
455 long unsigned int is1_spin,
456 long unsigned int is2_spin,
457 long unsigned int sum_spin,
458 long unsigned int diff_spin,
459 double complex *tmp_V
460) {
461 long unsigned int ibit_tmp_1, ibit_tmp_2;
462 long unsigned int bit, iexchg, off;
463 long unsigned int org_bit;
464 int sgn;
465 double complex dmv, dam_pr;
466 dam_pr = 0.0;
467
468 // For GC mode, use (j-1) as the bit representation; for canonical, use list_1[j]
469 if (X->Def.iCalcModel == SpinlessFermionGC) {
470 org_bit = j - 1;
471 } else {
472 org_bit = list_1[j];
473 }
474
475 ibit_tmp_1 = org_bit & sum_spin;
476 if (ibit_tmp_1 == 0 || ibit_tmp_1 == sum_spin) {
477 return (0.0);
478 }
479
480 ibit_tmp_1 = (org_bit & is1_spin);
481 ibit_tmp_2 = (org_bit & is2_spin);
482 if (ibit_tmp_1 != 0 && ibit_tmp_2 == 0) {
483 *tmp_V = conj(*tmp_V);
484 } else if (!(ibit_tmp_1 == 0 && ibit_tmp_2 != 0)) {
485 return (0.0);
486 }
487 bit = org_bit & diff_spin;
488 SgnBit(bit, &sgn);
489 iexchg = org_bit ^ sum_spin;
490
491 // For GC mode, calculate off directly; for canonical, use GetOffComp
492 if (X->Def.iCalcModel == SpinlessFermionGC) {
493 off = iexchg + 1;
494 } else {
495 if (GetOffComp(list_2_1, list_2_2, iexchg, X->Large.irght, X->Large.ilft, X->Large.ihfbit, &off) != TRUE) {
496 return (0.0);
497 }
498 }
499
500 dmv = sgn * tmp_v1[j];
501 // Build the output vector both for H|psi> (M_MLTPLY) and for the excited state
502 // in spectrum calculations (M_CALCSPEC); M_CORR only needs dam_pr.
503 if (X->Large.mode == M_MLTPLY || X->Large.mode == M_CALCSPEC) {
504 tmp_v0[off] += *tmp_V * dmv;
505 }
506 dam_pr = dmv * conj(tmp_v1[off]);
507
508 return dam_pr;
509}
