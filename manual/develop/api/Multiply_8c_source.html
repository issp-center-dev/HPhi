44int Multiply
45(
46 struct BindStruct *X
47 )
48{
49 long int i,i_max;
50 double complex dnorm;
51 double Ns;
52
53 int invalid = X == NULL || v0 == NULL || v1 == NULL || !isfinite(LargeValue);
54 if (!invalid && (X->Check.idim_max > LONG_MAX || X->Def.NsiteMPI == 0)) invalid = 1;
55 if (SumMPI_i(invalid) != 0) {
56 fprintf(stdoutMPI, "Error: invalid mTPQ step storage or scaling parameters.\n");
57 return -1;
58 }
59 i_max=X->Check.idim_max;
60 Ns = 1.0*X->Def.NsiteMPI;
61 // mltply is in expec_energy.c v0=H*v1
62 dnorm=0.0;
63#pragma omp parallel for default(none) reduction(+: dnorm) private(i) shared(v0, v1) firstprivate(i_max, Ns, LargeValue)
64 for(i = 1; i <= i_max; i++){
65 v0[i]=LargeValue*v1[i]-v0[i]/Ns; //v0=(l-H/Ns)*v1
66 dnorm += conj(v0[i])*v0[i];
67 }
68 dnorm=SumMPI_dc(dnorm);
69 if (!isfinite(creal(dnorm)) || !isfinite(cimag(dnorm)) || creal(dnorm) <= 0.0) {
70 fprintf(stdoutMPI, "Error: mTPQ step has zero or non-finite global norm.\n");
71 return -1;
72 }
73 dnorm=sqrt(dnorm);
74 global_norm = dnorm;
75#pragma omp parallel for default(none) private(i) shared(v0) firstprivate(i_max, dnorm)
76 for(i=1;i<=i_max;i++){
77 v0[i] = v0[i]/dnorm;
78 }
79 return 0;
80}

91int MultiplyForTEM
92(
93 struct BindStruct *X
94 )
95{
96
97 long int i,i_max;
98 int coef;
99 double complex dnorm=0.0;
100 double complex tmp1 = 1.0;
101 double complex tmp2=0.0;
102 double dt = 0;
103 int invalid = X == NULL || v0 == NULL || v1 == NULL || v2 == NULL;
104 if (!invalid) {
105 dt = X->Def.Param.TimeSlice;
106 invalid = !isfinite(dt) || dt < 0 || X->Check.idim_max > LONG_MAX ||

108 }
109 if (SumMPI_i(invalid) != 0) {
110 fprintf(stdoutMPI, "Error: invalid TE step storage, time step or Taylor order.\n");
111 return -1;
112 }
113
114 //Make |v0> = |psi(t+dt)> from |v1> = |psi(t)> and |v0> = H |psi(t)>
115 i_max=X->Check.idim_max;
116 // mltply is in expec_energy.c v0=H*v1
117 if(dt == 0.0){
118#pragma omp parallel for default(none) reduction(+: dnorm) private(i) shared(v0, v1, v2) firstprivate(i_max, dt, tmp2)
119 for(i = 1; i <= i_max; i++){
120 tmp2 = v0[i];
121 v0[i]=v1[i]; //v0=(1-i*dt*H)*v1
122 v1[i]=tmp2;
123 v2[i]= 0.0 + I*0.0;
124 }
125 if (SumMPI_i(mltply(X, v2, v1) != 0) != 0) return -1;
126 }
127 else {
128 tmp1 *= -I * dt;
129#pragma omp parallel for default(none) reduction(+: dnorm) private(i) shared(v0, v1, v2) firstprivate(i_max, dt, tmp1, tmp2)
130 for (i = 1; i <= i_max; i++) {
131 tmp2 = v0[i];
132 v0[i] = v1[i] + tmp1 * tmp2; //v0=(1-i*dt*H)*v1
133 v1[i] = tmp2;
134 v2[i] = 0.0 + I * 0.0;
135 }
136 for (coef = 2; coef <= X->Def.Param.ExpandCoef; coef++) {
137 tmp1 *= -I * dt / (double complex) coef;
138 //v2 = H*v1 = H^coef |psi(t)>
139 if (SumMPI_i(mltply(X, v2, v1) != 0) != 0) return -1;
140
141#pragma omp parallel for default(none) private(i) shared(v0, v1, v2) firstprivate(i_max, tmp1, myrank)
142 for (i = 1; i <= i_max; i++) {
143 v0[i] += tmp1 * v2[i];
144 v1[i] = v2[i];
145 v2[i] = 0.0 + I * 0.0;
146 }
147 }
148 }
149 dnorm=0.0;
150#pragma omp parallel for default(none) reduction(+: dnorm) private(i) shared(v0) firstprivate(i_max, dt)
151 for(i = 1; i <= i_max; i++){
152 dnorm += conj(v0[i])*v0[i];
153 }
154 dnorm=SumMPI_dc(dnorm);
155 if (!isfinite(creal(dnorm)) || !isfinite(cimag(dnorm)) || creal(dnorm) <= 0) {
156 fprintf(stdoutMPI, "Error: TE step has zero or non-finite global norm.\n");

158 }

160 global_norm = dnorm;
161#pragma omp parallel for default(none) private(i) shared(v0) firstprivate(i_max, dnorm)
162 for(i=1;i<=i_max;i++){
163 v0[i] = v0[i]/dnorm;
164 }
165 return 0;
