24int TPQRestartPlanSets(struct BindStruct *X, int numave,
25 struct TPQRestartPlan *plan)
26{
27 int failed;
28 if (plan == NULL) return -1;
29 memset(plan, 0, sizeof(*plan));
30 plan->numave = numave;
31 failed = X == NULL || numave <= 0;
32 if (!failed) {
33 plan->sources = calloc((size_t)numave, sizeof(*plan->sources));
34 plan->legacy_steps = calloc((size_t)numave, sizeof(*plan->legacy_steps));
35 failed = plan->sources == NULL || plan->legacy_steps == NULL;
36 }
37 if (SumMPI_i(failed) != 0) {
38 fprintf(stdoutMPI, "Error: TPQ restart plan allocation failed.\n");
39 TPQRestartFreePlan(plan);
40 return -1;
41 }
42 if (X->Def.iReStart != RESTART_INOUT && X->Def.iReStart != RESTART_IN) {
43 plan->boundaries_validated = 1;
44 return 0;
45 }
46 if (RestartManifestLoad(X, &plan->manifest, &plan->manifest_exists) != 0)
47 goto fail;
48 if (plan->manifest_exists && plan->manifest.method != X->Def.iCalcType) {
49 fprintf(stdoutMPI,
50 "Error: restart manifest belongs to another method.\n");
51 goto fail;
52 }
53 for (int set = 0; set < numave; ++set) {
54 if (RestartClassifySet(X,
55 plan->manifest_exists ? &plan->manifest : NULL,
56 (uint64_t)set, cFileNameInputVector, &plan->sources[set],
57 &plan->legacy_steps[set]) != 0)
58 goto fail;
59 if (plan->sources[set] == RESTART_SOURCE_NONE &&
60 X->Def.iReStart == RESTART_IN) {
61 fprintf(stdoutMPI,
62 "Error: ReStart=3 requires a checkpoint for every set "
63 "(set %d is missing).\n", set);
64 goto fail;
65 }
66 plan->any_resume |= plan->sources[set] != RESTART_SOURCE_NONE;
67 }
68 /* Each set is an independent publication unit. Its headers have already
69 * been checked against the runtime logical basis; compatible metadata
70 * layouts may differ between sets after an interrupted ReStart=2. */
71 plan->boundaries_validated = 1;
72 return 0;
73fail:
74 TPQRestartFreePlan(plan);
75 return -1;
76}

97int TPQRestartRecoverOutputs(struct BindStruct *X,
98 const struct TPQRestartPlan *plan,
99 int aggregate)
100{
101 int failed = X == NULL || plan == NULL || !plan->boundaries_validated;
102 if (SumMPI_i(failed) != 0) return -1;
103 if (!plan->any_resume) return 0;
104 struct RestartSeriesRule *rules =
105 calloc((size_t)plan->numave, sizeof(*rules));
106 failed = rules == NULL;
107#ifdef HPHI_TEST_SYMMETRY_PROBE
108 if (SymmetryFullDiagTestFail("restart_recover_alloc")) failed = 1;
109#endif
110 if (SumMPI_i(failed) != 0) {
111 free(rules);
112 fprintf(stdoutMPI, "Error: restart recovery allocation failed.\n");
113 return -1;
114 }
115 int nrule = 0;
116 for (int set = 0; set < plan->numave; ++set) {
117 if (plan->sources[set] == RESTART_SOURCE_NONE) continue;
118 rules[nrule].set = (uint64_t)set;
119 rules[nrule].keep_below_step = keep_below(plan, set);
120 ++nrule;
121 }
122 if (aggregate) {
123 const GreenOutputTPQDataKind kinds[] = {
124 GreenOutputTPQDataSS, GreenOutputTPQDataNorm, GreenOutputTPQDataFlct
125 };
126 struct RestartSeriesSpec spec = {"#", 0, 1, nrule, rules, 1};
127 for (int kind = 0; kind < 3; ++kind) {
128 char name[D_FileNameMax];
129 failed = GreenOutputTPQDataFileName(X, kinds[kind], name) != 0;
130 if (SumMPI_i(failed) != 0 ||
131 RestartRecoverTimeSeries(X, name, &spec) != 0) {
132 free(rules);
133 return -1;
134 }
135 }
136 } else {
137 const char *patterns[] = {
138 cFileNameSSRand, cFileNameNormRand, cFileNameFlctRand
139 };
140 for (int set = 0; set < plan->numave; ++set) {
141 if (plan->sources[set] == RESTART_SOURCE_NONE) continue;
142 struct RestartSeriesRule one = {(uint64_t)set, keep_below(plan, set)};
143 struct RestartSeriesSpec spec = {"#", -1, -1, 1, &one, 0};
144 for (int kind = 0; kind < 3; ++kind) {
145 char name[D_FileNameMax];
146 int n;
147 if (X->Def.iOutputDataHead == 1) {
148 n = snprintf(name, sizeof(name), "%s_", X->Def.CDataFileHead);
149 if (n >= 0 && n < (int)sizeof(name))
150 n += snprintf(name + n, sizeof(name) - (size_t)n,
151 patterns[kind], set);
152 } else {
153 n = snprintf(name, sizeof(name), patterns[kind], set);
154 }
155 failed = n < 0 || n >= (int)sizeof(name);
156 if (SumMPI_i(failed) != 0 ||
157 RestartRecoverTimeSeries(X, name, &spec) != 0) {
158 free(rules);
159 return -1;
160 }
161 }
162 }
163 }
164 if (GreenOutputUsesAggregate(X) &&
165 recover_green_aggregates(X, rules, nrule) != 0) {
166 free(rules);
167 return -1;
168 }
169 free(rules);
170 return 0;
171}
