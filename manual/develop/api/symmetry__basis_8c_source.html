43unsigned long int SymmetryApplyToSpinBits(unsigned long int state,
44 const int *perm,
45 unsigned int nsite)
46{
47 const unsigned int max_bits =
48 (unsigned int)(sizeof(unsigned long int) * CHAR_BIT);
49 unsigned long int out = 0;
50 if (perm == NULL || nsite > max_bits) return 0UL;
51 if (nsite < max_bits) state &= (1UL << nsite) - 1UL;
52 while (state != 0UL) {
53 unsigned int site = symmetry_ctz_ulong(state);
54 int target = perm[site];
55#ifdef HPHI_SYMMETRY_SPIN_VERIFY
56 if (site != portable_ctz_ulong(state)) return 0UL;
57#endif
58 if (target < 0 || (unsigned int)target >= nsite) return 0UL;
59 out |= 1UL << (unsigned int)target;
60 state &= state - 1UL;
61 }
62 return out;
63}

102static int apply_fermion_site_permutation(unsigned long int state,
103 const int *perm,
104 unsigned int nsite,
105 unsigned int orbitals_per_site,
106 struct SymmetryTransformResult *result)
107{
108 const unsigned int max_bits = (unsigned int)(sizeof(unsigned long int) * CHAR_BIT);
109 unsigned int norb = nsite * orbitals_per_site;
110 unsigned int orb;
111#ifdef HPHI_SYMMETRY_PARITY_VERIFY
112 unsigned int count = 0;
113 unsigned int mapped_orbitals[sizeof(unsigned long int) * CHAR_BIT];
114#endif
115 unsigned int inversion_parity = 0U;
116 unsigned long int seen = 0UL;
117 unsigned long int out = 0UL;
118 if (orbitals_per_site == 0U || nsite > max_bits / orbitals_per_site) return -1;
119 if (norb > max_bits) return -1;
120 for (orb = 0; orb < norb; orb++) {
121 if ((state & (1UL << orb)) != 0UL) {
122 unsigned int site = orb / orbitals_per_site;
123 unsigned int spin = orb % orbitals_per_site;
124 unsigned int target_site = (unsigned int)perm[site];
125 unsigned int mapped = orbitals_per_site * target_site + spin;
126 unsigned long int higher_seen;
127 if (mapped >= max_bits) return -1;
128#ifdef HPHI_SYMMETRY_PARITY_VERIFY
129 mapped_orbitals[count++] = mapped;
130#endif
131 /* Split the shift so mapped == max_bits - 1 never shifts by the
132 word width. */
133 higher_seen = (seen >> mapped) >> 1U;
134 inversion_parity ^= symmetry_popcount_ulong(higher_seen) & 1U;
135#ifdef HPHI_SYMMETRY_PARITY_VERIFY
136 if (symmetry_popcount_ulong(higher_seen) !=
137 portable_popcount_ulong(higher_seen)) {
138 return -1;
139 }
140#endif
141 seen |= 1UL << mapped;
142 out |= (1UL << mapped);
143 }
144 }
145 result->state = out;
146 result->amplitude = inversion_parity == 0U ? 1.0 : -1.0;
147#ifdef HPHI_SYMMETRY_PARITY_VERIFY
148 if ((int)result->amplitude !=
149 reference_parity_sign_from_mapped_orbitals(mapped_orbitals, count)) {
150 return -1;
151 }
152#endif
153 return 0;
154}

156int SymmetryApplyToState(const struct DefineList *def,
157 unsigned long int state,
158 unsigned int op,
159 struct SymmetryTransformResult *result)
160{
161 if (def == NULL || result == NULL || op >= def->NSymTrans) return -1;
162 memset(result, 0, sizeof(*result));
163 switch (def->iCalcModel) {
164 case Spin:
165 result->state = SymmetryApplyToSpinBits(state, def->SymTrans[op], def->Nsite);
166 result->amplitude = 1.0;
167 return 0;
168 case SpinlessFermion:
169 return apply_fermion_site_permutation(state, def->SymTrans[op], def->Nsite, 1U, result);
170 case Hubbard:
171 return apply_fermion_site_permutation(state, def->SymTrans[op], def->Nsite, 2U, result);
172 default:
173 return -1;
174 }
175}

196 struct SymmetryBasisRuntime *sym)
197{
198 unsigned int g, h, site;
199 if (def->NSymTrans == 0U) return -1;
200#if SIZE_MAX <= UINT_MAX
201 if (def->NSymTrans > SIZE_MAX / sizeof(*sym->group_inverse)) return -1;

216 if (site == def->Nsite) {
217 sym->group_inverse[g] = h;
218 found = TRUE;
219 break;
220 }
221 }

268int ValidateSymmetryGroupInput(const struct DefineList *def)
269{
270 const double eps_ch = 1.0e-10;
271 unsigned int g, h, site;
272 int identity = -1;
273 int *composed;
274
275 if (def->iFlgSymmetryBasis == FALSE) return 0;
276 if (def->NSymTrans == 0) {
277 fprintf(stdoutMPI, "Error: TransSym requires NQPTrans > 0.\n");
278 return -1;
279 }
280 if (validate_bijection_and_anti(def) != 0) return -1;
281
282 for (g = 0; g < def->NSymTrans; g++) {
283 unsigned int prev;
284 double norm = cabs(def->SymTransChar[g]);
285 if (!isfinite(creal(def->SymTransChar[g])) ||
286 !isfinite(cimag(def->SymTransChar[g]))) {
287 fprintf(stdoutMPI, "Error: TransSym character must be finite; op=%u.\n", g);
288 return -1;
289 }
290 if (fabs(norm - 1.0) > eps_ch) {
291 fprintf(stdoutMPI, "Error: TransSym character must have unit norm; op=%u abs=% .16e.\n",
292 g, norm);
293 return -1;
294 }
295 for (prev = 0; prev < g; prev++) {
296 if (same_perm(def->SymTrans[g], def->SymTrans[prev], def->Nsite) == TRUE) {
297 fprintf(stdoutMPI, "Error: duplicate TransSym operation: op %u and op %u are identical.\n",
298 prev, g);
299 return -1;
300 }
301 }
302 if (is_identity_perm(def->SymTrans[g], def->Nsite) == TRUE) {
303 if (identity >= 0) {
304 fprintf(stdoutMPI, "Error: TransSym contains duplicate identity operations.\n");
305 return -1;
306 }
307 identity = (int)g;
308 }
309 }
310 if (identity < 0) {
311 fprintf(stdoutMPI, "Error: TransSym group must contain identity permutation.\n");
312 return -1;
313 }
314 if (cabs(def->SymTransChar[identity] - 1.0) > eps_ch) {
315 fprintf(stdoutMPI, "Error: TransSym identity character must be 1.\n");
316 return -1;
317 }
318
319 composed = (int *)malloc(sizeof(int) * def->Nsite);
320 if (composed == NULL) return -1;
321 for (g = 0; g < def->NSymTrans; g++) {
322 for (h = 0; h < def->NSymTrans; h++) {
323 int gh;
324 for (site = 0; site < def->Nsite; site++) {
325 composed[site] = def->SymTrans[g][def->SymTrans[h][site]];
326 }
327 gh = find_perm(def, composed);
328 if (gh < 0) {
329 free(composed);
330 fprintf(stdoutMPI, "Error: TransSym operations are not closed; op %u * op %u is missing.\n",
331 g, h);
332 return -1;
333 }
334 if (cabs(def->SymTransChar[gh] - def->SymTransChar[g] * def->SymTransChar[h]) > eps_ch) {
335 free(composed);
336 fprintf(stdoutMPI, "Error: TransSym character is not multiplicative for op %u * op %u.\n",
337 g, h);
338 return -1;
339 }
340 }
341 }
342 free(composed);
343 return 0;
344}

397static int ensure_collector_capacity(struct SymmetryBasisCollector *collector,
398 unsigned long int needed)
399{
400 struct SymmetryBasisVector *next;
401 unsigned long int next_capacity;
402 size_t element_count;
403 if (needed <= collector->capacity) return 0;
404 next_capacity = (collector->capacity == 0UL) ? 16UL : collector->capacity;

406 if (next_capacity > ULONG_MAX / 2UL) return -1;

428static int find_representative_state(const struct DefineList *def,
429 const struct SymmetryBasisRuntime *sym,
430 unsigned long int state,
431 unsigned long int *rep,
432 unsigned int *op_rep_to_state,
433 unsigned long long *transform_calls)
434{
435 unsigned int g;
436 if (rep == NULL) return -1;
437 *rep = state;
438 if (op_rep_to_state != NULL) *op_rep_to_state = UINT_MAX;
439 for (g = 0; g < def->NSymTrans; g++) {
440 struct SymmetryTransformResult moved;
441 if (SymmetryApplyToState(def, state, g, &moved) != 0) return -1;
442 if (transform_calls != NULL) (*transform_calls)++;
443 if (moved.state < *rep) {
444 *rep = moved.state;
445 if (op_rep_to_state != NULL && sym != NULL &&
446 sym->group_inverse != NULL) {
447 *op_rep_to_state = sym->group_inverse[g];
448 }
449 } else if (moved.state == *rep && op_rep_to_state != NULL &&
450 sym != NULL && sym->group_inverse != NULL &&
451 sym->group_inverse[g] < *op_rep_to_state) {
452 *op_rep_to_state = sym->group_inverse[g];
453 }
454 }
455 return 0;
456}

458static int analyze_basis_candidate(const struct DefineList *def,
459 unsigned long int state,
460 int *is_representative,
461 unsigned int *orbit_size,
462 unsigned int *stabilizer_size,
463 double complex *stabilizer_sum,
464 unsigned long long *transform_calls)
465{
466 unsigned int g;
467 *is_representative = FALSE;
468 *stabilizer_size = 0;
469 *stabilizer_sum = 0.0;
470 for (g = 0; g < def->NSymTrans; g++) {
471 struct SymmetryTransformResult moved;
472 if (SymmetryApplyToState(def, state, g, &moved) != 0) return -1;
473 if (transform_calls != NULL) (*transform_calls)++;
474 if (moved.state < state) return 0;
475 if (moved.state == state) {
476 (*stabilizer_size)++;
477 *stabilizer_sum += conj(def->SymTransChar[g]) * moved.amplitude;
478 }
479 }
480 if (*stabilizer_size == 0U) return -1;
481 *is_representative = TRUE;
482 *orbit_size = def->NSymTrans / *stabilizer_size;
483 return 0;
484}

498static unsigned long int find_basis_index_by_rep(const struct SymmetryBasisRuntime *sym,
499 unsigned long int rep_state)
500{
501 if (sym->rep_hash_size > 0UL && sym->rep_hash_values != NULL &&
502 sym->rep_hash_keys != NULL) {
503 unsigned long int mask = sym->rep_hash_size - 1UL;
504 unsigned long int slot = rep_state_hash(rep_state) & mask;
505 unsigned long int probes;
506 for (probes = 0; probes < sym->rep_hash_size; probes++) {
507 unsigned long int value = sym->rep_hash_values[slot];
508 if (value == 0UL) return 0UL;
509 if (sym->rep_hash_keys[slot] == rep_state) return value;
510 slot = (slot + 1UL) & mask;
511 }
512 return 0UL;
513 }
514
515 unsigned long int lo = 1;
516 unsigned long int hi = sym->dim;
517 while (lo <= hi) {
518 unsigned long int mid = lo + (hi - lo) / 2;
519 if (sym->basis[mid].rep_state == rep_state) return mid;
520 if (sym->basis[mid].rep_state < rep_state) {
521 lo = mid + 1;
522 } else {
523 hi = mid - 1;
524 }
525 }
526 return 0;
527}

576static int build_rep_hash(struct SymmetryBasisRuntime *sym)
577{
578 unsigned long int i;
579 unsigned long int target_size;
580 if (sym->dim == 0UL) return 0;
581 if (sym->dim > (ULONG_MAX - 1UL) / 2UL) return -1;
582 target_size = next_power_of_two(sym->dim * 2UL + 1UL);
583 if (target_size == 0UL) return -1;
584 if (target_size < 4UL) target_size = 4UL;
585 if (target_size > SIZE_MAX / sizeof(*sym->rep_hash_keys)) return -1;
586
587 sym->rep_hash_keys = (unsigned long int *)calloc((size_t)target_size,
588 sizeof(*sym->rep_hash_keys));
589 sym->rep_hash_values = (unsigned long int *)calloc((size_t)target_size,
590 sizeof(*sym->rep_hash_values));
591 if (sym->rep_hash_keys == NULL || sym->rep_hash_values == NULL) return -1;
592 sym->rep_hash_size = target_size;
593
594 for (i = 1; i <= sym->dim; i++) {
595 if (insert_rep_hash(sym, sym->basis[i].rep_state, i) != 0) return -1;
596 }
597 return 0;
598}

698{

700#ifdef MPI

725 local_error = 1;

732 if (global_error != 0) goto fail;

743 global_error = SumMPI_i(local_error);

776int BuildSymmetryBasis(struct BindStruct *X)
777{
778 unsigned long int raw, full_dim;
779 unsigned long int local_raw_index;
780 unsigned long int rank_raw_count;
781 unsigned long int distribution_chunk;
782 unsigned long int basis_offset;
783 unsigned long int global_representative_candidates;
784 int collector_count = 1;
785 int actual_thread_count = 1;
786 int local_error = 0;
787 int global_error;
788 int thread;
789 struct SymmetryBasisCollector *collectors = NULL;
790 struct SymmetryBasisRuntime *sym;
791
792 if (X->Def.iFlgSymmetryBasis == FALSE) return 0;
793 full_dim = X->Check.idim_max;
794 sym = (struct SymmetryBasisRuntime *)calloc(1, sizeof(*sym));
795 global_error = SumMPI_i(sym == NULL ? 1 : 0);
796 if (global_error != 0) {
797 free(sym);
798 return -1;
799 }
800
801 sym->enabled = TRUE;
802 sym->nsite = X->Def.Nsite;
803 sym->group_order = X->Def.NSymTrans;
804 sym->full_dim = full_dim;
805 local_error = build_group_inverse(&X->Def, sym) != 0 ? 1 : 0;
806 global_error = SumMPI_i(local_error);
807 if (global_error != 0) goto fail;
808 distribution_chunk = symmetry_distribution_chunk(full_dim, nproc);
809 rank_raw_count = symmetry_rank_raw_count(full_dim, distribution_chunk,
810 myrank, nproc);
811
812#ifdef _OPENMP
813 collector_count = omp_get_max_threads();
814#endif
815 if (collector_count < 1) goto fail;
816 collectors = (struct SymmetryBasisCollector *)calloc(
817 (size_t)collector_count, sizeof(*collectors));
818 global_error = SumMPI_i(collectors == NULL ? 1 : 0);
819 if (global_error != 0) goto fail;
820
821 StartTimer(1110);
822#ifdef _OPENMP
823#pragma omp parallel shared(actual_thread_count, collectors, X, rank_raw_count, distribution_chunk)
824#endif
825 {
826 int thread_id = 0;
827 struct SymmetryBasisCollector *collector;
828#ifdef _OPENMP
829 thread_id = omp_get_thread_num();
830#pragma omp single
831 actual_thread_count = omp_get_num_threads();
832#endif
833 collector = &collectors[thread_id];
834#ifdef _OPENMP
835 /* Cyclic chunks spread representative-heavy regions while retaining
836 locality in list_1 and list_Diagonal. */
837#pragma omp for schedule(static, distribution_chunk)
838#endif
839 for (local_raw_index = 0UL; local_raw_index < rank_raw_count;
840 local_raw_index++) {
841 unsigned long int raw_index;
842 unsigned long int state;
843 int is_representative = FALSE;
844 unsigned int orbit_size = 0;
845 unsigned int stabilizer_size = 0;
846 double complex stabilizer_sum = 0.0;
847 double diagonal;
848 if (collector->error != 0) continue;
849 collector->raw_states++;
850 raw_index = symmetry_rank_raw_index(local_raw_index, distribution_chunk,
851 myrank, nproc);
852 state = list_1[raw_index];
853 diagonal = (list_Diagonal != NULL) ? list_Diagonal[raw_index] : 0.0;
854 if (analyze_basis_candidate(&X->Def, state, &is_representative,
855 &orbit_size, &stabilizer_size,
856 &stabilizer_sum,
857 &collector->transform_calls) != 0) {
858 collector->error = 1;
859 continue;
860 }
861 if (is_representative != TRUE) continue;
862 collector->representative_candidates++;
863 if (cabs(stabilizer_sum) < 0.5) continue;
864 if (append_basis_vector(collector, state, orbit_size, stabilizer_size,
865 stabilizer_sum, diagonal) != 0) {
866 collector->error = 1;
867 continue;
868 }
869 collector->compatible_survivors++;
870 }
871 }
872
873 for (thread = 0; thread < actual_thread_count; thread++) {
874 struct SymmetryBasisCollector *collector = &collectors[thread];
875 if (collector->error != 0 ||
876 sym->dim > ULONG_MAX - collector->count) {
877 local_error = 1;
878 continue;
879 }
880 sym->dim += collector->count;
881 sym->basis_raw_states += collector->raw_states;
882 sym->basis_representative_candidates +=
883 collector->representative_candidates;
884 sym->basis_compatible_survivors += collector->compatible_survivors;
885 sym->basis_transform_calls += collector->transform_calls;
886 if (collector->raw_states > sym->basis_thread_raw_states_max)
887 sym->basis_thread_raw_states_max = collector->raw_states;
888 if (collector->representative_candidates >
889 sym->basis_thread_representative_candidates_max)
890 sym->basis_thread_representative_candidates_max =
891 collector->representative_candidates;
892 if (collector->compatible_survivors >
893 sym->basis_thread_compatible_survivors_max)
894 sym->basis_thread_compatible_survivors_max =
895 collector->compatible_survivors;
896 if (collector->transform_calls > sym->basis_thread_transform_calls_max)
897 sym->basis_thread_transform_calls_max = collector->transform_calls;
898 }
899 global_error = SumMPI_i(local_error);
900 if (global_error != 0) {
901 StopTimer(1110);
902 free_basis_collectors(collectors, collector_count);
903 collectors = NULL;
904 goto fail;
905 }
906 sym->basis_thread_count = (unsigned int)actual_thread_count;
907 sym->basis_orbit_metadata_calls =
908 sym->basis_representative_candidates;
909
910 if (sym->dim > SIZE_MAX / sizeof(*sym->basis) - 1UL) {
911 local_error = 1;
912 } else {
913 sym->basis = (struct SymmetryBasisVector *)calloc(
914 (size_t)sym->dim + 1U, sizeof(*sym->basis));
915 if (sym->basis == NULL) local_error = 1;
916 }
917 global_error = SumMPI_i(local_error);
918 if (global_error != 0) {
919 StopTimer(1110);
920 free_basis_collectors(collectors, collector_count);
921 collectors = NULL;
922 goto fail;
923 }
924 sym->capacity = sym->dim;
925 basis_offset = 1UL;
926 for (thread = 0; thread < actual_thread_count; thread++) {
927 struct SymmetryBasisCollector *collector = &collectors[thread];
928 if (collector->count > 0UL) {
929 memcpy(sym->basis + basis_offset, collector->entries,
930 (size_t)collector->count * sizeof(*sym->basis));
931 basis_offset += collector->count;
932 }
933 }
934 free_basis_collectors(collectors, collector_count);
935 collectors = NULL;
936 StartTimer(1115);
937 if (gather_symmetry_basis(sym) != 0) {
938 StopTimer(1115);
939 StopTimer(1110);
940 goto fail;
941 }
942 global_representative_candidates = SumMPI_li(
943 (unsigned long int)sym->basis_representative_candidates);
944 StopTimer(1115);
945 StopTimer(1110);
946
947 StartTimer(1111);
948 if (sym->dim > 1) {
949 qsort(sym->basis + 1, sym->dim, sizeof(struct SymmetryBasisVector),
950 compare_basis_rep_state);
951 }
952 StopTimer(1111);
953 StartTimer(1112);
954 local_error = build_rep_hash(sym) != 0 ? 1 : 0;
955 global_error = SumMPI_i(local_error);
956 if (global_error != 0) {
957 StopTimer(1112);
958 goto fail;
959 }
960 StopTimer(1112);
961
962 StartTimer(1113);
963 if (sym->dim > SIZE_MAX / sizeof(*sym->sym_diagonal) - 1UL) {
964 StopTimer(1113);
965 goto fail;
966 }
967 sym->sym_diagonal = (double *)calloc((size_t)sym->dim + 1U,
968 sizeof(*sym->sym_diagonal));
969 local_error = sym->sym_diagonal == NULL ? 1 : 0;
970 global_error = SumMPI_i(local_error);
971 if (global_error != 0) {
972 StopTimer(1113);
973 goto fail;
974 }
975 for (raw = 1; raw <= sym->dim; raw++) {
976 sym->sym_diagonal[raw] = sym->basis[raw].diagonal;
977 }
978 StopTimer(1113);
979
980 fprintf(stdoutMPI, "Symmetry basis: raw_dim=%lu sector_dim=%lu group_order=%u\n",
981 sym->full_dim, sym->dim, sym->group_order);
982 fprintf(stdoutMPI,
983 "Symmetry basis build: raw_states=%lu representative_candidates=%lu "
984 "compatible_survivors=%lu\n",
985 sym->full_dim,
986 global_representative_candidates, sym->dim);
987 if (sym->dim == 0) {
988 fprintf(stdoutMPI, "Error: TransSym sector has zero basis dimension.\n");
989 FreeSymmetryBasis(sym);
990 return -1;
991 }
992 X->Sym = sym;
993 return 0;
994
995fail:
996 free_basis_collectors(collectors, collector_count);
997 FreeSymmetryBasis(sym);
998 return -1;
999}

1001int SymmetryCanonicalizeState(const struct BindStruct *X,
1002 unsigned long int state,
1003 struct SymmetryCanonicalResult *result)
1004{
1005 unsigned int g;
1006 unsigned int op_rep_to_state;
1007 unsigned long int rep_state;
1008 unsigned long int basis_index;
1009 if (result == NULL) return -1;
1010 memset(result, 0, sizeof(*result));
1011 if (X == NULL || X->Sym == NULL || X->Sym->enabled != TRUE) return -1;
1012
1013 if (find_representative_state(&X->Def, X->Sym, state, &rep_state,
1014 &op_rep_to_state, NULL) != 0) {
1015 return -1;
1016 }
1017 basis_index = find_basis_index_by_rep(X->Sym, rep_state);
1018 if (basis_index == 0) return 0;
1019
1020 if (op_rep_to_state != UINT_MAX &&
1021 op_rep_to_state < X->Def.NSymTrans) {
1022 struct SymmetryTransformResult moved;
1023 if (SymmetryApplyToState(&X->Def, rep_state, op_rep_to_state,
1024 &moved) != 0 || moved.state != state) {
1025 return -1;
1026 }
1027 result->found = TRUE;
1028 result->basis_index = basis_index;
1029 result->op_rep_to_state = op_rep_to_state;
1030 result->phase = X->Def.SymTransChar[op_rep_to_state] * moved.amplitude;
1031#ifdef HPHI_SYMMETRY_CANONICAL_VERIFY
1032 for (g = 0U; g < X->Def.NSymTrans; g++) {
1033 struct SymmetryTransformResult reference;
1034 if (SymmetryApplyToState(&X->Def, rep_state, g, &reference) != 0) {
1035 return -1;
1036 }
1037 if (reference.state == state) {
1038 double complex reference_phase =
1039 X->Def.SymTransChar[g] * reference.amplitude;
1040 if (g != result->op_rep_to_state ||
1041 reference_phase != result->phase) {
1042 return -1;
1043 }
1044 break;
1045 }
1046 }
1047 if (g == X->Def.NSymTrans) return -1;
1048#endif
1049 return 0;
1050 }
1051
1052 for (g = 0; g < X->Def.NSymTrans; g++) {
1053 struct SymmetryTransformResult moved;
1054 if (SymmetryApplyToState(&X->Def, rep_state, g, &moved) != 0) return -1;
1055 if (moved.state == state) {
1056 result->found = TRUE;
1057 result->basis_index = basis_index;
1058 result->op_rep_to_state = g;
1059 result->phase = X->Def.SymTransChar[g] * moved.amplitude;
1060 return 0;
1061 }
1062 }
1063 return 0;
1064}
