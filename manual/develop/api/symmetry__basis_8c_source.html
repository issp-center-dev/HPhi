102 return -1;
103 }
104 if (fabs(norm - 1.0) > eps_ch) {
105 fprintf(stdoutMPI, "Error: TransSym character must have unit norm; op=%u abs=% .16e.\n",
106 g, norm);
107 return -1;
108 }
109 for (prev = 0; prev < g; prev++) {
110 if (same_perm(def->SymTrans[g], def->SymTrans[prev], def->Nsite) == TRUE) {
111 fprintf(stdoutMPI, "Error: duplicate TransSym operation: op %u and op %u are identical.\n",
112 prev, g);
113 return -1;
114 }
115 }
116 if (is_identity_perm(def->SymTrans[g], def->Nsite) == TRUE) {
117 if (identity >= 0) {
118 fprintf(stdoutMPI, "Error: TransSym contains duplicate identity operations.\n");
119 return -1;
120 }
121 identity = (int)g;
122 }
123 }
124 if (identity < 0) {
125 fprintf(stdoutMPI, "Error: TransSym group must contain identity permutation.\n");
126 return -1;
127 }
128 if (cabs(def->SymTransChar[identity] - 1.0) > eps_ch) {
129 fprintf(stdoutMPI, "Error: TransSym identity character must be 1.\n");
130 return -1;
131 }
132
133 composed = (int *)malloc(sizeof(int) * def->Nsite);
134 if (composed == NULL) return -1;
135 for (g = 0; g < def->NSymTrans; g++) {
136 for (h = 0; h < def->NSymTrans; h++) {
137 int gh;
138 for (site = 0; site < def->Nsite; site++) {
139 composed[site] = def->SymTrans[g][def->SymTrans[h][site]];
140 }
141 gh = find_perm(def, composed);
142 if (gh < 0) {
143 free(composed);

145 g, h);
146 return -1;
147 }
148 if (cabs(def->SymTransChar[gh] - def->SymTransChar[g] * def->SymTransChar[h]) > eps_ch) {
149 free(composed);
150 fprintf(stdoutMPI, "Error: TransSym character is not multiplicative for op %u * op %u.\n",
151 g, h);
152 return -1;
153 }
154 }
155 }
156 free(composed);
157 return 0;

160static int ensure_basis_capacity(struct SymmetryBasisRuntime *sym,
161 unsigned long int needed)
162{
163 struct SymmetryBasisVector *next;
164 unsigned long int next_capacity;
165 if (needed <= sym->capacity) return 0;
166 next_capacity = (sym->capacity == 0UL) ? 16UL : sym->capacity;

169 next_capacity *= 2UL;
