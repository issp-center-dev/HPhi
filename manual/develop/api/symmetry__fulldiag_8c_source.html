48static int validate_context(const struct BindStruct *X,
49 const struct SymmetryHamiltonianInfo *info)
50{
51 const struct SymmetryBasisRuntime *sym;
52 unsigned long offset, count;
53 if (X == NULL || info == NULL || !X->Def.iFlgSymmetryBasis ||
54 X->Def.iCalcType != FullDiag) return -1;
55 sym = X->Sym;
56 if (sym == NULL || sym->enabled != TRUE ||
57 sym->basis_layout != SYMMETRY_BASIS_REPLICATED || sym->dim == 0 ||
58 sym->dim > INT_MAX || X->Check.idim_max != sym->dim) return -1;
59 if (SymmetryBlockRange(sym->dim, myrank, nproc, &offset, &count) != 0 ||
60 offset != sym->local_offset || count != sym->local_dim) return -1;
61 if (sym->local_dim >= SIZE_MAX / sizeof(double complex) - 1) return -1;
62 return info->identity.sector_dim == sym->dim ? 0 : -1;
63}

65/* LAPACK keeps L_vec[state][component] (both 0-based) on every rank; the
66 * distributed solvers gather the state on rank 0 and scatter the blocks. */
67static int copy_state(const struct BindStruct *X, unsigned long state,
68 struct StateBuffers *buffers)
69{
70 const struct SymmetryBasisRuntime *sym = X->Sym;
71 if (sym->dim == 1) {
72 /* The normalized eigenvector of a 1x1 Hermitian matrix is 1 up to a
73 * phase; LAPACK zheev returns exactly 1. ScaLAPACK pzheev has no N=1
74 * quick return and was observed to return 1+1i, so the backend vector
75 * is not used for a one-state sector. */
76 if (sym->local_dim == 1) buffers->owned[1] = 1.0;
77 return 0;
78 }
79#ifdef _SCALAPACK
80 if (use_scalapack) {
81 if (myrank == 0)
82 for (unsigned long i = 0; i < sym->dim; ++i) buffers->root[i] = NAN;
83 /* The existing API always returns 0. A missed transfer leaves NaN on
84 * rank 0, which the finite/norm check of the scattered state rejects. */
85 (void)GetEigenVectorBlock((long)state, (long)sym->dim, Z_vec, descZ_vec,
86 buffers->root);
87 return MPI_Scatterv(buffers->root, buffers->counts, buffers->displacements,
88 MPI_DOUBLE_COMPLEX, buffers->owned + 1, (int)sym->local_dim,
89 MPI_DOUBLE_COMPLEX, 0, MPI_COMM_WORLD) == MPI_SUCCESS ? 0 : -1;
90 }
91#endif
92 if (L_vec == NULL || L_vec[state] == NULL) return -1;
93 memcpy(buffers->owned + 1, L_vec[state] + sym->local_offset,
94 sym->local_dim * sizeof(*buffers->owned));
95 return 0;
96}

98/* The checkpoint contract: finite components and |norm^2 - 1| <= 1e-8.
99 * The solver result is not renormalized. */
100static int check_state(const struct BindStruct *X, unsigned long state,
101 const double complex *owned)
102{
103 double norm = 0.0;
104 int nonfinite = 0;
105 for (unsigned long i = 1; i <= X->Sym->local_dim; ++i) {
106 double re = creal(owned[i]), im = cimag(owned[i]);
107 if (!isfinite(re) || !isfinite(im)) { nonfinite = 1; break; }
108 norm += re*re + im*im;
109 }
110 if (SumMPI_i(nonfinite) != 0) {
111 fprintf(stdoutMPI, "Error: sector FullDiag eigenstate %lu has a non-finite component.\n", state);
112 return -1;
113 }
114 norm = SumMPI_d(norm);
115 if (!isfinite(norm) || fabs(norm - 1.0) > 1e-8) {
116 fprintf(stdoutMPI, "Error: sector FullDiag eigenstate %lu is not normalized (norm^2=%.17g).\n",
117 state, norm);
118 return -1;
119 }
120 return 0;
121}

123static int allocate_buffers(const struct BindStruct *X, struct StateBuffers *buffers)
124{
125 int invalid;
126 buffers->owned = calloc(X->Sym->local_dim + 1, sizeof(*buffers->owned));
127#ifdef HPHI_TEST_SYMMETRY_PROBE
128 if (SymmetryFullDiagTestFail("vector_alloc")) { free(buffers->owned); buffers->owned = NULL; }
129#endif
130 invalid = buffers->owned == NULL;
131#ifdef _SCALAPACK
132 if (use_scalapack) {
133 buffers->root = calloc(myrank == 0 ? X->Sym->dim : 1, sizeof(*buffers->root));
134 buffers->counts = malloc((size_t)nproc * sizeof(*buffers->counts));
135 buffers->displacements = malloc((size_t)nproc * sizeof(*buffers->displacements));
136 invalid |= buffers->root == NULL || buffers->counts == NULL ||
137 buffers->displacements == NULL;
138 for (int rank = 0; !invalid && rank < nproc; ++rank) {
139 unsigned long offset, count;
140 if (SymmetryBlockRange(X->Sym->dim, rank, nproc, &offset, &count) != 0) {
141 invalid = 1;
142 break;
143 }
144 buffers->counts[rank] = (int)count;
145 buffers->displacements[rank] = (int)offset;
146 }
147 }
148#endif
149 return SumMPI_i(invalid) != 0 ? -1 : 0;
150}

180int ProcessSymmetryFullDiagStates(const struct BindStruct *X,
181 const struct SymmetryHamiltonianInfo *info)
182{
183 struct SymmetryHamiltonianIdentity current;
184 struct StateBuffers buffers = { NULL, NULL, NULL, NULL };
185 struct BindStruct local;
186 unsigned int outputs;
187 int status = -1;
188 if (SumMPI_i(validate_context(X, info) != 0) != 0) {
189 fprintf(stdoutMPI, "Error: sector FullDiag state output requires the replicated FullDiag context of the prepared matrix.\n");
190 return -1;
191 }
192 if (BuildSymmetryHamiltonianIdentity(X, &current) != 0) return -1;
193 if (SumMPI_i(!same_identity(&current, &info->identity)) != 0) {
194 fprintf(stdoutMPI, "Error: sector FullDiag basis identity changed after matrix preparation.\n");
195 return -1;
196 }
197 outputs = SymmetryFullDiagOutputMask(&X->Def);
198 local = *X;
199 local.Check.idim_max = X->Sym->local_dim;
200 local.Check.idim_maxMPI = X->Sym->dim;
201 StartTimer(5700);
202 if (allocate_buffers(X, &buffers) != 0) {
203 fprintf(stdoutMPI, "Error: sector FullDiag state buffer allocation failed.\n");
204 goto cleanup;
205 }
206 /* Aggregate Green files are truncated once; states then append in order. */
207 if ((outputs & SYM_FD_CORRELATIONS) && GreenOutputInitializeAggregateFiles(&local) != 0)
208 goto cleanup;
209 for (unsigned long state = 0; state < X->Sym->dim; ++state) {
210 int failed;
211 local.Phys.eigen_num = (int)state;
212 StartTimer(5710);
213 failed = copy_state(X, state, &buffers) != 0;
214 StopTimer(5710);
215 if (SumMPI_i(failed) != 0) {
216 fprintf(stdoutMPI, "Error: sector FullDiag eigenstate %lu could not be distributed.\n", state);
217 goto cleanup;
218 }
219 if (check_state(X, state, buffers.owned) != 0) goto cleanup;
220 if (outputs & SYM_FD_CORRELATIONS) {
221 /* Each wrapper returns the same result on every rank, so the
222 * short-circuit keeps the collective sequence aligned. */
223 StartTimer(5720);
224 failed = expec_cisajs(&local, buffers.owned) != 0 ||
225 expec_cisajscktaltdc(&local, buffers.owned) != 0 ||
226 expec_nbodyg(&local, buffers.owned) != 0;
227 StopTimer(5720);
228 if (failed) {
229 fprintf(stdoutMPI, "Error: correlations of sector FullDiag eigenstate %lu failed.\n", state);
230 goto cleanup;
231 }
232 }
233 /* A state is saved only after its correlations succeeded. */
234 if (outputs & SYM_FD_EIGENVECTORS) {
235 StartTimer(5730);
236 failed = write_state(&local, state, buffers.owned, info) != 0;
237 StopTimer(5730);
238 if (failed) {
239 fprintf(stdoutMPI, "Error: sector FullDiag eigenstate %lu could not be saved as %s_eigenvec_%lu_rank_*.dat.\n",
240 state, X->Def.CDataFileHead, state);
241 goto cleanup;
242 }
243 }
244 }
245 status = 0;
246cleanup:
247 StopTimer(5700);
248 free_buffers(&buffers);
249 return status;
250}
