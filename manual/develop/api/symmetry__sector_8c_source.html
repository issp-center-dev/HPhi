18static uint64_t hash_integer(uint64_t hash, uint64_t value, unsigned int bytes)
19{
20 unsigned int b;
21 for (b = 0; b < bytes; ++b) {
22 hash = (hash ^ (value & UINT64_C(255))) * FNV_PRIME;
23 value >>= 8;
24 }
25 return hash;
26}

28static uint64_t hash_tag(uint64_t hash, const char *tag)
29{
30 do {
31 hash = (hash ^ (unsigned char)*tag) * FNV_PRIME;
32 } while (*tag++ != '\0');
33 return hash;
34}

36static int hash_double(uint64_t *hash, double value)
37{
38 uint64_t bits;
39 if (sizeof(value) != sizeof(bits) || DBL_MANT_DIG != 53 ||
40 DBL_MAX_EXP != 1024 || !isfinite(value)) return -1;
41 memcpy(&bits, &value, sizeof(bits));
42 *hash = hash_integer(*hash, bits, 8);
43 return 0;
44}

53static int compare_group_entries(const void *a, const void *b)
54{
55 const struct GroupDigestEntry *left = a, *right = b;
56 unsigned int site;
57 for (site = 0; site < left->nsite; ++site) {
58 if (left->perm[site] < right->perm[site]) return -1;
59 if (left->perm[site] > right->perm[site]) return 1;
60 }
61 return 0;

64int ComputeSymmetryGroupDigest(const struct DefineList *def, uint64_t *digest)
65{
66 struct GroupDigestEntry *entries;
67 unsigned int g, site;
68 uint64_t hash = hash_tag(FNV_OFFSET, "hphi-group-fnv1a64-v1");
69 if (digest == NULL) return -1;
70 *digest = 0;
71 if (def == NULL || def->Nsite == 0 || def->NSymTrans == 0 ||
72 def->SymTrans == NULL || def->SymTransAnti == NULL ||
73 def->SymTransChar == NULL) return -1;
74 entries = calloc(def->NSymTrans, sizeof(*entries));
75 if (entries == NULL) return -1;
76 for (g = 0; g < def->NSymTrans; ++g) {
77 double complex ch = def->SymTransChar[g];
78 if (def->SymTrans[g] == NULL || def->SymTransAnti[g] == NULL ||
79 !isfinite(creal(ch)) || !isfinite(cimag(ch)) ||
80 fabs(creal(ch)) > 2.0 || fabs(cimag(ch)) > 2.0) {
81 free(entries);
82 return -1;
83 }
84 entries[g].perm = def->SymTrans[g];
85 entries[g].anti = def->SymTransAnti[g];
86 entries[g].character = ch;
87 entries[g].nsite = def->Nsite;
88 }
89 qsort(entries, def->NSymTrans, sizeof(*entries), compare_group_entries);
90 hash = hash_integer(hash, def->Nsite, 4);
91 hash = hash_integer(hash, def->NSymTrans, 4);
92 for (g = 0; g < def->NSymTrans; ++g) {
93 for (site = 0; site < def->Nsite; ++site) {
94 hash = hash_integer(hash, (uint32_t)entries[g].perm[site], 4);
95 hash = hash_integer(hash, (uint32_t)entries[g].anti[site], 4);
96 }
97 hash = hash_integer(hash,
98 (uint64_t)(int64_t)llround(1e10 * creal(entries[g].character)), 8);
99 hash = hash_integer(hash,
100 (uint64_t)(int64_t)llround(1e10 * cimag(entries[g].character)), 8);
101 }
102 free(entries);
103 *digest = hash;
104 return 0;
105}

107static int hash_terms(uint64_t *hash, const char *tag, unsigned int count,
108 unsigned int width, int *const *indices,
109 const double *real_values,
110 const double complex *complex_values)
111{
112 unsigned int term, col;
113 *hash = hash_tag(*hash, tag);
114 *hash = hash_integer(*hash, count, 4);
115 if (count > 0 && (indices == NULL ||
116 (real_values == NULL && complex_values == NULL))) return -1;
117 for (term = 0; term < count; ++term) {
118 if (indices[term] == NULL) return -1;
119 for (col = 0; col < width; ++col)
120 *hash = hash_integer(*hash, (uint32_t)indices[term][col], 4);
121 if (hash_double(hash, real_values != NULL ? real_values[term] :
122 creal(complex_values[term])) != 0) return -1;
123 if (complex_values != NULL &&
124 hash_double(hash, cimag(complex_values[term])) != 0) return -1;
125 }
126 return 0;
127}

129int ComputeSymmetryHamiltonianDigest(const struct DefineList *def, uint64_t *digest)
130{
131 uint64_t hash = hash_tag(FNV_OFFSET, "hphi-parsed-hamiltonian-fnv1a64-v2");
132 if (digest == NULL) return -1;
133 *digest = 0;
134 if (def == NULL) return -1;
135 /* Extend this scope when enabling additional term families. Never silently
136 * omit an unsupported family from an otherwise successful fingerprint. */
137 if (def->NNBodyInterAll || def->NAnomalousTerm || def->NPairLiftCoupling)
138 return -1;
139 hash = hash_integer(hash, (uint32_t)def->iCalcModel, 4);
140 hash = hash_integer(hash, def->Nsite, 4);
141 hash = hash_integer(hash, def->NIsingCoupling, 4);
142 if (hash_terms(&hash, "transfer", def->EDNTransfer, 4,
143 def->EDGeneralTransfer, NULL, def->EDParaGeneralTransfer) ||
144 hash_terms(&hash, "coulomb_intra", def->NCoulombIntra, 1,
145 def->CoulombIntra, def->ParaCoulombIntra, NULL) ||
146 hash_terms(&hash, "coulomb_inter", def->NCoulombInter, 2,
147 def->CoulombInter, def->ParaCoulombInter, NULL) ||
148 hash_terms(&hash, "hund", def->NHundCoupling, 2,
149 def->HundCoupling, def->ParaHundCoupling, NULL) ||
150 hash_terms(&hash, "exchange", def->NExchangeCoupling, 2,
151 def->ExchangeCoupling, def->ParaExchangeCoupling, NULL))
152 return -1;
153 if (hash_terms(&hash, "pair_hopping", def->NPairHopping, 2,
154 def->PairHopping, def->ParaPairHopping, NULL) ||
155 hash_terms(&hash, "interall_diagonal", def->NInterAll_Diagonal, 4,
156 def->InterAll_Diagonal, def->ParaInterAll_Diagonal, NULL) ||
157 hash_terms(&hash, "interall_offdiagonal", def->NInterAll_OffDiagonal, 8,
158 def->InterAll_OffDiagonal, NULL, def->ParaInterAll_OffDiagonal)) return -1;
159 hash = hash_tag(hash, "chemical_potential");
160 hash = hash_integer(hash, def->EDNChemi, 4);
161 if (def->EDNChemi && (!def->EDChemi || !def->EDSpinChemi || !def->EDParaChemi)) return -1;
162 for (unsigned int i = 0; i < def->EDNChemi; ++i) {
163 hash = hash_integer(hash, (uint32_t)def->EDChemi[i], 4);
164 hash = hash_integer(hash, (uint32_t)def->EDSpinChemi[i], 4);
165 if (hash_double(&hash, def->EDParaChemi[i])) return -1;
166 }
167 *digest = hash;
168 return 0;
169}

171int ComputeSymmetrySectorDigest(const struct SymmetryBasisRuntime *sym,
172 struct SymmetrySectorDigest *digest)
173{
174 unsigned long int i, count = 0;
175 struct SymmetrySectorDigest local = {0, 0, 0};
176 int error = digest == NULL || sym == NULL || sym->enabled != TRUE;
177 if (digest != NULL) memset(digest, 0, sizeof(*digest));
178 if (!error) {
179 if (sym->basis_layout == SYMMETRY_BASIS_REPLICATED) count = sym->dim;
180 else if (sym->basis_layout == SYMMETRY_BASIS_DISTRIBUTED) count = sym->local_dim;
181 else error = 1;
182 }
183 for (i = 1; !error && i <= count; ++i) {
184 const struct SymmetryBasisVector *entry =
185 sym->basis_layout == SYMMETRY_BASIS_REPLICATED ?
186 SymmetryBasisReplicatedGlobalEntry(sym, i) : SymmetryBasisLocalEntry(sym, i);
187 uint64_t hash = hash_tag(FNV_OFFSET, "hphi-sector-multiset-v1");
188 if (entry == NULL) { error = 1; break; }
189 hash = hash_integer(hash, (uint64_t)entry->rep_state, 8);
190 hash = hash_integer(hash, entry->orbit_size, 4);
191 hash = hash_integer(hash, entry->stabilizer_size, 4);
192 local.count++;
193 local.xor_hash ^= hash;
194 local.sum_hash += hash;
195 }
196 if (SumMPI_i(error) != 0) return -1;
197#ifdef MPI
198 if (sym->basis_layout == SYMMETRY_BASIS_DISTRIBUTED) {
199 uint64_t sums[2] = {local.count, local.sum_hash}, global_sums[2], global_xor;
200 if (MPI_Allreduce(sums, global_sums, 2, MPI_UINT64_T, MPI_SUM,
201 MPI_COMM_WORLD) != MPI_SUCCESS ||
202 MPI_Allreduce(&local.xor_hash, &global_xor, 1, MPI_UINT64_T, MPI_BXOR,
203 MPI_COMM_WORLD) != MPI_SUCCESS) return -1;
204 local.count = global_sums[0];
205 local.sum_hash = global_sums[1];
206 local.xor_hash = global_xor;
207 }
208#endif
209 if (SumMPI_i(local.count != (uint64_t)sym->dim) != 0) return -1;
210 *digest = local;
211 return 0;
212}
