19static int expand_local(const struct Expansion *expansion, unsigned int site,
20 struct SymmetryKondoMonomial monomial)
21{
22 unsigned int choice;
23 if (site == expansion->nlocal) {
24 if (monomial.value == 0.0) return 0;
25 return expansion->callback(&monomial, expansion->context) ? -1 : 0;
26 }
27 for (choice = 0; choice < 4; ++choice) {
28 struct SymmetryKondoMonomial next = monomial;
29 double coefficient = expansion->local[site].coefficient[choice];
30 if (coefficient == 0.0) continue;
31 next.value *= coefficient;
32 if (choice) {
33 unsigned int n = next.nlocal++;
34 next.local_site[n] = expansion->local[site].site;
35 next.local_out[n] = choice == 1 ? 0 : 1;
36 next.local_in[n] = choice == 2 ? 0 : 1;
37 }
38 if (expand_local(expansion, site + 1, next)) return -1;
39 }
40 return 0;
41}

43static int normal_order(const struct Expansion *expansion,
44 const struct Fermion *ops, unsigned int count,
45 double complex value)
46{
47 struct SymmetryKondoMonomial monomial = {0};
48 unsigned int i, j;
49 /* a_p c^dagger_q = delta_pq - c^dagger_q a_p. Recursion strictly
50 * decreases either the string length or its number of inversions. */
51 for (i = 0; i + 1 < count; ++i) {
52 if (!ops[i].create && ops[i+1].create) {
53 struct Fermion next[4];
54 memcpy(next, ops, count * sizeof(*ops));
55 if (ops[i].orbital == ops[i+1].orbital) {
56 for (j = i; j + 2 < count; ++j) next[j] = ops[j+2];
57 if (normal_order(expansion, next, count - 2, value)) return -1;
58 }
59 memcpy(next, ops, count * sizeof(*ops));
60 next[i] = ops[i+1]; next[i+1] = ops[i];
61 return normal_order(expansion, next, count, -value);
62 }
63 }
64 for (i = 0; i < count; ++i) {
65 unsigned int *n = ops[i].create ? &monomial.ncreate : &monomial.nannihilate;
66 int *orbitals = ops[i].create ? monomial.create : monomial.annihilate;
67 if (*n >= 2) return -1;
68 for (j = 0; j < *n; ++j) if (orbitals[j] == ops[i].orbital) return 0;
69 j = (*n)++;
70 while (j && orbitals[j-1] > ops[i].orbital) {
71 orbitals[j] = orbitals[j-1]; --j; value = -value;
72 }
73 orbitals[j] = ops[i].orbital;
74 }
75 monomial.value = value;
76 return expand_local(expansion, 0, monomial);
77}

79/* Evaluate a local product in the four-state Fock space, retaining only its
80 * final singly occupied block. In particular, intermediate empty/double
81 * states are essential for the crossed Exchange encoding. */
82static void local_block(const struct Fermion *ops, unsigned int count,
83 double coefficient[4])
84{
85 double block[2][2] = {{0}};
86 unsigned int input;
87 for (input = 0; input < 2; ++input) {
88 unsigned int digit = 1U << input, i = count;
89 double sign = 1.0;
90 while (i-- > 0) {
91 unsigned int spin = (unsigned int)ops[i].orbital % 2;
92 unsigned int mask = 1U << spin;
93 if (((digit & mask) != 0) == ops[i].create) { sign = 0; break; }
94 if (spin && (digit & 1U)) sign = -sign;
95 digit ^= mask;
96 }
97 if (digit == 1U || digit == 2U) block[digit == 2U][input] = sign;
98 }
99 coefficient[0] = block[0][0];
100 coefficient[1] = block[0][1];
101 coefficient[2] = block[1][0];
102 coefficient[3] = block[1][1] - block[0][0];
103}

105int CanonicalizeSymmetryKondoTerm(const struct DefineList *def,
106 const struct SymmetryTerm *term, const int *permutation,
107 SymmetryKondoMonomialCallback callback, void *context)
108{
109 struct Fermion ops[4];
110 struct Expansion expansion = {0};
111 unsigned int count, i, begin;
112 double complex value;
113 int permutation_sign;
114 if (!def || !term || !callback || term->factors < 1 || term->factors > 2 ||
115 !isfinite(creal(term->value)) || !isfinite(cimag(term->value)) ||
116 ValidateSymmetryKondoSpace(def)) return -1;
117 if (permutation && SymmetryKondoPermutationSign(def, permutation, &permutation_sign)) return -1;
118 count = 2 * term->factors;
119 value = term->value;
120 expansion.callback = callback; expansion.context = context;
121 for (i = 0; i < count; ++i) {
122 int site = term->index[2*i], spin = term->index[2*i+1];
123 if (site < 0 || (unsigned int)site >= def->Nsite || spin < 0 || spin > 1) return -1;
124 if (permutation) site = permutation[site];
125 ops[i].orbital = 2 * site + spin;
126 ops[i].create = i % 2 == 0;
127 }
128 /* Stable insertion sort: local sites first in ascending order; all
129 * conduction operators share a key and keep their original order. */
130 for (i = 1; i < count; ++i) {
131 struct Fermion op = ops[i];
132 unsigned int j = i;
133 int site = op.orbital / 2;
134 unsigned int key = def->LocSpn[site] == LOCSPIN ? (unsigned int)site : def->Nsite;
135 while (j) {
136 int previous = ops[j-1].orbital / 2;
137 unsigned int previous_key = def->LocSpn[previous] == LOCSPIN ? (unsigned int)previous : def->Nsite;
138 if (previous_key <= key) break;
139 ops[j] = ops[j-1]; --j; value = -value;
140 }
141 ops[j] = op;
142 }
143 begin = 0;
144 while (begin < count && def->LocSpn[ops[begin].orbital / 2] == LOCSPIN) {
145 unsigned int end = begin + 1;
146 struct LocalBlock *block;
147 while (end < count && ops[end].orbital / 2 == ops[begin].orbital / 2) ++end;
148 /* Odd local products have a zero singly occupied block. This also
149 * bounds the number of nonzero local blocks to two. */
150 if ((end - begin) % 2) return 0;
151 block = &expansion.local[expansion.nlocal++];
152 block->site = ops[begin].orbital / 2;
153 local_block(ops + begin, end - begin, block->coefficient);
154 begin = end;
155 }
156 return normal_order(&expansion, ops + begin, count - begin, value);
157}

165static int append(const struct SymmetryKondoMonomial *term, void *context)
166{
167 struct Polynomial *p = context;
168 if (p->count == p->capacity) {
169 size_t capacity = p->capacity ? 2 * p->capacity : 32;
170 struct SymmetryKondoMonomial *next;
171 if (capacity < p->capacity || capacity > SIZE_MAX / sizeof(*next)) return -1;
172 next = realloc(p->terms, capacity * sizeof(*next));
173 if (!next) return -1;
174 p->terms = next; p->capacity = capacity;
175 }
176 p->terms[p->count++] = *term;
177 return 0;

200static int collect(struct Polynomial *p)
201{
202 size_t i, count = 0;
203 p->count = 0;
204 if (EnumerateSymmetryTerms(p->def, -1, canonical_term, p)) return -1;
205 if (p->count) qsort(p->terms, p->count, sizeof(*p->terms), compare);
206 for (i = 0; i < p->count; ++i) {
207 if (count && compare(&p->terms[count-1], &p->terms[i]) == 0)
208 p->terms[count-1].value += p->terms[i].value;
209 else p->terms[count++] = p->terms[i];
210 }
211 p->count = 0;
212 for (i = 0; i < count; ++i) {
213 double complex value = p->terms[i].value;
214 if (!isfinite(creal(value)) || !isfinite(cimag(value))) return -1;
215 if (cabs(value) > 1e-10) p->terms[p->count++] = p->terms[i];
216 }

219int ValidateSymmetryKondoTerms(const struct DefineList *def)
220{
221 struct Polynomial base = {0}, mapped = {0};
222 size_t i;
223 unsigned int g, f;
224 int error = 1;
225 base.def = mapped.def = def;
226 if (ValidateSymmetryKondoSpace(def) || (def->NSymTrans && !def->SymTrans) ||
227 collect(&base)) goto done;
228 for (i = 0; i < base.count; ++i) {
229 const struct SymmetryKondoMonomial *m = &base.terms[i];
230 int delta_sz = 0;
231 for (f = 0; f < m->nlocal; ++f) delta_sz += 2 * (m->local_in[f] - m->local_out[f]);
232 for (f = 0; f < m->ncreate; ++f) delta_sz += 1 - 2 * (m->create[f] % 2);
233 for (f = 0; f < m->nannihilate; ++f) delta_sz -= 1 - 2 * (m->annihilate[f] % 2);
234 if ((def->iCalcModel == Kondo && delta_sz) ||
235 (def->iCalcModel == KondoNConserved && m->ncreate != m->nannihilate)) {
236 fprintf(stdoutMPI, "Error: TransSym Kondo Hamiltonian does not conserve fixed quantum numbers.\n");
237 goto done;
238 }
239 }
240 for (g = 0; g < def->NSymTrans; ++g) {
241 int sign;
242 mapped.permutation = def->SymTrans[g];
243 if (SymmetryKondoPermutationSign(def, mapped.permutation, &sign) || collect(&mapped)) goto done;
244 if (base.count != mapped.count) goto invariant_error;
245 for (i = 0; i < base.count; ++i)
246 if (compare(&base.terms[i], &mapped.terms[i]) ||
247 cabs(base.terms[i].value - mapped.terms[i].value) > 1e-10)
248 goto invariant_error;
249 }
250 error = 0;
251 goto done;
252invariant_error:
253 fprintf(stdoutMPI, "Error: TransSym Kondo Hamiltonian invariance failed for normalized terms under op %u.\n", g);
254done:
255 free(base.terms); free(mapped.terms);
256 if (error) fprintf(stdoutMPI, "Error: invalid or unsupported TransSym Kondo Hamiltonian terms.\n");
257 return error ? -1 : 0;
