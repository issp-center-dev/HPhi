95/* Same criterion as emit() in symmetry_terms.c: the occupation change at
96 * every out-orbital vanishes. */
97static int term_is_diagonal(unsigned int factors, const int *index)
98{
99 unsigned int i, j;
100 for (i = 0; i < factors; ++i) {
101 int balance = 0;
102 for (j = 0; j < factors; ++j) {
103 if (index[4*i] == index[4*j] && index[4*i+1] == index[4*j+1]) ++balance;
104 if (index[4*i] == index[4*j+2] && index[4*i+1] == index[4*j+3]) --balance;
105 }
106 if (balance != 0) return 0;
107 }
108 return 1;
109}

111static int validate_operator(const struct DefineList *def,
112 const struct SymmetryCorrelationOperator *op)
113{
114 unsigned int f, nint;
115 int spin_max = def->iCalcModel == SpinlessFermion ? 0 : 1;
116 if (op->factors < 1U || op->factors > UINT_MAX / 4U || op->index == NULL) return -1;
117 nint = 4U * op->factors;
118 for (f = 0; f < nint; f += 2U)
119 if (op->index[f] < 0 || (unsigned int)op->index[f] >= def->Nsite ||
120 op->index[f+1] < 0 || op->index[f+1] > spin_max) return -1;
121 if (def->iCalcModel == Spin)
122 for (f = 0; f < op->factors; ++f)
123 if (op->index[4*f] != op->index[4*f+2]) return -1;
124 return 0;
125}

127static void permute_tuple(const struct DefineList *def, unsigned int g,
128 unsigned int factors, const int *in, int *out)
129{
130 unsigned int f;
131 for (f = 0; f < factors; ++f) {
132 out[4*f] = def->SymTrans[g][in[4*f]];
133 out[4*f+1] = in[4*f+1];
134 out[4*f+2] = def->SymTrans[g][in[4*f+2]];
135 out[4*f+3] = in[4*f+3];
136 }
137}

151static int build_orbit_table(const struct DefineList *def,
152 const struct SymmetryCorrelationOperator *ops,
153 size_t count, struct CorrelationOrbitTable *table)
154{
155 size_t t, k;
156 int *sorted = NULL, *tuple = NULL;
157 size_t sorted_capacity = 0;
158 int status = -1;
159 memset(table, 0, sizeof(*table));
160 if (def == NULL || ops == NULL || def->NSymTrans == 0U || def->SymTrans == NULL) return -1;
161 table->orbits = (struct CorrelationOrbit *)calloc(count, sizeof(*table->orbits));
162 table->op_to_orbit = (size_t *)calloc(count, sizeof(*table->op_to_orbit));
163 if (table->orbits == NULL || table->op_to_orbit == NULL) goto done;
164 table->bytes = (uint64_t)count * (sizeof(*table->orbits) + sizeof(*table->op_to_orbit));
165 for (t = 0; t < count; t++) {
166 unsigned int factors = ops[t].factors;
167 size_t len, n = 0, pos, need;
168 unsigned int g;
169 uint64_t scratch;
170 if (validate_operator(def, &ops[t]) != 0) goto done;
171 len = 4U * (size_t)factors;
172 if (SymmetryCheckedSizeMul(len, (size_t)def->NSymTrans, &need) != 0) goto done;
173 if (need > sorted_capacity) {
174 free(sorted); free(tuple);
175 sorted = (int *)malloc(need * sizeof(int));
176 tuple = (int *)malloc(len * sizeof(int));
177 if (sorted == NULL || tuple == NULL) goto done;
178 sorted_capacity = need;
179 }
180 scratch = ((uint64_t)sorted_capacity + (uint64_t)len) * sizeof(int);
181 if (scratch > table->scratch_peak) table->scratch_peak = scratch;
182 for (g = 0; g < def->NSymTrans; g++) {
183 int c = 1;
184 permute_tuple(def, g, factors, ops[t].index, tuple);
185 for (pos = 0; pos < n; pos++) {
186 c = compare_tuple(tuple, sorted + pos * len, len);
187 if (c <= 0) break;
188 }
189 if (pos < n && c == 0) continue; /* duplicate member */
190 memmove(sorted + (pos + 1) * len, sorted + pos * len, (n - pos) * len * sizeof(int));
191 memcpy(sorted + pos * len, tuple, len * sizeof(int));
192 n++;
193 }
194 for (k = 0; k < table->orbit_count; k++)
195 if (table->orbits[k].factors == factors &&
196 compare_tuple(table->orbits[k].members, sorted, len) == 0) break;
197 if (k == table->orbit_count) {
198 struct CorrelationOrbit *orbit = &table->orbits[k];
199 size_t m;
200 orbit->factors = factors;
201 orbit->member_count = n;
202 orbit->members = (int *)malloc(n * len * sizeof(int));
203 orbit->offdiag = (unsigned char *)calloc(n, sizeof(unsigned char));
204 table->orbit_count++;
205 if (orbit->members == NULL || orbit->offdiag == NULL) goto done;
206 table->bytes += (uint64_t)n * len * sizeof(int) + (uint64_t)n;
207 memcpy(orbit->members, sorted, n * len * sizeof(int));
208 for (m = 0; m < n; m++) {
209 orbit->offdiag[m] = term_is_diagonal(factors, orbit->members + m * len) ? 0U : 1U;
210 orbit->offdiag_count += orbit->offdiag[m];
211 }
212 table->offdiag_total += orbit->offdiag_count;
213 }
214 table->op_to_orbit[t] = k;
215 }
216 status = 0;
217done:
218 free(sorted); free(tuple);
219 if (status != 0) free_orbit_table(table);
220 return status;
221}

223int SymmetryCorrelationOrbitCount(const struct DefineList *def,
224 const struct SymmetryCorrelationOperator *ops,
225 size_t count, size_t *orbit_count, size_t *member_total)
226{
227 struct CorrelationOrbitTable table;
228 size_t k, members = 0;
229 if (orbit_count == NULL || member_total == NULL) return -1;
230 if (count == 0) { *orbit_count = 0; *member_total = 0; return 0; }
231 if (build_orbit_table(def, ops, count, &table) != 0) return -1;
232 for (k = 0; k < table.orbit_count; k++) members += table.orbits[k].member_count;
233 *orbit_count = table.orbit_count;
234 *member_total = members;
235 free_orbit_table(&table);
236 return 0;
237}

301static int find_ghost(const struct SymmetryVectorHaloPlan *halo, unsigned long int global_index, size_t *position)
302{
303 size_t left = 0U, right = halo->ghost_count;
304 while (left < right) {
305 size_t middle = left + (right - left) / 2U;
306 if (halo->ghost_global_index[middle] < global_index) left = middle + 1U; else right = middle;
307 }
308 if (left >= halo->ghost_count || halo->ghost_global_index[left] != global_index) return -1;
309 *position = left;
310 return 0;
311}

315/* Bytes that live simultaneously per block row, from the arrays allocated in
316 * process_wave(): row_counts, and per off-diagonal member one transition, one
317 * key, one resolved index, one norm, one fetched value and one halo column. */
318static int bytes_per_row(uint64_t offdiag_total, uint64_t *out)
319{
320 uint64_t per_member = sizeof(struct CorrelationTransition) + sizeof(unsigned long int) +
321 sizeof(unsigned long int) + sizeof(double) + sizeof(double complex) +
322 sizeof(unsigned long int);
323 uint64_t products;
324 if (checked_mul64(offdiag_total, per_member, &products) != 0) return -1;
325 return checked_add64(products, sizeof(size_t), out);
326}

328static int choose_block_rows(const struct CorrelationContext *ctx, unsigned long int *rows)
329{
330 uint64_t b = (uint64_t)HPHI_SYMMETRY_PLAN_LOCAL_ROWS_PER_BLOCK;
331 uint64_t offdiag_total = ctx->table->offdiag_total;
332 if (offdiag_total > 0U) {
333 uint64_t cap = (uint64_t)HPHI_SYMMETRY_CORRELATION_BLOCK_TRANSITIONS / offdiag_total;
334 if (cap == 0U) return -1; /* one row does not fit the transition cap */
335 if (cap < b) b = cap;
336 }
337 if (ctx->policy.hard_byte_limit != 0U) {
338 uint64_t per_row, fit;
339 if (bytes_per_row(offdiag_total, &per_row) != 0) return -1;
340 if (ctx->fixed_bytes >= ctx->policy.hard_byte_limit) return -1;
341 fit = (ctx->policy.hard_byte_limit - ctx->fixed_bytes) / per_row;
342 if (fit == 0U) return -1; /* one row does not fit the hard limit */
343 if (fit < b) b = fit;
344 }
345 if (b > (uint64_t)ULONG_MAX) b = (uint64_t)ULONG_MAX;
346 *rows = (unsigned long int)b;
347 return 0;
348}

364static int collect_block(const struct CorrelationContext *ctx,
365 unsigned long int row_begin, unsigned long int row_count,
366 struct CorrelationTransition *transitions, size_t *row_counts,
367 size_t *transition_count, double complex *acc)
368{
369 const struct BindStruct *X = ctx->X;
370 const struct DefineList *def = &X->Def;
371 const struct CorrelationOrbitTable *table = ctx->table;
372 const size_t per_row = (size_t)table->offdiag_total;
373 const size_t orbit_count = table->orbit_count;
374 int error = 0;
375 *transition_count = 0;
376#pragma omp parallel
377 {
378 double complex *partial = (double complex *)calloc(orbit_count > 0U ? orbit_count : 1U, sizeof(*partial));
379 int thread_error = partial == NULL;
380 long i;
381#pragma omp for schedule(dynamic, 16)
382 for (i = 0; i < (long)row_count; i++) {
383 unsigned long int local_index = row_begin + (unsigned long int)i + 1UL;
384 const struct SymmetryBasisVector *entry = SymmetryBasisLocalEntry(X->Sym, local_index);
385 double complex a;
386 struct CorrelationTransition *slot = per_row > 0U ? transitions + (size_t)i * per_row : NULL;
387 size_t written = 0, k, m;
388 if (thread_error != 0) continue;
389 if (entry == NULL || entry->norm == 0.0) { thread_error = 1; continue; }
390 a = ctx->vec[local_index];
391 for (k = 0; k < orbit_count && thread_error == 0; k++) {
392 const struct CorrelationOrbit *orbit = &table->orbits[k];
393 size_t len = 4U * (size_t)orbit->factors;
394 for (m = 0; m < orbit->member_count; m++) {
395 unsigned long int out = 0UL;
396 double sign = 0.0;
397 int status = ApplySymmetryFactors(def, orbit->factors, orbit->members + m * len, entry->rep_state, &out, &sign);
398 if (status < 0) { thread_error = 1; break; }
399 if (status == 0) continue;
400 if (out == entry->rep_state) { partial[k] += sign * conj(a) * a; continue; }
401 if (orbit->offdiag[m] == 0U || slot == NULL || written >= per_row) { thread_error = 1; break; }
402 if (ctx->replicated) {
403 struct SymmetryCanonicalResult res;
404 const struct SymmetryBasisVector *target;
405 if (SymmetryCanonicalizeState(X, out, &res) != 0) { thread_error = 1; break; }
406 if (res.found != TRUE) continue;
407 target = SymmetryBasisReplicatedGlobalEntry(X->Sym, res.basis_index);
408 if (target == NULL) { thread_error = 1; break; }
409 slot[written].key = res.basis_index;
410 slot[written].coefficient = a * sign * res.phase * (target->norm / entry->norm);
411 } else {
412 struct SymmetryRepresentativeResult rep;
413 if (SymmetryFindRepresentative(X, out, &rep) != 0) { thread_error = 1; break; }
414 slot[written].key = rep.rep_state;
415 slot[written].coefficient = a * sign * rep.phase / entry->norm;
416 }
417 slot[written].orbit = (uint32_t)k;
418 slot[written].reserved = 0U;
419 written++;
420 }
421 }
422 row_counts[i] = written;
423 }
424#pragma omp critical
425 {
426 size_t k;
427 if (thread_error != 0) error = 1;
428 else for (k = 0; k < orbit_count; k++) acc[k] += partial[k];
429 }
430 free(partial);
431 }
432 if (error == 0) {
433 size_t total = 0;
434 unsigned long int r;
435 for (r = 0; r < row_count; r++) {
436 if (row_counts[r] > 0U && total != (size_t)r * per_row)
437 memmove(transitions + total, transitions + (size_t)r * per_row, row_counts[r] * sizeof(*transitions));
438 total += row_counts[r];
439 }
440 *transition_count = total;
441 }
442 return error ? -1 : 0;
443}

447static int process_wave(struct CorrelationContext *ctx,
448 unsigned long int row_begin, unsigned long int row_count,
449 double complex *acc)
450{
451 const struct BindStruct *X = ctx->X;
452 struct CorrelationTransition *transitions = NULL;
453 size_t *row_counts = NULL;
454 unsigned long int *keys = NULL, *beta = NULL, *columns = NULL;
455 double *norm = NULL;
456 double complex *value = NULL;
457 struct SymmetryVectorHaloPlan halo;
458 struct SymmetryGlobalColumnSpan span;
459 size_t capacity = 0, count = 0, unique = 0, found = 0, u, p, local_columns = 0, remote_columns = 0;
460 uint64_t observed;
461 int local_error = 0, halo_ready = 0;
462 memset(&halo, 0, sizeof(halo));
463
464 row_counts = (size_t *)calloc(row_count > 0UL ? row_count : 1UL, sizeof(*row_counts));
465 if (row_counts == NULL) local_error = 1;
466 else if (SymmetryCheckedSizeMul((size_t)row_count, (size_t)ctx->table->offdiag_total, &capacity) != 0) local_error = 1;
467 else if (capacity > 0U) {
468 transitions = (struct CorrelationTransition *)malloc(capacity * sizeof(*transitions));
469 if (transitions == NULL) local_error = 1;
470 }
471 if (local_error == 0 && row_count > 0UL &&
472 collect_block(ctx, row_begin, row_count, transitions, row_counts, &count, acc) != 0) local_error = 1;
473 if (SymmetryMpiAgreeError(ctx->mpi_active, local_error) != 0) goto fail;
474 ctx->stats.transitions += count;
475
476 /* Diagonal-only requests: no keys, no directory, no halo. The condition is
477 * global (offdiag_total is identical on every rank), so every rank skips. */
478 if (ctx->table->offdiag_total == 0U) {
479 observed = ctx->fixed_bytes + (uint64_t)row_count * sizeof(*row_counts);
480 if (check_memory(ctx, observed, "diagonal-wave") != 0) goto fail;
481 free(row_counts);
482 return 0;
483 }
484
485 keys = (unsigned long int *)malloc((count > 0U ? count : 1U) * sizeof(*keys));
486 if (keys == NULL) local_error = 1;
487 else {
488 for (p = 0; p < count; p++) keys[p] = transitions[p].key;
489 qsort(keys, count, sizeof(*keys), compare_ulong);
490 for (p = 0; p < count; p++) if (p == 0U || keys[p] != keys[unique - 1U]) keys[unique++] = keys[p];
491 }
492 if (local_error == 0) {
493 size_t n = unique > 0U ? unique : 1U;
494 beta = (unsigned long int *)calloc(n, sizeof(*beta));
495 norm = (double *)calloc(n, sizeof(*norm));
496 value = (double complex *)calloc(n, sizeof(*value));
497 columns = (unsigned long int *)calloc(n, sizeof(*columns));
498 if (beta == NULL || norm == NULL || value == NULL || columns == NULL) local_error = 1;
499 }
500 if (SymmetryMpiAgreeError(ctx->mpi_active, local_error) != 0) goto fail;
501 ctx->stats.unique_keys += unique;
502
503 if (ctx->replicated) {
504 for (u = 0; u < unique; u++) { beta[u] = keys[u]; norm[u] = 1.0; }
505 } else if (SymmetryResolveRepresentativeBatchWithOptions(
506 ctx->directory, keys, (uint64_t)unique, beta, norm, NULL) != 0) {
507 local_error = 1; /* collective; called with unique == 0 as well */
508 }
509 if (SymmetryMpiAgreeError(ctx->mpi_active, local_error) != 0) goto fail;
510
511 for (u = 0; u < unique; u++) if (beta[u] != 0UL) columns[found++] = beta[u];
512 span.columns = columns;
513 span.count = found;
514 if (BuildSymmetryVectorHaloPlan(&halo, X->Sym->dim, X->Sym->local_offset, X->Sym->local_dim,
515 &span, 1U, nproc, myrank, &local_columns, &remote_columns) != 0) goto fail;
516 halo_ready = 1;
517 if (ExchangeSymmetryVectorHalo(&halo, ctx->vec) != 0) local_error = 1;
518 if (SymmetryMpiAgreeError(ctx->mpi_active, local_error) != 0) goto fail;
519
520 for (u = 0; u < unique && local_error == 0; u++) {
521 unsigned long int g = beta[u];
522 if (g == 0UL) continue;
523 if (g > X->Sym->local_offset && g <= X->Sym->local_offset + X->Sym->local_dim) {
524 value[u] = ctx->vec[g - X->Sym->local_offset];
525 } else {
526 size_t position;
527 if (find_ghost(&halo, g, &position) != 0) local_error = 1;
528 else value[u] = halo.ghost_values[position];
529 }
530 }
531 for (p = 0; p < count && local_error == 0; p++) {
532 const unsigned long int *hit = (const unsigned long int *)bsearch(
533 &transitions[p].key, keys, unique, sizeof(*keys), compare_ulong);
534 if (hit == NULL) { local_error = 1; break; }
535 u = (size_t)(hit - keys);
536 if (beta[u] == 0UL) continue;
537 acc[transitions[p].orbit] += transitions[p].coefficient * norm[u] * conj(value[u]);
538 }
539 observed = ctx->fixed_bytes +
540 (uint64_t)row_count * sizeof(*row_counts) +
541 (uint64_t)capacity * sizeof(*transitions) +
542 (uint64_t)count * sizeof(*keys) +
543 (uint64_t)unique * (sizeof(*beta) + sizeof(*norm) + sizeof(*value) + sizeof(*columns)) +
544 (uint64_t)halo.topology_scratch_bytes + (uint64_t)halo.schedule_bytes + (uint64_t)halo.runtime_buffer_bytes;
545 if (check_memory(ctx, observed, "block-wave") != 0) local_error = 1; /* collective */
546 if (SymmetryMpiAgreeError(ctx->mpi_active, local_error) != 0) goto fail;
547
548 FreeSymmetryVectorHaloPlan(&halo);
549 free(transitions); free(row_counts); free(keys); free(beta); free(norm); free(value); free(columns);
550 return 0;
551fail:
552 if (halo_ready) FreeSymmetryVectorHaloPlan(&halo);
553 free(transitions); free(row_counts); free(keys); free(beta); free(norm); free(value); free(columns);
554 return -1;
555}

559static int expectation_impl(const struct BindStruct *X, const double complex *vec,
560 const struct SymmetryCorrelationOperator *ops, size_t count,
561 double complex *values, struct SymmetryCorrelationStats *stats_out,
562 int verify)
563{
564 struct CorrelationOrbitTable table;
565 struct CorrelationContext ctx;
566 double complex *acc = NULL;
567 unsigned long long local_waves = 0ULL, waves = 0ULL, wave;
568 uint64_t setup_bytes = 0U, product = 0U;
569 int local_error = 0;
570 size_t t;
571 memset(&table, 0, sizeof(table));
572 memset(&ctx, 0, sizeof(ctx));
573 if (count == 0U) return 0;
574 ctx.mpi_active = SymmetryMpiCollectivesActive();
575 if (X == NULL || X->Sym == NULL || X->Sym->enabled != TRUE || vec == NULL || ops == NULL || values == NULL ||
576 build_orbit_table(&X->Def, ops, count, &table) != 0) local_error = 1;
577 if (SymmetryMpiAgreeError(ctx.mpi_active, local_error) != 0) goto fail;
578 ctx.X = X; ctx.vec = vec; ctx.table = &table;
579 ctx.replicated = X->Sym->basis_layout == SYMMETRY_BASIS_REPLICATED;
580 ctx.threads = team_size();
581 ctx.stats.threads = ctx.threads;
582 if (SymmetryLoadMemoryPolicy((uint64_t)HPHI_SYMMETRY_PLAN_BLOCK_MEMORY_BYTES, ctx.mpi_active, myrank,
583 "correlation", &ctx.policy) != 0) goto fail; /* collective */
584
585 /* Stage 1: orbit table (plus its construction scratch, already freed) under the policy. */
586 if (checked_add64(table.bytes, table.scratch_peak, &setup_bytes) != 0) local_error = 1;
587 if (SymmetryMpiAgreeError(ctx.mpi_active, local_error) != 0) goto fail;
588 if (check_memory(&ctx, setup_bytes, "orbit-table") != 0) goto fail;
589
590 /* Stage 2: temporary directory for distributed layouts with off-diagonal members.
591 * The solver releases X->Sym->representative_directory after its plan is built, so
592 * never rely on it; rebuild from the retained rank-local basis, as RebuildSymmetryTEPlan does. */
593 if (!ctx.replicated && table.offdiag_total > 0U) {
594 struct SymmetryRepresentativeDirectoryInfo info;
595 struct SymmetryLocalRepresentativeIndexStats index_stats;
596 if (X->Sym->local_basis == NULL || X->Sym->rank_offsets == NULL ||
597 BuildSymmetryRepresentativeDirectory(X->Sym->local_basis, X->Sym->dim, X->Sym->local_dim,
598 X->Sym->local_capacity, X->Sym->local_offset,
599 X->Sym->rank_offsets, myrank, nproc, &ctx.directory) != 0 ||
600 GetSymmetryRepresentativeDirectoryInfo(ctx.directory, &info) != 0 ||
601 GetSymmetryRepresentativeDirectoryLocalIndexStats(ctx.directory, &index_stats) != 0) local_error = 1;
602 else ctx.directory_bytes = (uint64_t)info.splitter_bytes + (uint64_t)index_stats.table_bytes;
603 if (SymmetryMpiAgreeError(ctx.mpi_active, local_error) != 0) goto fail;
604 }
605
606 /* Stage 3: accumulators and per-thread partials; fixed bytes for block sizing. */
607 acc = (double complex *)calloc(table.orbit_count, sizeof(*acc));
608 if (acc == NULL) local_error = 1;
609 else if (checked_mul64((uint64_t)table.orbit_count * sizeof(*acc), (uint64_t)ctx.threads + 1U, &product) != 0 ||
610 checked_add64(table.bytes, ctx.directory_bytes, &ctx.fixed_bytes) != 0 ||
611 checked_add64(ctx.fixed_bytes, product, &ctx.fixed_bytes) != 0) local_error = 1;
612 if (SymmetryMpiAgreeError(ctx.mpi_active, local_error) != 0) goto fail;
613 /* Directory tables and OpenMP team sizes may differ by rank. Size every
614 * block against the largest fixed allocation so all ranks choose one safe
615 * row count under a finite hard limit. */
616 if (max_fixed_bytes(ctx.mpi_active, ctx.fixed_bytes, &ctx.fixed_bytes) != 0) goto fail;
617 if (choose_block_rows(&ctx, &ctx.block_rows) != 0) local_error = 1;
618 if (SymmetryMpiAgreeError(ctx.mpi_active, local_error) != 0) goto fail;
619 if (agree_block_rows(ctx.mpi_active, ctx.block_rows) != 0) goto fail;
620 if (check_memory(&ctx, ctx.fixed_bytes, "setup") != 0) goto fail;
621
622 /* Stage 4: bounded waves, the same number on every rank. Diagonal-only
623 * requests walk the same waves without directory or halo traffic. */
624 local_waves = (X->Sym->local_dim + ctx.block_rows - 1UL) / ctx.block_rows;
625 if (max_wave_count(ctx.mpi_active, local_waves, &waves) != 0) goto fail;
626 ctx.stats.waves = waves;
627 for (wave = 0ULL; wave < waves; wave++) {
628 unsigned long int row_begin = 0UL, row_count = 0UL;
629 if (wave < local_waves) {
630 row_begin = (unsigned long int)wave * ctx.block_rows;
631 row_count = X->Sym->local_dim - row_begin;
632 if (row_count > ctx.block_rows) row_count = ctx.block_rows;
633 }
634 if (process_wave(&ctx, row_begin, row_count, acc) != 0) goto fail;
635 }
636 if (allreduce_sum(ctx.mpi_active, acc, table.orbit_count) != 0) goto fail;
637 for (t = 0; t < count; t++) {
638 const struct CorrelationOrbit *orbit = &table.orbits[table.op_to_orbit[t]];
639 values[t] = acc[table.op_to_orbit[t]] / (double)orbit->member_count;
640 }
641 if (verify) {
642 /* Verify the orbit grouping only: each row evaluated alone (without
643 * verification, so this cannot recurse) must give the same value. */
644 for (t = 0; t < count; t++) {
645 double complex single = 0.0;
646 if (expectation_impl(X, vec, &ops[t], 1U, &single, NULL, 0) != 0 ||
647 cabs(single - values[t]) > 1.0e-10 * (1.0 + cabs(single))) {
648 fprintf(stdoutMPI, "Error: symmetry correlation orbit grouping mismatch at row %zu.\n", t);
649 goto fail;
650 }
651 }
652 }
653 if (stats_out != NULL) *stats_out = ctx.stats;
654 fprintf(stdoutMPI,
655 "Symmetry correlation: rows=%zu orbits=%zu waves=%llu transitions=%llu "
656 "peak_bytes=%llu threads=%u layout=%s\n",
657 count, table.orbit_count, ctx.stats.waves, ctx.stats.transitions,
658 ctx.stats.peak_bytes, ctx.stats.threads, ctx.replicated ? "replicated" : "distributed");
659 FreeSymmetryRepresentativeDirectory(ctx.directory);
660 free(acc);
661 free_orbit_table(&table);
662 return 0;
663fail:
664 FreeSymmetryRepresentativeDirectory(ctx.directory);
665 free(acc);
666 free_orbit_table(&table);
667 return -1;
668}
