84int mltply(struct BindStruct *X, double complex *tmp_v0,double complex *tmp_v1) {
85 long unsigned int j=0;
86 long unsigned int irght=0;
87 long unsigned int ilft=0;
88 long unsigned int ihfbit=0;
89
90 double complex dam_pr;
91
92 long unsigned int i_max;
93
94 StartTimer(1);
95 i_max = X->Check.idim_max;
96 X->Large.prdct = 0.0;
97 dam_pr = 0.0;
98
99 if (X->Def.iFlgGeneralSpin == FALSE) {
100 if (GetSplitBitByModel(X->Def.Nsite, X->Def.iCalcModel, &irght, &ilft, &ihfbit) != 0) {
101 return -1;
102 }
103 }
104 else if (i_max != 0 && X->Def.iCalcModel == Spin) {
105 if (GetSplitBitForGeneralSpin(X->Def.Nsite, &ihfbit, X->Def.SiteToBit) != 0) {
106 return -1;
107 }
108 }
109 X->Large.i_max = i_max;
110 X->Large.irght = irght;
111 X->Large.ilft = ilft;
112 X->Large.ihfbit = ihfbit;
113 X->Large.mode = M_MLTPLY;
114
115 StartTimer(100);
116#pragma omp parallel for default(none) reduction(+:dam_pr) firstprivate(i_max) shared(tmp_v0, tmp_v1, list_Diagonal)
117 for (j = 1; j <= i_max; j++) {
118 tmp_v0[j] += (list_Diagonal[j]) * tmp_v1[j];
119 dam_pr += (list_Diagonal[j]) * conj(tmp_v1[j]) * tmp_v1[j];
120 }
121 X->Large.prdct += dam_pr;
122 StopTimer(100);
123 if (X->Def.iCalcType == TimeEvolution) diagonalcalcForTE(step_i, X, tmp_v0, tmp_v1);
124
125 switch (X->Def.iCalcModel) {
126 case HubbardGC:
127 mltplyHubbardGC(X, tmp_v0, tmp_v1);
128 break;
129
130 case Hubbard:
131 case tJ:
132 case tJGC:
133 case Kondo:
134 case KondoGC:
135 mltplyHubbard(X, tmp_v0, tmp_v1);
136 break;
137
138 case Spin:
139 mltplySpin(X, tmp_v0, tmp_v1);
140 break;
141
142 case SpinGC:
143 mltplySpinGC(X, tmp_v0, tmp_v1);
144 break;
145
146 case SpinlessFermion:
147 case SpinlessFermionGC:
148 mltplySpinlessFermion(X, tmp_v0, tmp_v1);
149 break;
150
151 default:
152 return -1;
153 }
154
155 X->Large.prdct = SumMPI_dc(X->Large.prdct);
156 StopTimer(1);
157 return 0;
158}
