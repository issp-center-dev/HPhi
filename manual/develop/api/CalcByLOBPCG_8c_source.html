154static double calc_preshift(
155 double eig,
156 double res,
157 double eps_LOBPCG
158)
159{
160 double k, i;
161 double preshift;
162
163 if (res < 1.0) {
164 k = trunc(log10(fabs(eig)));
165 if (eps_LOBPCG > res) i = ceil(log10(eps_LOBPCG));
166 else i = ceil(log10(res));
167
168 preshift = trunc(eig / pow(10.0, k + i))*pow(10.0, k + i);
169 }
170 else preshift = 0.0;
171
172 return(preshift);
