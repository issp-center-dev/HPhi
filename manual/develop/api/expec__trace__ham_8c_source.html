55/* ------------------------------------------------------------------ *
56 * File-local collector state (shared with the two static sinks).
57 * TraceHamCollect() runs the two makeHam() passes strictly serially, so
58 * a single set of module globals is sufficient; the collector is not
59 * re-entrant, matching every other makeHam()-driven code path.
60 * ------------------------------------------------------------------ */
61static long int g_n; /* matrix dimension (idim_max) */
62static long int *g_rowptr; /* counts -> prefix-sum offsets, n+1 */
63static long int *g_cursor; /* per-row fill cursors, n+1 */
64static long int *g_colidx; /* raw then merged column indices */
65static double complex *g_val; /* raw then merged values */
66static int g_overflow; /* set if a per-row count overflows */
67
68/* One sort-workspace entry: a (column, value) pair kept together so the
69 * stable merge sort permutes both arrays in lockstep. */
70typedef struct { long int col; double complex val; } TraceEnt;
71
72/* Counting sink: bump the raw entry count of row `irow` (1-based). */
73static void trace_count_sink(long int irow, long int jcol, double complex val) {
74 (void)jcol;
75 (void)val;
76 if (irow < 1 || irow > g_n) { g_overflow = 1; return; }
77 if (g_rowptr[irow] == LONG_MAX) { g_overflow = 1; return; }
78 g_rowptr[irow]++;
79}
80
81/* Fill sink: append (jcol, val) into row `irow`'s segment (0-based column).
82 * Bounded exactly like trace_count_sink: an out-of-range irow, or a cursor
83 * that has already reached the row's segment end g_rowptr[irow] (the
84 * end-offset representation after the pass-1 prefix sum), means pass 2 is
85 * emitting MORE entries for this row than pass 1 counted -- a pass-1/pass-2
86 * divergence. Flag g_overflow and drop the write rather than clobber the next
87 * row's segment; TraceHamCollect() checks the flag after pass 2 and demotes,
88 * exactly like the pass-1 overflow path. O(1), allocation-free. */
89static void trace_fill_sink(long int irow, long int jcol, double complex val) {
90 long int pos;
91 if (irow < 1 || irow > g_n) { g_overflow = 1; return; }
92 if (g_cursor[irow] >= g_rowptr[irow]) { g_overflow = 1; return; }
93 pos = g_cursor[irow]++;
94 g_colidx[pos] = jcol - 1;
95 g_val[pos] = val;
96}

98/* ------------------------------------------------------------------ *
99 * Checked byte arithmetic (mirrors phase-3b's guards).
100 * ------------------------------------------------------------------ */
101static int checked_bytes(uintmax_t a, uintmax_t b, size_t *out) {
102 if (a != 0 && b > UINTMAX_MAX / a) return 0;
103 uintmax_t r = a * b;
104 if (r > SIZE_MAX) return 0;
105 *out = (size_t)r;
106 return 1;
107}

109static int add_size(size_t *acc, size_t add) {
110 if (*acc > SIZE_MAX - add) return 0;
111 *acc += add;
112 return 1;
113}

115size_t TraceHamGatedBytes(long int n, long int nnz_raw, long int k_max,
116 int n_diag) {
117 size_t total = 0, t;
118 uintmax_t np1 = (uintmax_t)n + 1u;
119 uintmax_t nnz = (uintmax_t)nnz_raw;
120 uintmax_t nn = (uintmax_t)n;
121 uintmax_t nd = (uintmax_t)(n_diag > 0 ? n_diag : 0);
122 uintmax_t km = (uintmax_t)k_max;
123 /* One merge-workspace element is a whole TraceEnt (the type the actual
124 alloc at Alloc 4 uses); NOT the bare sizeof(long int)+sizeof(double
125 complex) sum, which undercounts on any ABI that pads TraceEnt (equal on
126 LP64, but the gate must match the real allocation on every ABI). */
127 uintmax_t ws_elem = (uintmax_t)sizeof(TraceEnt);
128
129 /* rowptr */
130 if (!checked_bytes(np1, sizeof(long int), &t) || !add_size(&total, t)) return SIZE_MAX;
131 /* cursors */
132 if (!checked_bytes(np1, sizeof(long int), &t) || !add_size(&total, t)) return SIZE_MAX;
133 /* colidx */
134 if (!checked_bytes(nnz, sizeof(long int), &t) || !add_size(&total, t)) return SIZE_MAX;
135 /* val */
136 if (!checked_bytes(nnz, sizeof(double complex), &t) || !add_size(&total, t)) return SIZE_MAX;
137 /* y */
138 if (!checked_bytes(nn, sizeof(double complex), &t) || !add_size(&total, t)) return SIZE_MAX;
139 /* diagonal coefficient arrays: n_diag * n * sizeof(double) */
140 {
141 size_t nd_n;
142 if (!checked_bytes(nd, nn, &nd_n)) return SIZE_MAX;
143 if (!checked_bytes((uintmax_t)nd_n, sizeof(double), &t) || !add_size(&total, t)) return SIZE_MAX;
144 }
145 /* sort workspace: k_max * sizeof(TraceEnt) (see ws_elem above) */
146 if (!checked_bytes(km, ws_elem, &t) || !add_size(&total, t)) return SIZE_MAX;
147
148 return total;
149}

151/* ------------------------------------------------------------------ *
152 * Frozen per-model number of diagonal coefficient arrays (design §3b).
153 * ------------------------------------------------------------------ */
154static int trace_model_n_diag(int iCalcModel) {
155 switch (iCalcModel) {
156 case Hubbard:
157 case HubbardNConserved:
158 case Kondo:
159 case KondoNConserved:
160 case tJ:
161 case tJNConserved:
162 case HubbardGC:
163 case KondoGC:
164 case tJGC:
165 return 3; /* D(k), N(k), S(k) */
166 case SpinGC:
167 return 1; /* S(k) */
168 case Spin:
169 default:
170 return 0; /* canonical Spin (and any ineligible model) */
171 }
172}

182/* ------------------------------------------------------------------ *
183 * Gated allocator: allocation number *nalloc (0-based) fails when it
184 * equals fail_at, driving the demotion path from the unit test.
185 * ------------------------------------------------------------------ */
186static void *galloc(size_t sz, int *nalloc, int fail_at) {
187 int idx = (*nalloc)++;
188 if (idx == fail_at) return NULL;
189 if (sz == 0) sz = 1;
190 return malloc(sz);
191}

193/* Stable insertion sort of a (col,val) segment. */
194static void insertion_seg(long int *col, double complex *v, long int len) {
195 long int i, j;
196 for (i = 1; i < len; i++) {
197 long int c = col[i];
198 double complex val = v[i];
199 j = i - 1;
200 while (j >= 0 && col[j] > c) {
201 col[j + 1] = col[j];
202 v[j + 1] = v[j];
203 j--;
204 }
205 col[j + 1] = c;
206 v[j + 1] = val;
207 }
208}

210/* Bottom-up stable merge sort of a (col,val) segment, ws sized >= len. */
211static void merge_sort_seg(long int *col, double complex *v, long int len,
212 TraceEnt *ws) {
213 long int width;
214 for (width = 1; width < len; width *= 2) {
215 long int i;
216 for (i = 0; i < len; i += 2 * width) {
217 long int lo = i;
218 long int mid = (i + width < len) ? i + width : len;
219 long int hi = (i + 2 * width < len) ? i + 2 * width : len;
220 long int a = lo, b = mid, k = lo;
221 while (a < mid && b < hi) {
222 if (col[a] <= col[b]) { ws[k].col = col[a]; ws[k].val = v[a]; a++; }
223 else { ws[k].col = col[b]; ws[k].val = v[b]; b++; }
224 k++;
225 }
226 while (a < mid) { ws[k].col = col[a]; ws[k].val = v[a]; a++; k++; }
227 while (b < hi) { ws[k].col = col[b]; ws[k].val = v[b]; b++; k++; }
228 for (k = lo; k < hi; k++) { col[k] = ws[k].col; v[k] = ws[k].val; }
229 }
230 }
231}

233/* ------------------------------------------------------------------ *
234 * Fill the energy-family diagonal coefficient arrays (Task 3).
235 *
236 * diag[q][i] holds the RAW per-basis quantity for CSR row i (0-based), i.e.
237 * the evaluator's basis index k == i + 1. The SAME per-k helpers the Mode-1
238 * evaluators use are called here, so no algebra is duplicated and the arrays
239 * reproduce expec_energy_flct()'s fluctuation fields under the frozen-table
240 * scalings. Serial (single-threaded), run once after the CSR is built.
241 *
242 * Dispatch mirrors expec_energy_flct()'s switch EXACTLY: only iCalcModel ==
243 * HubbardGC uses the GC (bare (k-1)) helper; Hubbard/tJ/tJGC/Kondo/KondoGC
244 * (and the N-conserved variants) use the canonical list_1 helper.
245 * ------------------------------------------------------------------ */
246static void trace_fill_diag(struct BindStruct *X, long int n, int n_diag,
247 double *diag[3]) {
248 long int i;
249 if (n_diag == 3) {
250 int gc = (X->Def.iCalcModel == HubbardGC);
251 double D, N, S;
252 for (i = 0; i < n; i++) {
253 if (gc) EnergyFlctCoeff_HubbardGC(X, i + 1, &D, &N, &S);
254 else EnergyFlctCoeff_Hubbard(X, i + 1, &D, &N, &S);
255 diag[0][i] = D;
256 diag[1][i] = N;
257 diag[2][i] = S;
258 }
259 } else if (n_diag == 1) {
260 double S;
261 for (i = 0; i < n; i++) {
262 if (X->Def.iFlgGeneralSpin == TRUE) EnergyFlctCoeff_GeneralSpinGC(X, i + 1, &S);
263 else EnergyFlctCoeff_HalfSpinGC(X, i + 1, &S);
264 diag[0][i] = S;
265 }
266 }
267}

274 iHamSinkMode = prev_mode;

277int TraceHamCollect(struct BindStruct *X, size_t cap_bytes,
278 int fail_alloc_at, TraceHamCsr *csr) {
279 int prev_mode;
280 void (*prev_hook)(long int, long int, double complex);
281 long int n, nnz_raw, k_max, w;
282 int n_diag, nalloc = 0, di;
283 size_t want;
284 TraceEnt *ws = NULL;
285 double complex *y = NULL;
286 double *diag[3] = { NULL, NULL, NULL };
287 long int r;
288
289 memset(csr, 0, sizeof(*csr));
290 n = (long int)X->Check.idim_max;
291 n_diag = trace_model_n_diag(X->Def.iCalcModel);
292
293 prev_mode = iHamSinkMode;
294 prev_hook = hamCollectSink;
295
296 /* Alloc 0: rowptr (also the per-row counter during pass 1). */
297 g_rowptr = (long int *)galloc(sizeof(long int) * (size_t)(n + 1), &nalloc, fail_alloc_at);
298 /* Alloc 1: cursors. */
299 g_cursor = (long int *)galloc(sizeof(long int) * (size_t)(n + 1), &nalloc, fail_alloc_at);
300 if (g_rowptr == NULL || g_cursor == NULL) {
301 free(g_rowptr); g_rowptr = NULL;
302 free(g_cursor); g_cursor = NULL;
303 return 0; /* sink not yet touched; nothing to restore */
304 }
305
306 /* -------- Pass 1: count raw entries per row. -------- */
307 g_n = n;
308 g_overflow = 0;
309 for (r = 0; r <= n; r++) g_rowptr[r] = 0;
310
311 iHamSinkMode = HAM_SINK_TRACE_COLLECT;
312 hamCollectSink = trace_count_sink;
313 if (makeHam(X) != 0) {
314 fprintf(stdoutMPI, "ERROR: ExpecMode 2 Hamiltonian re-enumeration failed\n");
315 trace_restore_sink(prev_mode, prev_hook);
316 exitMPI(-1);
317 }
318
319 if (g_overflow) {
320 trace_restore_sink(prev_mode, prev_hook);
321 free(g_rowptr); g_rowptr = NULL;
322 free(g_cursor); g_cursor = NULL;
323 return 0;
324 }
325
326 /* k_max = largest raw row; prefix-sum counts into rowptr end-offsets.
327 The prefix sum is guarded exactly like trace_count_sink's per-increment
328 LONG_MAX check: a running total that would overflow signed long int is a
329 demotion (free, restore sink, return 0), never signed-overflow UB. */
330 k_max = 0;
331 for (r = 1; r <= n; r++) if (g_rowptr[r] > k_max) k_max = g_rowptr[r];
332 g_rowptr[0] = 0;
333 for (r = 1; r <= n; r++) {
334 if (g_rowptr[r] > LONG_MAX - g_rowptr[r - 1]) {
335 trace_restore_sink(prev_mode, prev_hook);
336 free(g_rowptr); g_rowptr = NULL;
337 free(g_cursor); g_cursor = NULL;
338 return 0;
339 }
340 g_rowptr[r] += g_rowptr[r - 1];
341 }
342 nnz_raw = g_rowptr[n];
343
344 /* -------- Memory gate. -------- */
345 want = TraceHamGatedBytes(n, nnz_raw, k_max, n_diag);
346 if (want > cap_bytes) {
347 trace_restore_sink(prev_mode, prev_hook);
348 free(g_rowptr); g_rowptr = NULL;
349 free(g_cursor); g_cursor = NULL;
350 return 0;
351 }
352
353 /* Alloc 2: colidx. Alloc 3: val. Alloc 4: sort workspace. Alloc 5: y.
354 Alloc 6..: the n_diag coefficient arrays. */
355 g_colidx = (long int *)galloc(sizeof(long int) * (size_t)nnz_raw, &nalloc, fail_alloc_at);
356 g_val = (double complex *)galloc(sizeof(double complex) * (size_t)nnz_raw, &nalloc, fail_alloc_at);
357 ws = (TraceEnt *)galloc(sizeof(TraceEnt) * (size_t)k_max, &nalloc, fail_alloc_at);
358 y = (double complex *)galloc(sizeof(double complex) * (size_t)n, &nalloc, fail_alloc_at);
359 for (di = 0; di < n_diag; di++) {
360 diag[di] = (double *)galloc(sizeof(double) * (size_t)n, &nalloc, fail_alloc_at);
361 }
362 {
363 int alloc_ok = (g_colidx != NULL && g_val != NULL && ws != NULL && y != NULL);
364 for (di = 0; di < n_diag; di++) if (diag[di] == NULL) alloc_ok = 0;
365 if (!alloc_ok) {
366 trace_restore_sink(prev_mode, prev_hook);
367 free(g_rowptr); g_rowptr = NULL;
368 free(g_cursor); g_cursor = NULL;
369 free(g_colidx); g_colidx = NULL;
370 free(g_val); g_val = NULL;
371 free(ws);
372 free(y);
373 for (di = 0; di < 3; di++) free(diag[di]);
374 return 0;
375 }
376 }
377
378 /* Zero the energy-family coefficient arrays (Task 3 fills them). */
379 for (di = 0; di < n_diag; di++) {
380 memset(diag[di], 0, sizeof(double) * (size_t)n);
381 }
382
383 /* -------- Pass 2: fill row segments. -------- */
384 for (r = 1; r <= n; r++) g_cursor[r] = g_rowptr[r - 1];
385 g_overflow = 0; /* trace_fill_sink raises it on a pass-1/pass-2 divergence */
386 hamCollectSink = trace_fill_sink;
387 if (makeHam(X) != 0) {
388 fprintf(stdoutMPI, "ERROR: ExpecMode 2 Hamiltonian re-enumeration failed\n");
389 trace_restore_sink(prev_mode, prev_hook);
390 exitMPI(-1);
391 }
392
393 /* Detect a pass-1/pass-2 divergence rather than trust the counts blindly:
394 (1) trace_fill_sink set g_overflow if any row OVER-filled (pass 2 emitted
395 more than pass 1 counted) or saw an out-of-range irow; (2) every
396 row's cursor must now sit exactly at its segment end g_rowptr[r] --
397 a cursor short of the end means pass 2 UNDER-filled that row. Either
398 way the CSR would be malformed, so demote (free everything, restore
399 the sink) exactly like the pass-1 overflow path. Cheap O(N),
400 allocation-free. */
401 if (!g_overflow) {
402 for (r = 1; r <= n; r++) {
403 if (g_cursor[r] != g_rowptr[r]) { g_overflow = 1; break; }
404 }
405 }
406 if (g_overflow) {
407 trace_restore_sink(prev_mode, prev_hook);
408 free(g_rowptr); g_rowptr = NULL;
409 free(g_cursor); g_cursor = NULL;
410 free(g_colidx); g_colidx = NULL;
411 free(g_val); g_val = NULL;
412 free(ws);
413 free(y);
414 for (di = 0; di < 3; di++) free(diag[di]);
415 return 0;
416 }
417
418 /* Sinks no longer needed; restore the caller's dense-mode state. */
419 trace_restore_sink(prev_mode, prev_hook);
420
421 /* -------- Per-row stable sort + adjacent-duplicate sum + compaction. --------
422 g_cursor now holds the raw END of each row (== raw rowptr). Snapshot the
423 raw boundaries there while rowptr is overwritten with merged offsets. */
424 for (r = 0; r <= n; r++) g_cursor[r] = g_rowptr[r];
425 w = 0;
426 for (r = 1; r <= n; r++) {
427 long int raw_s = g_cursor[r - 1];
428 long int raw_e = g_cursor[r];
429 long int len = raw_e - raw_s;
430 long int *col = g_colidx + raw_s;
431 double complex *v = g_val + raw_s;
432 long int p;
433 if (len < 16) insertion_seg(col, v, len);
434 else merge_sort_seg(col, v, len, ws);
435 p = 0;
436 while (p < len) {
437 long int c = col[p];
438 double complex acc = v[p];
439 p++;
440 while (p < len && col[p] == c) { acc += v[p]; p++; }
441 g_colidx[w] = c;
442 g_val[w] = acc;
443 w++;
444 }
445 g_rowptr[r] = w;
446 }
447 g_rowptr[0] = 0;
448
449 free(ws);
450 free(g_cursor); g_cursor = NULL;
451
452 /* Debug invariants. */
453 assert(g_rowptr[0] == 0);
454#ifndef NDEBUG
455 for (r = 1; r <= n; r++) assert(g_rowptr[r] >= g_rowptr[r - 1]);
456#endif
457 assert(g_rowptr[n] == w);
458
459 /* Fill the energy-family diagonal coefficient arrays (Task 3). */
460 trace_fill_diag(X, n, n_diag, diag);
461
462 csr->n = n;
463 csr->nnz = w;
464 csr->rowptr = g_rowptr;
465 csr->colidx = g_colidx; /* raw capacity retained */
466 csr->val = g_val;
467 csr->y = y;
468 csr->n_diag = n_diag;
469 for (di = 0; di < 3; di++) csr->diag[di] = diag[di];
470
471 g_rowptr = NULL;
472 g_colidx = NULL;
473 g_val = NULL;
474 return 1;
475}

477void TraceHamFree(TraceHamCsr *csr) {
478 int di;
479 if (csr == NULL) return;
480 free(csr->rowptr);
481 free(csr->colidx);
482 free(csr->val);
483 free(csr->y);
484 for (di = 0; di < 3; di++) free(csr->diag[di]);
485 memset(csr, 0, sizeof(*csr));
486}

488/* ================================================================== *
489 * Phase-3c Task 4: the streaming energy-family kernel.
490 *
491 * For the single eigenstate in the global v1 (== x, 1-based) this
492 * (1) forms y = H.x by a CSR SpMV (row i of y[] from row i of the CSR;
493 * column colidx[k] (0-based) picks x == v1[colidx[k] + 1]);
494 * (2) reduces energy = Re(x^dagger y) and <H^2> = y^dagger y = sum |y_i|^2;
495 * (3) writes the fluctuation-family Phys fields prescribed by the frozen
496 * per-model table from the precomputed csr->diag[] arrays, applying the
497 * SAME scalings expec_energy_flct() applies.
498 *
499 * The trace kernel runs in the replicated FullDiag layout (no MPI site
500 * decomposition -- see expec_trace.c), so x and H are wholly rank-local and no
501 * SumMPI is taken; the reductions are the same reproducibility class as the
502 * Mode-1 evaluators. The SpMV writes only csr->y; the field writes touch only
503 * X->Phys. v0 and v1 are never written. Diagonal H(i,i) is a normal CSR entry
504 * and is included in the SpMV automatically; csr->diag[] carry the SEPARATE
505 * energy-family coefficients (D/N/S), never the matrix diagonal.
506 * ================================================================== */
507int TraceEnergyEvalState(struct BindStruct *X, const TraceHamCsr *csr) {
508 long int n, i, k;
509 const long int *rowptr, *colidx;
510 const double complex *val;
511 double complex *y;
512 double complex e_acc;
513 double var_acc;
514
515 if (X == NULL || csr == NULL) return 1;
516 n = csr->n;
517 if (n < 0) return 1;
518 if (n > 0) {
519 if (csr->rowptr == NULL || csr->y == NULL) return 1;
520 if (csr->nnz > 0 && (csr->colidx == NULL || csr->val == NULL)) return 1;
521 }
522
523 rowptr = csr->rowptr;
524 colidx = csr->colidx;
525 val = csr->val;
526 y = csr->y;
527
528 /* -------- (1) SpMV: y = H.x, per-row dot product (no cross-row accum). --- */
529#pragma omp parallel for default(none) \
530 shared(rowptr, colidx, val, v1, y) firstprivate(n) private(i, k)
531 for (i = 0; i < n; i++) {
532 double complex acc = 0.0;
533 for (k = rowptr[i]; k < rowptr[i + 1]; k++) {
534 acc += val[k] * v1[colidx[k] + 1];
535 }
536 y[i] = acc;
537 }
538
539 /* -------- (2) energy = Re(x^dagger y), var = y^dagger y (== <H^2>). ------ */
540 e_acc = 0.0;
541 var_acc = 0.0;
542#pragma omp parallel for default(none) \
543 shared(v1, y) firstprivate(n) private(i) reduction(+:e_acc, var_acc)
544 for (i = 0; i < n; i++) {
545 e_acc += conj(v1[i + 1]) * y[i]; /* x^dagger y */
546 var_acc += creal(conj(y[i]) * y[i]); /* |y_i|^2 (imag part is zero) */
547 }
548 X->Phys.energy = creal(e_acc);
549 X->Phys.var = var_acc; /* store <H^2>; NOT clamped */
550
551 /* -------- (3) frozen-table fluctuation fields from csr->diag[]. ---------- */
552 if (csr->n_diag == 3) {
553 /* Hubbard family / HubbardGC: diag[0]=D(k), diag[1]=N(k), diag[2]=S(k). */
554 const double *Dc = csr->diag[0], *Nc = csr->diag[1], *Sc = csr->diag[2];
555 double sumD = 0.0, sumD2 = 0.0, sumN = 0.0, sumN2 = 0.0, sumS = 0.0, sumS2 = 0.0;
556#pragma omp parallel for default(none) shared(v1, Dc, Nc, Sc) firstprivate(n) \
557 private(i) reduction(+:sumD, sumD2, sumN, sumN2, sumS, sumS2)
558 for (i = 0; i < n; i++) {
559 double w = creal(conj(v1[i + 1]) * v1[i + 1]);
560 double D = Dc[i], N = Nc[i], S = Sc[i];
561 sumD += w * D; sumD2 += w * D * D;
562 sumN += w * N; sumN2 += w * N * N;
563 sumS += w * S; sumS2 += w * S * S;
564 }
565 X->Phys.doublon = sumD;
566 X->Phys.doublon2 = sumD2;
567 X->Phys.num = sumN;
568 X->Phys.num2 = sumN2;
569 X->Phys.Sz = sumS * 0.5;
570 X->Phys.Sz2 = sumS2 * 0.25;
571 X->Phys.num_up = 0.5 * (sumN + sumS);
572 X->Phys.num_down = 0.5 * (sumN - sumS);
573 } else if (csr->n_diag == 1) {
574 /* SpinGC (half or general): diag[0]=S(k); num is the NsiteMPI constant. */
575 const double *Sc = csr->diag[0];
576 double Ns = (double)X->Def.NsiteMPI;
577 double sumS = 0.0, sumS2 = 0.0;
578#pragma omp parallel for default(none) shared(v1, Sc) firstprivate(n) \
579 private(i) reduction(+:sumS, sumS2)
580 for (i = 0; i < n; i++) {
581 double w = creal(conj(v1[i + 1]) * v1[i + 1]);
582 double S = Sc[i];
583 sumS += w * S; sumS2 += w * S * S;
584 }
585 X->Phys.doublon = 0.0;
586 X->Phys.doublon2 = 0.0;
587 X->Phys.num = Ns;
588 X->Phys.num2 = Ns * Ns;
589 X->Phys.Sz = sumS * 0.5;
590 X->Phys.Sz2 = sumS2 * 0.25;
591 X->Phys.num_up = 0.5 * (Ns + sumS);
592 X->Phys.num_down = 0.5 * (Ns - sumS);
593 } else {
594 /* n_diag == 0: canonical Spin -- constant row; num_up/num_down UNWRITTEN
595 (frozen table), left stale exactly like expec_energy_flct()'s case Spin. */
596 double Ns = (double)X->Def.NsiteMPI;
597 double sz = 0.5 * (double)X->Def.Total2SzMPI;
598 X->Phys.doublon = 0.0;
599 X->Phys.doublon2 = 0.0;
600 X->Phys.num = Ns;
601 X->Phys.num2 = Ns * Ns;
602 X->Phys.Sz = sz;
603 X->Phys.Sz2 = sz * sz;
604 }
605
606 return 0;
