72/* Debug assertions (both branches): the row index is 1-based, so hs_i_ >= 1
73 * is required in either storage mode. A row of 0 -- produced when a Hamiltonian
74 * element helper reports an annihilated/out-of-sector transition and sets its
75 * out-param to 0 (or leaves it stale) -- underran the distributed panel
76 * (Ham_local[... + (0 - 1)], heap-buffer-overflow) while the replicated matrix
77 * silently absorbed it into the never-read Ham[0][j]. All such sites now guard
78 * the call, and this assert is the runtime safety net. The panel branch also
79 * checks the row against the panel leading dimension (HamPanelLd == idim_max);
80 * no cheap idim_max upper bound is in macro scope for the replicated branch
81 * (X->Large.i_max is not visible here), so only the lower bound is asserted
82 * there. */
83#define AddHamElem(irow, jcol, val) \
84 do { \
85 long int hs_i_ = (long int)(irow); \
86 long int hs_j_ = (long int)(jcol); \
87 assert(hs_i_ >= 1); \
88 if (iHamSinkMode == HAM_SINK_TRACE_COLLECT) { \
89 hamCollectSink(hs_i_, hs_j_, (val)); \
90 } else if (iHamPanelActive) { \

92 assert(hs_i_ <= HamPanelLd); \
93 Ham_local[(hs_j_ - HamColBegin) * HamPanelLd + (hs_i_ - 1)] \
94 += (val); \
95 } else { \
96 Ham[hs_i_][hs_j_] += (val); \
97 } \
98 } while (0)
