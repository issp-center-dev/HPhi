68int expec_energy_flct(struct BindStruct *X){
69
70 long unsigned int i,j;
71 long unsigned int irght,ilft,ihfbit;
72 double complex dam_pr,dam_pr1;
73 long unsigned int i_max;
74
75 switch(X->Def.iCalcType){
76 case Lanczos:
77 fprintf(stdoutMPI, "%s", cLogExpecEnergyStart);
78 TimeKeeper(X, cFileNameTimeKeep, cExpecStart, "a");
79 break;
80 case TPQCalc:
81 case cTPQ:
82 case TimeEvolution:
83#ifdef _DEBUG
84 fprintf(stdoutMPI, "%s", cLogExpecEnergyStart);
85 TimeKeeperWithStep(X, cFileNameTimeKeep, cTPQExpecStart, "a", step_i);
86#endif
87 break;
88 case FullDiag:
89 case CG:
90 break;
91 default:
92 return -1;
93 }
94
95 i_max=X->Check.idim_max;
96 if(GetSplitBitByModel(X->Def.Nsite, X->Def.iCalcModel, &irght, &ilft, &ihfbit)!=0){
97 return -1;
98 }
99
100 X->Large.i_max = i_max;
101 X->Large.irght = irght;
102 X->Large.ilft = ilft;
103 X->Large.ihfbit = ihfbit;
104 X->Large.mode = M_ENERGY;
105 X->Phys.energy=0.0;
106
107 int nCalcFlct;
108 if(X->Def.iCalcType == Lanczos){
109 nCalcFlct=4301;
110 }
111 else if (X->Def.iCalcType == TPQCalc){
112 nCalcFlct=3201;
113 }
114 else{//For FullDiag
115 nCalcFlct=5301;
116 }
117 StartTimer(nCalcFlct);
118
119 switch(X->Def.iCalcModel){
120 case HubbardGC:
121 expec_energy_flct_HubbardGC(X);
122 break;
123 case Hubbard:
124 case tJ:
125 case tJGC:
126 case Kondo:
127 case KondoGC:
128 expec_energy_flct_Hubbard(X);
129 break;
130
131 case SpinGC:
132 if(X->Def.iFlgGeneralSpin == FALSE) {
133 expec_energy_flct_HalfSpinGC(X);
134 }
135 else{//for generalspin
136 expec_energy_flct_GeneralSpinGC(X);
137 }
138 break;/*case SpinGC*/
139 /* SpinGCBoost */
140 case Spin:
141 /*
142 if(X->Def.iFlgGeneralSpin == FALSE){
143 expec_energy_flct_HalfSpin(X);
144 }
145 else{
146 expec_energy_flct_GeneralSpin(X);
147 }
148 */
149 X->Phys.doublon = 0.0;
150 X->Phys.doublon2 = 0.0;
151 X->Phys.num = X->Def.NsiteMPI;
152 X->Phys.num2 = X->Def.NsiteMPI*X->Def.NsiteMPI;
153 X->Phys.Sz = 0.5 * (double)X->Def.Total2SzMPI;
154 X->Phys.Sz2 = X->Phys.Sz * X->Phys.Sz;
155 break;
156
157 case SpinlessFermion:
158 // Canonical ensemble: particle number is fixed (use total across MPI)
159 X->Phys.doublon = 0.0;
160 X->Phys.doublon2 = 0.0;
161 X->Phys.num = X->Def.NeMPI;
162 X->Phys.num2 = X->Def.NeMPI * X->Def.NeMPI;
163 X->Phys.Sz = 0.0;
164 X->Phys.Sz2 = 0.0;
165 break;
166
167 case SpinlessFermionGC:
168 // Grand canonical: calculate particle number from wavefunction
169 expec_energy_flct_SpinlessFermionGC(X);
170 break;
171
172 default:
173 return -1;
174 }
175
176 StopTimer(nCalcFlct);
177
178#pragma omp parallel for default(none) private(i) shared(v1,v0) firstprivate(i_max)
179 for(i = 1; i <= i_max; i++){
180 v1[i]=v0[i];
181 v0[i]=0.0;
182 }
183
184 int nCalcExpec;
185 if(X->Def.iCalcType == Lanczos){
186 nCalcExpec=4302;
187 }
188 else if (X->Def.iCalcType == TPQCalc){
189 nCalcExpec=3202;
190 }
191 else{//For FullDiag
192 nCalcExpec=5302;
193 }
194 StartTimer(nCalcExpec);
195 mltply(X, v0, v1); // v0+=H*v1
196 StopTimer(nCalcExpec);
197/* switch -> SpinGCBoost */
198
199 dam_pr=0.0;
200 dam_pr1=0.0;
201 #pragma omp parallel for default(none) reduction(+:dam_pr, dam_pr1) private(j) shared(v0, v1)firstprivate(i_max)
202 for(j=1;j<=i_max;j++){
203 dam_pr += conj(v1[j])*v0[j]; // E = <v1|H|v1>=<v1|v0>
204 dam_pr1 += conj(v0[j])*v0[j]; // E^2 = <v1|H*H|v1>=<v0|v0>
205 //v0[j]=v1[j]; v1-> orginal v0=H*v1
206 }
207 dam_pr = SumMPI_dc(dam_pr);
208 dam_pr1 = SumMPI_dc(dam_pr1);
209 //fprintf(stdoutMPI, "Debug: ene=%lf, var=%lf\n", creal(dam_pr), creal(dam_pr1));
210
211 X->Phys.energy = dam_pr;
212 X->Phys.var = dam_pr1;
213
214 switch(X->Def.iCalcType) {
215 case Lanczos:
216 fprintf(stdoutMPI, "%s", cLogExpecEnergyEnd);
217 TimeKeeper(X, cFileNameTimeKeep, cExpecEnd, "a");
218 break;
219 case TPQCalc:
220 case TimeEvolution:
221#ifdef _DEBUG
222 fprintf(stdoutMPI, "%s", cLogExpecEnergyEnd);
223 TimeKeeperWithStep(X, cFileNameTimeKeep, cTPQExpecEnd, "a", step_i);
224#endif
225 break;
226 default:
227 break;
228 }
229
230 return 0;
231}

238int expec_energy_flct_HubbardGC(struct BindStruct *X) {
239 long unsigned int j;
240 long unsigned int isite1;
241 long unsigned int is1_up_a, is1_up_b;
242 long unsigned int is1_down_a, is1_down_b;
243 int bit_up, bit_down, bit_D;
244 long unsigned int ibit_up, ibit_down, ibit_D;
245 double D, tmp_D, tmp_D2;
246 double N, tmp_N, tmp_N2;
247 double Sz, tmp_Sz, tmp_Sz2;
248 double tmp_v02;
249 long unsigned int i_max;
250 unsigned int l_ibit1, u_ibit1, i_32;
251 i_max=X->Check.idim_max;
252
253 i_32 = 0xFFFFFFFF; //2^32 - 1
254 // tentative doublon
255 tmp_D = 0.0;
256 tmp_D2 = 0.0;
257 tmp_N = 0.0;
258 tmp_N2 = 0.0;
259 tmp_Sz = 0.0;
260 tmp_Sz2 = 0.0;
261
262//[s] for bit count
263 is1_up_a = 0;
264 is1_up_b = 0;
265 is1_down_a = 0;
266 is1_down_b = 0;
267 for (isite1 = 1; isite1 <= X->Def.NsiteMPI; isite1++) {
268 if (isite1 > X->Def.Nsite) {
269 is1_up_a += X->Def.Tpow[2 * isite1 - 2];
270 is1_down_a += X->Def.Tpow[2 * isite1 - 1];
271 } else {
272 is1_up_b += X->Def.Tpow[2 * isite1 - 2];
273 is1_down_b += X->Def.Tpow[2 * isite1 - 1];
274 }
275 }
276//[e]
277#pragma omp parallel for reduction(+:tmp_D,tmp_D2,tmp_N,tmp_N2,tmp_Sz,tmp_Sz2) default(none) shared(v0,list_1) \
278 firstprivate(i_max, X,myrank,is1_up_a,is1_down_a,is1_up_b,is1_down_b,i_32) \
279 private(j, tmp_v02,D,N,Sz,isite1,bit_up,bit_down,bit_D,u_ibit1,l_ibit1,ibit_up,ibit_down,ibit_D)
280 for (j = 1; j <= i_max; j++) {
281 tmp_v02 = conj(v0[j]) * v0[j];
282 bit_up = 0;
283 bit_down = 0;
284 bit_D = 0;
285// isite1 > X->Def.Nsite
286 ibit_up = (unsigned long int) myrank & is1_up_a;
287 u_ibit1 = ibit_up >> 32;
288 l_ibit1 = ibit_up & i_32;
289 bit_up += pop(u_ibit1);
290 bit_up += pop(l_ibit1);
291
292 ibit_down = (unsigned long int) myrank & is1_down_a;
293 u_ibit1 = ibit_down >> 32;
294 l_ibit1 = ibit_down & i_32;
295 bit_down += pop(u_ibit1);
296 bit_down += pop(l_ibit1);
297
298 ibit_D = (ibit_up) & (ibit_down >> 1);
299 u_ibit1 = ibit_D >> 32;
300 l_ibit1 = ibit_D & i_32;
301 bit_D += pop(u_ibit1);
302 bit_D += pop(l_ibit1);
303
304// isite1 <= X->Def.Nsite
305 ibit_up = (unsigned long int) (j - 1) & is1_up_b;
306 u_ibit1 = ibit_up >> 32;
307 l_ibit1 = ibit_up & i_32;
308 bit_up += pop(u_ibit1);
309 bit_up += pop(l_ibit1);
310
311 ibit_down = (unsigned long int) (j - 1) & is1_down_b;
312 u_ibit1 = ibit_down >> 32;
313 l_ibit1 = ibit_down & i_32;
314 bit_down += pop(u_ibit1);
315 bit_down += pop(l_ibit1);
316
317 ibit_D = (ibit_up) & (ibit_down >> 1);
318 u_ibit1 = ibit_D >> 32;
319 l_ibit1 = ibit_D & i_32;
320 bit_D += pop(u_ibit1);
321 bit_D += pop(l_ibit1);
322
323 D = bit_D;
324 N = bit_up + bit_down;
325 Sz = bit_up - bit_down;
326
327 tmp_D += tmp_v02 * D;
328 tmp_D2 += tmp_v02 * D * D;
329 tmp_N += tmp_v02 * N;
330 tmp_N2 += tmp_v02 * N * N;
331 tmp_Sz += tmp_v02 * Sz;
332 tmp_Sz2 += tmp_v02 * Sz * Sz;
333 }
334 tmp_D = SumMPI_d(tmp_D);
335 tmp_D2 = SumMPI_d(tmp_D2);
336 tmp_N = SumMPI_d(tmp_N);
337 tmp_N2 = SumMPI_d(tmp_N2);
338 tmp_Sz = SumMPI_d(tmp_Sz);
339 tmp_Sz2 = SumMPI_d(tmp_Sz2);
340
341 X->Phys.doublon = tmp_D;
342 X->Phys.doublon2 = tmp_D2;
343 X->Phys.num = tmp_N;
344 X->Phys.num2 = tmp_N2;
345 X->Phys.Sz = tmp_Sz*0.5;
346 X->Phys.Sz2 = tmp_Sz2*0.25;
347 X->Phys.num_up = 0.5*(tmp_N+tmp_Sz);
348 X->Phys.num_down = 0.5*(tmp_N-tmp_Sz);
349
350 return 0;
351}

737
738 tmp_N = 0.0;
739 tmp_N2 = 0.0;
740
741 // Separate sites into inter-process (a) and intra-process (b)
742 is_a = 0;
743 is_b = 0;
744 for (isite1 = 1; isite1 <= X->Def.NsiteMPI; isite1++) {
745 if (isite1 > X->Def.Nsite) {
746 is_a += X->Def.Tpow[isite1 - 1]; // Inter-process sites
747 } else {
748 is_b += X->Def.Tpow[isite1 - 1]; // Intra-process sites
749 }
750 }
751
752#pragma omp parallel for reduction(+:tmp_N,tmp_N2) default(none) shared(v0) \
753 firstprivate(i_max, X, myrank, is_a, is_b, i_32) \
754 private(j, tmp_v02, N, bit_n, u_ibit1, l_ibit1, ibit_n)
755 for (j = 1; j <= i_max; j++) {
756 tmp_v02 = conj(v0[j]) * v0[j];
757 bit_n = 0;
758
759 // Count occupied inter-process sites (from myrank)
760 ibit_n = (unsigned long int) myrank & is_a;
761 u_ibit1 = ibit_n >> 32;
762 l_ibit1 = ibit_n & i_32;
763 bit_n += pop(u_ibit1);
764 bit_n += pop(l_ibit1);
765
766 // Count occupied intra-process sites (from j-1)
767 ibit_n = (unsigned long int) (j - 1) & is_b;
768 u_ibit1 = ibit_n >> 32;
769 l_ibit1 = ibit_n & i_32;
770 bit_n += pop(u_ibit1);
771 bit_n += pop(l_ibit1);
772
773 N = bit_n;
774 tmp_N += tmp_v02 * N;
775 tmp_N2 += tmp_v02 * N * N;
776 }
777

779 tmp_N2 = SumMPI_d(tmp_N2);
780

782 X->Phys.doublon2 = 0.0;
783 X->Phys.num = tmp_N;
784 X->Phys.num2 = tmp_N2;
785 X->Phys.Sz = 0.0; // No spin for spinless
786 X->Phys.Sz2 = 0.0;
787 X->Phys.num_up = 0.0;
788 X->Phys.num_down = 0.0;
789
