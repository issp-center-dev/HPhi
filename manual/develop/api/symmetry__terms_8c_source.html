20static int emit(const struct DefineList *def, int kind, unsigned int factors,
21 const int *index, double complex value,
22 SymmetryTermCallback callback, void *context)
23{
24 struct SymmetryTerm term = {0};
25 unsigned int i, j;
26 int diagonal = 1;
27 term.factors = factors;
28 term.value = value;
29 if (!isfinite(creal(value)) || !isfinite(cimag(value))) return -1;
30 for (i = 0; i < 4 * factors; i += 2) {
31 if (index[i] < 0 || (unsigned int)index[i] >= def->Nsite ||
32 index[i+1] < 0 || index[i+1] > (def->iCalcModel == SpinlessFermion ? 0 : 1))
33 return -1;
34 term.index[i] = index[i];
35 term.index[i+1] = index[i+1];
36 }
37 if (def->iCalcModel == Spin) {
38 for (i = 0; i < factors; ++i)
39 if (index[4*i] != index[4*i+2]) return -1;
40 }
41 /* A product is diagonal precisely when the occupation change at each
42 * orbital (or the local Spin state) vanishes. */
43 for (i = 0; i < factors; ++i) {
44 int balance = 0;
45 for (j = 0; j < factors; ++j) {
46 if (index[4*i] == index[4*j] && index[4*i+1] == index[4*j+1]) ++balance;
47 if (index[4*i] == index[4*j+2] && index[4*i+1] == index[4*j+3]) --balance;
48 }
49 if (balance != 0) diagonal = 0;
50 }
51 if (kind >= 0 && kind != !diagonal) return 0;
52 return callback(&term, context);
53}

134int ApplySymmetryTerm(const struct DefineList *def,
135 const struct SymmetryTerm *term, unsigned long state,
136 unsigned long *out, double complex *value)
137{
138 int f;
139 double sign = 1;
140 unsigned int width = (def && (def->iCalcModel == Hubbard || def->iCalcModel == tJ)) ? 2U : 1U;
141 if (!def || !term || !out || !value || term->factors < 1 || term->factors > 2 ||
142 def->Nsite == 0 || def->Nsite > CHAR_BIT * sizeof(state) / width ||
143 (def->iCalcModel != Spin && def->iCalcModel != SpinlessFermion &&
144 def->iCalcModel != Hubbard && def->iCalcModel != tJ)) return -1;
145 for (f = 0; f < (int)(4*term->factors); f += 2)
146 if (term->index[f] < 0 || (unsigned int)term->index[f] >= def->Nsite ||
147 term->index[f+1] < 0 || term->index[f+1] > (def->iCalcModel == SpinlessFermion ? 0 : 1))
148 return -1;
149 if (def->iCalcModel == Spin)
150 for (f = 0; f < (int)term->factors; ++f)
151 if (term->index[4*f] != term->index[4*f+2]) return -1;
152 for (f = (int)term->factors - 1; f >= 0; --f) {
153 const int *x = term->index + 4*f;
154 if (def->iCalcModel == Spin) {
155 unsigned long mask = 1UL << x[0];
156 if ((int)((state >> x[0]) & 1UL) != x[3]) return 0;
157 state = (state & ~mask) | ((unsigned long)x[1] << x[0]);
158 } else {
159 int op;
160 for (op = 1; op >= 0; --op) { /* annihilation, then creation */
161 unsigned int orbital = width * (unsigned int)x[2*op] + (unsigned int)x[2*op+1];
162 unsigned long mask = 1UL << orbital, below = state & (mask - 1UL);
163 if (((state & mask) != 0) != (op == 1)) return 0;
164 while (below) { sign = -sign; below &= below - 1UL; }
165 state ^= mask;
166 }
167 }
168 }
169 /* Project the final state onto the physical tJ Hilbert space. */
170 if (def->iCalcModel == tJ && (state & (state >> 1U) & (ULONG_MAX / 3UL))) return 0;
171 *out = state;
172 *value = sign * term->value;
173 return 1;
174}

187static int append_monomial(struct Polynomial *poly, const int key[7], double complex value)
188{
189 if (poly->count == poly->capacity) {
190 size_t cap = poly->capacity ? poly->capacity * 2 : 32;
191 struct Monomial *next;
192 if (cap < poly->capacity || cap > SIZE_MAX / sizeof(*next)) return -1;
193 next = realloc(poly->terms, cap * sizeof(*next));
194 if (!next) return -1;
195 poly->terms = next; poly->capacity = cap;
196 }
197 memcpy(poly->terms[poly->count].key, key, 7 * sizeof(int));
198 poly->terms[poly->count++].value = value;
199 return 0;

202static int canonical_term(const struct SymmetryTerm *input, void *context)
203{
204 struct Polynomial *poly = context;
205 int x[8], key[7] = {0}, n = (int)input->factors, i;
206 double complex value = input->value;
207 memcpy(x, input->index, sizeof(x));
208 if (poly->permutation)
209 for (i = 0; i < 4*n; i += 2) x[i] = poly->permutation[x[i]];
210 if (poly->def->iCalcModel == Spin) {
211 if (n == 2 && x[0] == x[4]) {
212 if (x[3] != x[5]) return 0;
213 x[3] = x[7]; n = 1;
214 }
215 if (n == 2 && x[0] > x[4])
216 for (i = 0; i < 4; ++i) { int tmp = x[i]; x[i] = x[i+4]; x[i+4] = tmp; }
217 /* At most two E00 factors: expand their identity/E11 choices. */
218 for (i = 0; i < (1 << n); ++i) {
219 int f, count = 0, valid = 1;
220 double complex coefficient = value;
221 memset(key, 0, sizeof(key));
222 for (f = 0; f < n; ++f) {
223 int zero = x[4*f+1] == 0 && x[4*f+3] == 0;
224 int choose = (i >> f) & 1;
225 if (!zero && choose) { valid = 0; break; }
226 if (zero && !choose) continue;
227 key[1+3*count] = x[4*f];
228 key[2+3*count] = zero ? 1 : x[4*f+1];
229 key[3+3*count] = zero ? 1 : x[4*f+3];
230 if (zero) coefficient = -coefficient;
231 ++count;
232 }
233 key[0] = count;
234 if (valid && append_monomial(poly, key, coefficient)) return -1;
235 }
236 return 0;
237 } else {
238 int width = (poly->def->iCalcModel == Hubbard || poly->def->iCalcModel == tJ) ? 2 : 1;
239 int a = width*x[0]+x[1], b = width*x[2]+x[3];
240 if (n == 1) {
241 key[0] = 1; key[1] = a; key[2] = b;
242 return append_monomial(poly, key, value);
243 } else {
244 int c = width*x[4]+x[5], d = width*x[6]+x[7], tmp;
245 if (b == c) {
246 key[0] = 1; key[1] = a; key[2] = d;
247 if (append_monomial(poly, key, value)) return -1;
248 }
249 if (a == c || b == d) return 0;
250 /* P O P vanishes if a normal-ordered string creates or annihilates
251 * a doublon. Retain any contraction emitted above before discarding it. */
252 if (poly->def->iCalcModel == tJ && (a / 2 == c / 2 || b / 2 == d / 2)) return 0;
253 value = -value;
254 if (a > c) { tmp = a; a = c; c = tmp; value = -value; }
255 if (b > d) { tmp = b; b = d; d = tmp; value = -value; }
256 memset(key, 0, sizeof(key));
257 key[0] = 2; key[1] = a; key[2] = c; key[3] = b; key[4] = d;
258 return append_monomial(poly, key, value);
259 }
260 }

274static int collect_polynomial(struct Polynomial *poly)
275{
276 size_t i, next = 0;
277 if (EnumerateSymmetryTerms(poly->def, -1, canonical_term, poly)) return -1;
278 if (poly->count) qsort(poly->terms, poly->count, sizeof(*poly->terms), compare_monomials);
279 for (i = 0; i < poly->count; ++i) {
280 if (next && compare_monomials(&poly->terms[next-1], &poly->terms[i]) == 0)
281 poly->terms[next-1].value += poly->terms[i].value;
282 else poly->terms[next++] = poly->terms[i];
283 }
284 poly->count = 0;
285 for (i = 0; i < next; ++i) {
286 if (!isfinite(creal(poly->terms[i].value)) || !isfinite(cimag(poly->terms[i].value))) return -1;
287 if (cabs(poly->terms[i].value) > 1e-10) poly->terms[poly->count++] = poly->terms[i];
288 }
289 return 0;
290}

292int ValidateSymmetryTerms(const struct DefineList *def)
293{
294 struct Polynomial base = {0}, mapped = {0};
295 size_t i;
296 unsigned int g;
297 int error = 1;
298 base.def = mapped.def = def;
299 if (collect_polynomial(&base)) goto done;
300 for (i = 0; i < base.count; ++i) {
301 const int *key = base.terms[i].key;
302 int delta = 0, f;
303 if (def->iCalcModel == Spin)
304 for (f = 0; f < key[0]; ++f) delta += key[2+3*f] - key[3+3*f];
305 else if (def->iCalcModel == Hubbard || def->iCalcModel == tJ)
306 for (f = 0; f < key[0]; ++f) delta += key[1+f]%2 - key[1+key[0]+f]%2;
307 if (delta) {
308 fprintf(stdoutMPI, "Error: TransSym Hamiltonian does not conserve fixed Sz/Nup/Ndown.\n");
309 goto done;
310 }
311 }
312 for (g = 0; g < def->NSymTrans; ++g) {
313 mapped.count = 0;
314 mapped.permutation = def->SymTrans[g];
315 if (collect_polynomial(&mapped)) goto done;
316 if (base.count != mapped.count) goto invariant_error;
317 for (i = 0; i < base.count; ++i)
318 if (compare_monomials(&base.terms[i], &mapped.terms[i]) ||
319 cabs(base.terms[i].value - mapped.terms[i].value) > 1e-10)
320 goto invariant_error;
321 }
322 error = 0;
323 goto done;
324invariant_error:
325 fprintf(stdoutMPI, "Error: TransSym Hamiltonian invariance failed for normalized terms under op %u.\n", g);
326done:
327 free(base.terms); free(mapped.terms);
328 if (error) fprintf(stdoutMPI, "Error: invalid or unsupported TransSym Hamiltonian terms.\n");
329 return error ? -1 : 0;
