15 * IEEE binary64 real/imaginary components (index zero is not serialized).
16 * Phase 1: normalized sum_g conjugate(chi(g)) T_g |smallest representative>,
17 * with a positive real coefficient at that representative.
18 * Kondo version 2 appends the local-site mask and fixed-quantity flags,
19 * using phase 2 (physical Kondo translation). */

56{
57 unsigned char bytes[8];

61static int get_word(FILE *fp, uint64_t *word)
62{
63 unsigned char bytes[8];
64 if (fread(bytes, 1, 8, fp) != 8) return -1;
65 *word = 0;
66 for (int i = 7; i >= 0; --i) *word = (*word << 8) | bytes[i];

129static int consistent_header(const uint64_t *header, int words)
130{
131 uint64_t local[MAX_WORDS], root[MAX_WORDS];
132 memcpy(local, header, words * sizeof(*local));
133 local[RANK] = local[OFFSET] = local[LOCAL_DIM] = local[ORDER] = 0;
134 memcpy(root, local, words * sizeof(*root));
135#ifdef MPI
136 if (MPI_Bcast(root, words, MPI_UINT64_T, 0, MPI_COMM_WORLD) != MPI_SUCCESS) return -1;
137#endif
138 return failure(memcmp(root, local, words * sizeof(*root)) != 0, "cross-rank metadata consistency");
139}
140static int vector_digest(const struct BindStruct *X, const double complex *vector,
141 uint64_t *xor_hash, uint64_t *sum_hash)
142{
143 uint64_t hash = hash_word(FNV_OFFSET, (uint64_t)myrank);
144 double norm = 0;
145 int invalid = vector == NULL;
146 hash = hash_word(hash, X->Sym->local_offset);
147 hash = hash_word(hash, X->Sym->local_dim);
148 for (unsigned long i = 1; !invalid && i <= X->Sym->local_dim; ++i) {
149 double re = creal(vector[i]), im = cimag(vector[i]);
150 if (!isfinite(re) || !isfinite(im)) { invalid = 1; break; }
151 hash = hash_word(hash_word(hash, double_word(re)), double_word(im));
152 norm += re*re + im*im;
153 }
154 if (failure(invalid, "finite vector validation")) return -1;
155 norm = SumMPI_d(norm);
156 if (failure(!isfinite(norm) || fabs(norm - 1.0) > 1e-8, "global norm validation")) return -1;
157 *xor_hash = *sum_hash = hash;
158#ifdef MPI
159 if (MPI_Allreduce(&hash, xor_hash, 1, MPI_UINT64_T, MPI_BXOR, MPI_COMM_WORLD) != MPI_SUCCESS ||
160 MPI_Allreduce(&hash, sum_hash, 1, MPI_UINT64_T, MPI_SUM, MPI_COMM_WORLD) != MPI_SUCCESS) return -1;
161#endif
162 return 0;
163}
