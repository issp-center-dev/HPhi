65/* Phase 3c Task 5: rank-local energy-kernel streaming time (seconds), reset at
66 the top of every phys_stateparallel_local_loop() and read back by the
67 orchestrator via phys_stateparallel_energy_stream_seconds(). This is a plain
68 file-static double with a plain accessor -- no MPI, so the ExpecLocal guard
69 (which scans this file) is satisfied. */
70static double g_energy_stream_seconds = 0.0;
71
72double phys_stateparallel_energy_stream_seconds(void) {
73 return g_energy_stream_seconds;
74}

78static double local_now_seconds(void) {
79 struct timespec ts;
80 clock_gettime(CLOCK_MONOTONIC, &ts);
81 return (double)ts.tv_sec + (double)ts.tv_nsec * 1e-9;
82}

96int phys_stateparallel_local_loop(struct BindStruct *X,
97 double complex *panel,
98 long int jb, long int je, long int NN,
99 const TraceExecutionPlan *plan,
100 const TraceHamCsr *ham_csr) {
101 int rc = 0;
102 long int n, j;
103
104 g_energy_stream_seconds = 0.0; /* reset the per-loop kernel-stream accumulator */
105
106 for (n = jb; n <= je && rc == 0; n++) {
107 X->Phys.eigen_num = (int)(n - 1); /* 0-based, as in the serial phys.c */
108
109 for (j = 0; j < NN; j++) {
110 v0[j + 1] = panel[(n - jb) * NN + j];
111 }
112
113 /* Energy/fluctuation family: the trace kernel (when plan->kernel[ENERGY])
114 or the ExpecMode-1 fallback. In the FALLBACK branch expec_energy_flct()
115 consumes x from v0 and copies v0->v1 itself (byte-identical to ExpecMode
116 0/1). In the KERNEL branch the driver reproduces that v0->v1 copy (the
117 postcondition v1==x every downstream evaluator relies on), then streams
118 the energy family from the CSR; the kernel reads v1/CSR only and writes
119 neither v0 nor v1 (design spec 3b step 1). */
120 if (!plan->kernel[TRACE_Q_ENERGY]) {
121 if (expec_energy_flct(X) != 0) { rc = -1; break; }
122 } else {
123 double t0;
124 for (j = 0; j < NN; j++) v1[j + 1] = v0[j + 1];
125 t0 = local_now_seconds();
126 if (TraceEnergyEvalState(X, ham_csr) != 0) { rc = -1; break; }
127 g_energy_stream_seconds += local_now_seconds() - t0;
128 }
129 if (!plan->kernel[TRACE_Q_ONEBODY]) {
130 if (expec_cisajs(X, v1) != 0) { rc = -1; break; }
131 }
132 if (!plan->kernel[TRACE_Q_TWOBODY]) {
133 if (expec_cisajscktaltdc(X, v1) != 0) { rc = -1; break; }
134 }
135 if (expec_nbodyg(X, v1) != 0) { rc = -1; break; }
136 if (expec_anomalousg(X, v1) != 0) { rc = -1; break; }
137 if (X->Def.iCalcType == FullDiag) {
138 if (expec_totalspin(X, v1) != 0) { rc = -1; break; }
139 }
140 if (ExpecLocalError()) { rc = -1; break; }
141
142 /* Record this rank's owned segment (index n-1), same assignment group as
143 the tail of the serial phys.c loop. */
144 X->Phys.all_energy[n - 1] = X->Phys.energy;
145 X->Phys.all_doublon[n - 1] = X->Phys.doublon;
146 X->Phys.all_sz[n - 1] = X->Phys.Sz;
147 X->Phys.all_s2[n - 1] = X->Phys.s2;
148 X->Phys.all_num_up[n - 1] = X->Phys.num_up;
149 X->Phys.all_num_down[n - 1] = X->Phys.num_down;
150 }
151
152 return rc;
153}
