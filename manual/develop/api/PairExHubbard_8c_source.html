37int GetPairExcitedStateHubbardGC(
38 struct BindStruct *X,
39 double complex *tmp_v0,
40 double complex *tmp_v1
42){
43
44 long unsigned int i,j;
45 long unsigned int isite1;
46 long unsigned int org_isite1,org_isite2,org_sigma1,org_sigma2;
47
48 double complex tmp_trans=0;
49 long int i_max;
50 long int ibit;
51 long unsigned int is;
52 i_max = X->Check.idim_maxOrg;
53 for(i=0;i<X->Def.NPairExcitationOperator;i++) {
54 org_isite1 = X->Def.PairExcitationOperator[i][0] + 1;
55 org_isite2 = X->Def.PairExcitationOperator[i][2] + 1;
56 org_sigma1 = X->Def.PairExcitationOperator[i][1];
57 org_sigma2 = X->Def.PairExcitationOperator[i][3];
58 tmp_trans = X->Def.ParaPairExcitationOperator[i];
59
60 if (org_isite1 > X->Def.Nsite &&
61 org_isite2 > X->Def.Nsite) {
62 if (org_isite1 == org_isite2 && org_sigma1 == org_sigma2) {
63 if (X->Def.PairExcitationOperator[i][4] == 0) {
64 if (org_sigma1 == 0) {
65 is = X->Def.Tpow[2 * org_isite1 - 2];
66 }
67 else {
68 is = X->Def.Tpow[2 * org_isite1 - 1];
69 }
70 ibit = (unsigned long int) myrank & is;
71 if (ibit != is) {
72 //minus sign comes from negative tmp_trans due to readdef
73#pragma omp parallel for default(none) shared(tmp_v0, tmp_v1) \
74 firstprivate(i_max, tmp_trans) private(j)
75 for (j = 1; j <= i_max; j++) tmp_v0[j] += -tmp_trans * tmp_v1[j];
76 }
77 }
78 else {//X->Def.PairExcitationOperator[i][4]==1
79 if (org_sigma1 == 0) {
80 is = X->Def.Tpow[2 * org_isite1 - 2];
81 }
82 else {
83 is = X->Def.Tpow[2 * org_isite1 - 1];
84 }
85 ibit = (unsigned long int) myrank & is;
86 if (ibit == is) {
87#pragma omp parallel for default(none) shared(tmp_v0, tmp_v1) \
88 firstprivate(i_max, tmp_trans) private(j)
89 for (j = 1; j <= i_max; j++) tmp_v0[j] += tmp_trans * tmp_v1[j];
90 }
91 }
92 }
93 else {
94 child_GC_general_hopp_MPIdouble(org_isite1 - 1, org_sigma1, org_isite2 - 1, org_sigma2, -tmp_trans, X,
95 tmp_v0, tmp_v1);
96 }
97 }
98 else if (org_isite2 > X->Def.Nsite || org_isite1 > X->Def.Nsite) {
99 if (org_isite1 < org_isite2) {
100 child_GC_general_hopp_MPIsingle(org_isite1 - 1, org_sigma1, org_isite2 - 1, org_sigma2, -tmp_trans, X,
101 tmp_v0, tmp_v1);
102 }
103 else {
104 child_GC_general_hopp_MPIsingle(org_isite2 - 1, org_sigma2, org_isite1 - 1, org_sigma1,
105 -conj(tmp_trans), X, tmp_v0, tmp_v1);
106 }
107 }
108 else {
109
110 if (org_isite1 == org_isite2 && org_sigma1 == org_sigma2 && X->Def.PairExcitationOperator[i][4] == 0) {
111 isite1=X->Def.Tpow[2 * org_isite1 - 2 + org_sigma1];
112#pragma omp parallel for default(none) private(j) firstprivate(i_max,X,isite1, tmp_trans) shared(tmp_v0, tmp_v1)
113 for(j=1;j<=i_max;j++){
114 GC_AisCis(j, tmp_v0, tmp_v1, X, isite1, -tmp_trans);
115 }
116 }
117 else {
118 if (general_hopp_GetInfo(X, org_isite1, org_isite2, org_sigma1, org_sigma2) != 0) {
119 return -1;
120 }
121 GC_general_hopp(tmp_v0, tmp_v1, X, tmp_trans);
122 }
123 }
124 }
125 return TRUE;
126}

146 long unsigned int irght,ilft,ihfbit;
147 long unsigned int org_isite1,org_isite2,org_sigma1,org_sigma2;
148 long unsigned int tmp_off=0;
149
150 double complex tmp_trans=0;
151 long int i_max;
152 int tmp_sgn, num1;
153 long int ibit;
154 long unsigned int is, Asum, Adiff;
155 long unsigned int ibitsite1, ibitsite2;
156
157 // i_max = X->Check.idim_max;
158 i_max = X->Check.idim_maxOrg;
159 if(GetSplitBitByModel(X->Def.Nsite, X->Def.iCalcModel, &irght, &ilft, &ihfbit)!=0){

161 }

165 X->Large.ihfbit = ihfbit;
166 X->Large.mode=M_CALCSPEC;

170 //set size
171#ifdef MPI
172 idim_maxMPI = MaxMPI_li(X->Check.idim_maxOrg);

174#endif // MPI

176 for(i=0;i<NPairExcitationOperator;i++){
177 org_isite1 = PairExcitationOperator[i][0]+1;
178 org_isite2 = PairExcitationOperator[i][2]+1;
179 org_sigma1 = PairExcitationOperator[i][1];
180 org_sigma2 = PairExcitationOperator[i][3];
181 tmp_trans = ParaPairExcitationOperator[i];
182 ibitsite1 = X->Def.OrgTpow[2*org_isite1-2+org_sigma1] ;
183 ibitsite2 = X->Def.OrgTpow[2*org_isite2-2+org_sigma2] ;
184 general_hopp_GetInfo(X, org_isite1, org_isite2, org_sigma1, org_sigma2);
185 Asum = X->Large.isA_spin;
186 Adiff = X->Large.A_spin;
187
188 if(X->Def.iFlagListModified == TRUE // Not to adopt HubbrdNConserved
189 && org_sigma1 != org_sigma2){
190 if (org_isite1 > X->Def.Nsite &&
191 org_isite2 > X->Def.Nsite)
192 {
193 child_CisAjt_MPIdouble(org_isite1-1, org_sigma1, org_isite2-1, org_sigma2, -tmp_trans, X, tmp_v0, tmp_v1, tmp_v1bufOrg, list_1_org, list_1buf_org, list_2_1, list_2_2);
194 }
195 else if (org_isite2 > X->Def.Nsite
196 || org_isite1 > X->Def.Nsite)
197 {
198 if(org_isite1 < org_isite2) {
199 child_CisAjt_MPIsingle(org_isite1 - 1, org_sigma1, org_isite2 - 1, org_sigma2, -tmp_trans, X, tmp_v0,
200 tmp_v1, tmp_v1bufOrg, list_1_org, list_1buf_org, list_2_1, list_2_2);
201 } else{
202 child_CisAjt_MPIsingle(org_isite2 - 1, org_sigma2, org_isite1 - 1, org_sigma1, -conj(tmp_trans), X, tmp_v0,
203 tmp_v1, tmp_v1bufOrg, list_1_org, list_1buf_org, list_2_1, list_2_2); }
204 }
205 else{
206#pragma omp parallel for default(none) shared(tmp_v0, tmp_v1,stdoutMPI) \
207 firstprivate(i_max, tmp_trans, Asum, Adiff, ibitsite1, ibitsite2, X, list_1_org, list_1, myrank) \
208 private(j, tmp_sgn, tmp_off)
209 for (j = 1; j <= i_max; j++){
210 tmp_sgn=child_CisAjt(list_1_org[j], X, ibitsite1, ibitsite2, Asum, Adiff, &tmp_off);
211 tmp_v0[tmp_off] += tmp_trans * tmp_sgn*tmp_v1[j];
212 }
213 }
214 }
215 else{
216 if (org_isite1 > X->Def.Nsite &&
217 org_isite2 > X->Def.Nsite) {
218 if(org_isite1==org_isite2 && org_sigma1==org_sigma2){//diagonal
219 is = X->Def.Tpow[2 * org_isite1 - 2 + org_sigma1];
220 ibit = (unsigned long int) myrank & is;
221 if( PairExcitationOperator[i][4]==0) {
222 if (ibit != is) {
223#pragma omp parallel for default(none) shared(tmp_v0, tmp_v1) \
224 firstprivate(i_max, tmp_trans) private(j)
225 for (j = 1; j <= i_max; j++) tmp_v0[j] += -tmp_trans * tmp_v1[j];
226 }
227 }
228 else{
229 if (ibit == is) {
230#pragma omp parallel for default(none) shared(tmp_v0, tmp_v1) \
231 firstprivate(i_max, tmp_trans) private(j)
232 for (j = 1; j <= i_max; j++) tmp_v0[j] += tmp_trans * tmp_v1[j];
233 }
234 }
235 }
236 else{
237 child_general_hopp_MPIdouble(org_isite1-1, org_sigma1, org_isite2-1, org_sigma2, -tmp_trans, X, tmp_v0, tmp_v1);
238 }
239 }
240 else if (org_isite2 > X->Def.Nsite || org_isite1 > X->Def.Nsite){
241 if(org_isite1 < org_isite2){
242 child_general_hopp_MPIsingle(org_isite1-1, org_sigma1,org_isite2-1, org_sigma2, -tmp_trans, X, tmp_v0, tmp_v1);
243 }
244 else{
245 child_general_hopp_MPIsingle(org_isite2-1, org_sigma2, org_isite1-1, org_sigma1, -conj(tmp_trans), X, tmp_v0, tmp_v1);
246 }
247 }
248 else{
249 if(general_hopp_GetInfo( X,org_isite1,org_isite2,org_sigma1,org_sigma2)!=0){
250 return -1;
251 }
252 if(org_isite1==org_isite2 && org_sigma1==org_sigma2){
253 is = X->Def.Tpow[2 * org_isite1 - 2 + org_sigma1];
254 if( PairExcitationOperator[i][4]==0) {
255#pragma omp parallel for default(none) shared(list_1, tmp_v0, tmp_v1) firstprivate(i_max, is, tmp_trans) private(num1, ibit)
256 for (j = 1; j <= i_max; j++) {
257 ibit = list_1[j] & is;
258 num1 = (1-ibit / is);
259 tmp_v0[j] += -tmp_trans * num1 * tmp_v1[j];
260 }
261 }
262 else{
263#pragma omp parallel for default(none) shared(list_1, tmp_v0, tmp_v1) firstprivate(i_max, is, tmp_trans) private(num1, ibit)
264 for (j = 1; j <= i_max; j++) {
265 ibit = list_1[j] & is;
266 num1 = ibit / is;
267 tmp_v0[j] += tmp_trans * num1 * tmp_v1[j];
268 }
269 }
270 }
271 else{
272 general_hopp(tmp_v0, tmp_v1,X,tmp_trans);
273 }
274 }
275 }
276 }
277
278#ifdef MPI
279 free_cd_1d_allocate(tmp_v1bufOrg);
280#endif // MPI
281
282 return TRUE;
283}
