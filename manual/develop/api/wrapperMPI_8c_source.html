132void InitializeMPI(int argc, char *argv[]){
133 int ierr;
134
135#ifdef MPI
136 ierr = MPI_Init(&argc, &argv);
137 ierr = MPI_Comm_size(MPI_COMM_WORLD, &nproc);
138 ierr = MPI_Comm_rank(MPI_COMM_WORLD, &myrank);
139 if(ierr != 0) exitMPI(ierr);
140 /* Parse HPHI_MPI_NOBATCH on rank 0 and broadcast so every rank agrees
141 (a per-rank divergence would deadlock the collective MPI_Sendrecv pattern). */
142 if (myrank == 0) {
143 const char *e = getenv("HPHI_MPI_NOBATCH");
144 iFlgMPIBatch = (e != NULL && atoi(e) != 0) ? 0 : 1;
145 }
146 MPI_Bcast(&iFlgMPIBatch, 1, MPI_INT, 0, MPI_COMM_WORLD);
147#else
148 nproc = 1;
149 myrank = 0;
150 {
151 const char *e = getenv("HPHI_MPI_NOBATCH");
152 iFlgMPIBatch = (e != NULL && atoi(e) != 0) ? 0 : 1;
153 }
154#endif
155 if (myrank == 0) stdoutMPI = stdout;
156 else stdoutMPI = fopen("/dev/null", "w");
157 splash();
158
159#pragma omp parallel default(none) shared(nthreads)
160#pragma omp master
161#ifdef _OPENMP
162 nthreads = omp_get_num_threads();
163#else
164 nthreads=1;
165#endif
166 fprintf(stdoutMPI, "\n\n##### Parallelization Info. #####\n\n");
167 fprintf(stdoutMPI, " OpenMP threads : %d\n", nthreads);
168 fprintf(stdoutMPI, " MPI PEs : %d \n\n", nproc);
169 fprintf(stdoutMPI, " MPI batching : %s\n\n", iFlgMPIBatch ? "ON" : "OFF");
170}/*void InitializeMPI(int argc, char *argv[])*/

197static void WaitAbortGrace(void)
198{
199 struct timespec req, rem;
200 req.tv_sec = HPHI_MPI_ABORT_GRACE_MS / 1000;
201 req.tv_nsec = (long)(HPHI_MPI_ABORT_GRACE_MS % 1000) * 1000000L;
202 while (nanosleep(&req, &rem) != 0 && errno == EINTR) req = rem;
203}

270char* fgetsMPI(
271 char* InputString,
272 int maxcount,
273 FILE* fp
274){
275 int inull;
276 char *ctmp;
277
278 /* Not reachable from the FullDiag expec_* evaluation layer (see
279 docs/superpowers/specs/2026-07-11-expec-call-inventory.md §1); a debug
280 assert catches an accidental future call during ExpecLocal mode
281 instead of inventing untested no-communication semantics for it (same
282 style as the other assert-forbidden wrappers below, e.g. BarrierMPI). */
283 assert(!iExpecLocal);
284 ctmp = InputString;
285 inull = 0;
286 if (myrank == 0) {
287 ctmp = fgets(InputString, maxcount, fp);
288 if (ctmp == NULL){
289 inull = 1;
290 }
291
292 while(*InputString == '\n' || strncmp(InputString, "#", 1)==0){
293 ctmp = fgets(InputString, maxcount, fp);
294 if (ctmp == NULL){
295 inull=1;
296 break;
297 }
298 }
299 }
300#ifdef MPI
301 MPI_Bcast(InputString, maxcount, MPI_CHAR, 0, MPI_COMM_WORLD);
302 MPI_Bcast(&inull, 1, MPI_INT, 0, MPI_COMM_WORLD);
303#endif
304 if (myrank != 0 && inull == 1) {
305 ctmp = NULL;
306 }
307
308 return ctmp;

497double NormMPI_dc(
498 unsigned long int idim,
499 double complex *_v1
500){
501 double complex cdnorm=0;
502 double dnorm =0;
503 unsigned long int i;
504 /* Not reachable from the FullDiag expec_* evaluation layer (see
505 docs/superpowers/specs/2026-07-11-expec-call-inventory.md §1); a debug
506 assert catches an accidental future call during ExpecLocal mode
507 instead of inventing untested no-communication semantics for it.
508 (The inner SumMPI_dc call below is already a no-op under
509 ExpecLocal, so if this guard is ever relaxed, the reduction itself
510 needs no further change.) */

512 //DEBUG
513#pragma omp parallel for default(none) private(i) firstprivate(myrank) shared(_v1, idim) reduction(+: cdnorm)
514 for(i=1;i<=idim;i++){
515 cdnorm += conj(_v1[i])*_v1[i];
516 }
517#ifdef MPI
518 cdnorm = SumMPI_dc(cdnorm);
519#endif
520 dnorm=creal(cdnorm);
521 dnorm=sqrt(dnorm);
522
523 return dnorm;

530double complex VecProdMPI(
531 long unsigned int ndim,
532 double complex *v1,
533 double complex *v2
534){
535 long unsigned int idim;
536 double complex prod;
537
538 /* Not reachable from the FullDiag expec_* evaluation layer (see
539 docs/superpowers/specs/2026-07-11-expec-call-inventory.md §1); a debug
540 assert catches an accidental future call during ExpecLocal mode
