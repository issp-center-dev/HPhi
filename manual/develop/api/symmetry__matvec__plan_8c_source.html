21static int apply_exchange_halfspin(unsigned long int state,
22 int site0,
23 int site1,
24 unsigned long int *out_state)
25{
26 unsigned long int b0 = (state >> (unsigned int)site0) & 1UL;
27 unsigned long int b1 = (state >> (unsigned int)site1) & 1UL;
28 if (b0 == b1) return FALSE;
29 *out_state = state ^ (1UL << (unsigned int)site0) ^ (1UL << (unsigned int)site1);
30 return TRUE;
31}

33static unsigned long int mask_between_sites(unsigned int site0, unsigned int site1)
34{
35 unsigned int lo = site0 < site1 ? site0 : site1;
36 unsigned int hi = site0 < site1 ? site1 : site0;
37 if (hi <= lo + 1U) return 0UL;
38 return (1UL << hi) - (1UL << (lo + 1U));
39}

41static int apply_spinless_hopping_hermite(unsigned long int state,
42 unsigned int site1,
43 unsigned int site2,
44 double complex trans,
45 unsigned long int *out_state,
46 double complex *hval)
47{
48 unsigned long int mask1 = 1UL << site1;
49 unsigned long int mask2 = 1UL << site2;
50 unsigned long int occupied1 = state & mask1;
51 unsigned long int occupied2 = state & mask2;
52 int sgn = 1;
53 if (site1 == site2) return FALSE;
54 if ((occupied1 == 0UL && occupied2 == 0UL) ||
55 (occupied1 != 0UL && occupied2 != 0UL)) {
56 return FALSE;
57 }
58 SgnBit(state & mask_between_sites(site1, site2), &sgn);
59 *out_state = state ^ mask1 ^ mask2;
60 if (occupied1 != 0UL && occupied2 == 0UL) {
61 *hval = (double)sgn * conj(trans);
62 } else {
63 *hval = (double)sgn * trans;
64 }
65 return TRUE;
66}

68static int apply_hubbard_hopping_hermite(unsigned long int state,
69 unsigned int site1,
70 unsigned int spin1,
71 unsigned int site2,
72 unsigned int spin2,
73 double complex trans,
74 unsigned long int *out_state,
75 double complex *hval)
76{
77 const unsigned int max_bits = (unsigned int)(sizeof(unsigned long int) * CHAR_BIT);
78 unsigned int orbital1;
79 unsigned int orbital2;
80 if (spin1 > 1U || spin2 > 1U) return FALSE;
81 if (site1 > (max_bits - spin1) / 2U ||
82 site2 > (max_bits - spin2) / 2U) {
83 return FALSE;
84 }
85 orbital1 = 2U * site1 + spin1;
86 orbital2 = 2U * site2 + spin2;
87 if (orbital1 >= max_bits || orbital2 >= max_bits) return FALSE;
88 return apply_spinless_hopping_hermite(state, orbital1, orbital2, trans,
89 out_state, hval);
90}

355 const struct BindStruct *X,
356 unsigned long int local_row_begin,
357 unsigned long int local_row_count,
358 uint64_t memory_byte_limit,
359 struct SymmetryUnresolvedMatvecBlock *block_out)
360{

362 size_t *row_counts = NULL;
363 size_t row_ptr_bytes = 0U;
364 size_t row_count_bytes = 0U;
365 size_t transition_bytes = 0U;
366 size_t request_bytes = 0U;
367 size_t count_peak = 0U;
368 size_t storage_bytes = 0U;
369 size_t temporary_peak = 0U;
370 size_t request_write = 0U;

375 block_out->transitions != NULL || block_out->request_keys != NULL ||
376 block_out->local_row_begin != 0UL ||

389 local_row_count > X->Sym->local_dim - local_row_begin ||

395 next.local_row_begin = local_row_begin;
396 next.local_row_count = local_row_count;
397 next.memory_byte_limit = memory_byte_limit;

399 (size_t)local_row_count + 1U, sizeof(*next.row_ptr),

401 SymmetryCheckedSizeMul(
402 (size_t)local_row_count, sizeof(*row_counts),
403 &row_count_bytes) != 0 ||
404 SymmetryCheckedSizeAdd(

413 (size_t *)calloc((size_t)local_row_count + 1U,

416 if (local_row_count > 0UL) {
417 row_counts =

419 sizeof(*row_counts));
420 if (row_counts == NULL) goto fail;
421 }
422#pragma omp parallel for default(none) schedule(static) reduction(|:build_error) \
423 shared(X, local_row_begin, local_row_count, row_counts)

433 }

435 for (block_row = 0UL; block_row < local_row_count; block_row++) {
436 if (next.row_ptr[block_row] >
437 SIZE_MAX - row_counts[block_row]) {
438 goto fail;
439 }

441 next.row_ptr[block_row] + row_counts[block_row];
442 }
443 next.transition_count = next.row_ptr[local_row_count];
444 if (next.transition_count < (size_t)local_row_count) goto fail;
445 next.offdiagonal_count =

448 next.transition_count, sizeof(*next.transitions),
449 &transition_bytes) != 0 ||
450 SymmetryCheckedSizeMul(
451 next.offdiagonal_count, sizeof(*next.request_keys),

453 SymmetryCheckedSizeAdd(

455 SymmetryCheckedSizeAdd(
456 storage_bytes, request_bytes, &storage_bytes) != 0) {
457 goto fail;
458 }

461#if SIZE_MAX > UINT64_MAX
462 if (temporary_peak > (size_t)UINT64_MAX) goto fail;
463#endif
464 if ((uint64_t)temporary_peak > memory_byte_limit) {
465 goto fail;
466 }
467 free(row_counts);
468 row_counts = NULL;
469 if (next.transition_count > 0U) {
470 next.transitions =
471 (struct SymmetryUnresolvedTransition *)malloc(transition_bytes);
472 if (next.transitions == NULL) goto fail;
473 }
474 if (next.offdiagonal_count > 0U) {

477 if (next.request_keys == NULL) goto fail;
478 }

480#pragma omp parallel for default(none) schedule(static) reduction(|:build_error) \

484 SymmetryBasisLocalEntry(
485 X->Sym, local_row_begin + block_row + 1UL);

492 }
493 }

496 transition_index < next.transition_count;
497 transition_index++) {

502 }
503 if (request_write != next.offdiagonal_count) goto fail;
504 if (request_write > 1U) {
505 size_t input;
506 size_t output = 1U;

508 sizeof(*next.request_keys),
509 compare_unresolved_request_key);
510 for (input = 1U; input < request_write; input++) {
511 if (next.request_keys[output - 1U] ==
512 next.request_keys[input]) {
513 continue;
514 }
515 next.request_keys[output++] = next.request_keys[input];
516 }
517 next.request_count = output;
518 } else {
519 next.request_count = request_write;
520 }
521 next.storage_bytes = storage_bytes;
522 next.temporary_peak_bytes = temporary_peak;
523 *block_out = next;
524 return 0;
525
526fail:
527 free(row_counts);
528 FreeSymmetryUnresolvedMatvecBlock(&next);
529 return -1;
530}

663 while (left < right) {
664 size_t middle = left + (right - left) / 2U;
665 if (unresolved->request_keys[middle] < rep_state) {
666 left = middle + 1U;
667 } else {
668 right = middle;
669 }
670 }
671 if (left >= unresolved->request_count ||
672 unresolved->request_keys[left] != rep_state) {
673 return -1;
674 }
675 if (global_beta[left] == 0UL) {
676 if (norm[left] != 0.0 || signbit(norm[left])) return -1;
677 } else if (!isfinite(norm[left]) || norm[left] <= 0.0) {
678 return -1;
679 }
680 *resolved_beta = global_beta[left];
681 *resolved_norm = norm[left];
682 return 0;
683}

718 unresolved->transition_count,

731 if (peak > (size_t)UINT64_MAX) return -1;

738static int finalize_distributed_matvec_block(
739 const struct BindStruct *X,
740 const struct SymmetryUnresolvedMatvecBlock *unresolved,
741 const unsigned long int *resolved_beta,
742 const double *resolved_norm,
743 struct SymmetryMatvecBlock *block_out)
744{
745 struct SymmetryMatvecBlock next;
746 size_t column_bytes = 0U;
747 size_t value_bytes = 0U;
748 unsigned long int block_row;
749 int finalize_error = 0;
750 if (X == NULL || X->Sym == NULL || unresolved == NULL ||
751 block_out == NULL || block_out->row_ptr != NULL ||
752 block_out->global_columns != NULL ||
753 block_out->column_slot32 != NULL ||
754 block_out->column_slot64 != NULL ||
755 block_out->values != NULL ||
756 unresolved->row_ptr == NULL ||
757 unresolved->local_row_begin > X->Sym->local_dim ||
758 unresolved->local_row_count >
759 X->Sym->local_dim - unresolved->local_row_begin ||
760 unresolved->row_ptr[0] != 0U ||
761 unresolved->row_ptr[unresolved->local_row_count] !=
762 unresolved->transition_count) {
763 return -1;
764 }
765 memset(&next, 0, sizeof(next));
766 next.local_row_begin = unresolved->local_row_begin;
767 next.local_row_count = unresolved->local_row_count;
768 next.row_ptr =
769 (size_t *)calloc(
770 (size_t)next.local_row_count + 1U,
771 sizeof(*next.row_ptr));
772 if (next.row_ptr == NULL) goto fail;
773#pragma omp parallel for default(none) schedule(static) \
774 reduction(|:finalize_error) \
775 shared(X, unresolved, resolved_beta, resolved_norm, next)
776 for (block_row = 0UL;
777 block_row < unresolved->local_row_count;
778 block_row++) {
779 size_t transition;
780 size_t count = 0U;
781 size_t begin = unresolved->row_ptr[block_row];
782 size_t end = unresolved->row_ptr[block_row + 1UL];
783 if (begin >= end ||
784 unresolved->transitions == NULL ||
785 unresolved->transitions[begin].is_diagonal != TRUE) {
786 finalize_error = 1;
787 continue;
788 }
789 for (transition = begin; transition < end; transition++) {
790 const struct SymmetryUnresolvedTransition *candidate =
791 &unresolved->transitions[transition];
792 if (transition == begin) {
793 if (candidate->is_diagonal != TRUE) {
794 finalize_error = 1;
795 break;
796 }
797 count++;
798 } else {
799 unsigned long int global_beta;
800 double target_norm;
801 if (candidate->is_diagonal != FALSE ||
802 find_resolved_request(
803 unresolved, resolved_beta, resolved_norm,
804 candidate->rep_state, &global_beta,
805 &target_norm) != 0) {
806 finalize_error = 1;
807 break;
808 }
809 (void)target_norm;
810 if (global_beta != 0UL) {
811 if (count == SIZE_MAX) {
812 finalize_error = 1;
813 break;
814 }
815 count++;
816 }
817 }
818 }
819 next.row_ptr[block_row + 1UL] = count;
820 }
821 if (finalize_error != 0) goto fail;
822 for (block_row = 0UL; block_row < next.local_row_count; block_row++) {
823 if (next.row_ptr[block_row] >
824 SIZE_MAX - next.row_ptr[block_row + 1UL]) {
825 goto fail;
826 }
827 next.row_ptr[block_row + 1UL] += next.row_ptr[block_row];
828 }
829 next.nnz = next.row_ptr[next.local_row_count];
830 if (SymmetryCheckedSizeMul(
831 next.nnz, sizeof(*next.global_columns),
832 &column_bytes) != 0 ||
833 SymmetryCheckedSizeMul(
834 next.nnz, sizeof(*next.values),
835 &value_bytes) != 0) {
836 goto fail;
837 }
838 if (next.nnz > 0U) {
839 next.global_columns =
840 (unsigned long int *)malloc(column_bytes);
841 next.values = (double complex *)malloc(value_bytes);
842 if (next.global_columns == NULL || next.values == NULL) goto fail;
843 }
844 finalize_error = 0;
845#pragma omp parallel for default(none) schedule(static) \
846 reduction(|:finalize_error) \
847 shared(X, unresolved, resolved_beta, resolved_norm, next)
848 for (block_row = 0UL;
849 block_row < unresolved->local_row_count;
850 block_row++) {
851 const struct SymmetryBasisVector *source =
852 SymmetryBasisLocalEntry(
853 X->Sym,
854 unresolved->local_row_begin + block_row + 1UL);
855 size_t transition;
856 size_t write = next.row_ptr[block_row];
857 size_t write_end = next.row_ptr[block_row + 1UL];
858 size_t begin = unresolved->row_ptr[block_row];
859 size_t end = unresolved->row_ptr[block_row + 1UL];
860 if (source == NULL || !isfinite(source->norm) ||
861 source->norm <= 0.0 || begin >= end) {
862 finalize_error = 1;
863 continue;
864 }
865 for (transition = begin; transition < end; transition++) {
866 const struct SymmetryUnresolvedTransition *candidate =
867 &unresolved->transitions[transition];
868 unsigned long int global_beta;
869 double target_norm;
870 double complex coefficient;
871 if (transition == begin) {
872 global_beta =
873 X->Sym->local_offset +
874 unresolved->local_row_begin + block_row + 1UL;
875 target_norm = source->norm;
876 coefficient = candidate->phased_hval;
877 } else {
878 if (find_resolved_request(
879 unresolved, resolved_beta, resolved_norm,
880 candidate->rep_state, &global_beta,
881 &target_norm) != 0) {
882 finalize_error = 1;
883 break;
884 }
885 if (global_beta == 0UL) continue;
886 coefficient =
887 candidate->phased_hval *
888 (target_norm / source->norm);
889 }
890 if (write >= write_end || global_beta == 0UL ||
891 global_beta > X->Sym->dim) {
892 finalize_error = 1;
893 break;
894 }
895 next.global_columns[write] = global_beta;
896 next.values[write] = conj(coefficient);
897 write++;
898 }
899 if (write != write_end) finalize_error = 1;
900 }
901 if (finalize_error != 0) goto fail;
902 *block_out = next;
903 return 0;
904
905fail:
906 free(next.row_ptr);
907 free(next.global_columns);
908 free(next.values);
909 return -1;
910}

1003 }

1016 plan->block_count = local_block_count;
1017 if (plan->block_count > SIZE_MAX / sizeof(*plan->blocks)) {
1018 local_error = 1;
1019 } else {
1020 plan->build_local_wave_count = local_wave_count;
1021 plan->build_max_wave_count = max_wave_count;

1023 memory_policy.warning_byte_threshold;

1025 memory_policy.hard_byte_limit;

1027 (struct SymmetryMatvecBlock *)calloc(
1028 plan->block_count, sizeof(*plan->blocks));
1029 if (plan->blocks == NULL) local_error = 1;
1030 }
1031 }
1032 if (SymmetryMpiAgreeError(mpi_active, local_error) != 0) goto fail;

1054 BuildSymmetryUnresolvedMatvecBlock(
1055 X, local_row_begin, local_row_count,
1056 effective_memory_limit,
1057 &unresolved) != 0;

1061 &unresolved, effective_memory_limit,
1062 &round_peak) != 0 ||
1063 (size_t)(uint64_t)unresolved.request_count !=

1074 plan->build_temporary_peak_bytes = round_peak;

1078 &memory_policy, (uint64_t)round_peak,
1079 mpi_active, myrank, "matvec-plan", "block-wave",
1080 memory_warning_emitted == 0);

1082 FreeSymmetryUnresolvedMatvecBlock(&unresolved);
1083 goto fail;
1084 }
1085 if (memory_status > 0) memory_warning_emitted = 1;
1086 }
1087 if (SymmetryCheckedSizeMul(
1088 unresolved.request_count, sizeof(*resolved_beta),
1089 &response_beta_bytes) != 0 ||
1090 SymmetryCheckedSizeMul(
1091 unresolved.request_count, sizeof(*resolved_norm),
1092 &response_norm_bytes) != 0) {
1093 local_error = 1;
1094 } else if (unresolved.request_count > 0U) {
1095 resolved_beta =
1096 (unsigned long int *)malloc(response_beta_bytes);
1097 resolved_norm = (double *)malloc(response_norm_bytes);
1098 if (resolved_beta == NULL || resolved_norm == NULL) {
1099 local_error = 1;
1100 }
1101 }
1102 if (SymmetryMpiAgreeError(mpi_active, local_error) != 0) {
1103 free(resolved_beta);
1104 free(resolved_norm);
1105 FreeSymmetryUnresolvedMatvecBlock(&unresolved);
1106 goto fail;
1107 }
1108 if (SymmetryResolveRepresentativeBatchWithOptions(
1109 X->Sym->representative_directory,

1114 }

1116 free(resolved_beta);
1117 free(resolved_norm);

1142 (size_t)block->local_row_count + 1U,

1148 block->nnz, sizeof(*block->values),

1157 plan->column_storage_bytes >
1158 SIZE_MAX - column_bytes) {
1159 local_error = 1;

1166 size_t row_nnz =
1167 block->row_ptr[row + 1UL] -
1168 block->row_ptr[row];
1169 if (row_nnz > plan->row_nnz_max) {
1170 plan->row_nnz_max = row_nnz;
1171 }
1172 }
1173 block_write++;
1174 next_local_row_begin += local_row_count;
1175 }
1176 }
1177 }
1178 free(resolved_beta);
1179 free(resolved_norm);
1180 FreeSymmetryUnresolvedMatvecBlock(&unresolved);
1181 if (SymmetryMpiAgreeError(mpi_active, local_error) != 0) goto fail;
1182 }
1183 if (plan->local_dim == 0UL) {
1184 plan->blocks[0].row_ptr =
1185 (size_t *)calloc(1U, sizeof(*plan->blocks[0].row_ptr));
1186 if (plan->blocks[0].row_ptr == NULL) {
1187 local_error = 1;
1188 } else {
1189 plan->matrix_storage_bytes =
1190 sizeof(*plan->blocks[0].row_ptr);
1191 block_write = 1U;
1192 }
1193 }
1194 if (block_write != plan->block_count ||
1195 next_local_row_begin != plan->local_dim) {
1196 local_error = 1;
1197 }
1198 if (SymmetryMpiAgreeError(mpi_active, local_error) != 0) goto fail;
1199 plan->ready = TRUE;
1200 sync_single_block_aliases(plan);
1201 X->Sym->matvec_plan = plan;
1202 return 0;
1203
1204fail:
1205 FreeSymmetryMatvecPlan(plan);
1206 return -1;
1207}
