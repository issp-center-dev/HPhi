258{

271static int validate_exchange_invariance(const struct DefineList *def)
272{
273 unsigned int g, i;
274 for (g = 0; g < def->NSymTrans; g++) {
275 for (i = 0; i < def->NExchangeCoupling; i++) {
276 int src0 = def->ExchangeCoupling[i][0];
277 int src1 = def->ExchangeCoupling[i][1];
278 if (src0 < 0 || src1 < 0 ||
279 (unsigned int)src0 >= def->Nsite ||
280 (unsigned int)src1 >= def->Nsite) {
281 fprintf(stdoutMPI,
282 "Error: TransSym Exchange term %u has a site outside [0, Nsite).\n",
283 i);
284 return -1;
285 }
286 if (src0 == src1) continue;
287 int a = def->SymTrans[g][src0];
288 int b = def->SymTrans[g][src1];
289 double src_sum = sum_exchange_pair(def, src0, src1);
290 double mapped_sum = sum_exchange_pair(def, a, b);

292 fprintf(stdoutMPI,
