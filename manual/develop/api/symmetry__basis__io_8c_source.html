321{

393 for (g = 0; g < def->NSymTrans; g++) {

678 for (i = 0; i < def->NCoulombIntra; i++) {

686 i);

688 }
689 mapped = def->SymTrans[g][src];
690 src_sum = sum_hubbard_coulomb_intra_site(def, src);
691 mapped_sum = sum_hubbard_coulomb_intra_site(def, mapped);
692 if (count_hubbard_coulomb_intra_site(def, mapped) == 0) {
693 fprintf(stdoutMPI,
694 "Error: TransSym Hubbard CoulombIntra term %u maps to a missing site under op %u.\n",
695 i, g);
696 return -1;
697 }
698 if (fabs(mapped_sum - src_sum) > 1.0e-10) {
699 fprintf(stdoutMPI,
700 "Error: TransSym Hubbard CoulombIntra invariance failed for term %u under op %u.\n",
701 i, g);
702 return -1;
703 }
704 }
705 }

710{
711 const struct DefineList *def = &X->Def;
712 if (def->iFlgSymmetryBasis == FALSE) return 0;
713 if (def->iCalcModel == Spin) {

715 if (validate_ising_diagonal_invariance(def) != 0) return -1;
716 } else if (def->iCalcModel == SpinlessFermion) {
717 if (validate_spinless_transfer_invariance(def) != 0) return -1;
718 if (validate_spinless_coulomb_invariance(def) != 0) return -1;
719 } else if (def->iCalcModel == Hubbard) {
720 if (validate_hubbard_transfer_invariance(def) != 0) return -1;
721 if (validate_hubbard_coulomb_intra_invariance(def) != 0) return -1;
722 }
723 return 0;
