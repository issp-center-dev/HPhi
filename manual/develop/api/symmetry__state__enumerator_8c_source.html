138 const struct SymmetryStateEnumerator *enumerator,
139 unsigned long int rank,
140 unsigned long int *state)
141{
142 unsigned int position = enumerator->bit_count;
143 unsigned int remaining_up = enumerator->nup;
144 unsigned int remaining_down = enumerator->ndown;
145 unsigned long int result = 0UL;
146 while (position-- > 0U) {
147 unsigned int lower_up = (position + 1U) / 2U;
148 unsigned int lower_down = position / 2U;
149 unsigned int *remaining =
150 (position % 2U == 0U) ? &remaining_up : &remaining_down;
151 unsigned long int up_count, down_count, zero_count;
152 up_count = enumerator_binomial(enumerator, lower_up, remaining_up);
153 down_count = enumerator_binomial(enumerator, lower_down, remaining_down);

155 return -1;
