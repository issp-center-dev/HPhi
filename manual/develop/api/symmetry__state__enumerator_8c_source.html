8static int initialize_binomial_table(
9 struct SymmetryStateEnumerator *enumerator)
10{
11 unsigned int n, k;
12 if (enumerator == NULL ||
13 enumerator->nsite > HPHI_SYMMETRY_STATE_WORD_BITS) {
14 return -1;
15 }
16 memset(enumerator->binomial, 0, sizeof(enumerator->binomial));
17 enumerator->binomial[0][0] = 1UL;
18 for (n = 1U; n <= enumerator->nsite; n++) {
19 enumerator->binomial[n][0] = 1UL;
20 enumerator->binomial[n][n] = 1UL;
21 for (k = 1U; k < n; k++) {
22 if (enumerator->binomial[n - 1U][k] >
23 ULONG_MAX - enumerator->binomial[n - 1U][k - 1U]) {
24 return -1;
25 }
26 enumerator->binomial[n][k] =
27 enumerator->binomial[n - 1U][k] +
28 enumerator->binomial[n - 1U][k - 1U];
29 }
30 }
31 return 0;
32}

60 unsigned long int up_dim, down_dim, raw_dim;

84 initialized.bit_count = def->Nsite;
85 break;

87 case tJ:

89 def->Nup > def->Nsite ||

91 def->Ne != def->Nup + def->Ndown ||

98 return -1;
99 }

103 if (initialized.model == Hubbard || initialized.model == tJ) {
104 down_dim = enumerator_binomial(

108 } else {
109 raw_dim = up_dim;
110 }
111 if (raw_dim == 0UL || raw_dim != expected_raw_dim) return -1;
112 initialized.raw_dim = raw_dim;
113 *enumerator = initialized;
114 return 0;

117static int fixed_popcount_state_at(
118 const struct SymmetryStateEnumerator *enumerator,
119 unsigned long int rank,
120 unsigned long int *state)
121{
122 unsigned int position = enumerator->nsite;
123 unsigned int remaining = enumerator->nup;
124 unsigned long int result = 0UL;
125 while (position-- > 0U) {
126 unsigned long int zero_count =
127 enumerator_binomial(enumerator, position, remaining);
128 if (rank >= zero_count) {
129 result |= 1UL << position;
130 rank -= zero_count;
131 if (remaining == 0U) return -1;
132 remaining--;
133 }
134 }
135 if (rank != 0UL || remaining != 0U) return -1;
136 *state = result;
137 return 0;
138}

140static int hubbard_state_at(
141 const struct SymmetryStateEnumerator *enumerator,
142 unsigned long int rank,
143 unsigned long int *state)
144{
145 unsigned int position = enumerator->bit_count;
146 unsigned int remaining_up = enumerator->nup;
147 unsigned int remaining_down = enumerator->ndown;
148 unsigned long int result = 0UL;
149 while (position-- > 0U) {
150 unsigned int lower_up = (position + 1U) / 2U;
151 unsigned int lower_down = position / 2U;
152 unsigned int *remaining =
153 (position % 2U == 0U) ? &remaining_up : &remaining_down;
154 unsigned long int up_count, down_count, zero_count;
155 up_count = enumerator_binomial(enumerator, lower_up, remaining_up);
156 down_count = enumerator_binomial(enumerator, lower_down, remaining_down);
157 if (checked_product(up_count, down_count, &zero_count) != 0) {
158 return -1;
159 }
160 if (rank >= zero_count) {
161 result |= 1UL << position;
162 rank -= zero_count;
163 if (*remaining == 0U) return -1;
164 (*remaining)--;
165 }
166 }
167 if (rank != 0UL || remaining_up != 0U || remaining_down != 0U) return -1;
168 *state = result;
169 return 0;
170}

181/* Lexicographic base-4 state order, with local digits 0 (empty), 1 (up),
182 * 2 (down). Count completions instead of constructing a Hubbard list and
183 * removing doubly occupied states. This also covers empty/full sectors. */
184static int tj_state_at(const struct SymmetryStateEnumerator *enumerator,
185 unsigned long rank, unsigned long *state)
186{
187 unsigned int site = enumerator->nsite;
188 unsigned int up = enumerator->nup, down = enumerator->ndown;
189 unsigned long result = 0UL;
190 while (site-- > 0U) {
191 unsigned long empty_count, up_count = 0UL;
192 if (tj_completion_count(enumerator, site, up, down, &empty_count)) return -1;
193 if (rank < empty_count) continue;
194 rank -= empty_count;
195 if (up && tj_completion_count(enumerator, site, up - 1U, down, &up_count)) return -1;
196 if (rank < up_count) {
197 result |= 1UL << (2U * site);
198 --up;
199 } else {
200 rank -= up_count;
201 if (!down) return -1;
202 result |= 1UL << (2U * site + 1U);
203 --down;
204 }
205 }
206 if (rank || up || down) return -1;
207 *state = result;
208 return 0;
209}
