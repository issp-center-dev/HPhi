10static int initialize_binomial_table(
11 struct SymmetryStateEnumerator *enumerator, unsigned int maximum)
12{
13 unsigned int n, k;
14 if (enumerator == NULL ||
15 maximum > HPHI_SYMMETRY_STATE_WORD_BITS) {
16 return -1;
17 }
18 memset(enumerator->binomial, 0, sizeof(enumerator->binomial));
19 enumerator->binomial[0][0] = 1UL;
20 for (n = 1U; n <= maximum; n++) {
21 enumerator->binomial[n][0] = 1UL;
22 enumerator->binomial[n][n] = 1UL;
23 for (k = 1U; k < n; k++) {
24 if (enumerator->binomial[n - 1U][k] >
25 ULONG_MAX - enumerator->binomial[n - 1U][k - 1U]) {
26 return -1;
27 }
28 enumerator->binomial[n][k] =
29 enumerator->binomial[n - 1U][k] +
30 enumerator->binomial[n - 1U][k - 1U];
31 }
32 }
33 return 0;
34}

71static int kondo_completions(const struct SymmetryStateEnumerator *e, unsigned int l,
72 unsigned int c, int up, int down, int nc,
73 unsigned long *count)
74{
75 unsigned long total = 0, a, b, term;
76 unsigned int u, exponent;
77 if (e->model == Kondo) {
78 if (up < 0 || down < 0 || (unsigned int)up > l + c || (unsigned int)down > l + c) {
79 *count = 0;
80 return 0;
81 }
82 for (u = 0; u <= l; ++u) {
83 a = kondo_binomial(e, c, up - (int)u);
84 b = kondo_binomial(e, c, down - (int)l + (int)u);
85 if (!a || !b)
86 continue;
87 if (dimension_product(a, b, &term) ||
88 dimension_product(term, e->binomial[l][u], &term) ||
89 term > (unsigned long)LONG_MAX || total > (unsigned long)LONG_MAX - term)
90 return -1;
91 total += term;
92 }
93 } else {
94 total = e->model == KondoGC ? 1UL : kondo_binomial(e, 2 * c, nc);
95 exponent = l + (e->model == KondoGC ? 2 * c : 0U);
96 for (u = 0; u < exponent; ++u)
97 if (dimension_product(total, 2UL, &total))
98 return -1;
99 }
100 *count = total;
101 return 0;
102}

198static int initialize_kondo(const struct DefineList *def, unsigned long expected,
199 struct SymmetryStateEnumerator *e)
200{
201 unsigned int site, maximum, l = def->NLocSpn, c = def->Nsite - l;
202 unsigned long dimension;

204 SymmetryKondoLocalMask(def, &e->local_site_mask))
205 return -1;

214 e->prefix_local[site + 1] = e->prefix_local[site] + local;
215 e->prefix_conduction[site + 1] = e->prefix_conduction[site] + 1U - local;

220static int kondo_state_at(const struct SymmetryStateEnumerator *e, unsigned long rank,
221 unsigned long *state)
222{
223 unsigned int site = e->nsite;
224 int up = (int)e->nup, down = (int)e->ndown, nc = (int)e->ncond;
225 unsigned long word = 0;
226 while (site-- > 0U) {
227 unsigned int digit, local = (unsigned int)((e->local_site_mask >> site) & 1UL);
228 for (digit = 0; digit < 4; ++digit) {
229 unsigned long completions;
230 int next_up = up - (int)(digit & 1U), next_down = down - (int)(digit >> 1);
231 int next_nc = nc - (local ? 0 : (int)((digit & 1U) + (digit >> 1)));
232 if (local && (digit == 0 || digit == 3))
233 continue;
234 if (kondo_completions(e, e->prefix_local[site], e->prefix_conduction[site], next_up,
235 next_down, next_nc, &completions))
236 return -1;
237 if (rank > completions) {
238 rank -= completions;
239 continue;
240 }
241 word |= (unsigned long)digit << (2 * site);
242 up = next_up;
243 down = next_down;
244 nc = next_nc;
245 break;
246 }
247 if (digit == 4)
248 return -1;
249 }
250 if (rank != 1 || (e->model == Kondo && (up || down)) ||
251 (e->model == KondoNConserved && nc))
252 return -1;
253 *state = word;
254 return 0;
255}

259 unsigned long int expected_raw_dim,

271 initialized.ndown = def->Ndown;

275 *enumerator = initialized;

283 initialized.raw_dim = 1UL << def->Nsite;

297 initialized.raw_dim = raw_dim;

308 initialized.raw_dim = raw_dim;
309 *enumerator = initialized;
310 return 0;
311 }

317 def->Nup > def->Nsite ||

324 break;

328 initialized.ndown = 0U;
329 initialized.bit_count = def->Nsite;

334 def->Nup > def->Nsite ||
335 def->Ndown > def->Nsite ||
336 def->Ne != def->Nup + def->Ndown ||

339 }

341 break;

344 }

348 if (initialized.model == Hubbard || initialized.model == tJ) {
349 down_dim = enumerator_binomial(

351 initialized.ndown);
352 if (checked_product(up_dim, down_dim, &raw_dim) != 0) return -1;
353 } else {
354 raw_dim = up_dim;
355 }
356 if (raw_dim == 0UL || raw_dim != expected_raw_dim) return -1;
357 initialized.raw_dim = raw_dim;
358 *enumerator = initialized;

362static int fixed_popcount_state_at(
363 const struct SymmetryStateEnumerator *enumerator,
364 unsigned long int rank,
365 unsigned long int *state)
366{
367 unsigned int position = enumerator->bit_count;
368 unsigned int remaining = enumerator->nup;
369 unsigned long int result = 0UL;
370 while (position-- > 0U) {
371 unsigned long int zero_count =
372 enumerator_binomial(enumerator, position, remaining);
373 if (rank >= zero_count) {
374 result |= 1UL << position;
375 rank -= zero_count;
376 if (remaining == 0U) return -1;
377 remaining--;
378 }
379 }
380 if (rank != 0UL || remaining != 0U) return -1;
381 *state = result;
382 return 0;
383}

385static int hubbard_state_at(
386 const struct SymmetryStateEnumerator *enumerator,
387 unsigned long int rank,
388 unsigned long int *state)
389{
390 unsigned int position = enumerator->bit_count;
391 unsigned int remaining_up = enumerator->nup;
392 unsigned int remaining_down = enumerator->ndown;
393 unsigned long int result = 0UL;
394 while (position-- > 0U) {
395 unsigned int lower_up = (position + 1U) / 2U;
396 unsigned int lower_down = position / 2U;
397 unsigned int *remaining =
398 (position % 2U == 0U) ? &remaining_up : &remaining_down;
399 unsigned long int up_count, down_count, zero_count;
400 up_count = enumerator_binomial(enumerator, lower_up, remaining_up);
401 down_count = enumerator_binomial(enumerator, lower_down, remaining_down);
402 if (checked_product(up_count, down_count, &zero_count) != 0) {
403 return -1;
404 }
405 if (rank >= zero_count) {
406 result |= 1UL << position;
407 rank -= zero_count;
408 if (*remaining == 0U) return -1;
409 (*remaining)--;
410 }
411 }
412 if (rank != 0UL || remaining_up != 0U || remaining_down != 0U) return -1;
413 *state = result;
414 return 0;
415}

426/* Lexicographic base-4 state order, with local digits 0 (empty), 1 (up),
427 * 2 (down). Count completions instead of constructing a Hubbard list and
428 * removing doubly occupied states. This also covers empty/full sectors. */
429static int tj_state_at(const struct SymmetryStateEnumerator *enumerator,
430 unsigned long rank, unsigned long *state)
431{
432 unsigned int site = enumerator->nsite;
433 unsigned int up = enumerator->nup, down = enumerator->ndown;
434 unsigned long result = 0UL;
435 while (site-- > 0U) {
436 unsigned long empty_count, up_count = 0UL;
437 if (tj_completion_count(enumerator, site, up, down, &empty_count)) return -1;
438 if (rank < empty_count) continue;
439 rank -= empty_count;
440 if (up && tj_completion_count(enumerator, site, up - 1U, down, &up_count)) return -1;
441 if (rank < up_count) {
442 result |= 1UL << (2U * site);
443 --up;
444 } else {
445 rank -= up_count;
446 if (!down) return -1;
447 result |= 1UL << (2U * site + 1U);
448 --down;
449 }
450 }
451 if (rank || up || down) return -1;
452 *state = result;
453 return 0;
454}

457static int tj_nconserved_state_at(const struct SymmetryStateEnumerator *enumerator,
458 unsigned long rank, unsigned long *state)
459{
460 unsigned int site = enumerator->nsite, n = enumerator->nup;
461 unsigned long result = 0UL;
462 while (site-- > 0U) {
463 unsigned long empty_count, spin_count;
464 if (tj_nconserved_completion_count(enumerator, site, n, &empty_count)) return -1;
465 if (rank < empty_count) continue;
466 rank -= empty_count;
467 if (!n || tj_nconserved_completion_count(enumerator, site, n - 1U, &spin_count))
468 return -1;
469 if (rank < spin_count) {
470 result |= 1UL << (2U * site);
471 } else {
472 rank -= spin_count;
473 result |= 1UL << (2U * site + 1U);
474 }
475 --n;
476 }
477 if (rank || n) return -1;
478 *state = result;
479 return 0;
480}

482int SymmetryStateEnumeratorStateAt(
483 const struct SymmetryStateEnumerator *enumerator,
484 unsigned long int raw_index,
485 unsigned long int *state)
486{
487 const unsigned int word_bits =
488 (unsigned int)(CHAR_BIT * sizeof(unsigned long int));
489 unsigned long int result;
490 int status;
491 if (enumerator == NULL || state == NULL ||
492 enumerator->raw_dim == 0UL ||
493 raw_index == 0UL || raw_index > enumerator->raw_dim) {
494 return -1;
495 }
496 if (IsSymmetryKondoModel(enumerator->model)) {
497 if (!enumerator->nsite || enumerator->nsite > word_bits / 2U ||
498 (enumerator->model == KondoGC && enumerator->nsite == word_bits / 2U) ||
499 enumerator->bit_count != 2U * enumerator->nsite) return -1;
500 return kondo_state_at(enumerator, raw_index, state);
501 }
502 if (enumerator->model == SpinGC) {
503 if (enumerator->nsite == 0U || enumerator->nsite >= word_bits ||
504 enumerator->bit_count != enumerator->nsite ||
505 enumerator->raw_dim != (1UL << enumerator->nsite)) {
506 return -1;
507 }
508 *state = raw_index - 1UL;
509 return 0;
510 }
511 if (IsSymmetryFermionGCModel(enumerator->model)) {
512 unsigned long int rank = raw_index - 1UL, parity = 0UL, rest;
513 unsigned int selected = enumerator->particle_parity != 0 ? 1U : 0U;
514 if (enumerator->nsite == 0U || enumerator->bit_count == 0U ||
515 enumerator->bit_count >= word_bits ||
516 enumerator->bit_count !=
517 fermion_gc_orbitals(enumerator->model, enumerator->nsite) ||
518 enumerator->particle_parity < -1 || enumerator->particle_parity > 1 ||
519 enumerator->raw_dim != (1UL << (enumerator->bit_count - selected))) {
520 return -1;
521 }
522 if (!selected) {
523 *state = rank;
524 return 0;
525 }
526 /* The lowest orbital completes the requested popcount parity, which
527 * keeps the selected states in increasing integer order. */
528 for (rest = rank; rest != 0UL; rest &= rest - 1UL) parity ^= 1UL;
529 *state = (rank << 1) | (parity ^ (enumerator->particle_parity < 0 ? 1UL : 0UL));
530 return 0;
531 }
532 if (enumerator->model == HubbardNConserved || enumerator->model == tJNConserved) {
533 if (enumerator->nsite == 0U ||
534 enumerator->nsite > word_bits / 2U ||
535 enumerator->bit_count != 2U * enumerator->nsite ||
536 enumerator->ndown != 0U ||
537 enumerator->nup > (enumerator->model == tJNConserved
538 ? enumerator->nsite : enumerator->bit_count)) {
539 return -1;
540 }
541 status = enumerator->model == tJNConserved
542 ? tj_nconserved_state_at(enumerator, raw_index - 1UL, &result)
543 : fixed_popcount_state_at(enumerator, raw_index - 1UL, &result);
544 } else if (enumerator->model == Hubbard || enumerator->model == tJ) {
545 if (enumerator->nsite == 0U ||
546 enumerator->nsite > word_bits / 2U ||
547 enumerator->bit_count != 2U * enumerator->nsite ||
548 enumerator->nup > enumerator->nsite ||
549 enumerator->ndown > enumerator->nsite ||
550 (enumerator->model == tJ && enumerator->nup > enumerator->nsite - enumerator->ndown)) {
551 return -1;
552 }
553 status = enumerator->model == tJ
554 ? tj_state_at(enumerator, raw_index - 1UL, &result)
555 : hubbard_state_at(enumerator, raw_index - 1UL, &result);
556 } else if (enumerator->model == Spin ||
557 enumerator->model == SpinlessFermion) {
558 if (enumerator->nsite == 0U ||
559 enumerator->nsite > word_bits ||
560 enumerator->bit_count != enumerator->nsite ||
561 enumerator->nup > enumerator->nsite) {
562 return -1;
563 }
564 status = fixed_popcount_state_at(enumerator, raw_index - 1UL, &result);
565 } else {
566 return -1;
567 }
568 if (status != 0) return -1;
569 *state = result;
570 return 0;
