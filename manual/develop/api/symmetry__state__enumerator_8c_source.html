9static int initialize_binomial_table(
10 struct SymmetryStateEnumerator *enumerator, unsigned int maximum)
11{
12 unsigned int n, k;
13 if (enumerator == NULL ||
14 maximum > HPHI_SYMMETRY_STATE_WORD_BITS) {
15 return -1;
16 }
17 memset(enumerator->binomial, 0, sizeof(enumerator->binomial));
18 enumerator->binomial[0][0] = 1UL;
19 for (n = 1U; n <= maximum; n++) {
20 enumerator->binomial[n][0] = 1UL;
21 enumerator->binomial[n][n] = 1UL;
22 for (k = 1U; k < n; k++) {
23 if (enumerator->binomial[n - 1U][k] >
24 ULONG_MAX - enumerator->binomial[n - 1U][k - 1U]) {
25 return -1;
26 }
27 enumerator->binomial[n][k] =
28 enumerator->binomial[n - 1U][k] +
29 enumerator->binomial[n - 1U][k - 1U];
30 }
31 }
32 return 0;
33}

70static int kondo_completions(const struct SymmetryStateEnumerator *e, unsigned int l,
71 unsigned int c, int up, int down, int nc,
72 unsigned long *count)
73{
74 unsigned long total = 0, a, b, term;
75 unsigned int u, exponent;
76 if (e->model == Kondo) {
77 if (up < 0 || down < 0 || (unsigned int)up > l + c || (unsigned int)down > l + c) {
78 *count = 0;
79 return 0;
80 }
81 for (u = 0; u <= l; ++u) {
82 a = kondo_binomial(e, c, up - (int)u);
83 b = kondo_binomial(e, c, down - (int)l + (int)u);
84 if (!a || !b)
85 continue;
86 if (dimension_product(a, b, &term) ||
87 dimension_product(term, e->binomial[l][u], &term) ||
88 term > (unsigned long)LONG_MAX || total > (unsigned long)LONG_MAX - term)
89 return -1;
90 total += term;
91 }
92 } else {
93 total = e->model == KondoGC ? 1UL : kondo_binomial(e, 2 * c, nc);
94 exponent = l + (e->model == KondoGC ? 2 * c : 0U);
95 for (u = 0; u < exponent; ++u)
96 if (dimension_product(total, 2UL, &total))
97 return -1;
98 }
99 *count = total;
100 return 0;
101}

136 struct SymmetryStateEnumerator *e)
137{

157static int kondo_state_at(const struct SymmetryStateEnumerator *e, unsigned long rank,
158 unsigned long *state)
159{
160 unsigned int site = e->nsite;
161 int up = (int)e->nup, down = (int)e->ndown, nc = (int)e->ncond;
162 unsigned long word = 0;
163 while (site-- > 0U) {
164 unsigned int digit, local = (unsigned int)((e->local_site_mask >> site) & 1UL);
165 for (digit = 0; digit < 4; ++digit) {
166 unsigned long completions;
167 int next_up = up - (int)(digit & 1U), next_down = down - (int)(digit >> 1);
168 int next_nc = nc - (local ? 0 : (int)((digit & 1U) + (digit >> 1)));
169 if (local && (digit == 0 || digit == 3))
170 continue;
171 if (kondo_completions(e, e->prefix_local[site], e->prefix_conduction[site], next_up,
172 next_down, next_nc, &completions))
173 return -1;
174 if (rank > completions) {
175 rank -= completions;
176 continue;
177 }
178 word |= (unsigned long)digit << (2 * site);
179 up = next_up;
180 down = next_down;
181 nc = next_nc;
182 break;
183 }
184 if (digit == 4)
185 return -1;
186 }
187 if (rank != 1 || (e->model == Kondo && (up || down)) ||
188 (e->model == KondoNConserved && nc))
189 return -1;
190 *state = word;
191 return 0;
192}

202 unsigned long int up_dim, down_dim, raw_dim;

204 memset(&initialized, 0, sizeof(initialized));

208 initialized.ndown = def->Ndown;

218 initialized.bit_count = def->Nsite;

222 *enumerator = initialized;
223 return 0;
224 }

229 def->Nsite > word_bits ||
230 def->Nup > def->Nsite ||
231 (def->iFlgSzConserved != TRUE &&

237 break;

240 initialized.nup = def->Ne;
241 initialized.ndown = 0U;
242 initialized.bit_count = def->Nsite;
243 break;

245 case tJ:

247 def->Nup > def->Nsite ||
248 def->Ndown > def->Nsite ||
249 def->Ne != def->Nup + def->Ndown ||

251 return -1;
252 }

254 break;
255 default:
256 return -1;
257 }
258 if (initialize_binomial_table(&initialized, initialized.nsite) != 0) return -1;
259 up_dim = enumerator_binomial(
260 &initialized, initialized.nsite, initialized.nup);
261 if (initialized.model == Hubbard || initialized.model == tJ) {
262 down_dim = enumerator_binomial(
263 &initialized, initialized.nsite - (initialized.model == tJ ? initialized.nup : 0U),
264 initialized.ndown);
265 if (checked_product(up_dim, down_dim, &raw_dim) != 0) return -1;
266 } else {
267 raw_dim = up_dim;
268 }
269 if (raw_dim == 0UL || raw_dim != expected_raw_dim) return -1;
270 initialized.raw_dim = raw_dim;
271 *enumerator = initialized;
272 return 0;

275static int fixed_popcount_state_at(
276 const struct SymmetryStateEnumerator *enumerator,
277 unsigned long int rank,
278 unsigned long int *state)
279{

281 unsigned int remaining = enumerator->nup;
282 unsigned long int result = 0UL;
283 while (position-- > 0U) {
284 unsigned long int zero_count =
285 enumerator_binomial(enumerator, position, remaining);
286 if (rank >= zero_count) {

288 rank -= zero_count;

291 }

298static int hubbard_state_at(
299 const struct SymmetryStateEnumerator *enumerator,
300 unsigned long int rank,
301 unsigned long int *state)
302{
303 unsigned int position = enumerator->bit_count;
304 unsigned int remaining_up = enumerator->nup;
305 unsigned int remaining_down = enumerator->ndown;
306 unsigned long int result = 0UL;
307 while (position-- > 0U) {
308 unsigned int lower_up = (position + 1U) / 2U;
309 unsigned int lower_down = position / 2U;
310 unsigned int *remaining =
311 (position % 2U == 0U) ? &remaining_up : &remaining_down;
312 unsigned long int up_count, down_count, zero_count;
313 up_count = enumerator_binomial(enumerator, lower_up, remaining_up);
314 down_count = enumerator_binomial(enumerator, lower_down, remaining_down);
315 if (checked_product(up_count, down_count, &zero_count) != 0) {
316 return -1;
317 }
318 if (rank >= zero_count) {
319 result |= 1UL << position;
320 rank -= zero_count;
321 if (*remaining == 0U) return -1;
322 (*remaining)--;
323 }
324 }
325 if (rank != 0UL || remaining_up != 0U || remaining_down != 0U) return -1;
326 *state = result;
327 return 0;
328}

339/* Lexicographic base-4 state order, with local digits 0 (empty), 1 (up),
340 * 2 (down). Count completions instead of constructing a Hubbard list and
341 * removing doubly occupied states. This also covers empty/full sectors. */
342static int tj_state_at(const struct SymmetryStateEnumerator *enumerator,
343 unsigned long rank, unsigned long *state)
344{
345 unsigned int site = enumerator->nsite;
346 unsigned int up = enumerator->nup, down = enumerator->ndown;
347 unsigned long result = 0UL;
348 while (site-- > 0U) {
349 unsigned long empty_count, up_count = 0UL;
350 if (tj_completion_count(enumerator, site, up, down, &empty_count)) return -1;
351 if (rank < empty_count) continue;
352 rank -= empty_count;
353 if (up && tj_completion_count(enumerator, site, up - 1U, down, &up_count)) return -1;
354 if (rank < up_count) {
355 result |= 1UL << (2U * site);
356 --up;
357 } else {
358 rank -= up_count;
359 if (!down) return -1;
360 result |= 1UL << (2U * site + 1U);
361 --down;
362 }
363 }
364 if (rank || up || down) return -1;
365 *state = result;
366 return 0;
367}

369int SymmetryStateEnumeratorStateAt(
370 const struct SymmetryStateEnumerator *enumerator,
371 unsigned long int raw_index,
372 unsigned long int *state)
373{
374 const unsigned int word_bits =
375 (unsigned int)(CHAR_BIT * sizeof(unsigned long int));
376 unsigned long int result;
377 int status;
378 if (enumerator == NULL || state == NULL ||
379 enumerator->raw_dim == 0UL ||
380 raw_index == 0UL || raw_index > enumerator->raw_dim) {
381 return -1;
382 }
383 if (IsSymmetryKondoModel(enumerator->model)) {
384 if (!enumerator->nsite || enumerator->nsite > word_bits / 2U ||
385 (enumerator->model == KondoGC && enumerator->nsite == word_bits / 2U) ||
386 enumerator->bit_count != 2U * enumerator->nsite) return -1;
387 return kondo_state_at(enumerator, raw_index, state);
388 }
389 if (enumerator->model == SpinGC) {
390 if (enumerator->nsite == 0U || enumerator->nsite >= word_bits ||
391 enumerator->bit_count != enumerator->nsite ||
392 enumerator->raw_dim != (1UL << enumerator->nsite)) {
393 return -1;
394 }
395 *state = raw_index - 1UL;
396 return 0;
397 }
398 if (enumerator->model == Hubbard || enumerator->model == tJ) {
399 if (enumerator->nsite == 0U ||
400 enumerator->nsite > word_bits / 2U ||
401 enumerator->bit_count != 2U * enumerator->nsite ||
402 enumerator->nup > enumerator->nsite ||
403 enumerator->ndown > enumerator->nsite ||
404 (enumerator->model == tJ && enumerator->nup > enumerator->nsite - enumerator->ndown)) {
405 return -1;
406 }
407 status = enumerator->model == tJ
408 ? tj_state_at(enumerator, raw_index - 1UL, &result)
409 : hubbard_state_at(enumerator, raw_index - 1UL, &result);
410 } else if (enumerator->model == Spin ||
411 enumerator->model == SpinlessFermion) {
412 if (enumerator->nsite == 0U ||
413 enumerator->nsite > word_bits ||
414 enumerator->bit_count != enumerator->nsite ||
415 enumerator->nup > enumerator->nsite) {
416 return -1;
417 }
418 status = fixed_popcount_state_at(enumerator, raw_index - 1UL, &result);
419 } else {
420 return -1;
421 }
422 if (status != 0) return -1;
423 *state = result;
424 return 0;
