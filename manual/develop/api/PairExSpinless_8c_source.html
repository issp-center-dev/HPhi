33int GetPairExcitedState_SpinlessFermion(
34 struct BindStruct *X,
35 double complex *tmp_v0,
36 double complex *tmp_v1
37) {
38 long int i;
39 int isite1, isite2;
40 complex double dam_pr, tmp_trans;
41 //Transfer
42 for (i = 0; i < X->Def.NPairExcitationOperator; i++) {
43 tmp_trans = X->Def.ParaPairExcitationOperator[i];
44
45 // Diagonal operator c^+_i c_i = n_i (or 1 - n_i for itype 0): it is diagonal
46 // in the occupation basis, so it needs no MPI communication regardless of
47 // whether site i is held by this process. Apply it directly; the hopping
48 // kernels return 0 for creation and annihilation on the same site.
49 if (X->Def.PairExcitationOperator[i][0] == X->Def.PairExcitationOperator[i][2]) {
50 int isite0 = X->Def.PairExcitationOperator[i][0]; // 0-based site
51 int itype = X->Def.PairExcitationOperator[i][4];
52 double complex coef = (itype == 0) ? -tmp_trans : tmp_trans;
53 long unsigned int jj, i_max = X->Large.i_max;
54 if (isite0 + 1 > X->Def.Nsite) {
55 // Inter-process site: occupation is fixed by this process rank.
56 int occ = (myrank >> (isite0 - X->Def.Nsite)) & 1;
57 double complex num1 = (itype == 0) ? (double)(1 - occ) : (double)occ;
58 for (jj = 1; jj <= i_max; jj++) tmp_v0[jj] += coef * num1 * tmp_v1[jj];
59 } else {
60 // Local site: occupation read from the basis bit pattern.
61 long unsigned int is = X->Def.Tpow[isite0];
62 int isGC = (X->Def.iCalcModel == SpinlessFermionGC);
63 for (jj = 1; jj <= i_max; jj++) {
64 long unsigned int bitrep = isGC ? (jj - 1) : list_1[jj];
65 long unsigned int ibit = (bitrep & is) / is;
66 double complex num1 = (itype == 0) ? (double)(1 - ibit) : (double)ibit;
67 tmp_v0[jj] += coef * num1 * tmp_v1[jj];
68 }
69 }
70 continue;
71 }
72
73 if (X->Def.PairExcitationOperator[i][0] + 1 > X->Def.Nsite &&
74 X->Def.PairExcitationOperator[i][2] + 1 > X->Def.Nsite) {
75 dam_pr = X_child_general_hopp_Spinless_MPIdouble(X->Def.PairExcitationOperator[i][0],
76 X->Def.PairExcitationOperator[i][2],
77 tmp_trans, X,
78 tmp_v0,
79 tmp_v1);
80 } else if (X->Def.PairExcitationOperator[i][2] + 1 > X->Def.Nsite) {
81 dam_pr = X_child_general_hopp_Spinless_MPIsingle(X->Def.PairExcitationOperator[i][0],
82 X->Def.PairExcitationOperator[i][2],
83 tmp_trans, X,
84 tmp_v0,
85 tmp_v1);
86 } else if (X->Def.PairExcitationOperator[i][0] + 1 > X->Def.Nsite) {
87 dam_pr = X_child_general_hopp_Spinless_MPIsingle(X->Def.PairExcitationOperator[i][2],
88 X->Def.PairExcitationOperator[i][0],
89 conj(tmp_trans), X,
90 tmp_v0,
91 tmp_v1);
92 } else {
93 isite1 = X->Def.PairExcitationOperator[i][0] + 1;
94 isite2 = X->Def.PairExcitationOperator[i][2] + 1;
95 if (child_general_hopp_GetInfo_Spinless(X, isite1, isite2) != 0) {
96 return -1;
97 }
98 dam_pr = child_general_hopp_Spinless(tmp_v0, tmp_v1, X, tmp_trans);
99 }
100 X->Large.prdct += dam_pr;
101 }
102 return 0;
103}
