27static int canonical_sector_exists(int64_t local_count,
28 int64_t conduction_count,
29 int64_t nup, int64_t ndown)
30{
31 int64_t lower = 0;
32 int64_t upper = local_count;
33 int64_t candidate;
34
35 candidate = nup - conduction_count;
36 if (candidate > lower) lower = candidate;
37 candidate = local_count - ndown;
38 if (candidate > lower) lower = candidate;
39
40 if (nup < upper) upper = nup;
41 candidate = local_count + conduction_count - ndown;
42 if (candidate < upper) upper = candidate;
43 return lower <= upper;
44}

252int SymmetryKondoPermutationSign(const struct DefineList *def,
253 const int *permutation, int *sign)
254{
255 unsigned int source;
256 unsigned int other;
257 unsigned long seen = 0UL;
258 unsigned int inversion_parity = 0U;
259 int result;
260
261 if (permutation == NULL || sign == NULL)
262 return kondo_error("missing Kondo permutation or sign output");
263 if (ValidateSymmetryKondoSpace(def) != 0) return -1;
264 for (source = 0U; source < def->Nsite; ++source) {
265 int target = permutation[source];
266 if (target < 0 || (unsigned int)target >= def->Nsite)
267 return kondo_error("Kondo site permutation is out of range");
268 if ((seen & (1UL << (unsigned int)target)) != 0UL)
269 return kondo_error("Kondo site permutation is not bijective");
270 seen |= 1UL << (unsigned int)target;
271 if (def->LocSpn[source] != def->LocSpn[target])
272 return kondo_error("Kondo site permutation changes site type");
273 }
274 for (source = 0U; source < def->Nsite; ++source) {
275 if (def->LocSpn[source] != LOCSPIN) continue;
276 for (other = source + 1U; other < def->Nsite; ++other) {
277 if (def->LocSpn[other] == LOCSPIN &&
278 permutation[source] > permutation[other])
279 inversion_parity ^= 1U;
280 }
281 }
282 result = inversion_parity == 0U ? 1 : -1;
283 *sign = result;
284 return 0;
285}
