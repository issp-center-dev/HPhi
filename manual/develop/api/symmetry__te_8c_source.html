118int ValidateSymmetryTESchedule(struct BindStruct *X, struct SymmetryTEHamiltonian *view)
119{
120 const struct DefineList *def = &view->base;
121 int failed = (def->NTETransferMax && def->NTEInterAllMax) ||
122 (def->NLaser && (def->NTETransferMax || def->NTEInterAllMax));
123 if (def->NLaser) {
124 failed |= def->NLaser != 9 || !def->ParaLaser;
125 if (!failed) {
126 for (int i = 0; i < 9; ++i) failed |= !isfinite(def->ParaLaser[i]);
127 double mode = def->ParaLaser[0], lx = def->ParaLaser[5], ly = def->ParaLaser[6];
128 failed |= mode < 0 || mode > 5 || floor(mode) != mode ||
129 lx < 1 || lx >= INT_MAX || floor(lx) != lx ||
130 ly < 1 || ly >= INT_MAX || floor(ly) != ly;
131 if (mode == 0 || mode == 4 || mode == 5) failed |= def->ParaLaser[3] <= 0;
132 if (mode == 5) failed |= def->ParaLaser[4] <= 0;
133 }
134 }
135 if (SumMPI_i(failed) != 0) {
136 fprintf(stdoutMPI, "Error: invalid or mixed symmetry TE driving definitions.\n");
137 return -1;
138 }
139 view->digests = calloc(def->Lanczos_max, sizeof(*view->digests));
140 if (SumMPI_i(view->digests == NULL) != 0) return -1;
141 for (unsigned int step = 0; step < def->Lanczos_max; ++step) {
142 double time = SymmetryTETime(def, step);
143 failed = !isfinite(time) ||
144 (step && (!isfinite(time - SymmetryTETime(def, step-1)) ||
145 time < SymmetryTETime(def, step-1)));
146 if (SumMPI_i(failed) != 0 || SelectSymmetryTEHamiltonian(X, view, step, time) != 0)
147 return -1;
148 failed = ValidateSymmetryHamiltonian(X) != 0 ||
149 ComputeSymmetryHamiltonianDigest(&X->Def, &view->digests[step]) != 0;
150 X->Def = *def;
151 if (SumMPI_i(failed) != 0) {
152 fprintf(stdoutMPI, "Error: symmetry TE preflight failed at step %u (time %.17g).\n", step, time);
153 return -1;
154 }
155 }
156 free_arrays(view);
157 fprintf(stdoutMPI, "Symmetry TE: all %u Hamiltonians passed preflight.\n", def->Lanczos_max);
