209static size_t TraceGbufBytes(long int nops, long int nc_uniform, size_t max_bytes) {
210 size_t sn, sc;
211 if (nops <= 0 || nc_uniform <= 0) return 0;
212 /* Rule out truncation on platforms where long int is wider than size_t,
213 via uintmax_t (do not assume a 64-bit long/size_t). */
214 if ((uintmax_t)nops > (uintmax_t)SIZE_MAX ||
215 (uintmax_t)nc_uniform > (uintmax_t)SIZE_MAX) return 0;
216 sn = (size_t)nops; sc = (size_t)nc_uniform;
217 if (sc > SIZE_MAX / sizeof(double complex)) return 0; /* ncols*16 overflow */
218 if (sn > SIZE_MAX / (sc * sizeof(double complex))) return 0; /* nops*(ncols*16) overflow */
219 if (sn * sc * sizeof(double complex) > max_bytes) return 0; /* cap exceeded */
220 return sn * sc * sizeof(double complex);
221}

223size_t TraceGbufMaxBytesFromEnv(void) {
224 const size_t default_bytes = ((size_t)1024) << 20; /* 1024 MiB */
225 const char *e = getenv("HPHI_TRACE_BUF_MAX_MB");
226 long int val;
227 char *endptr;
228
229 if (e == NULL || e[0] == '\0') return default_bytes;
230
231 errno = 0;
232 val = strtol(e, &endptr, 10);
233 if (errno != 0 || endptr == e || *endptr != '\0' || val < 1 || val > 1048576) {
234 fprintf(stderr,
235 " Warning: HPHI_TRACE_BUF_MAX_MB='%s' is not an integer in "
236 "[1,1048576]; using the default 1024 MiB.\n", e);
237 return default_bytes;
238 }
239 /* MiB->byte conversion, checked (value_mb > SIZE_MAX>>20 falls back). */
240 if ((uintmax_t)val > (uintmax_t)(SIZE_MAX >> 20)) {
241 fprintf(stderr,
242 " Warning: HPHI_TRACE_BUF_MAX_MB=%ld overflows size_t after "
243 "MiB->byte conversion; using the default 1024 MiB.\n", val);
244 return default_bytes;
245 }
246 return ((size_t)val) << 20;
247}

249/* Positive whitelist of the models whose per-basis fluctuation semantics the
250 energy trace kernel (trace_fill_diag/TraceEnergyEvalState in
251 expec_trace_ham.c) actually implements. Declared in expec_trace_ham.h; kept
252 HERE (next to its sole caller TraceBuildPlan) rather than in
253 expec_trace_ham.c so the minimal expec_trace_map_check unit test -- which
254 links expec_trace.c but not the heavy expec_trace_ham.c
255 (makeHam/expec_energy_flct) closure -- resolves it without pulling that
256 closure in.
257
258 INTENTIONALLY separate from trace_model_n_diag() (expec_trace_ham.c): that
259 function returns 0 for BOTH canonical Spin (supported) and
260 SpinlessFermion/unknown (unsupported), so n_diag cannot distinguish
261 supported from unsupported. This whitelist MUST stay in sync with
262 trace_model_n_diag: every model it handles as a real (non-default) case,
263 PLUS canonical Spin, is supported here. SpinlessFermion(GC) and any other
264 model fall through to default: return 0, forcing the ExpecMode-1 fallback
265 (their num/Sz semantics differ from canonical Spin's and are not
266 implemented in the kernel). */
267int TraceModelEnergySupported(int iCalcModel) {
268 switch (iCalcModel) {
269 case Hubbard:
270 /* HubbardNConserved/KondoNConserved/tJNConserved are whitelisted only
271 because sz() remaps each back to its base enum (Hubbard/Kondo/tJ) before
272 the FullDiag output() path runs -- see src/sz.c's "NConserved -> Normal"
273 block (~lines 615-626). The energy kernel (trace_fill_diag /
274 TraceEnergyEvalState in expec_trace_ham.c) and its output therefore only
275 ever see the base enum for these three. If a future change to sz() stops
276 remapping them, these three cases would strand (no base-enum handling
277 downstream) and must be revisited here in lockstep. */
278 case HubbardNConserved:
279 case Kondo:
280 case KondoNConserved:
281 case tJ:
282 case tJNConserved:
283 case HubbardGC:
284 case KondoGC:
285 case tJGC:
286 case SpinGC:
287 case Spin:
288 return 1;
289 default:
290 return 0; /* SpinlessFermion, SpinlessFermionGC, and any other model */
291 }

294void TraceBuildPlan(const struct BindStruct *X, long int nc_uniform,
295 size_t gbuf_max_bytes, TraceExecutionPlan *plan) {
296 int q, i;
297 int cap_q[TRACE_Q_NQUANT] = {0, 0, 0};
298
299 memset(plan, 0, sizeof(*plan));
300 plan->nc_uniform = nc_uniform;
301
302 if (X == NULL || X->Def.iExpecMode != EXPECMODE_TRACE) {
303 /* ExpecMode!=2: all-fallback plan (every field already zeroed above). */
304 return;
305 }
306
307 for (i = 0; i < N_TRACE_CAP; i++) {
308 if (kTraceCap[i].calc_model == X->Def.iCalcModel &&
309 kTraceCap[i].flg_general_spin == X->Def.iFlgGeneralSpin) {
310 cap_q[TRACE_Q_ONEBODY] = kTraceCap[i].q[TRACE_Q_ONEBODY];
311 cap_q[TRACE_Q_TWOBODY] = kTraceCap[i].q[TRACE_Q_TWOBODY];
312 break;
313 }
314 }
315
316 /* Energy/fluctuation slot (phase 3c): broader than the GF capability table
317 (it is NOT gated on cap_q[]), but still restricted to the models whose
318 per-basis fluctuation semantics the energy kernel actually implements.
319 TWO static demotions, checked in order:
320 (1) InputHam: re-running makeHam in the CSR collector would build a
321 matrix DIFFERENT from the one diagonalized, so the energy family
322 cannot use the trace kernel.
323 (2) unsupported model: TraceModelEnergySupported() whitelists exactly
324 the Hubbard/Kondo/tJ family (canonical + GC + N-conserved), SpinGC,
325 and canonical Spin. SpinlessFermion(GC) -- and any unknown model --
326 are NOT whitelisted: canonical/GC spinless fluctuation semantics
327 (num/Sz) differ from canonical Spin's and are not implemented in
328 TraceEnergyEvalState/trace_fill_diag, so activating the kernel would
329 write WRONG num/Sz fields. These models must fall back. A FullDiag
330 symmetry basis is unreachable (Task-1 audit (d)), so no further
331 config predicate is needed. Otherwise the slot is PROVISIONALLY
332 kernel=1; the runtime CSR collector plus TraceFinalizeEnergyPlan()
333 make the final decision. */
334 if (X->Def.iInputHam != 0) {
335 plan->demoted_input_ham[TRACE_Q_ENERGY] = 1;
336 plan->kernel[TRACE_Q_ENERGY] = 0;
337 } else if (!TraceModelEnergySupported(X->Def.iCalcModel)) {
338 plan->demoted_unsupported_model[TRACE_Q_ENERGY] = 1;
339 plan->kernel[TRACE_Q_ENERGY] = 0;
340 } else {
341 plan->kernel[TRACE_Q_ENERGY] = 1; /* provisional; finalize may demote */
342 }
343
344 for (q = 0; q < TRACE_Q_NQUANT; q++) {
345 long int nops;
346
347 if (q == TRACE_Q_ENERGY) continue; /* handled above; not a GF slot */
348
349 if (!cap_q[q]) {
350 /* unsupported model: stays kernel=0, demoted_memory=0 --
351 TraceReportPlan() reports this as "unsupported model". */
352 continue;
353 }
354
355 /* Shared-evaluator demotion (design-gap fix, see the field's doc comment
356 in expec_trace.h): expec_cisajscktaltdc() computes the two-body GF and
357 the ThreeBody/FourBody/SixBody GFs in ONE evaluator
358 (expec_cisajscktaltdc.c:115 runs it when ANY of the four counts is
359 positive), and the multibody GFs are always-fallback in 3b. If the
360 trace kernel took the two-body GF, the fallback loop would skip the
361 whole evaluator and silently drop the multibody outputs; calling the
362 evaluator anyway would double-write the two-body files. So whenever
363 any multibody GF is defined, the two-body GF falls back together with
364 them. Checked BEFORE the no-operators check and the memory gate so
365 gbuf_bytes[TWOBODY]/no_operators[TWOBODY]/demoted_memory[TWOBODY] all
366 stay 0 (the reasons are exclusive). ONEBODY is unaffected:
367 expec_cisajs() handles one-body only (no NTBody/NFBody/NSBody
368 reference exists in expec_cisajs.c). */
369 if (q == TRACE_Q_TWOBODY &&
370 (X->Def.NTBody > 0 || X->Def.NFBody > 0 || X->Def.NSBody > 0)) {
371 plan->demoted_shared_evaluator[q] = 1;
372 plan->kernel[q] = 0;
373 continue;
374 }
375
376 nops = (q == TRACE_Q_ONEBODY) ? (long int)X->Def.NCisAjt
377 : (long int)X->Def.NCisAjtCkuAlvDC;
378
379 /* No-operators demotion (final whole-branch review fix, see the field's
380 doc comment in expec_trace.h): a supported model with zero operators
381 of this kind defined has nothing to stream. Checked BEFORE the memory
382 gate -- TraceGbufBytes(0, ...) also returns 0, so without this
383 short-circuit the memory gate would fire instead and
384 TraceReportPlan() would misreport "result buffer would exceed
385 HPHI_TRACE_BUF_MAX_MB" for a quantity that never had a buffer to size
386 in the first place. */
387 if (nops <= 0) {
388 plan->no_operators[q] = 1;
389 plan->kernel[q] = 0;
390 continue;
391 }
392
393 plan->gbuf_bytes[q] = TraceGbufBytes(nops, nc_uniform, gbuf_max_bytes);
394 if (plan->gbuf_bytes[q] == 0) {
395 plan->demoted_memory[q] = 1;
396 plan->kernel[q] = 0;
397 } else {
398 plan->kernel[q] = 1;
399 }
400 }

403void TraceReportPlan(const TraceExecutionPlan *plan, FILE *fp) {
404 static const char *kQuantityName[TRACE_Q_NQUANT] = { "one-body", "two-body",
405 "energy" };
406 int q;
407
408 for (q = 0; q < TRACE_Q_NQUANT; q++) {
409 if (q == TRACE_Q_ENERGY) {
410 /* The energy/fluctuation family has its own verbatim INFO wording
411 (design spec 3c). kernel==1: trace kernel; else the ExpecMode-1
412 fallback with a reason -- InputHam (static, from TraceBuildPlan) or
413 the HPHI_TRACE_BUF_MAX_MB memory gate (from TraceFinalizeEnergyPlan).
414 These substrings are the single source for the Task-6 equivalence
415 script's greps -- keep them byte-exact. */
416 if (plan->kernel[q]) {
417 fprintf(fp,
418 " INFO: ExpecMode 2: the energy/fluctuation family uses the "
419 "trace kernel.\n");
420 } else if (plan->demoted_input_ham[q]) {
421 fprintf(fp,
422 " INFO: ExpecMode 2: the energy/fluctuation family uses the "
423 "ExpecMode-1 fallback (the Hamiltonian was read from "
424 "InputHam).\n");
425 } else if (plan->demoted_unsupported_model[q]) {
426 fprintf(fp,
427 " INFO: ExpecMode 2: the energy/fluctuation family uses the "
428 "ExpecMode-1 fallback (unsupported model).\n");
429 } else {
430 fprintf(fp,
431 " INFO: ExpecMode 2: the energy/fluctuation family uses the "
432 "ExpecMode-1 fallback (the Hamiltonian buffer would exceed "
433 "HPHI_TRACE_BUF_MAX_MB).\n");
434 }
435 continue;
436 }
437 if (plan->kernel[q]) {
438 fprintf(fp, " INFO: ExpecMode 2: %s Green functions use the trace kernel.\n",
439 kQuantityName[q]);
440 } else if (plan->demoted_shared_evaluator[q]) {
441 fprintf(fp,
442 " INFO: ExpecMode 2: %s Green functions use the ExpecMode-1 "
443 "fallback (they share their evaluator with three-/four-/six-body "
444 "Green functions).\n",
445 kQuantityName[q]);
446 } else if (plan->no_operators[q]) {
447 fprintf(fp,
448 " INFO: ExpecMode 2: %s Green functions use the ExpecMode-1 "
449 "fallback (no operators of this kind are defined).\n",
450 kQuantityName[q]);
451 } else if (plan->demoted_memory[q]) {
452 fprintf(fp,
453 " INFO: ExpecMode 2: %s Green functions use the ExpecMode-1 "
454 "fallback (result buffer would exceed HPHI_TRACE_BUF_MAX_MB).\n",
455 kQuantityName[q]);
456 } else {
457 fprintf(fp,
458 " INFO: ExpecMode 2: %s Green functions use the ExpecMode-1 "
459 "fallback (unsupported model).\n",
460 kQuantityName[q]);
461 }
462 }
463 fprintf(fp,
464 " INFO: ExpecMode 2: S2, NBodyG, and AnomalousG always use the "
465 "ExpecMode-1 path in this version.\n");
466}

595static int expec_trace_twobody_output(struct BindStruct *X, long int jb, long int je,
596 long int ncols, const double complex *gbuf) {
597 long int nops = (long int)X->Def.NCisAjtCkuAlvDC;
598 long int n, p;
599 int model = X->Def.iCalcModel;
600 int is_gc = (model == HubbardGC || model == tJGC || model == KondoGC);
601 int is_hubbard_family = (model == Hubbard || model == HubbardGC || model == tJ ||
602 model == tJGC || model == Kondo || model == KondoGC);
603
604 for (n = jb; n <= je; n++) {
605 FILE *fp = NULL;
606 char sdt[D_FileNameMax];
607
608 X->Phys.eigen_num = (int)(n - 1); /* 0-based, same convention as Mode 1 */
609
610 if (GreenOutputKindUsesAggregate(X, GreenOutputTwoBody)) {
611 if (GreenOutputOpenAggregate(X, GreenOutputTwoBody, &fp) != 0) return -1;
612 } else {
613 sprintf(sdt, cFileName2BGreen_FullDiag, X->Def.CDataFileHead, X->Phys.eigen_num);
614 if (childfopenMPI(sdt, "w", &fp) != 0) return -1;
615 }
616
617 for (p = 0; p < nops; p++) {
618 long unsigned int i1 = (long unsigned int)X->Def.CisAjtCkuAlvDC[p][0];
619 long unsigned int s1 = (long unsigned int)X->Def.CisAjtCkuAlvDC[p][1];
620 long unsigned int i2 = (long unsigned int)X->Def.CisAjtCkuAlvDC[p][2];
621 long unsigned int s2 = (long unsigned int)X->Def.CisAjtCkuAlvDC[p][3];
622 long unsigned int i3 = (long unsigned int)X->Def.CisAjtCkuAlvDC[p][4];
623 long unsigned int s3 = (long unsigned int)X->Def.CisAjtCkuAlvDC[p][5];
624 long unsigned int i4 = (long unsigned int)X->Def.CisAjtCkuAlvDC[p][6];
625 long unsigned int s4 = (long unsigned int)X->Def.CisAjtCkuAlvDC[p][7];
626 double complex val = gbuf[p * ncols + (n - jb)];
627 const char *fmt;
628
629 if (is_hubbard_family) {
630 fmt = (!is_gc && X->Def.iFlgSzConserved == TRUE && (s1 + s3 != s2 + s4))

659 /* Task 8: reset this rank's phase-timing accumulators at the top of every

662 an earlier call. See g_trace_timings' doc comment above. */
663 memset(g_trace_timings, 0, sizeof(g_trace_timings));

722int TraceStreamOneBody(struct BindStruct *X, const double complex *panel,
723 long int jb, long int je, long int NN,
724 long int ncols, double complex *gbuf) {
725 long int nops = (long int)X->Def.NCisAjt;
726 long int p, n, k;
727 double t0;
728
729 /* The panel's stride NN must equal the Hilbert-space dimension every
730 TraceMap is built over -- see src/phys_distributed_local.c:70-81's
731 `v0[j + 1] = panel[(n - jb) * NN + j]` (NN doubles as both the
732 FullDiag state count and, here, the per-state vector length; the two
733 coincide because neig == idim_max in the replicated FullDiag driver). */
734 assert((long int)X->Check.idim_max == NN);
735
736 for (p = 0; p < nops; p++) {
737 TraceMap map;
738
739 t0 = TraceNowSeconds();
740 if (TraceMapExtractOneBody(X, (int)p, &map) != 0) return -1;
741 g_trace_timings[TRACE_Q_ONEBODY][0] += TraceNowSeconds() - t0;
742
743 t0 = TraceNowSeconds();

749 if (map.kprime[k] >= 0)
750 acc += conj(z[map.kprime[k]]) * map.amp[k] * z[k];
751 }
752 gbuf[p * ncols + (n - jb)] = acc;
753 }
754 g_trace_timings[TRACE_Q_ONEBODY][1] += TraceNowSeconds() - t0;
755
756 TraceMapFree(&map); /* only one TraceMap alive at a time (spec Sec.3.1) */
757 }
758 return 0;
759}

798/* Basis-mapping extraction (phase 3b Task 2). */
799/* */
800/* These drivers replicate the LOCAL (intra-process) dispatch of */
801/* expec_cisajs.c / expec_cisajscktaltdc.c with X->Large.mode=M_CORR */
802/* and, per operator pair, stream the co-located `*_TraceProbe` */
803/* adapters over k=1..idim_max to fill a TraceMap. In the replicated */
804/* FullDiag layout the trace kernel runs in (iFlgScaLAPACK, no MPI */
805/* site decomposition) every operator site is intra-process, so the */
806/* inter-PE branches of the Mode-1 dispatch are unreachable and are */
807/* intentionally not mirrored here. */
808/* */
809/* Purity: X->Large is snapshot on entry and restored on exit, so the */
810/* GetInfo scratch writes leave X byte-identical (see the purity */
811/* unit test). The only writes any reachable helper makes are to */
812/* X->Large; list_1/list_2_* and the Def/Tpow tables are read-only. */

816static int trace_map_alloc(TraceMap *map, long int n) {
817 long int k;
818 map->n = n;
819 map->is_diagonal = 0;
820 /* Representability guards, same rigor as TraceGbufBytes(): reject a
821 non-positive or size_t-overflowing dimension before multiplying. */
822 if (n <= 0 ||
823 (uintmax_t)n > (uintmax_t)SIZE_MAX ||
824 (size_t)n > SIZE_MAX / sizeof(double complex) ||
825 (size_t)n > SIZE_MAX / sizeof(long int)) {
826 map->kprime = NULL;
827 map->amp = NULL;
828 return -1;
829 }
830 map->kprime = (long int *)malloc(sizeof(long int) * (size_t)n);
831 map->amp = (double complex *)malloc(sizeof(double complex) * (size_t)n);
832 if (map->kprime == NULL || map->amp == NULL) {
833 TraceMapFree(map);
834 return -1;
835 }
836 for (k = 0; k < n; k++) {
837 map->kprime[k] = -1;
838 map->amp[k] = 0.0;
839 }
840 return 0;
841}

908 case tJ:
909 case tJGC:
910 case Kondo:

918 break;

934 }

936 }

945 } /* else off-diagonal spin hopping -> empty map (GF = 0) */
946 break;
947 }
948 case SpinGC: {
949 if (X->Def.iFlgGeneralSpin != FALSE) { rc = -1; break; }
950 if (org_isite1 == org_isite2) {
951 long unsigned int isite1 = X->Def.Tpow[org_isite1 - 1];
952 if (org_sigma1 == org_sigma2) {

961 }
962 } /* else empty map (GF = 0) */
963 break;
964 }
965 default:
966 rc = -1;
967 break;
968 }
969
970 if (rc == 0) trace_map_assert_range(map);
971 X->Large = saved; /* purity restore */
972 if (rc != 0) TraceMapFree(map);
973 return rc;
974}
