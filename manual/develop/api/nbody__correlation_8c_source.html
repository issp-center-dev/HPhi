21static int parse_unsigned_token(const char **pp, unsigned int *value)
22{
23 char *end = NULL;
24 unsigned long v;
25 const char *p = *pp;
26 while (isspace((unsigned char)*p)) p++;
27 if (*p == '\0' || *p == '-' || *p == '+') return -1;
28 errno = 0;
29 v = strtoul(p, &end, 10);
30 if (errno != 0 || end == p || v > UINT_MAX) return -1;
31 *value = (unsigned int)v;
32 *pp = end;
33 return 0;
34}

51int ParseNBodyGLine(
52 const char *line,
53 unsigned int *N,
54 int **factors
55) {
56 const char *p = line;
57 unsigned int n;
58 unsigned int k;
59 int *buf;
60 size_t nints;
61
62 if (parse_unsigned_token(&p, &n) != 0 || n == 0) {
63 fprintf(stdoutMPI, "Error: NBodyG line has an invalid factor count.\n");
64 return -1;
65 }
66 if (n > UINT_MAX / 4) {
67 fprintf(stdoutMPI, "Error: NBodyG line is too large.\n");
68 return -1;
69 }
70#if SIZE_MAX < UINT_MAX
71 if ((size_t)n > SIZE_MAX / 4 / sizeof(int)) {
72 fprintf(stdoutMPI, "Error: NBodyG line is too large.\n");
73 return -1;
74 }
75#endif
76 nints = (size_t)4 * n;
77 buf = (int *)malloc(nints * sizeof(int));
78 if (buf == NULL) {
79 fprintf(stdoutMPI, "Error: Failed to allocate NBodyG parser buffer.\n");
80 return -1;
81 }
82 for (k = 0; k < 4 * n; k++) {
83 if (parse_int_token(&p, &buf[k]) != 0) {
84 fprintf(stdoutMPI, "Error: NBodyG line has too few integer fields.\n");
85 free(buf);
86 return -1;
87 }
88 }
89 while (isspace((unsigned char)*p)) p++;
90 if (*p != '\0') {
91 fprintf(stdoutMPI, "Error: NBodyG line has extra fields.\n");
92 free(buf);
93 return -1;
94 }
95
96 *N = n;
97 *factors = buf;
98 return 0;
99}

101int ValidateNBodyGScope(const struct DefineList *D)
102{
103 unsigned int t, k;
104 if (D->NNBodyG == 0) return 0;
105 if (D->iCalcModel != SpinGC) {
106 fprintf(stdoutMPI, "Error: NBodyG is currently supported only for SpinGC.\n");
107 return -1;
108 }
109 if (D->iFlgGeneralSpin != FALSE) {
110 fprintf(stdoutMPI, "Error: NBodyG is currently supported only for spin-1/2 SpinGC.\n");
111 return -1;
112 }
113 for (t = 0; t < D->NNBodyG; t++) {
114 const unsigned int off = D->NBodyG_Offset[t];
115 for (k = 0; k < D->NBodyG_N[t]; k++) {
116 const int *f = D->NBodyG_Factors[off + k];
117 if (f[0] < 0 || f[0] >= (int)D->Nsite || f[2] < 0 || f[2] >= (int)D->Nsite) {
118 fprintf(stdoutMPI, "Error: Site index of NBodyG is incorrect.\n");
119 return -1;
120 }
121 if (f[0] != f[2]) {
122 fprintf(stdoutMPI, "Error: NBodyG currently requires site_out == site_in for every factor.\n");
123 return -1;
124 }
125 if (f[1] < 0 || f[1] > 1 || f[3] < 0 || f[3] > 1) {
126 fprintf(stdoutMPI, "Error: Spin index of NBodyG is incorrect.\n");
127 return -1;
128 }
129 }
130 }
131 return 0;
132}

143static void sort_canonical(int *sites, int *outs, int *ins, unsigned int n)
144{
145 unsigned int i, j;
146 for (i = 0; i < n; i++) {
147 for (j = i + 1; j < n; j++) {
148 if (sites[j] < sites[i]) {
149 int ts = sites[i], to = outs[i], ti = ins[i];
150 sites[i] = sites[j];
151 outs[i] = outs[j];
152 ins[i] = ins[j];
153 sites[j] = ts;
154 outs[j] = to;
155 ins[j] = ti;
156 }
157 }
158 }

161int NormalizeNBodyGTerms(struct DefineList *D)
162{
163 unsigned int t, k;
164 unsigned int total = 0;
165
166 D->NBodyG_TotalCanonicalFactors = 0;
167 for (t = 0; t < D->NNBodyG; t++) {
168 const unsigned int nraw = D->NBodyG_N[t];
169 const unsigned int off = D->NBodyG_Offset[t];
170 int *sites = (int *)malloc(nraw * sizeof(int));
171 int *outs = (int *)malloc(nraw * sizeof(int));
172 int *ins = (int *)malloc(nraw * sizeof(int));
173 unsigned int ncanon = 0;
174 int is_zero = FALSE;
175 if (sites == NULL || outs == NULL || ins == NULL) {
176 fprintf(stdoutMPI, "Error: Failed to allocate NBodyG normalization buffer.\n");
177 free(sites);
178 free(outs);
179 free(ins);
180 return -1;
181 }
182
183 for (k = 0; k < nraw; k++) {
184 const int *f = D->NBodyG_Factors[off + k];
185 const int site = f[0];
186 const int spin_out = f[1];
187 const int spin_in = f[3];
188 const int pos = find_site(sites, ncanon, site);
189 if (pos < 0) {
190 sites[ncanon] = site;
191 outs[ncanon] = spin_out;
192 ins[ncanon] = spin_in;
193 ncanon++;
194 }
195 else {
196 if (ins[pos] != spin_out) {
197 is_zero = TRUE;
198 break;
199 }
200 ins[pos] = spin_in;
201 }
202 }
203
204 D->NBodyG_IsZero[t] = is_zero;
205 D->NBodyG_CanonicalOffset[t] = total;
206 if (is_zero == TRUE) {
207 D->NBodyG_CanonicalN[t] = 0;
208 free(sites);
209 free(outs);
210 free(ins);
211 continue;
212 }
213
214 sort_canonical(sites, outs, ins, ncanon);
215 D->NBodyG_CanonicalN[t] = ncanon;
216 for (k = 0; k < ncanon; k++) {
217 D->NBodyG_CanonicalFactors[total + k][0] = sites[k];
218 D->NBodyG_CanonicalFactors[total + k][1] = outs[k];
219 D->NBodyG_CanonicalFactors[total + k][2] = sites[k];
220 D->NBodyG_CanonicalFactors[total + k][3] = ins[k];
221 }
222 total += ncanon;
223 free(sites);
224 free(outs);
225 free(ins);
226 }
227 D->NBodyG_TotalCanonicalFactors = total;
228 return 0;
229}

231static int apply_nbodyg_spingc(
232 const struct BindStruct *X,
233 unsigned int term,
234 unsigned long int local_in,
235 int rank_in,
236 unsigned long int *local_out,
237 int *rank_out
238) {
239 const struct DefineList *D = &X->Def;
240 unsigned int k;
241 unsigned long int lo = local_in;
242 int ro = rank_in;
243 const unsigned int n = D->NBodyG_CanonicalN[term];
244 const unsigned int off = D->NBodyG_CanonicalOffset[term];
245
246 for (k = 0; k < n; k++) {
247 const int *f = D->NBodyG_CanonicalFactors[off + k];
248 const unsigned int site = (unsigned int)f[0];
249 const int spin_out = f[1];
250 const int spin_in = f[3];
251 const unsigned long int mask = D->Tpow[site];
252 int bit;
253
254 if (site < D->Nsite) {
255 bit = (lo & mask) ? 1 : 0;
256 if (bit != spin_in) return 0;
257 if (spin_out != spin_in) {
258 if (spin_out == 1) lo |= mask;
259 else lo &= ~mask;
260 }
261 }
262 else {
263 bit = (ro & (int)mask) ? 1 : 0;
264 if (bit != spin_in) return 0;
265 if (spin_out != spin_in) {
266 if (spin_out == 1) ro |= (int)mask;
267 else ro &= ~((int)mask);
268 }
269 }
270 }
271
272 *local_out = lo;
273 *rank_out = ro;
274 return 1;
275}

277static int nbodyg_rank_flip_mask(const struct BindStruct *X, unsigned int term, int *mask)
278{
279 unsigned int k;
280 int m = 0;
281 const unsigned int n = X->Def.NBodyG_CanonicalN[term];
282 const unsigned int off = X->Def.NBodyG_CanonicalOffset[term];
283 for (k = 0; k < n; k++) {
284 const int *f = X->Def.NBodyG_CanonicalFactors[off + k];
285 const unsigned int site = (unsigned int)f[0];
286 if (site >= X->Def.Nsite && f[1] != f[3]) {
287 m ^= (int)X->Def.Tpow[site];
288 }
289 }
290 *mask = m;
291 return 0;
292}

294static double complex expec_nbodyg_term_to_rank(
295 struct BindStruct *X,
296 unsigned int term,
297 const double complex *src_vec,
298 const double complex *bra_vec,
299 int rank_in
300) {
301 unsigned long int j;
302 double complex dam_pr = 0.0;
303 const unsigned long int i_max = X->Check.idim_max;
304
305#pragma omp parallel for default(none) reduction(+:dam_pr) \
306 shared(X, src_vec, bra_vec) firstprivate(i_max, term, rank_in, myrank) private(j)
307 for (j = 1; j <= i_max; j++) {
308 unsigned long int local_out = 0;
309 int rank_out = 0;
310 int ret = apply_nbodyg_spingc(X, term, j - 1, rank_in, &local_out, &rank_out);
311 if (ret == 1 && rank_out == myrank) {
312 dam_pr += conj(bra_vec[local_out + 1]) * src_vec[j];
313 }
314 }
315 return dam_pr;
316}

318static double complex calc_nbodyg_term(struct BindStruct *X, unsigned int term, double complex *vec)
319{
320 int mask = 0;
321 int origin;
322 double complex dam_pr = 0.0;
323
324 nbodyg_rank_flip_mask(X, term, &mask);
325 origin = myrank ^ mask;
326 if (origin == myrank) {
327 dam_pr = expec_nbodyg_term_to_rank(X, term, vec, vec, myrank);
328 return SumMPI_dc(dam_pr);
329 }
330
331#ifdef MPI
332 {
333 MPI_Status statusMPI;
334 int ierr = MPI_Sendrecv(vec, X->Check.idim_max + 1, MPI_DOUBLE_COMPLEX, origin, 0,
335 v1buf, X->Check.idim_max + 1, MPI_DOUBLE_COMPLEX, origin, 0,
336 MPI_COMM_WORLD, &statusMPI);
337 if (ierr != 0) exitMPI(-1);
338 dam_pr = expec_nbodyg_term_to_rank(X, term, v1buf, vec, origin);
339 }
340#else
341 fprintf(stdoutMPI, "Error: NBodyG reached an MPI-only rank flip path without MPI.\n");
342 return 0.0;
343#endif
344 return SumMPI_dc(dam_pr);
345}

361static int get_nbodyg_filename(struct BindStruct *X, char *sdt)
362{
363 switch (X->Def.iCalcType) {
364 case Lanczos:
365 sprintf(sdt, cFileNameNBodyG_Lanczos, X->Def.CDataFileHead);
366 break;
367 case TPQCalc:
368 case cTPQ:
369 sprintf(sdt, cFileNameNBodyG_TPQ, X->Def.CDataFileHead, X->Def.irand, X->Def.istep);
370 break;
371 case TimeEvolution:
372 sprintf(sdt, cFileNameNBodyG_TE, X->Def.CDataFileHead, X->Def.istep);
373 break;

377 break;
378 default:
379 fprintf(stdoutMPI, "Error: NBodyG does not support this calculation type.\n");
380 return -1;
381 }
382 return 0;
383}

385int expec_nbodyg(struct BindStruct *X, double complex *vec)
386{
387 FILE *fp;
388 char sdt[D_FileNameMax];
389 unsigned int t;
390
391 if (X->Def.NNBodyG < 1) return 0;
392 if (X->Def.iCalcModel != SpinGC || X->Def.iFlgGeneralSpin != FALSE) {
393 fprintf(stdoutMPI, "Error: NBodyG is currently supported only for spin-1/2 SpinGC.\n");
394 return -1;
395 }
396 if (get_nbodyg_filename(X, sdt) != 0) return -1;
397 if (childfopenMPI(sdt, "w", &fp) != 0) return -1;
398
399 for (t = 0; t < X->Def.NNBodyG; t++) {
400 double complex value = 0.0;
401 if (X->Def.NBodyG_IsZero[t] == FALSE) {
402 value = calc_nbodyg_term(X, t, vec);
403 }
404 write_nbodyg_line(fp, &X->Def, t, value);
405 }
406
407 fclose(fp);
408 return 0;
409}
