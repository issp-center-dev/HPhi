92int TransferWithPeierls(struct BindStruct *X, const double time) {
93 int i;
94 int ri_x, rj_x;
95 int ri_y, rj_y;
96 double complex dir;
97 const int Mode = (int) (X->Def.ParaLaser[0]);
98 const double Avp = X->Def.ParaLaser[1];
99 const double omega = X->Def.ParaLaser[2];
100 const double time_d = X->Def.ParaLaser[3];
101 const double time_c = X->Def.ParaLaser[4];
102 const int Lx = (int) (X->Def.ParaLaser[5]);
103 const int Ly = (int) (X->Def.ParaLaser[6]);
104 const double dirX = X->Def.ParaLaser[7];
105 const double dirY = X->Def.ParaLaser[8];
106 const double dt = time - time_c;
107 const double dt2 = time - (time_c + time_d);
108 const double td = time_c / 3.0;
109 double VecPot = 0.0;
110
111 if (Mode == 0) {//Gaussian Wave
112 VecPot = Avp * cos(omega * dt) * exp(-dt * dt / (2.0 * time_d * time_d));
113 } else if (Mode == 1) {//Cosine Wave
114 VecPot = Avp * sin(omega * dt);
115 } else if (Mode == 2) {//DC Limit
116 VecPot = Avp * dt;
117 } else if (Mode == 3) {//Pulse
118 VecPot = Avp * exp(-time_d * dt);
119 } else if (Mode == 4) {//Linear
120 if (dt <= 0.0) {
121 VecPot = 0.0;
122 } else if (dt < time_d) {
123 VecPot = Avp * cos(omega * dt) * (dt / time_d);
124 } else {
125 VecPot = Avp * cos(omega * dt);
126 }
127 } else if (Mode == 5) {//Linear
128 if (time <= 0.0) {
129 VecPot = 0.0;
130 } else if (time < time_c) {
131 VecPot = Avp * cos(omega * dt) * exp(-dt * dt / (2.0 * td * td));
132 } else if (time < time_c + time_d) {
133 VecPot = Avp * cos(omega * dt);
134 } else {
135 VecPot = Avp * cos(omega * dt) * exp(-dt2 * dt2 / (2.0 * td * td));
136 }
137 }
138
139 for (i = 0; i < X->Def.EDNTransfer; i++) {
140 ri_x = X->Def.EDGeneralTransfer[i][0] % Lx;
141 rj_x = X->Def.EDGeneralTransfer[i][2] % Lx;
142 ri_y = X->Def.EDGeneralTransfer[i][0] / Lx;
143 rj_y = X->Def.EDGeneralTransfer[i][2] / Lx;
144 if (ri_x - rj_x > 1) {
145 rj_x += Lx;
146 } else if (ri_x - rj_x < -1) {
147 rj_x -= Lx;
148 }
149 if (ri_y - rj_y > 1) {
150 rj_y += Ly;
151 } else if (ri_y - rj_y < -1) {
152 rj_y -= Ly;
153 }
154 dir = dirX * (ri_x - rj_x) + dirY * (ri_y - rj_y);
155
156 X->Def.EDParaGeneralTransfer[i] = X->Def.ParaGeneralTransfer[i] * cexp(-I * VecPot * dir);
157 }
158
159 return 0;
160}

172int TransferForQuench(struct BindStruct *X, const double time) {
173 int i;
174 const int Mode = (int) (X->Def.ParaLaser[0]);
175 const double Avp = X->Def.ParaLaser[1];
176 const double time_d = X->Def.ParaLaser[3];
177 const double time_c = X->Def.ParaLaser[4];
178 const double dt = time - time_c;
179 double Bessel = 0.0;
180
181 if (Mode == 0) {//Gaussian Wave
182 if (dt <= 0.0) {
183 Bessel = 0.0;
184 } else if (dt < time_d) {
185 Bessel = j0(Avp * dt / time_d);
186 } else {
187 Bessel = j0(Avp);
188 }
189 }
190
191 for (i = 0; i < X->Def.EDNTransfer; i++) {
192 X->Def.EDParaGeneralTransfer[i] = X->Def.ParaGeneralTransfer[i] * Bessel;
193 }
194
195 return 0;
196}
