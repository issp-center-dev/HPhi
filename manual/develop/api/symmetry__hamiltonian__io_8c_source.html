38/* Collective: 0 when every rank holds the root's words. */
39static int agree_with_root(const uint64_t *local, int count)
40{
41 uint64_t root[IDENTITY_WORDS];
42 if (count < 1 || count > IDENTITY_WORDS) return -1;
43 memcpy(root, local, (size_t)count * sizeof(*root));
44#ifdef MPI
45 if (MPI_Bcast(root, count, MPI_UINT64_T, 0, MPI_COMM_WORLD) != MPI_SUCCESS) return -1;
46#endif
47 return SumMPI_i(memcmp(root, local, (size_t)count * sizeof(*root)) != 0) != 0 ? -1 : 0;
48}

50int BuildSymmetryHamiltonianIdentity(const struct BindStruct *X,
51 struct SymmetryHamiltonianIdentity *identity)
52{
53 struct SymmetryHamiltonianIdentity local;
54 struct SymmetrySectorDigest sector;
55 uint64_t words[IDENTITY_WORDS];
56 int invalid = identity == NULL || X == NULL || X->Sym == NULL ||
57 !X->Def.iFlgSymmetryBasis || sizeof(double) != 8 ||
58 DBL_MANT_DIG != 53 || DBL_MAX_EXP != 1024;
59 if (!invalid)
60 invalid = X->Sym->enabled != TRUE ||
61 X->Sym->basis_layout != SYMMETRY_BASIS_REPLICATED ||
62 X->Sym->basis == NULL || X->Sym->dim == 0 || X->Sym->dim > INT_MAX;
63 if (SumMPI_i(invalid) != 0) {
64 fprintf(stdoutMPI, "Error: sector Hamiltonian identity requires a replicated, nonempty TransSym basis with binary64 scalars.\n");
65 return -1;
66 }
67 if (ComputeSymmetrySectorDigest(X->Sym, &sector) != 0) {
68 fprintf(stdoutMPI, "Error: sector Hamiltonian identity: sector digest failed.\n");
69 return -1;
70 }
71 memset(&local, 0, sizeof(local));
72 local.model = (uint64_t)X->Def.iCalcModel;
73 local.nsite = X->Def.Nsite;
74 if (X->Def.iCalcModel != SpinGC) {
75 local.nup = X->Def.Nup;
76 local.ndown = X->Def.Ndown;
77 local.ne = X->Def.Ne;
78 }
79 local.phase = 1;
80 if (IsSymmetryKondoModel(X->Def.iCalcModel)) {
81 struct SymmetryKondoIdentity kondo;
82 if (GetSymmetryKondoIdentity(&X->Def, &kondo) != 0) {
83 invalid = 1;
84 } else {
85 local.nup = kondo.nup;
86 local.ndown = kondo.ndown;
87 local.ne = kondo.ne;
88 local.local_site_mask = kondo.local_site_mask;
89 local.fixed_flags = kondo.fixed_flags;
90 local.phase = kondo.phase;
91 }
92 }
93 local.full_dim = X->Sym->full_dim;
94 local.sector_dim = X->Sym->dim;
95 if (!invalid && ComputeSymmetryGroupDigest(&X->Def, &local.group_digest) != 0) invalid = 1;
96 local.sector_count = sector.count;
97 local.sector_xor = sector.xor_hash;
98 local.sector_sum = sector.sum_hash;
99 /* One global stream, the word encoding of the checkpoint ORDER field. */
100 local.basis_order = FNV_OFFSET;
101 for (unsigned long i = 1; !invalid && i <= X->Sym->dim; ++i) {
102 const struct SymmetryBasisVector *b = SymmetryBasisReplicatedGlobalEntry(X->Sym, i);
103 if (b == NULL) { invalid = 1; break; }
104 local.basis_order = SymmetryHashInteger(local.basis_order, i, 8);
105 local.basis_order = SymmetryHashInteger(local.basis_order, b->rep_state, 8);
106 local.basis_order = SymmetryHashInteger(local.basis_order, b->orbit_size, 8);
107 local.basis_order = SymmetryHashInteger(local.basis_order, b->stabilizer_size, 8);
108 }
109 if (SumMPI_i(invalid) != 0) {
110 fprintf(stdoutMPI, "Error: sector Hamiltonian identity: group or basis metadata is invalid.\n");
111 return -1;
112 }
113 pack_identity(&local, words);
114 if (agree_with_root(words, IDENTITY_WORDS) != 0) {
115 fprintf(stdoutMPI, "Error: sector Hamiltonian identity differs across MPI ranks.\n");
116 return -1;
117 }
118 *identity = local;
119 return 0;
120}

162static int validate_storage(unsigned long dim)
163{
164 if (!iHamPanelActive) {
165 if (Ham == NULL) return -1;
166 for (unsigned long row = 0; row <= dim; ++row)
167 if (Ham[row] == NULL) return -1;
168 return 0;
169 }
170 long chunk = ((long)dim + nproc - 1) / nproc;
171 long first = (long)myrank * chunk + 1, last = ((long)myrank + 1) * chunk;
172 if (last > (long)dim) last = (long)dim;
173 if (first > (long)dim) { first = 1; last = 0; }
174 if (Ham_local == NULL || HamPanelLd != (long)dim ||
175 HamColBegin != first || HamColEnd != last) return -1;
176 if (last >= first &&
177 (size_t)(last - first + 1) > SIZE_MAX / dim / sizeof(*Ham_local)) return -1;
178 return 0;

186/* Off diagonal: |a - conj(b)| <= tol max(1, |a|, |b|).
187 * Diagonal: |Im(a)| <= tol max(1, |a|), without the factor two from a-conj(a).
188 * Inputs have finite components, but their magnitudes/difference may overflow.
189 * Scale by a power of two first: norms/difference stay bounded by sqrt(8),
190 * without rounding each operand through division before subtracting. */
191static int hermitian_pair(double complex lower, double complex upper, int diagonal)
192{
193 double scale = fmax(1.0, fmax(fmax(fabs(creal(lower)), fabs(cimag(lower))),
194 fmax(fabs(creal(upper)), fabs(cimag(upper)))));
195 int exponent = ilogb(scale) + 1;
196 double lr = scalbn(creal(lower), -exponent), li = scalbn(cimag(lower), -exponent);
197 double ur = scalbn(creal(upper), -exponent), ui = scalbn(cimag(upper), -exponent);
198 double bound = HERMITIAN_TOLERANCE *
199 fmax(scalbn(1.0, -exponent), fmax(hypot(lr, li), hypot(ur, ui)));
200 return (diagonal ? fabs(li) : hypot(lr - ur, li + ui)) <= bound;
201}

235/* Every rank owns the full matrix: check, canonicalize and hash locally. */
236static int prepare_replicated(unsigned long dim, uint64_t *digest)
237{
238 int nonfinite = 0, nonhermitian = 0;
239 for (unsigned long col = 1; col <= dim && !nonfinite && !nonhermitian; ++col) {
240 for (unsigned long row = col; row <= dim; ++row) {
241 double complex lower = Ham[row][col], upper = Ham[col][row];
242 if (!finite_element(lower) || !finite_element(upper)) {
243 note_element("not finite", row, col);
244 nonfinite = 1;
245 break;
246 }
247 if (!hermitian_pair(lower, upper, row == col)) {
248 note_element("not Hermitian", row, col);
249 nonhermitian = 1;
250 break;
251 }
252 }
253 }
254 nonfinite = SumMPI_i(nonfinite) != 0;
255 nonhermitian = SumMPI_i(nonhermitian) != 0;
256 if (report_matrix(nonfinite, nonhermitian) != 0) return -1;
257 uint64_t hash = SymmetryHashInteger(FNV_OFFSET, MATRIX_DIGEST_VERSION, 8);
258 hash = SymmetryHashInteger(hash, dim, 8);
259 for (unsigned long col = 1; col <= dim; ++col) {
260 double complex diagonal = creal(Ham[col][col]);
261 Ham[col][col] = diagonal;
262 hash = hash_element(hash, diagonal);
263 for (unsigned long row = col + 1; row <= dim; ++row) {
264 Ham[col][row] = conj(Ham[row][col]);
265 hash = hash_element(hash, Ham[row][col]);
266 }
267 }
268 *digest = hash;
269 return 0;
270}

273/* ELPA panel: column c is sent once from its owner. The owner of column r
274 * compares its H[c,r] with the received H[r,c] before writing the conjugate,
275 * and every rank hashes the identical received lower triangle. */
276static int prepare_panel(unsigned long dim, uint64_t *digest)
277{
278 double complex *column = malloc(dim * sizeof(*column));
279 uint64_t hash = SymmetryHashInteger(FNV_OFFSET, MATRIX_DIGEST_VERSION, 8);
280#ifdef HPHI_TEST_SYMMETRY_PROBE
281 if (SymmetryFullDiagTestFail("matrix_alloc")) { free(column); column = NULL; }
282#endif
283 if (SumMPI_i(column == NULL) != 0) {
284 free(column);
285 fprintf(stdoutMPI, "Error: sector Hamiltonian column buffer allocation failed.\n");
286 return -1;
287 }
288 hash = SymmetryHashInteger(hash, dim, 8);
289 for (unsigned long col = 1; col <= dim; ++col) {
290 int flags = 0, any = 0;
291 if (SumMPI_i(broadcast_column(dim, col, column) != 0) != 0) {
292 free(column);
293 fprintf(stdoutMPI, "Error: sector Hamiltonian column broadcast failed.\n");
294 return -1;
295 }
296 for (unsigned long row = col; row <= dim && !flags; ++row) {
297 double complex lower = column[row - 1];
298 if (!finite_element(lower)) {
299 /* Every rank received this element; report it once. */
300 flags |= 1;
301 if (myrank == 0) note_element("not finite", row, col);
302 } else if (owns_column(row)) {
303 double complex upper = read_owned(col, row);
304 if (!finite_element(upper)) flags |= 1;
305 else if (!hermitian_pair(lower, upper, row == col)) flags |= 2;
306 if (flags) note_element(flags & 1 ? "not finite" : "not Hermitian", row, col);
307 }
308 }
309 if (MPI_Allreduce(&flags, &any, 1, MPI_INT, MPI_BOR, MPI_COMM_WORLD) != MPI_SUCCESS)
310 any = 1;
311 if (report_matrix(any & 1, any & 2) != 0) {
312 free(column);
313 return -1;
314 }
315 column[col - 1] = creal(column[col - 1]);
316 if (owns_column(col)) set_owned(col, col, column[col - 1]);
317 for (unsigned long row = col; row <= dim; ++row) {
318 if (row > col && owns_column(row)) set_owned(col, row, conj(column[row - 1]));
319 hash = hash_element(hash, column[row - 1]);
320 }
321 }
322 free(column);
323 *digest = hash;
324 return 0;
325}

328int PrepareSymmetryHamiltonian(const struct BindStruct *X, int from_file,
329 struct SymmetryHamiltonianInfo *info)
330{
331 struct SymmetryHamiltonianInfo local;
332 uint64_t digest = 0;
333 int status = -1;
334 StartTimer(5600);
335 memset(&local, 0, sizeof(local));
336 if (SumMPI_i(info == NULL || (from_file != 0 && from_file != 1)) != 0) {
337 fprintf(stdoutMPI, "Error: invalid sector Hamiltonian preparation arguments.\n");
338 goto done;
339 }
340 if (BuildSymmetryHamiltonianIdentity(X, &local.identity) != 0) goto done;
341 if (SumMPI_i(validate_storage(X->Sym->dim) != 0) != 0) {
342 fprintf(stdoutMPI, "Error: sector Hamiltonian storage does not match the FullDiag sector dimension.\n");
343 goto done;
344 }
345#ifdef MPI
346 if (iHamPanelActive) {
347 if (prepare_panel(X->Sym->dim, &digest) != 0) goto done;
348 } else
349#endif
350 if (prepare_replicated(X->Sym->dim, &digest) != 0) goto done;
351 if (agree_with_root(&digest, 1) != 0) {
352 fprintf(stdoutMPI, "Error: sector Hamiltonian digest differs across MPI ranks.\n");
353 goto done;
354 }
355 local.matrix_digest = digest;
356 local.from_file = from_file;
357 *info = local;
358 status = 0;
359done:
360 StopTimer(5600);
361 return status;
362}
