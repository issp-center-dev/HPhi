28/* Field order of the identity, also the cross-rank comparison order. */
29static void pack_identity(const struct SymmetryHamiltonianIdentity *id, uint64_t *words)
30{
31 const uint64_t fields[IDENTITY_WORDS] = {
32 id->model, id->nsite, id->nup, id->ndown, id->ne, id->local_site_mask,
33 id->fixed_flags, id->phase, id->full_dim, id->sector_dim, id->group_digest,
34 id->sector_count, id->sector_xor, id->sector_sum, id->basis_order
35 };
36 memcpy(words, fields, sizeof(fields));
37}

39/* Collective: 0 when every rank holds the root's words. */
40static int agree_with_root(const uint64_t *local, int count)
41{
42 uint64_t root[IDENTITY_WORDS];
43 if (count < 1 || count > IDENTITY_WORDS) return -1;
44 memcpy(root, local, (size_t)count * sizeof(*root));
45#ifdef MPI
46 if (MPI_Bcast(root, count, MPI_UINT64_T, 0, MPI_COMM_WORLD) != MPI_SUCCESS) return -1;
47#endif
48 return SumMPI_i(memcmp(root, local, (size_t)count * sizeof(*root)) != 0) != 0 ? -1 : 0;
49}

51int BuildSymmetryHamiltonianIdentity(const struct BindStruct *X,
52 struct SymmetryHamiltonianIdentity *identity)
53{
54 struct SymmetryHamiltonianIdentity local;
55 struct SymmetrySectorDigest sector;
56 uint64_t words[IDENTITY_WORDS];
57 int invalid = identity == NULL || X == NULL || X->Sym == NULL ||
58 !X->Def.iFlgSymmetryBasis || sizeof(double) != 8 ||
59 DBL_MANT_DIG != 53 || DBL_MAX_EXP != 1024;
60 if (!invalid)
61 invalid = X->Sym->enabled != TRUE ||
62 X->Sym->basis_layout != SYMMETRY_BASIS_REPLICATED ||
63 X->Sym->basis == NULL || X->Sym->dim == 0 || X->Sym->dim > INT_MAX;
64 if (SumMPI_i(invalid) != 0) {
65 fprintf(stdoutMPI, "Error: sector Hamiltonian identity requires a replicated, nonempty TransSym basis with binary64 scalars.\n");
66 return -1;
67 }
68 if (ComputeSymmetrySectorDigest(X->Sym, &sector) != 0) {
69 fprintf(stdoutMPI, "Error: sector Hamiltonian identity: sector digest failed.\n");
70 return -1;
71 }
72 memset(&local, 0, sizeof(local));
73 local.model = (uint64_t)X->Def.iCalcModel;
74 local.nsite = X->Def.Nsite;
75 if (X->Def.iCalcModel != SpinGC) {
76 local.nup = X->Def.Nup;
77 local.ndown = X->Def.Ndown;
78 local.ne = X->Def.Ne;
79 }
80 local.phase = 1;
81 if (IsSymmetryKondoModel(X->Def.iCalcModel)) {
82 struct SymmetryKondoIdentity kondo;
83 if (GetSymmetryKondoIdentity(&X->Def, &kondo) != 0) {
84 invalid = 1;
85 } else {
86 local.nup = kondo.nup;
87 local.ndown = kondo.ndown;
88 local.ne = kondo.ne;
89 local.local_site_mask = kondo.local_site_mask;
90 local.fixed_flags = kondo.fixed_flags;
91 local.phase = kondo.phase;
92 }
93 }
94 local.full_dim = X->Sym->full_dim;
95 local.sector_dim = X->Sym->dim;
96 if (!invalid && ComputeSymmetryGroupDigest(&X->Def, &local.group_digest) != 0) invalid = 1;
97 local.sector_count = sector.count;
98 local.sector_xor = sector.xor_hash;
99 local.sector_sum = sector.sum_hash;
100 /* One global stream, the word encoding of the checkpoint ORDER field. */
101 local.basis_order = FNV_OFFSET;
102 for (unsigned long i = 1; !invalid && i <= X->Sym->dim; ++i) {
103 const struct SymmetryBasisVector *b = SymmetryBasisReplicatedGlobalEntry(X->Sym, i);
104 if (b == NULL) { invalid = 1; break; }
105 local.basis_order = SymmetryHashInteger(local.basis_order, i, 8);
106 local.basis_order = SymmetryHashInteger(local.basis_order, b->rep_state, 8);
107 local.basis_order = SymmetryHashInteger(local.basis_order, b->orbit_size, 8);
108 local.basis_order = SymmetryHashInteger(local.basis_order, b->stabilizer_size, 8);
109 }
110 if (SumMPI_i(invalid) != 0) {
111 fprintf(stdoutMPI, "Error: sector Hamiltonian identity: group or basis metadata is invalid.\n");
112 return -1;
113 }
114 pack_identity(&local, words);
115 if (agree_with_root(words, IDENTITY_WORDS) != 0) {
116 fprintf(stdoutMPI, "Error: sector Hamiltonian identity differs across MPI ranks.\n");
117 return -1;
118 }
119 *identity = local;
120 return 0;
121}

163static int validate_storage(unsigned long dim)
164{
165 if (!iHamPanelActive) {
166 if (Ham == NULL) return -1;
167 for (unsigned long row = 0; row <= dim; ++row)
168 if (Ham[row] == NULL) return -1;
169 return 0;
170 }
171 long chunk = ((long)dim + nproc - 1) / nproc;
172 long first = (long)myrank * chunk + 1, last = ((long)myrank + 1) * chunk;
173 if (last > (long)dim) last = (long)dim;
174 if (first > (long)dim) { first = 1; last = 0; }
175 if (Ham_local == NULL || HamPanelLd != (long)dim ||
176 HamColBegin != first || HamColEnd != last) return -1;
177 if (last >= first &&
178 (size_t)(last - first + 1) > SIZE_MAX / dim / sizeof(*Ham_local)) return -1;
179 return 0;
180}

182static int finite_element(double complex value)
183{
184 return isfinite(creal(value)) && isfinite(cimag(value));
185}
186
187/* Off diagonal: |a - conj(b)| <= tol max(1, |a|, |b|).
188 * Diagonal: |Im(a)| <= tol max(1, |a|), without the factor two from a-conj(a).
189 * Inputs have finite components, but their magnitudes/difference may overflow.
190 * Scale by a power of two first: norms/difference stay bounded by sqrt(8),
191 * without rounding each operand through division before subtracting. */
192static int hermitian_pair(double complex lower, double complex upper, int diagonal)
193{
194 double scale = fmax(1.0, fmax(fmax(fabs(creal(lower)), fabs(cimag(lower))),
195 fmax(fabs(creal(upper)), fabs(cimag(upper)))));
196 int exponent = ilogb(scale) + 1;
197 double lr = scalbn(creal(lower), -exponent), li = scalbn(cimag(lower), -exponent);
198 double ur = scalbn(creal(upper), -exponent), ui = scalbn(cimag(upper), -exponent);
199 double bound = HERMITIAN_TOLERANCE *
200 fmax(scalbn(1.0, -exponent), fmax(hypot(lr, li), hypot(ur, ui)));
201 return (diagonal ? fabs(li) : hypot(lr - ur, li + ui)) <= bound;
202}

204static uint64_t hash_component(uint64_t hash, double value)
205{
206 uint64_t bits;
207 if (value == 0.0) value = 0.0; /* signed zero hashes as +0 */
208 memcpy(&bits, &value, sizeof(bits));
209 return SymmetryHashInteger(hash, bits, 8);
210}
211
212static uint64_t hash_element(uint64_t hash, double complex value)
213{
214 return hash_component(hash_component(hash, creal(value)), cimag(value));
215}

236/* Every rank owns the full matrix: check, canonicalize and hash locally. */
237static int prepare_replicated(unsigned long dim, uint64_t *digest)
238{
239 int nonfinite = 0, nonhermitian = 0;
240 for (unsigned long col = 1; col <= dim && !nonfinite && !nonhermitian; ++col) {
241 for (unsigned long row = col; row <= dim; ++row) {
242 double complex lower = Ham[row][col], upper = Ham[col][row];
243 if (!finite_element(lower) || !finite_element(upper)) {
244 note_element("not finite", row, col);
245 nonfinite = 1;
246 break;
247 }
248 if (!hermitian_pair(lower, upper, row == col)) {
249 note_element("not Hermitian", row, col);
250 nonhermitian = 1;
251 break;
252 }
253 }
254 }
255 nonfinite = SumMPI_i(nonfinite) != 0;
256 nonhermitian = SumMPI_i(nonhermitian) != 0;
257 if (report_matrix(nonfinite, nonhermitian) != 0) return -1;
258 uint64_t hash = SymmetryHashInteger(FNV_OFFSET, MATRIX_DIGEST_VERSION, 8);
259 hash = SymmetryHashInteger(hash, dim, 8);
260 for (unsigned long col = 1; col <= dim; ++col) {
261 double complex diagonal = creal(Ham[col][col]);
262 Ham[col][col] = diagonal;
263 hash = hash_element(hash, diagonal);
264 for (unsigned long row = col + 1; row <= dim; ++row) {
265 Ham[col][row] = conj(Ham[row][col]);
266 hash = hash_element(hash, Ham[row][col]);
267 }
268 }
269 *digest = hash;
270 return 0;
271}

274/* ELPA panel: column c is sent once from its owner. The owner of column r
275 * compares its H[c,r] with the received H[r,c] before writing the conjugate,
276 * and every rank hashes the identical received lower triangle. */
277static int prepare_panel(unsigned long dim, uint64_t *digest)
278{
279 double complex *column = malloc(dim * sizeof(*column));
280 uint64_t hash = SymmetryHashInteger(FNV_OFFSET, MATRIX_DIGEST_VERSION, 8);
281#ifdef HPHI_TEST_SYMMETRY_PROBE
282 if (SymmetryFullDiagTestFail("matrix_alloc")) { free(column); column = NULL; }
283#endif
284 if (SumMPI_i(column == NULL) != 0) {
285 free(column);
286 fprintf(stdoutMPI, "Error: sector Hamiltonian column buffer allocation failed.\n");
287 return -1;
288 }
289 hash = SymmetryHashInteger(hash, dim, 8);
290 for (unsigned long col = 1; col <= dim; ++col) {
291 int flags = 0, any = 0;
292 if (SumMPI_i(broadcast_column(dim, col, column) != 0) != 0) {
293 free(column);
294 fprintf(stdoutMPI, "Error: sector Hamiltonian column broadcast failed.\n");
295 return -1;
296 }
297 for (unsigned long row = col; row <= dim && !flags; ++row) {
298 double complex lower = column[row - 1];
299 if (!finite_element(lower)) {
300 /* Every rank received this element; report it once. */
301 flags |= 1;
302 if (myrank == 0) note_element("not finite", row, col);
303 } else if (owns_column(row)) {
304 double complex upper = read_owned(col, row);
305 if (!finite_element(upper)) flags |= 1;
306 else if (!hermitian_pair(lower, upper, row == col)) flags |= 2;
307 if (flags) note_element(flags & 1 ? "not finite" : "not Hermitian", row, col);
308 }
309 }
310 if (MPI_Allreduce(&flags, &any, 1, MPI_INT, MPI_BOR, MPI_COMM_WORLD) != MPI_SUCCESS)
311 any = 1;
312 if (report_matrix(any & 1, any & 2) != 0) {
313 free(column);
314 return -1;
315 }
316 column[col - 1] = creal(column[col - 1]);
317 if (owns_column(col)) set_owned(col, col, column[col - 1]);
318 for (unsigned long row = col; row <= dim; ++row) {
319 if (row > col && owns_column(row)) set_owned(col, row, conj(column[row - 1]));
320 hash = hash_element(hash, column[row - 1]);
321 }
322 }
323 free(column);
324 *digest = hash;
325 return 0;
326}
327#endif

329int PrepareSymmetryHamiltonian(const struct BindStruct *X, int from_file,
330 struct SymmetryHamiltonianInfo *info)
331{
332 struct SymmetryHamiltonianInfo local;
333 uint64_t digest = 0;
334 int status = -1;
335 StartTimer(5600);
336 memset(&local, 0, sizeof(local));
337 if (SumMPI_i(info == NULL || (from_file != 0 && from_file != 1)) != 0) {
338 fprintf(stdoutMPI, "Error: invalid sector Hamiltonian preparation arguments.\n");
339 goto done;
340 }
341 if (BuildSymmetryHamiltonianIdentity(X, &local.identity) != 0) goto done;
342 if (SumMPI_i(validate_storage(X->Sym->dim) != 0) != 0) {
343 fprintf(stdoutMPI, "Error: sector Hamiltonian storage does not match the FullDiag sector dimension.\n");
344 goto done;
345 }
346#ifdef MPI
347 if (iHamPanelActive) {
348 if (prepare_panel(X->Sym->dim, &digest) != 0) goto done;
349 } else
350#endif
351 if (prepare_replicated(X->Sym->dim, &digest) != 0) goto done;
352 if (agree_with_root(&digest, 1) != 0) {
353 fprintf(stdoutMPI, "Error: sector Hamiltonian digest differs across MPI ranks.\n");
354 goto done;
355 }
356 local.matrix_digest = digest;
357 local.from_file = from_file;
358 *info = local;
359 status = 0;
360done:
361 StopTimer(5600);
362 return status;
363}

418/* Digits only: no sign, space, prefix or overflow. */
419static int parse_decimal(const char *text, uint64_t *value)
420{
421 uint64_t result = 0;
422 if (*text == '\0') return -1;
423 for (; *text != '\0'; ++text) {
424 unsigned int digit;
425 if (*text < '0' || *text > '9') return -1;
426 digit = (unsigned int)(*text - '0');
427 if (result > (UINT64_MAX - digit) / 10) return -1;
428 result = result * 10 + digit;
429 }
430 *value = result;
431 return 0;
432}

434static int parse_hex16(const char *text, uint64_t *value)
435{
436 uint64_t result = 0;
437 if (strlen(text) != 16) return -1;
438 for (; *text != '\0'; ++text) {
439 int digit;
440 if (*text >= '0' && *text <= '9') digit = *text - '0';
441 else if (*text >= 'a' && *text <= 'f') digit = *text - 'a' + 10;
442 else if (*text >= 'A' && *text <= 'F') digit = *text - 'A' + 10;
443 else return -1;
444 result = (result << 4) | (uint64_t)digit;
445 }
446 *value = result;
447 return 0;
448}

457/* Splits on spaces/tabs; returns max+1 when there are more tokens. */
458static int split_tokens(char *text, char **tokens, int max)
459{
460 int count = 0;
461 for (;;) {

465 tokens[count++] = text;

485/* 1 for a line without its LF/CRLF, 0 at the end of the file, -1 for a
486 * read error, embedded NUL or a line that does not fit the buffer. */
487static int read_line(struct MatrixReader *reader)
488{
489 size_t length = 0;
490 int byte = fgetc(reader->fp);
491 if (byte == EOF)
492 return ferror(reader->fp) ? -1 : 0;
493 reader->line++;
494 do {
495 /* Count physical bytes, including the line end, before treating them as
496 * a C string. A NUL in the file must not hide records or extra tokens. */
497 if (byte == '\0' || length == sizeof(reader->text) - 1) return -1;
498 reader->text[length++] = (char)byte;
499 if (byte == '\n') break;
500 } while ((byte = fgetc(reader->fp)) != EOF);
501 if (ferror(reader->fp)) return -1;
502 if (length > 0 && reader->text[length - 1] == '\n') --length;
503 if (length > 0 && reader->text[length - 1] == '\r') --length;
504 reader->text[length] = '\0';
505 return 1;
506}

524/* "% HPHI key=value": each key once, values in their documented syntax. */
525static int parse_metadata(const struct MatrixReader *reader, char *entry,
526 uint64_t *words, unsigned long *seen)
527{
528 char *value = strchr(entry, '=');
529 int key, invalid;
530 if (value == NULL) {
531 reader_error(reader, "malformed metadata line (expected %skey=value)", METADATA_PREFIX);
532 return -1;
533 }
534 *value++ = '\0';
535 for (key = 0; key < REQUIRED_KEYS && strcmp(entry, required_keys[key]) != 0; ++key) ;
536 if (key == REQUIRED_KEYS) {
537 reader_error(reader, "unknown metadata key %s", entry);
538 return -1;
539 }
540 if (*seen & (1UL << key)) {
541 reader_error(reader, "duplicate metadata key %s", entry);
542 return -1;
543 }
544 *seen |= 1UL << key;
545 if (key == 0 || key == 1) {
546 const char *expected = key == 0 ? MATRIX_FORMAT : MATRIX_SCALAR;
547 if (strcmp(value, expected) == 0) return 0;
548 reader_error(reader, "unsupported %s %s (expected %s)", entry, value, expected);
549 return -1;
550 }
551 int field = key - FIRST_IDENTITY_KEY;
552 if (field == 0) {
553 invalid = 1;
554 for (size_t i = 0; i < sizeof(matrix_models) / sizeof(matrix_models[0]); ++i)
555 if (strcmp(value, matrix_models[i].name) == 0) {
556 words[0] = (uint64_t)matrix_models[i].model;
557 invalid = 0;
558 }
559 } else {
560 invalid = hex_field(field) ? parse_hex16(value, &words[field])
561 : parse_decimal(value, &words[field]);
562 }
563 if (invalid) reader_error(reader, "invalid metadata value for %s: %s", entry, value);
564 return invalid ? -1 : 0;
565}

568static int read_matrix_header(struct MatrixReader *reader, uint64_t *words, uint64_t *dims)

585 return -1;
586 }
587 if (blank(reader->text)) continue;
588 if (reader->text[0] != '%') break;
589 if (strncmp(reader->text, METADATA_PREFIX, strlen(METADATA_PREFIX)) == 0 &&
590 parse_metadata(reader, reader->text + strlen(METADATA_PREFIX), words, &seen) != 0)
591 return -1;
592 }
593 for (int key = 0; key < REQUIRED_KEYS; ++key) {
594 if (seen & (1UL << key)) continue;
595 fprintf(stdoutMPI, "Error: sector Hamiltonian file %s: missing metadata key %s.\n",
596 reader->path, required_keys[key]);
597 return -1;
598 }
599 if (split_tokens(reader->text, tokens, 3) != 3 || parse_decimal(tokens[0], &dims[0]) != 0 ||
600 parse_decimal(tokens[1], &dims[1]) != 0 || parse_decimal(tokens[2], &dims[2]) != 0) {
601 reader_error(reader, "dimension line must hold three nonnegative integers (rows columns records)");
602 return -1;
603 }
604 return 0;

607/* Root only: the basis identity and the dimension line of the file. */
608static int check_matrix_header(const struct MatrixReader *reader, const uint64_t *words,
609 const uint64_t *expected, const uint64_t *dims)
610{
611 uint64_t dim = expected[9];
612 for (int field = 0; field < IDENTITY_WORDS; ++field) {
613 char found[32], wanted[32];
614 if (words[field] == expected[field]) continue;
615 format_identity_value(field, words[field], found, sizeof(found));
616 format_identity_value(field, expected[field], wanted, sizeof(wanted));
617 fprintf(stdoutMPI, "Error: sector Hamiltonian file %s: metadata %s does not match this sector "
618 "(file=%s, expected=%s).\n", reader->path,
619 required_keys[FIRST_IDENTITY_KEY + field], found, wanted);
620 return -1;
621 }
622 if (dims[0] != dim || dims[1] != dim) {
623 reader_error(reader, "matrix %" PRIu64 " x %" PRIu64 " does not match sector dimension %" PRIu64,
624 dims[0], dims[1], dim);
625 return -1;
626 }
627 if (dims[2] > dim * (dim + 1) / 2) {
628 reader_error(reader, "%" PRIu64 " records declared, more records than the lower triangle "
629 "of dimension %" PRIu64 " holds", dims[2], dim);
630 return -1;
631 }
632 return 0;
633}

635/* Root only: one "row column real imaginary" record of the lower triangle. */
636static int read_record(struct MatrixReader *reader, uint64_t dim, uint64_t read, uint64_t nnz,
637 uint64_t *index, double *value)
638{
639 char *tokens[4];
640 int status = read_line(reader);
641 if (status == 0) {
642 fprintf(stdoutMPI, "Error: sector Hamiltonian file %s: fewer records than declared "
643 "(%" PRIu64 " of %" PRIu64 ").\n", reader->path, read, nnz);
644 return -1;
645 }
646 if (status < 0) {
647 unreadable_line(reader);
648 return -1;
649 }
650 if (split_tokens(reader->text, tokens, 4) != 4 || parse_decimal(tokens[0], &index[0]) != 0 ||
651 parse_decimal(tokens[1], &index[1]) != 0 || parse_double(tokens[2], &value[0]) != 0 ||
652 parse_double(tokens[3], &value[1]) != 0) {
653 reader_error(reader, "malformed record (expected: row column real imaginary)");
654 return -1;
655 }
656 if (index[0] < 1 || index[0] > dim || index[1] < 1 || index[1] > dim) {
657 reader_error(reader, "index out of range (%" PRIu64 ",%" PRIu64 ")", index[0], index[1]);
658 return -1;
659 }
660 if (index[1] > index[0]) {
661 reader_error(reader, "upper-triangle record (%" PRIu64 ",%" PRIu64 "); write row >= column only",
662 index[0], index[1]);
663 return -1;
664 }
665 if (!isfinite(value[0]) || !isfinite(value[1])) {
666 reader_error(reader, "non-finite coefficient");
667 return -1;
668 }
669 if (index[0] == index[1] && value[1] != 0.0) {
670 reader_error(reader, "nonzero diagonal imaginary part");
671 return -1;
672 }
673 return 0;
674}

678{
679 int status;

684 reader_error(reader, "unexpected content after the %" PRIu64 " declared records", nnz);
685 return -1;
686 }
687 if (status < 0) {
688 unreadable_line(reader);
689 return -1;
690 }
691 return 0;

721/* Collective: the root's status and count, then (on success) the records. */
722static int share_chunk(int *share, uint64_t *index, double *value)
723{
724#ifdef MPI
725 if (MPI_Bcast(share, 2, MPI_INT, 0, MPI_COMM_WORLD) != MPI_SUCCESS) return -1;
726 if (share[0] != 0 || share[1] == 0) return 0;
727 if (MPI_Bcast(index, 2 * share[1], MPI_UINT64_T, 0, MPI_COMM_WORLD) != MPI_SUCCESS ||
728 MPI_Bcast(value, 2 * share[1], MPI_DOUBLE, 0, MPI_COMM_WORLD) != MPI_SUCCESS) return -1;
729#else
730 (void)share; (void)index; (void)value;
731#endif
732 return 0;
733}

735/* Every rank stores the records it owns. Duplicates are detected by the
736 * unique column owner (col-1)/NC with a D x NC bitset, for every storage. */
737static int apply_chunk(unsigned long dim, unsigned long chunk_columns, unsigned char *seen,
738 const uint64_t *index, const double *value, int count)
739{
740 unsigned long first = (unsigned long)myrank * chunk_columns + 1;
741 int duplicate = 0;
742 for (int i = 0; i < count; ++i) {
743 unsigned long row = (unsigned long)index[2 * i], col = (unsigned long)index[2 * i + 1];
744 if ((col - 1) / chunk_columns == (unsigned long)myrank) {
745 uint64_t bit = (uint64_t)(col - first) * dim + (row - 1);
746 unsigned char mask = (unsigned char)(1u << (bit & 7));
747 if (seen[bit >> 3] & mask) {
748 if (!duplicate)
749 fprintf(stderr, "Error: rank %d sector Hamiltonian record (%lu,%lu) is duplicated.\n",
750 myrank, row, col);
751 duplicate = 1;
752 continue;
753 }
754 seen[bit >> 3] |= mask;
755 }
756 store_record(row, col, value[2 * i] + I * value[2 * i + 1]);
757 }
758 return duplicate;
759}

761int InputSymmetryHamiltonian(const struct BindStruct *X,
762 struct SymmetryHamiltonianInfo *info)
763{
764 struct SymmetryHamiltonianIdentity identity;
765 struct MatrixReader reader;
766 char path[D_FileNameMax];
767 uint64_t words[IDENTITY_WORDS] = {0}, expected[IDENTITY_WORDS], dims[3] = {0, 0, 0};
768 uint64_t *index = NULL;
769 double *value = NULL;
770 unsigned char *seen = NULL;
771 unsigned long dim, chunk_columns;
772 int status = -1, failed;
773 StartTimer(5800);
774 memset(&reader, 0, sizeof(reader));
775 failed = X == NULL || info == NULL || matrix_path(&X->Def, 0, path, sizeof(path)) != 0;
776 if (SumMPI_i(failed) != 0) {
777 fprintf(stdoutMPI, "Error: invalid sector Hamiltonian input arguments or path.\n");
778 goto done;
779 }
780 if (BuildSymmetryHamiltonianIdentity(X, &identity) != 0) goto done;
781 dim = X->Sym->dim;
782 if (SumMPI_i(validate_storage(dim) != 0) != 0) {
783 fprintf(stdoutMPI, "Error: sector Hamiltonian storage does not match the FullDiag sector dimension.\n");
784 goto done;
785 }
786 chunk_columns = (dim + (unsigned long)nproc - 1) / (unsigned long)nproc;
787 {
788 uint64_t bits = (uint64_t)dim * chunk_columns;
789 size_t bytes = (size_t)(bits / 8 + 1);
790 seen = bits / 8 + 1 <= SIZE_MAX ? calloc(bytes, 1) : NULL;
791 }
792 index = malloc(2 * RECORD_CHUNK * sizeof(*index));
793 value = malloc(2 * RECORD_CHUNK * sizeof(*value));
794 if (SumMPI_i(seen == NULL || index == NULL || value == NULL) != 0) {
795 fprintf(stdoutMPI, "Error: sector Hamiltonian input buffer allocation failed.\n");
796 goto done;
797 }
798 clear_storage(dim);
799 pack_identity(&identity, expected);
800 failed = 0;
801 if (myrank == 0) {
802 reader.path = path;
803 reader.fp = fopen(path, "r");
804 if (reader.fp == NULL) {
805 fprintf(stdoutMPI, "Error: could not open sector Hamiltonian input %s.\n", path);
806 failed = 1;
807 } else {
808 failed = read_matrix_header(&reader, words, dims) != 0 ||
809 check_matrix_header(&reader, words, expected, dims) != 0;
810 }
811 }
812 if (SumMPI_i(failed) != 0) goto done;
813#ifdef MPI
814 if (MPI_Bcast(&dims[2], 1, MPI_UINT64_T, 0, MPI_COMM_WORLD) != MPI_SUCCESS) failed = 1;
815 if (SumMPI_i(failed) != 0) goto done;
816#endif
817 for (uint64_t read = 0; read < dims[2];) {
818 int share[2] = {0, 0}; /* root status, record count */
819 if (myrank == 0) {
820 uint64_t left = dims[2] - read;
821 share[1] = left < RECORD_CHUNK ? (int)left : RECORD_CHUNK;
822 for (int i = 0; i < share[1] && share[0] == 0; ++i)
823 share[0] = read_record(&reader, dim, read + (uint64_t)i, dims[2],
824 &index[2 * i], &value[2 * i]) != 0;
825 }
826 failed = share_chunk(share, index, value) != 0;
827 if (SumMPI_i(failed || share[0] != 0) != 0) goto done;
828 if (SumMPI_i(apply_chunk(dim, chunk_columns, seen, index, value, share[1])) != 0) {
829 fprintf(stdoutMPI, "Error: sector Hamiltonian file %s has a duplicated record.\n", path);
830 goto done;
831 }
832 read += (uint64_t)share[1];
833 }
834 failed = 0;
835 if (myrank == 0) {
836 failed = read_trailer(&reader, dims[2]) != 0;
837 if (fclose(reader.fp) != 0) failed = 1;
838 reader.fp = NULL;
839 }
840 if (SumMPI_i(failed) != 0) goto done;
841 /* The lower triangle and its conjugate are stored: canonicalize and hash. */
842 if (PrepareSymmetryHamiltonian(X, 1, info) != 0) goto done;
843 status = 0;
844done:
845 if (reader.fp != NULL) fclose(reader.fp);
846 free(seen);
847 free(index);
848 free(value);
849 StopTimer(5800);
850 return status;
851}

905{
906 char path[D_FileNameMax], temporary[D_FileNameMax];
907 FILE *fp = NULL;
908 double complex *column = NULL;
909 uint64_t nnz = 0;
910 unsigned long dim = 0, col;
911 int status = -1, failed, created = 0;
912 StartTimer(5900);
913 failed = X == NULL || info == NULL || X->Sym == NULL ||
914 matrix_path(&X->Def, 0, path, sizeof(path)) != 0 ||
915 matrix_path(&X->Def, 1, temporary, sizeof(temporary)) != 0;
916 if (!failed) {
917 dim = X->Sym->dim;
918 failed = info->identity.sector_dim != dim || model_name(info->identity.model) == NULL ||
919 validate_storage(dim) != 0;
920 }
921 if (SumMPI_i(failed) != 0) {
922 fprintf(stdoutMPI, "Error: sector Hamiltonian output needs the prepared sector matrix and a valid path.\n");
923 goto done;
924 }

926 if (SumMPI_i(column == NULL) != 0) {
927 fprintf(stdoutMPI, "Error: sector Hamiltonian output column buffer allocation failed.\n");
928 goto done;
929 }
930 /* The open is agreed before any column is sent. */
931 failed = 0;
932 if (myrank == 0) {
933 fp = fopen(temporary, "w");
934 created = fp != NULL;
935 failed = fp == NULL;
936 if (failed) fprintf(stdoutMPI, "Error: could not open sector Hamiltonian output %s.\n", temporary);
937 }
938 if (SumMPI_i(failed) != 0) goto done;
939 /* Pass 1: the root counts the nonzero lower-triangle elements. */
940 for (col = 1; col <= dim && !failed; ++col) {
941 if (load_column(dim, col, column) != 0) {
942 fprintf(stdoutMPI, "Error: sector Hamiltonian column broadcast failed.\n");
943 failed = 1;
944 break;
945 }
946 if (myrank != 0) continue;
947 for (unsigned long row = col; row <= dim; ++row) {
948 double re, im;
949 canonical_parts(column[row - 1], &re, &im);

954 /* Pass 2: column-major lower triangle. Panel ranks agree before it and
955 * after each column, so a failed root write stops the broadcasts; only
956 * the root writes replicated storage. */
957 if (iHamPanelActive) failed = SumMPI_i(failed) != 0;
958 for (col = 1; col <= dim && !failed; ++col) {
959 if (load_column(dim, col, column) != 0) {
960 failed = 1;
961 break;
962 }
963 if (myrank == 0) {
964#ifdef HPHI_TEST_SYMMETRY_PROBE

967 failed = 1;
968 } else
969#endif
970 failed = write_column(fp, dim, col, column) != 0;
971 }
972 if (iHamPanelActive ? SumMPI_i(failed) != 0 : failed) break;
973 }
974 if (myrank == 0 && fp != NULL) {
975 if (fflush(fp) != 0 || ferror(fp)) failed = 1;
976 if (fclose(fp) != 0) failed = 1;
977 fp = NULL;
978#ifdef HPHI_TEST_SYMMETRY_PROBE
979 if (SymmetryFullDiagTestFail("matrix_close")) failed = 1;
980#endif
981 }
982 if (SumMPI_i(failed) != 0) {
983 if (created) remove(temporary);
984 fprintf(stdoutMPI, "Error: sector Hamiltonian output write failed for %s.\n", temporary);
985 goto done;
986 }
987 if (myrank == 0) {
988#ifdef HPHI_TEST_SYMMETRY_PROBE
989 if (SymmetryFullDiagTestFail("matrix_rename")) failed = 1;
990 else
991#endif
992 failed = rename(temporary, path) != 0;
993 if (failed) remove(temporary);
994 }
995 if (SumMPI_i(failed) != 0) {
996 fprintf(stdoutMPI, "Error: sector Hamiltonian output publish failed for %s.\n", path);
997 goto done;
998 }
999 status = 0;
1000done:
1001 if (fp != NULL) {
1002 fclose(fp);
1003 remove(temporary);
1004 }
1005 free(column);
1006 StopTimer(5900);
1007 return status;
1008}
