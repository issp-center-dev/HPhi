67int GetPairExcitedStateHalfSpinGC(
68 struct BindStruct *X,
69 unsigned int NPairExcitationOperator,
70 int **PairExcitationOperator,
71 double complex *ParaPairExcitationOperator,
72 double complex *tmp_v0,
73 double complex *tmp_v1
75){
76 long unsigned int i,j;
77 long unsigned int isite1;
78 long unsigned int org_isite1,org_isite2,org_sigma1,org_sigma2;
79 long unsigned int tmp_off=0;
80
81 double complex tmp_trans=0;
82 long int i_max;
83 int tmp_sgn;
84 i_max = X->Check.idim_maxOrg;
85
86 for(i=0;i<NPairExcitationOperator;i++){
87 org_isite1 = PairExcitationOperator[i][0]+1;
88 org_isite2 = PairExcitationOperator[i][2]+1;
89 org_sigma1 = PairExcitationOperator[i][1];
90 org_sigma2 = PairExcitationOperator[i][3];
91 tmp_trans = ParaPairExcitationOperator[i];
92 if(org_isite1 == org_isite2){
93 if(org_isite1 > X->Def.Nsite){
94 if(org_sigma1==org_sigma2){ // longitudinal magnetic field
95 if(PairExcitationOperator[i][4]==0) {
96 child_GC_AisCis_spin_MPIdouble(org_isite1 - 1, org_sigma1, -tmp_trans, X, tmp_v0, tmp_v1);
97 }
98 else{
99 child_GC_CisAis_spin_MPIdouble(org_isite1 - 1, org_sigma1, tmp_trans, X, tmp_v0, tmp_v1);
100 }
101 }
102 else{ // transverse magnetic field
103 //fprintf(stdoutMPI, "Debug: test, org_isite1=%d, org_sigma1=%d, orgsima_2=%d\n", org_isite1, org_sigma1, org_sigma2);
104 child_GC_CisAit_spin_MPIdouble(org_isite1-1, org_sigma1, org_sigma2, tmp_trans, X, tmp_v0, tmp_v1);
105 }
106 }else{
107 isite1 = X->Def.Tpow[org_isite1-1];
108 if(org_sigma1==org_sigma2) {
109 if (PairExcitationOperator[i][4] == 0) {
110 // longitudinal magnetic field
111#pragma omp parallel for default(none) private(j, tmp_sgn) firstprivate(i_max, isite1, org_sigma1, X,tmp_trans) shared(tmp_v0, tmp_v1)
112 for (j = 1; j <= i_max; j++) {
113 tmp_v0[j] += (1.0-child_SpinGC_CisAis(j, X, isite1, org_sigma1)) * tmp_v1[j] * (-tmp_trans);
114 }
115 }
116 else {
117 // longitudinal magnetic field
118#pragma omp parallel for default(none) private(j, tmp_sgn) firstprivate(i_max, isite1, org_sigma1, X,tmp_trans) shared(tmp_v0, tmp_v1)
119 for (j = 1; j <= i_max; j++) {
120 tmp_v0[j] += child_SpinGC_CisAis(j, X, isite1, org_sigma1) * tmp_v1[j] * tmp_trans;
121 }
122 }
123 }else{
124 // transverse magnetic field
125 // fprintf(stdoutMPI, "Debug: isite1=%d, org_sigma2=%d\n", isite1, org_sigma2);
126#pragma omp parallel for default(none) private(j, tmp_sgn, tmp_off) firstprivate(i_max, isite1, org_sigma2, X, tmp_trans) shared(tmp_v0, tmp_v1)
127 for(j=1;j<=i_max;j++){
128 tmp_sgn = child_SpinGC_CisAit(j,X, isite1,org_sigma2,&tmp_off);
129 if(tmp_sgn !=0){
130 tmp_v0[tmp_off+1]+= tmp_sgn*tmp_v1[j]*tmp_trans;
131 }
132 }
133 }
134 }
135 }else{
136 fprintf(stdoutMPI, "ERROR: hopping is not allowed in localized spin system\n");
137 return FALSE;
138 }
139 }
140 return TRUE;
141}

152int GetPairExcitedStateGeneralSpinGC(
153 struct BindStruct *X,

155 int **PairExcitationOperator,

157 double complex *tmp_v0,

207 }
208 }
209 }else{
210 // transverse magnetic field

216 }
217 }
218 }
219 }
220 }else{
221 fprintf(stdoutMPI, "ERROR: hopping is not allowed in localized spin system\n");
222 return FALSE;
223 }
224 }
225 return TRUE;

267int GetPairExcitedStateHalfSpin(
268 struct BindStruct *X,
269 unsigned int NPairExcitationOperator,
270 int **PairExcitationOperator,
271 double complex *ParaPairExcitationOperator,
272 double complex *tmp_v0,
273 double complex *tmp_v1
275)
276{
277 long unsigned int i,j, idim_maxMPI;
278 long unsigned int isite1;
279 long unsigned int org_isite1,org_isite2,org_sigma1,org_sigma2;
280 long unsigned int tmp_off=0;
281
282 double complex tmp_trans=0;
283 long int i_max;
284 int num1;
285 long int ibit1;
286 long unsigned int is1_up;
287
288 i_max = X->Check.idim_maxOrg;
289
290 double complex *tmp_v1bufOrg;
291 //set size
292#ifdef MPI
293 idim_maxMPI = MaxMPI_li(X->Check.idim_maxOrg);
294 tmp_v1bufOrg=cd_1d_allocate(idim_maxMPI + 1);
295#endif // MPI
296
297 for (i = 0; i < NPairExcitationOperator; i++) {
298 org_isite1 = PairExcitationOperator[i][0] + 1;
299 org_isite2 = PairExcitationOperator[i][2] + 1;
300 org_sigma1 = PairExcitationOperator[i][1];
301 org_sigma2 = PairExcitationOperator[i][3];
302 tmp_trans = ParaPairExcitationOperator[i];
303 if (org_sigma1 == org_sigma2) {
304 if (org_isite1 == org_isite2) {
305 if (org_isite1 > X->Def.Nsite) {
306 is1_up = X->Def.Tpow[org_isite1 - 1];
307 ibit1 = child_SpinGC_CisAis((unsigned long int) myrank + 1, X, is1_up, org_sigma1);
308 if (PairExcitationOperator[i][4] == 0) {
309 if (ibit1 == 0) {
310#pragma omp parallel for default(none) shared(tmp_v0, tmp_v1) \
311 firstprivate(i_max, tmp_trans) private(j)
312 for (j = 1; j <= i_max; j++) tmp_v0[j] += -tmp_trans * tmp_v1[j];
313 }
314 } else {
315 if (ibit1 != 0) {
316#pragma omp parallel for default(none) shared(tmp_v0, tmp_v1) \
317 firstprivate(i_max, tmp_trans) private(j)
318 for (j = 1; j <= i_max; j++) tmp_v0[j] += tmp_trans * tmp_v1[j];
319 }
320 }
321 }// org_isite1 > X->Def.Nsite
322 else {
323 isite1 = X->Def.Tpow[org_isite1 - 1];
324 if (org_isite1 == org_isite2 && org_sigma1 == org_sigma2 &&
325 PairExcitationOperator[i][4] == 0) {
326#pragma omp parallel for default(none) private(j) firstprivate(i_max, isite1, org_sigma1, X, tmp_trans) shared(tmp_v0, tmp_v1)
327 for (j = 1; j <= i_max; j++) {
328 tmp_v0[j] += (1.0 - child_Spin_CisAis(j, X, isite1, org_sigma1)) * tmp_v1[j] * (-tmp_trans);
329 }
330 } else {
331#pragma omp parallel for default(none) private(j) firstprivate(i_max, isite1, org_sigma1, X, tmp_trans) shared(tmp_v0, tmp_v1)
332 for (j = 1; j <= i_max; j++) {
333 tmp_v0[j] += child_Spin_CisAis(j, X, isite1, org_sigma1) * tmp_v1[j] * tmp_trans;
334 }
335 }
336 }
337 } else {
338 fprintf(stdoutMPI, "Error: isite1 must be equal to isite2 for Spin system. \n");
339 return FALSE;
340 }
341 } else { //org_sigma1 != org_sigma2 // for the canonical case
342 if (org_isite1 > X->Def.Nsite) {//For MPI
343 child_CisAit_spin_MPIdouble(org_isite1-1, org_sigma2, tmp_trans, X, tmp_v0, tmp_v1, tmp_v1bufOrg, i_max, X->Def.Tpow,list_1_org, list_1buf_org, list_2_1, list_2_2, X->Large.irght, X->Large.ilft,X->Large.ihfbit);
344
345 } else {
346 isite1 = X->Def.Tpow[org_isite1 - 1];
347#pragma omp parallel for default(none) private(j, tmp_off, num1) \
348 firstprivate(i_max, isite1, org_sigma2, X, tmp_trans, list_1_org, list_1, list_2_1, list_2_2) shared(tmp_v0, tmp_v1)
349 for (j = 1; j <= i_max; j++) {
350 num1=child_Spin_CisAit(j, X, isite1, org_sigma2, list_1_org, list_2_1, list_2_2, &tmp_off);
351 if (num1 != 0) tmp_v0[tmp_off] += tmp_v1[j] * tmp_trans*(double)num1;
352 }
353 }
354 }
355 }
356#ifdef MPI
357 free_cd_1d_allocate(tmp_v1bufOrg);
358#endif
359 return TRUE;
360}

470 }//org_isite1 != org_isite2
471 }
472#ifdef MPI
473 free_cd_1d_allocate(tmp_v1bufOrg);
474#endif // MPI
475
476 return TRUE;
477}
