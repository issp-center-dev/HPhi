335 fprintf(stdoutMPI, cErrDefFileParam, defname, ctmp);
336 return(-1);
337 }
338 }

340
341 /* Check values*/
342 if(ValidateValue(X->iCalcModel, 0, NUM_CALCMODEL-1)){
343 fprintf(stdoutMPI,cErrCalcModel, defname);
344 return (-1);
345 }

347 X->iCalcModel == tJNConserved ||
348 X->iCalcModel == KondoNConserved){

356 fprintf(stdoutMPI, cErrCalcType, defname);
357 return (-1);
358 }
359 if(ValidateValue(X->iOutputMode, 0, NUM_OUTPUTMODE-1)){
360 fprintf(stdoutMPI, cErrOutputMode, defname);
361 return (-1);
362 }
363

365 fprintf(stdoutMPI, cErrCalcEigenVec, defname);
366 return (-1);
367 }
368
369 if(ValidateValue(X->iInitialVecType, -1, NUM_SETINITAILVEC-1)){
370 fprintf(stdoutMPI, cErrSetIniVec, defname);
371 return (-1);
372 }
373

375 fprintf(stdoutMPI, cErrOutputHam, defname);
376 return (-1);
377 }

379 fprintf(stdoutMPI, cErrInputHam, defname);
380 return (-1);
381 }
382 if(X->iInputHam == 1 && X->iOutputHam==1){
383 fprintf(stdoutMPI, cErrInputOutputHam, defname);
384 return (-1);
385 }

390 if(X->iNGPU < 0){
391 fprintf(stdoutMPI, cErrCUDA, defname);
392 return (-1);
393 }

395 fprintf(stdoutMPI, cErrCUDA, defname);
396 return (-1);
397 }
398
399 /* In the case of Full Diagonalization method(iCalcType=2)*/
400 if(X->iCalcType==2 && ValidateValue(X->iFlgFiniteTemperature, 0, 1)){
401 fprintf(stdoutMPI, cErrFiniteTemp, defname);
402 return (-1);
403 }
404
405 if(X->iCalcType !=2 && X->iOutputHam ==TRUE) {
406 fprintf(stdoutMPI, cErrOutputHamForFullDiag, defname);
407 return (-1);
408 }
409
410 return 0;

423int GetFileName(
424 const char* cFileListNameFile,
425 char cFileNameList[][D_CharTmpReadDef]
426 )
427{
428 FILE *fplist;
429 int itmpKWidx=-1;
430 char ctmpFileName[D_FileNameMaxReadDef];
431 char ctmpKW[D_CharTmpReadDef], ctmp2[256];
432 int i;
433 for(i=0; i< D_iKWNumDef; i++){
434 strcpy(cFileNameList[i],"");
435 }
436
437 fplist = fopenMPI(cFileListNameFile, "r");
438 if(fplist==NULL) return ReadDefFileError(cFileListNameFile);
439
440 while(fgetsMPI(ctmp2, 256, fplist) != NULL){
441 memset(ctmpKW, '\0', strlen(ctmpKW));
442 memset(ctmpFileName, '\0', strlen(ctmpFileName));
443 sscanf(ctmp2,"%s %s\n", ctmpKW, ctmpFileName);
444
445 if(strncmp(ctmpKW, "#", 1)==0 || *ctmp2=='\n' || (strcmp(ctmpKW, "")&&strcmp(ctmpFileName,""))==0){
446 continue;
447 }
448 else if(strcmp(ctmpKW, "")*strcmp(ctmpFileName, "")==0){
449 fprintf(stdoutMPI, cErrKW_InCorPair, cFileListNameFile);
450 fclose(fplist);
451 return(-1);
452 }
454 if( CheckKW(ctmpKW, cKWListOfFileNameList, D_iKWNumDef, &itmpKWidx)!=0 ){
455 fprintf(stdoutMPI, cErrKW, ctmpKW, cFileListNameFile);
456 fprintf(stdoutMPI, "%s", cErrKW_ShowList);
457 for(i=0; i<D_iKWNumDef;i++){
458 fprintf(stdoutMPI, "%s \n", cKWListOfFileNameList[i]);
459 }
460 fclose(fplist);
461 return(-1);
462 }
464 if(strcmp(cFileNameList[itmpKWidx], "") !=0){
465 fprintf(stdoutMPI, cErrKW_Same, cFileListNameFile);
466 fclose(fplist);
467 return(-1);
468 }
469
471 strcpy(cFileNameList[itmpKWidx], ctmpFileName);
472 }
473 fclose(fplist);
474 return 0;
475}

484static char *ReadNBodyLineMPI(FILE *fp, size_t *len)
485{
486 char *line = NULL;
487 size_t cap = 0;
488 size_t n = 0;
489 unsigned long blen = 0;
490
491 if (myrank == 0) {
492 for (;;) {
493 int ch;
494 n = 0;
495 if (cap == 0) {
496 cap = 256;
497 line = (char *)malloc(cap);
498 if (line == NULL) {
499 fprintf(stdoutMPI, "Error: Failed to allocate NBody line buffer.\n");
500 exitMPI(-1);
501 }
502 }
503 while ((ch = fgetc(fp)) != EOF) {
504 if (n + 2 > cap) {
505 char *tmp;
506 cap *= 2;
507 tmp = (char *)realloc(line, cap);
508 if (tmp == NULL) {
509 fprintf(stdoutMPI, "Error: Failed to grow NBody line buffer.\n");
510 free(line);
511 exitMPI(-1);
512 }
513 line = tmp;
514 }
515 line[n++] = (char)ch;
516 if (ch == '\n') break;
517 }
518 if (n == 0 && ch == EOF) {
519 free(line);
520 line = NULL;
521 blen = 0;
522 break;
523 }
524 line[n] = '\0';
525 if (IsSkippableNBodyLine(line)) continue;
526 blen = (unsigned long)(n + 1);
527 break;
528 }
529 }
530
531#ifdef MPI
532 if (blen > (unsigned long)INT_MAX) {
533 fprintf(stdoutMPI, "Error: NBody line is too long for MPI broadcast.\n");
534 free(line);
535 exitMPI(-1);
536 }
537 if (MPI_Bcast(&blen, 1, MPI_UNSIGNED_LONG, 0, MPI_COMM_WORLD) != 0) exitMPI(-1);
538 if (blen == 0) {
539 if (len != NULL) *len = 0;
540 return NULL;
541 }
542 if (myrank != 0) {
543 line = (char *)malloc((size_t)blen);
544 if (line == NULL) exitMPI(-1);
545 }
546 if (MPI_Bcast(line, (int)blen, MPI_CHAR, 0, MPI_COMM_WORLD) != 0) exitMPI(-1);
547#else
548 if (blen == 0) {
549 if (len != NULL) *len = 0;
550 return NULL;
551 }
552#endif
553 if (len != NULL) *len = (size_t)blen - 1;
554 return line;
555}
