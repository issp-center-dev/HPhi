313 int GetlistSize
314 (
315 struct BindStruct *X
316 ) {
317 // unsigned int idim_maxMPI;
318
319// idim_maxMPI = MaxMPI_li(X->Check.idim_max);
320
321 switch (X->Def.iCalcModel) {
322 case Spin:
323 case SpinlessFermion:
324 case Hubbard:
325 case HubbardNConserved:
326 case Kondo:
327 case KondoNConserved:
328 case KondoGC:
329 case tJ:
330 case tJNConserved:
331 case tJGC:
332 if (X->Def.iFlgGeneralSpin == FALSE) {
333 if ((X->Def.iCalcModel == Spin || X->Def.iCalcModel == SpinlessFermion) && X->Def.Nsite % 2 == 1) {
334 X->Large.SizeOflist_2_1 = X->Check.sdim * 2 + 2;
335 } else {
336 X->Large.SizeOflist_2_1 = X->Check.sdim + 2;
337 }
338 X->Large.SizeOflist_2_2 = X->Check.sdim + 2;
339 X->Large.SizeOflistjb = X->Check.sdim + 2;
340 } else {//for spin-canonical general spin
341 X->Large.SizeOflist_2_1 = X->Check.sdim + 2;
342 X->Large.SizeOflist_2_2 =
343 X->Def.Tpow[X->Def.Nsite - 1] * X->Def.SiteToBit[X->Def.Nsite - 1] / X->Check.sdim + 2;
344 X->Large.SizeOflistjb =
345 X->Def.Tpow[X->Def.Nsite - 1] * X->Def.SiteToBit[X->Def.Nsite - 1] / X->Check.sdim + 2;
346 }
347 break;
348 default:
349 return FALSE;
350 }
351 return TRUE;
