45static int IsFiniteComplex(double complex z) {
46 return isfinite(creal(z)) && isfinite(cimag(z));
47}

49static void AnalyzeBiCGVector(const double complex *v, unsigned long n, BiCGVectorDiag *diag) {
50 unsigned long i;
51 diag->sum2 = 0.0;
52 diag->nbad = 0;
53 diag->nsample = 0;
54 for (i = 0; i < n; i++) {
55 const double real = creal(v[i]);
56 const double imag = cimag(v[i]);
57 if (IsFiniteComplex(v[i])) {
58 diag->sum2 += real * real + imag * imag;
59 } else {
60 if (diag->nsample < BICG_DIAG_SAMPLES) {
61 diag->sample_index[diag->nsample] = i + 1;
62 diag->sample_real[diag->nsample] = real;
63 diag->sample_imag[diag->nsample] = imag;
64 diag->nsample++;
65 }
66 diag->nbad++;
67 }
68 }
69}

71static void PrintBiCGVectorDiag(const char *name, const BiCGVectorDiag *diag) {
72 int i;
73 fprintf(stderr, "%s_sum2=%25.17e %s_nbad=%lu", name, diag->sum2, name, diag->nbad);
74 for (i = 0; i < diag->nsample; i++) {
75 fprintf(stderr, " %s_bad[%d]=(idx=%lu,value=%25.17e,%25.17e)",
76 name, i, diag->sample_index[i], diag->sample_real[i], diag->sample_imag[i]);
77 }
