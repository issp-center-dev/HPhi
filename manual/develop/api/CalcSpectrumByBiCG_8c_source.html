50static int IsFiniteComplex(double complex z) {
51 return isfinite(creal(z)) && isfinite(cimag(z));
52}

54static void AnalyzeBiCGVector(const double complex *v, unsigned long n, BiCGVectorDiag *diag) {
55 unsigned long i;
56 diag->sum2 = 0.0;
57 diag->sum_real = 0.0;
58 diag->sum_imag = 0.0;
59 diag->weighted_sum_real = 0.0;
60 diag->weighted_sum_imag = 0.0;
61 diag->nbad = 0;
62 diag->nsample = 0;
63 for (i = 0; i < n; i++) {
64 const double real = creal(v[i]);
65 const double imag = cimag(v[i]);
66 if (IsFiniteComplex(v[i])) {
67 const double weight = (double)(i + 1);
68 diag->sum2 += real * real + imag * imag;
69 diag->sum_real += real;
70 diag->sum_imag += imag;
71 diag->weighted_sum_real += weight * real;
72 diag->weighted_sum_imag += weight * imag;
73 } else {
74 if (diag->nsample < BICG_DIAG_SAMPLES) {
75 diag->sample_index[diag->nsample] = i + 1;
76 diag->sample_real[diag->nsample] = real;
77 diag->sample_imag[diag->nsample] = imag;
78 diag->nsample++;
79 }
80 diag->nbad++;
81 }
82 }
83}

185static double BiCGResidualRatio(double residual, double initial_residual) {
186 if (isfinite(residual) == FALSE || isfinite(initial_residual) == FALSE ||
187 initial_residual <= 0.0) {
188 return NAN;
189 }
190 return residual / initial_residual;
191}

214static void PrintBiCGStatusTrace(
215 FILE *fp,
216 int stp,
217 const int status[3],
218 double unshifted_residual,
219 double initial_residual,
220 double max_unshifted_residual,
221 int first_spike_iter,
222 int Nomega,
223 const double *shifted_residual
224) {
225 int iomega;
226 double ratio = BiCGResidualRatio(max_unshifted_residual, initial_residual);
227 double shifted_max = NAN;
228 double shifted_min = NAN;
229 int n_below_threshold = 0;
230 int n_nonfinite = 0;
231 for (iomega = 0; iomega < Nomega; iomega++) {
232 const double res = shifted_residual == NULL ? NAN : shifted_residual[iomega];
233 if (isfinite(res) == FALSE) {
234 n_nonfinite++;
235 continue;
236 }
237 if (isfinite(shifted_max) == FALSE || res > shifted_max) shifted_max = res;
238 if (isfinite(shifted_min) == FALSE || res < shifted_min) shifted_min = res;
239 if (res < eps_Lanczos) n_below_threshold++;
240 }
241 fprintf(fp,
242 "%7d %7d %7d %25.17e %25.17e %25.17e %25.17e %7d "
243 "%25.17e %25.17e %7d %7d\n",
244 stp, status[1], status[2], unshifted_residual,
245 initial_residual, max_unshifted_residual, ratio, first_spike_iter,
246 shifted_max, shifted_min, n_below_threshold, n_nonfinite);
247}

270void ReadTMComponents_BiCG(
271 struct EDMainCalStruct *X,

281 char sdt[D_FileNameMax];
282 char ctmp[256];
283
284 int status[3], idim_max2int, max_step, iter_old;
285 unsigned long int idx;
286 double complex *alphaCG, *betaCG, *res_save, z_seed;
287 double z_seed_r, z_seed_i, alpha_r, alpha_i, beta_r, beta_i, res_r, res_i;
288 FILE *fp;
289 int comm;
290
291#if defined(MPI)
292 comm = MPI_Comm_c2f(MPI_COMM_WORLD);
293#else
294 comm = 0;
295#endif
296 idim_max2int = (int)X->Bind.Check.idim_max;
297
298 if (X->Bind.Def.iFlgCalcSpec == RECALC_FROM_TMComponents ||
299 X->Bind.Def.iFlgCalcSpec == RECALC_FROM_TMComponents_VEC ||
300 X->Bind.Def.iFlgCalcSpec == RECALC_INOUT_TMComponents_VEC) {
301 sprintf(sdt, cFileNameTridiagonalMatrixComponents, X->Bind.Def.CDataFileHead);
302 if (childfopenALL(sdt, "rb", &fp) != 0) {
303 fprintf(stdoutMPI, "INFO: File for the restart is not found.\n");
304 fprintf(stdoutMPI, " Start from SCRATCH.\n");
305 max_step = (int)X->Bind.Def.Lanczos_max;
306 komega_bicg_init(&idim_max2int, &nBra, &Nomega, dcSpectrum, dcomega, &max_step, &eps_Lanczos, &comm);
307 }
308 else {
309 fgetsMPI(ctmp, sizeof(ctmp) / sizeof(char), fp);
310 sscanf(ctmp, "%d", &iter_old);
311 if (X->Bind.Def.iFlgCalcSpec > RECALC_FROM_TMComponents) {
312 alphaCG = (double complex*)malloc((iter_old + X->Bind.Def.Lanczos_max) * sizeof(double complex));
313 betaCG = (double complex*)malloc((iter_old + X->Bind.Def.Lanczos_max) * sizeof(double complex));
314 res_save = (double complex*)malloc((iter_old + X->Bind.Def.Lanczos_max) * sizeof(double complex));
315 }
316 else {
317 alphaCG = (double complex*)malloc(iter_old * sizeof(double complex));
318 betaCG = (double complex*)malloc(iter_old * sizeof(double complex));
319 res_save = (double complex*)malloc(iter_old * sizeof(double complex));
320 }
321 fgetsMPI(ctmp, sizeof(ctmp) / sizeof(char), fp);
322 sscanf(ctmp, "%lf %lf\n", &z_seed_r, &z_seed_i);
323 z_seed = z_seed_r + I * z_seed_i;
324
325 idx = 0;
326 while (fgetsMPI(ctmp, sizeof(ctmp) / sizeof(char), fp) != NULL) {
327 sscanf(ctmp, "%lf %lf %lf %lf %lf %lf\n",
328 &alpha_r, &alpha_i, &beta_r, &beta_i, &res_r, &res_i);
329 alphaCG[idx] = alpha_r + I * alpha_i;
330 betaCG[idx] = beta_r + I * beta_i;
331 res_save[idx] = res_r + I * res_i;
332 idx += 1;
333 }
334 fclose(fp);
335
336 if (X->Bind.Def.iFlgCalcSpec == RECALC_FROM_TMComponents) X->Bind.Def.Lanczos_max = 0;
337 max_step = (int)(iter_old + X->Bind.Def.Lanczos_max);
338
339 komega_bicg_restart(&idim_max2int, &nBra, &Nomega, dcSpectrum, dcomega, &max_step, &eps_Lanczos, status,
340 &iter_old, &v2[1], &v12[1], &v4[1], &v14[1], alphaCG, betaCG, &z_seed, res_save, &comm);
341 free(alphaCG);
342 free(betaCG);
343 free(res_save);
344 }/*if (childfopenALL(sdt, "rb", &fp) == 0)*/
345 }/*if (X->Bind.Def.iFlgCalcSpec > RECALC_NOT)*/
346 else {
347 max_step = (int)X->Bind.Def.Lanczos_max;
348 komega_bicg_init(&idim_max2int, &nBra, &Nomega, dcSpectrum, dcomega, &max_step, &eps_Lanczos, &comm);
349 }
