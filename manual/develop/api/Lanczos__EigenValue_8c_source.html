85int Lanczos_EigenValue(struct BindStruct *X) {
86
87 fprintf(stdoutMPI, "%s", cLogLanczos_EigenValueStart);
88 FILE *fp;
89 char sdt[D_FileNameMax], sdt_2[D_FileNameMax];
90 int stp;
91 long int i, i_max;
92 unsigned long int i_max_tmp;
93 int k_exct, Target;
94 int iconv = -1;
95 double beta1, alpha1; //beta,alpha1 should be real
96 double complex temp1, temp2;
97 double complex cbeta1;
98 double E[5], ebefor, E_target;
99
100// for GC
101 double dnorm=0.0;
102
103 double **tmp_mat;
104 double *tmp_E;
105 int int_i, int_j;
106 int iret=0;
107 sprintf(sdt_2, cFileNameLanczosStep, X->Def.CDataFileHead);
108
109 i_max = X->Check.idim_max;
110 k_exct = X->Def.k_exct;
111 unsigned long int liLanczosStp;
112 liLanczosStp = X->Def.Lanczos_max;
113 unsigned long int liLanczosStp_vec=0;

116 if(ReadInitialVector( X, v0, v1, &liLanczosStp_vec)!=0){

130 X->Def.Lanczos_restart=liLanczosStp;
131 //Calculate EigenValue

133 liLanczosStp = liLanczosStp+X->Def.Lanczos_max;
134 alpha1=alpha[X->Def.Lanczos_restart];
135 beta1=beta[X->Def.Lanczos_restart];
136 }/*X->Def.iReStart == RESTART_INOUT || X->Def.iReStart == RESTART_IN*/
137 else {
138 SetInitialVector(X, v0, v1);

150 cbeta1 = 0.0;

192 temp2 = (v0[i] - alpha1 * v1[i]) / beta1;

199 StopTimer(4101);

207 cbeta1 += conj(v0[i] - alpha1 * v1[i]) * (v0[i] - alpha1 * v1[i]);

218 tmp_E = d_1d_allocate(stp+1);

252 }

254 //if (stp > 2 && stp % 2 == 0) {
255 if (stp > 2) {
256 childfopenMPI(sdt_2, "a", &fp);
257 tmp_mat = d_2d_allocate(stp,stp);

276 StopTimer(4103);

278 E[2] = tmp_E[1];
279 E[3] = tmp_E[2];

282 if (stp > Target) {
283 E_target = tmp_E[Target];
284 }
285 free_d_1d_allocate(tmp_E);
286 free_d_2d_allocate(tmp_mat);
287 if (stp > Target) {
288 fprintf(stdoutMPI, " stp = %d %.10lf %.10lf %.10lf %.10lf %.10lf %.10lf\n", stp, E[1], E[2], E[3], E[4],
289 E_target, E[0] / (double) X->Def.NsiteMPI);
290 fprintf(fp, "stp=%d %.10lf %.10lf %.10lf %.10lf %.10lf %.10lf\n", stp, E[1], E[2], E[3], E[4], E_target,
291 E[0] / (double) X->Def.NsiteMPI);
292 } else {
293 fprintf(stdoutMPI, " stp = %d %.10lf %.10lf %.10lf %.10lf xxxxxxxxx %.10lf\n", stp, E[1], E[2], E[3], E[4],
294 E[0] / (double) X->Def.NsiteMPI);
295 fprintf(fp, "stp=%d %.10lf %.10lf %.10lf %.10lf xxxxxxxxx %.10lf\n", stp, E[1], E[2], E[3], E[4],
296 E[0] / (double) X->Def.NsiteMPI);
297 }
298 fclose(fp);
299 if (stp > Target) {
300 if (fabs((E_target - ebefor) / E_target) < eps_Lanczos || fabs(beta[stp]) < pow(10.0, -14)) {
301 /*
302 if(X->Def.iReStart == RESTART_INOUT ||X->Def.iReStart == RESTART_OUT){
303 break;
304 }
305 */
306 tmp_E = d_1d_allocate(stp+1);
307 StartTimer(4102);
308 vec12(alpha, beta, stp, tmp_E, X);
309 StopTimer(4102);
310 X->Large.itr = stp;
311 X->Phys.Target_energy = E_target;
312 X->Phys.Target_CG_energy = tmp_E[k_exct]; //for CG
313 iconv = 0;
314 free_d_1d_allocate(tmp_E);
315 break;
316 }
317 ebefor = E_target;
318 }
319
320 }
321 }
322 if (X->Def.iReStart == RESTART_INOUT ||X->Def.iReStart == RESTART_OUT ){
323 if(stp != X->Def.Lanczos_restart+2) { // 2 steps are needed to get the value: E[stp+2]-E[stp+1]
324 OutputTMComponents(X, alpha, beta, dnorm, stp - 1);
325 OutputLanczosVector(X, v0, v1, stp - 1);
326 }
327 if (iconv !=0){
328 sprintf(sdt, "%s", cLogLanczos_EigenValueNotConverged);
329 fprintf(stdoutMPI, "Lanczos Eigenvalue is not converged in this process (restart mode).\n");
330 return 1;
331 }
332 }
333
334 sprintf(sdt, cFileNameTimeKeep, X->Def.CDataFileHead);
335 if (iconv != 0) {
336 sprintf(sdt, "%s", cLogLanczos_EigenValueNotConverged);
337 fprintf(stdoutMPI, "Lanczos Eigenvalue is not converged in this process.\n");
338 return -1;
339 }
340
341 TimeKeeper(X, cFileNameTimeKeep, cLanczos_EigenValueFinish, "a");
342 fprintf(stdoutMPI, "%s", cLogLanczos_EigenValueEnd);
343
344 return 0;
345}

359int Lanczos_GetTridiagonalMatrixComponents(
360 struct BindStruct *X,
361 double *_alpha,
362 double *_beta,
363 double complex *tmp_v1,
364 unsigned long int *liLanczos_step
365 ) {
366
367 char sdt[D_FileNameMax];
368 int stp;
369 long int i, i_max;
370 i_max = X->Check.idim_max;
371
372 unsigned long int i_max_tmp;
373 double beta1, alpha1; //beta,alpha1 should be real
374 double complex temp1, temp2;
375 double complex cbeta1;
376
377 sprintf(sdt, cFileNameLanczosStep, X->Def.CDataFileHead);
378
379 /*
380 Set Maximum number of loop to the dimension of the Wavefunction
381 */
382 i_max_tmp = SumMPI_li(i_max);
383 if (i_max_tmp < *liLanczos_step || i_max_tmp < X->Def.LanczosTarget) {
384 *liLanczos_step = i_max_tmp;
385 }
386
387 if (X->Def.Lanczos_restart == 0) { // initial procedure (not restart)
388#pragma omp parallel for default(none) private(i) shared(v0, v1, tmp_v1) firstprivate(i_max)
389 for (i = 1; i <= i_max; i++) {
390 v0[i] = 0.0;
391 v1[i] = tmp_v1[i];
392 }
393 stp = 0;
394 mltply(X, v0, tmp_v1);
395 TimeKeeperWithStep(X, cFileNameTimeKeep, c_Lanczos_SpectrumStep, "a", stp);
396 alpha1 = creal(X->Large.prdct);// alpha = v^{\dag}*H*v
397 _alpha[1] = alpha1;
398 cbeta1 = 0.0;
399 fprintf(stdoutMPI, " Step / Step_max alpha beta \n");
400
401#pragma omp parallel for reduction(+:cbeta1) default(none) private(i) shared(v0, v1) firstprivate(i_max, alpha1)
402 for (i = 1; i <= i_max; i++) {
403 cbeta1 += conj(v0[i] - alpha1 * v1[i]) * (v0[i] - alpha1 * v1[i]);
404 }
405 cbeta1 = SumMPI_dc(cbeta1);
406 beta1 = creal(cbeta1);
407 beta1 = sqrt(beta1);
408 _beta[1] = beta1;
409 X->Def.Lanczos_restart = 1;
410 } else { // restart case
411 alpha1 = alpha[X->Def.Lanczos_restart];
412 beta1 = beta[X->Def.Lanczos_restart];
413 }
414
415 for (stp = X->Def.Lanczos_restart + 1; stp <= *liLanczos_step; stp++) {
416
417 if (fabs(_beta[stp - 1]) < pow(10.0, -14)) {
418 *liLanczos_step = stp - 1;
419 break;
420 }
421
422#pragma omp parallel for default(none) private(i, temp1, temp2) shared(v0, v1) firstprivate(i_max, alpha1, beta1)
423 for (i = 1; i <= i_max; i++) {
424 temp1 = v1[i];
425 temp2 = (v0[i] - alpha1 * v1[i]) / beta1;
426 v0[i] = -beta1 * temp1;
427 v1[i] = temp2;
428 }
429
430 mltply(X, v0, v1);
431 TimeKeeperWithStep(X, cFileNameTimeKeep, c_Lanczos_SpectrumStep, "a", stp);
432 alpha1 = creal(X->Large.prdct);
433 _alpha[stp] = alpha1;
434 cbeta1 = 0.0;
435
436#pragma omp parallel for reduction(+:cbeta1) default(none) private(i) shared(v0, v1) firstprivate(i_max, alpha1)
437 for (i = 1; i <= i_max; i++) {
438 cbeta1 += conj(v0[i] - alpha1 * v1[i]) * (v0[i] - alpha1 * v1[i]);
439 }
440 cbeta1 = SumMPI_dc(cbeta1);
441 beta1 = creal(cbeta1);
442 beta1 = sqrt(beta1);
443 _beta[stp] = beta1;
444 if(stp %10 == 0) {
445 fprintf(stdoutMPI, " stp = %d / %lu %.10lf %.10lf \n", stp, *liLanczos_step, alpha1, beta1);
446 }
447 }
448
449 return TRUE;
450}

534void SetInitialVector(struct BindStruct *X, double complex* tmp_v0, double complex *tmp_v1) {
535 int iproc;
536 long int i, iv, i_max;
537 unsigned long int i_max_tmp, sum_i_max;
538 int mythread;
539
540// for GC
541 double dnorm;
542 double complex cdnorm;
543 long unsigned int u_long_i;
544 dsfmt_t dsfmt;
545
546 i_max = X->Check.idim_max;
547 if (initial_mode == 0) {
548 if(X->Def.iFlgMPI==0) {
549 sum_i_max = SumMPI_li(X->Check.idim_max);
550 }
551 else{
552 sum_i_max =X->Check.idim_max;
553 }
554 X->Large.iv = (sum_i_max / 2 + X->Def.initial_iv) % sum_i_max + 1;
555 iv = X->Large.iv;
556 fprintf(stdoutMPI, " initial_mode=%d normal: iv = %ld i_max=%ld k_exct =%d \n\n", initial_mode, iv, i_max,
557 X->Def.k_exct);
558#pragma omp parallel for default(none) private(i) shared(tmp_v0, tmp_v1) firstprivate(i_max)
559 for (i = 1; i <= i_max; i++) {
560 tmp_v0[i] = 0.0;
561 tmp_v1[i] = 0.0;
562 }
563
564 sum_i_max = 0;
565 if(X->Def.iFlgMPI==0) {
566 for (iproc = 0; iproc < nproc; iproc++) {
567 i_max_tmp = BcastMPI_li(iproc, i_max);
568 if (sum_i_max <= iv && iv < sum_i_max + i_max_tmp) {
569 if (myrank == iproc) {
570 tmp_v1[iv - sum_i_max + 1] = 1.0;
571 if (X->Def.iInitialVecType == 0) {
572 tmp_v1[iv - sum_i_max + 1] += 1.0 * I;
573 tmp_v1[iv - sum_i_max + 1] /= sqrt(2.0);
574 }
575 }/*if (myrank == iproc)*/
576 }/*if (sum_i_max <= iv && iv < sum_i_max + i_max_tmp)*/
577
578 sum_i_max += i_max_tmp;
579
580 }/*for (iproc = 0; iproc < nproc; iproc++)*/
581 }
582 else {
583 tmp_v1[iv + 1] = 1.0;
584 if (X->Def.iInitialVecType == 0) {
585 tmp_v1[iv + 1] += 1.0 * I;
586 tmp_v1[iv + 1] /= sqrt(2.0);
587 }
588 }
589 }/*if(initial_mode == 0)*/
590 else if (initial_mode == 1) {
591 iv = X->Def.initial_iv;
592 fprintf(stdoutMPI, " initial_mode=%d (random): iv = %ld i_max=%ld k_exct =%d \n\n", initial_mode, iv, i_max,
593 X->Def.k_exct);
594#pragma omp parallel default(none) private(i, u_long_i, mythread, dsfmt) \
595 shared(tmp_v0, tmp_v1, iv, X, nthreads, myrank) firstprivate(i_max)
596 {
597
598#pragma omp for
599 for (i = 1; i <= i_max; i++) {
600 tmp_v0[i] = 0.0;
601 }
602 /*
603 Initialise MT
604 */
605#ifdef _OPENMP
606 mythread = omp_get_thread_num();
607#else
608 mythread = 0;
609#endif
610 if(X->Def.iFlgMPI==0) {
611 u_long_i = 123432 + labs(iv) + mythread + nthreads * myrank;
612 }
613 else{
614 u_long_i = 123432 + labs(iv)+ mythread ;
615 }
616 dsfmt_init_gen_rand(&dsfmt, u_long_i);
617
618 if (X->Def.iInitialVecType == 0) {
619#pragma omp for
620 for (i = 1; i <= i_max; i++)
621 tmp_v1[i] = 2.0 * (dsfmt_genrand_close_open(&dsfmt) - 0.5) +
622 2.0 * (dsfmt_genrand_close_open(&dsfmt) - 0.5) * I;
623 } else {
624#pragma omp for
625 for (i = 1; i <= i_max; i++)
626 tmp_v1[i] = 2.0 * (dsfmt_genrand_close_open(&dsfmt) - 0.5);
627 }
628
629 }/*#pragma omp parallel*/
630
631 cdnorm = 0.0;
632#pragma omp parallel for default(none) private(i) shared(tmp_v1, i_max) reduction(+: cdnorm)
633 for (i = 1; i <= i_max; i++) {
634 cdnorm += conj(tmp_v1[i]) * tmp_v1[i];
635 }
636 if(X->Def.iFlgMPI==0) {
637 cdnorm = SumMPI_dc(cdnorm);
638 }
639 dnorm = creal(cdnorm);
640 dnorm = sqrt(dnorm);
