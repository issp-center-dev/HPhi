90int CalcSpectrum(
91 struct EDMainCalStruct *X
92 ) {
93 char sdt[D_FileNameMax];
94 char *defname;
95 unsigned long int i;
96 unsigned long int i_max = 0;
97 int i_stp;
98 int iFlagListModified = FALSE;
99 FILE *fp;
100 double dnorm = 0.0;
101
102 //ToDo: Nomega should be given as a parameter
103 int Nomega;
104 double complex OmegaMax, OmegaMin;
105 double complex *dcSpectrum;
106 double complex *dcomega;
107 size_t byte_size;
108
109 //set omega
110 if (SetOmega(&(X->Bind.Def)) != TRUE) {
111 fprintf(stderr, "Error: Fail to set Omega.\n");
112 exitMPI(-1);
113 } else {
114 if (X->Bind.Def.iFlgSpecOmegaOrg == FALSE) {
115 X->Bind.Def.dcOmegaOrg = I*(X->Bind.Def.dcOmegaMax - X->Bind.Def.dcOmegaMin) / (double) X->Bind.Def.iNOmega;
116 }
117 }
118 /*
119 Set & malloc omega grid
120 */
121 Nomega = X->Bind.Def.iNOmega;
122 dcSpectrum = cd_1d_allocate(Nomega);
123 dcomega = cd_1d_allocate(Nomega);
124 OmegaMax = X->Bind.Def.dcOmegaMax + X->Bind.Def.dcOmegaOrg;
125 OmegaMin = X->Bind.Def.dcOmegaMin + X->Bind.Def.dcOmegaOrg;
126 for (i = 0; i < Nomega; i++) {
127 dcomega[i] = (OmegaMax - OmegaMin) / Nomega * i + OmegaMin;
128 }
129
130 fprintf(stdoutMPI, "\nFrequency range:\n");
131 fprintf(stdoutMPI, " Omega Max. : %15.5e %15.5e\n", creal(OmegaMax), cimag(OmegaMax));
132 fprintf(stdoutMPI, " Omega Min. : %15.5e %15.5e\n", creal(OmegaMin), cimag(OmegaMin));
133 fprintf(stdoutMPI, " Num. of Omega : %d\n", Nomega);
134
135 if (X->Bind.Def.NSingleExcitationOperator == 0 && X->Bind.Def.NPairExcitationOperator == 0) {
136 fprintf(stderr, "Error: Any excitation operators are not defined.\n");
137 exitMPI(-1);
138 }
139 //Make New Lists
140 if (MakeExcitedList(&(X->Bind), &iFlagListModified) == FALSE) {
141 return FALSE;
142 }
143 X->Bind.Def.iFlagListModified = iFlagListModified;
144
145 //Set Memory
146 v1Org = cd_1d_allocate(X->Bind.Check.idim_maxOrg+1);
147 for(i=0; i<X->Bind.Check.idim_maxOrg+1; i++){
148 v1Org[i]=0;
149 }
150
151 //Make excited state
152 StartTimer(6100);
153 if (X->Bind.Def.iFlgCalcSpec == RECALC_NOT ||
154 X->Bind.Def.iFlgCalcSpec == RECALC_OUTPUT_TMComponents_VEC ||
155 (X->Bind.Def.iFlgCalcSpec == RECALC_INOUT_TMComponents_VEC && X->Bind.Def.iCalcType == CG)) {
156 //input eigen vector
157 StartTimer(6101);
158 fprintf(stdoutMPI, " Start: An Eigenvector is inputted in CalcSpectrum.\n");
159 TimeKeeper(&(X->Bind), cFileNameTimeKeep, c_InputEigenVectorStart, "a");
160 GetFileNameByKW(KWSpectrumVec, &defname);
161 strcat(defname, "_rank_%d.dat");
162// sprintf(sdt, cFileNameInputEigen, X->Bind.Def.CDataFileHead, X->Bind.Def.k_exct - 1, myrank);
163 sprintf(sdt, defname, myrank);
164 childfopenALL(sdt, "rb", &fp);
165
166 if (fp == NULL) {
167 fprintf(stderr, "Error: A file of Input vector does not exist.\n");
168 return -1;
169 }
170
171 byte_size = fread(&i_stp, sizeof(i_stp), 1, fp);
172 X->Bind.Large.itr = i_stp; //For TPQ
173 byte_size = fread(&i_max, sizeof(i_max), 1, fp);
174 if (i_max != X->Bind.Check.idim_maxOrg) {
175 fprintf(stderr, "Error: myrank=%d, i_max=%ld\n", myrank, i_max);
176 fprintf(stderr, "Error: A file of Input vector is incorrect.\n");
177 return -1;
178 }
179 byte_size = fread(v1Org, sizeof(complex double), i_max + 1, fp);
180 fclose(fp);
181 StopTimer(6101);
182 if (byte_size == 0) printf("byte_size: %d \n", (int) byte_size);
183
184 for (i = 0; i <= X->Bind.Check.idim_max; i++) {
185 v0[i] = 0;
186 }
187 fprintf(stdoutMPI, " End: An Input vector is inputted in CalcSpectrum.\n\n");
188 TimeKeeper(&(X->Bind), cFileNameTimeKeep, c_InputEigenVectorEnd, "a");
189 TimeKeeper(&(X->Bind), cFileNameTimeKeep, c_CalcExcitedStateStart, "a");
190 fprintf(stdoutMPI, " Start: Calculating an excited vector.\n");
191
192 //Multiply Operator
193 StartTimer(6102);
194 GetExcitedState(&(X->Bind), v0, v1Org);
195 StopTimer(6102);
196
197 //calculate norm
198 dnorm = NormMPI_dc(X->Bind.Check.idim_max, v0);
199 if (fabs(dnorm) < pow(10.0, -15)) {
200 fprintf(stderr, "Warning: Norm of an excited vector becomes 0.\n");
201 fprintf(stdoutMPI, " End: Calculating an excited vector.\n\n");
202 TimeKeeper(&(X->Bind), cFileNameTimeKeep, c_CalcExcitedStateEnd, "a");
203 fprintf(stdoutMPI, " End: Calculating a spectrum.\n\n");
204 TimeKeeper(&(X->Bind), cFileNameTimeKeep, c_CalcSpectrumEnd, "a");
205 for (i = 0; i < Nomega; i++) {
206 dcSpectrum[i] = 0;
207 }
208 OutputSpectrum(X, Nomega, dcSpectrum, dcomega);
209 return TRUE;
210 }
211 //normalize vector
212#pragma omp parallel for default(none) private(i) shared(v1, v0) firstprivate(i_max, dnorm, X)
213 for (i = 1; i <= X->Bind.Check.idim_max; i++) {
214 v1[i] = v0[i] / dnorm;
215 }
216
217 //Output excited vector
218 if (X->Bind.Def.iOutputExVec == 1) {
219 fprintf(stdoutMPI, " Start: Output an excited vector.\n\n");
220 sprintf(sdt, cFileNameOutputExcitedVec, X->Bind.Def.CDataFileHead, myrank);
221 if(childfopenALL(sdt, "w", &fp)!=0){
222 return -1;
223 }
224 fprintf(fp, "%ld\n", X->Bind.Check.idim_max);
225 for (i = 1; i <= X->Bind.Check.idim_max; i++) {
226 fprintf(fp, "%.10lf, %.10lf\n", creal(v0[i]), cimag(v0[i]));
227 }
228 fclose(fp);
229 fprintf(stdoutMPI, " End: Output an excited vector.\n\n");
230 }
231
232 fprintf(stdoutMPI, " End: Calculating an excited vector.\n\n");
233 TimeKeeper(&(X->Bind), cFileNameTimeKeep, c_CalcExcitedStateEnd, "a");
234 }
235 StopTimer(6100);
236 //Reset list_1, list_2_1, list_2_2
237 if (iFlagListModified == TRUE) {
238 free(v1Org);
239 free(list_1_org);
240 free(list_2_1_org);
241 free(list_2_2_org);
242 }
243 //calculate Diagonal term
244 diagonalcalc(&(X->Bind));
245
246
247 int iret = TRUE;
248 fprintf(stdoutMPI, " Start: Calculating a spectrum.\n\n");
249 TimeKeeper(&(X->Bind), cFileNameTimeKeep, c_CalcSpectrumStart, "a");
250 StartTimer(6200);
251 switch (X->Bind.Def.iCalcType) {
252 case Lanczos:
253
254 iret = CalcSpectrumByLanczos(X, v1, dnorm, Nomega, dcSpectrum, dcomega);
255
256 if (iret != TRUE) {
257 //Error Message will be added.
258 return FALSE;
259 }
260
261 break;//Lanczos Spectrum
262
263 case CG:
264
265 iret = CalcSpectrumByBiCG(X, v0, v1, vg, Nomega, dcSpectrum, dcomega);
266
267 if (iret != TRUE) {
268 //Error Message will be added.
269 return FALSE;
270 }
271
272 break;//Lanczos Spectrum
273
274 case TPQCalc:
275 fprintf(stderr, " Error: TPQ is not supported for calculating spectrum mode.\n");
276 return FALSE;//TPQ is not supprted.
277#ifdef _CALCSPEC_TPQ
278 iret = CalcSpectrumByTPQ(X, v1, dnorm, Nomega, dcSpectrum, dcomega);
279 if (iret != TRUE) {
280 //Error Message will be added.
281 return FALSE;
282 }
283#endif
284
285 case FullDiag:
286 iret = CalcSpectrumByFullDiag(X, Nomega, dcSpectrum, dcomega);
287 break;
288
289 default:
290 break;
291 }
292 StopTimer(6200);
293
294 if (iret != TRUE) {
295 fprintf(stderr, " Error: The selected calculation type is not supported for calculating spectrum mode.\n");
296 return FALSE;
297 }
298
299 fprintf(stdoutMPI, " End: Calculating a spectrum.\n\n");
300 TimeKeeper(&(X->Bind), cFileNameTimeKeep, c_CalcSpectrumEnd, "a");
301 iret = OutputSpectrum(X, Nomega, dcSpectrum, dcomega);
302 free_cd_1d_allocate(dcSpectrum);
303 free_cd_1d_allocate(dcomega);
304 return TRUE;
305
306}/*int CalcSpectrum*/

315int GetExcitedState
316(
317 struct BindStruct *X,
318 double complex *tmp_v0,
319 double complex *tmp_v1
320) {
321 if (X->Def.NSingleExcitationOperator > 0 && X->Def.NPairExcitationOperator > 0) {
322 fprintf(stderr, "Error: Both single and pair excitation operators exist.\n");
323 return FALSE;
324 }
325
326
327 if (X->Def.NSingleExcitationOperator > 0) {
328 if (GetSingleExcitedState(X, tmp_v0, tmp_v1) != TRUE) {
329 return FALSE;
330 }
331 } else if (X->Def.NPairExcitationOperator > 0) {
332 if (GetPairExcitedState(X, tmp_v0, tmp_v1) != TRUE) {
333 return FALSE;
334 }
335 } else {
336 unsigned long int i_max, j;
337 i_max = X->Check.idim_maxOrg;
338#pragma omp parallel for default(none) shared(tmp_v0, tmp_v1) \
339 firstprivate(i_max) private(j)
340 for (j = 1; j <= i_max; j++) tmp_v0[j] = tmp_v1[j];
341 }
342
343 return TRUE;
344}

353int SetOmega
354(
355 struct DefineList *X
356){
357 FILE *fp;
358 char sdt[D_FileNameMax],ctmp[256];
359 int istp=4;
360 double E1, E2, E3, E4, Emax;
361 long unsigned int iline_countMax=2;
362 long unsigned int iline_count=2;
363
364
365 if(X->iFlgSpecOmegaMax == TRUE && X->iFlgSpecOmegaMin == TRUE){
366 return TRUE;
367 }
368 else{
369 if (X->iCalcType == Lanczos || X->iCalcType == FullDiag) {
370 sprintf(sdt, cFileNameLanczosStep, X->CDataFileHead);
371 childfopenMPI(sdt, "r", &fp);
372 if (fp == NULL) {
373 fprintf(stdoutMPI, "Error: xx_Lanczos_Step.dat does not exist.\n");
374 return FALSE;
375 }
376 fgetsMPI(ctmp, 256, fp); //1st line is skipped
377 fgetsMPI(ctmp, 256, fp); //2nd line is skipped
378 while (fgetsMPI(ctmp, 256, fp) != NULL) {
379 iline_count++;
380 }
381 iline_countMax = iline_count;
382 iline_count = 2;
383 rewind(fp);
384 fgetsMPI(ctmp, 256, fp); //1st line is skipped
385 fgetsMPI(ctmp, 256, fp); //2nd line is skipped
386
387 while (fgetsMPI(ctmp, 256, fp) != NULL) {
388 sscanf(ctmp, "stp=%d %lf %lf %lf %lf %lf\n",
389 &istp,
390 &E1,
391 &E2,
392 &E3,
393 &E4,
394 &Emax);
395 iline_count++;
396 if (iline_count == iline_countMax) break;
397 }
398 fclose(fp);
399 if (istp < 4) {
400 fprintf(stdoutMPI, "Error: Lanczos step must be greater than 4 for using spectrum calculation.\n");
401 return FALSE;
402 }
403 }/*if (X->iCalcType == Lanczos || X->iCalcType == FullDiag)*/
404 else
405 {
406 sprintf(sdt, cFileNameEnergy_Lanczos, X->CDataFileHead);
407 childfopenMPI(sdt, "r", &fp);
408 if (fp == NULL) {
409 fprintf(stdoutMPI, "Error: xx_energy.dat does not exist.\n");
410 return FALSE;
411 }/*if (fp == NULL)*/
412 fgetsMPI(ctmp, 256, fp); //1st line is skipped
413 fgetsMPI(ctmp, 256, fp); //1st line is skipped
414 sscanf(ctmp, " Energy %lf \n", &E1);
415 Emax = LargeValue;
416 }
417 //Read Lanczos_Step
418 if(X->iFlgSpecOmegaMax == FALSE){
419 X->dcOmegaMax= Emax*(double)X->Nsite;
420 }
421 if(X->iFlgSpecOmegaMin == FALSE){
422 X->dcOmegaMin= E1;
423 }
424 }/*Omegamax and omegamin is not specified in modpara*/
425
426 return TRUE;
427}

438int MakeExcitedList(

440 int *iFlgListModifed
441) {
442 long int j;
443 *iFlgListModifed = FALSE;
444 //To Get Original space
445 if (check(X) == MPIFALSE) {
446 FinalizeMPI();
447 return -1;
448 }
449
450 X->Check.idim_maxOrg = X->Check.idim_max;
451 X->Check.idim_maxMPIOrg = X->Check.idim_maxMPI;
452
453 if (X->Def.NSingleExcitationOperator > 0) {
454 switch (X->Def.iCalcModel) {
455 case HubbardGC:
456 break;
457 case HubbardNConserved:
458 case Hubbard:
459 case Kondo:
460 case KondoGC:
461 case tJ:
462 case tJNConserved:
463 case tJGC:
464 *iFlgListModifed = TRUE;
465 break;
466 case Spin:
467 case SpinGC:
468 return FALSE;
469 }
470 } else if (X->Def.NPairExcitationOperator > 0) {
471 switch (X->Def.iCalcModel) {
472 case SpinGC:
473 case HubbardNConserved:
474 case HubbardGC:
475 case KondoGC:
476 case tJNConserved:
477 case tJGC:
478 break;
479 case Hubbard:
480 case Kondo:
481 case tJ:
482 case Spin:
483 if (X->Def.PairExcitationOperator[0][1] != X->Def.PairExcitationOperator[0][3]) {
484 *iFlgListModifed = TRUE;
485 }
486 break;
487 }
488 } else {
489 return FALSE;
490 }
491
492 if (*iFlgListModifed == TRUE) {
493 if(GetlistSize(X)==TRUE) {
494 list_1_org = lui_1d_allocate(X->Check.idim_max + 1);
495#ifdef MPI
496 list_1buf_org = lui_1d_allocate(X->Check.idim_maxMPI + 1);
497 //lui_malloc1(list_1buf_org, X->Check.idim_maxMPI + 1);
498#endif // MPI
499 list_2_1_org = lui_1d_allocate(X->Large.SizeOflist_2_1);
500 list_2_2_org = lui_1d_allocate(X->Large.SizeOflist_2_2);
501 //lui_malloc1(list_2_1_org, X->Large.SizeOflist_2_1);
502 //lui_malloc1(list_2_2_org, X->Large.SizeOflist_2_2);
503 if(list_1_org==NULL
504 || list_2_1_org==NULL
505 || list_2_2_org==NULL
506 )
507 {
508 return -1;
509 }
510 for(j =0; j<X->Large.SizeOflist_2_1; j++){
511 list_2_1_org[j]=0;
512 }
513 for(j =0; j<X->Large.SizeOflist_2_2; j++){
514 list_2_2_org[j]=0;
515 }
516
517 }
518
519 if (sz(X, list_1_org, list_2_1_org, list_2_2_org) != 0) {
520 return FALSE;
521 }
522
523 if (X->Def.NSingleExcitationOperator > 0) {
524 switch (X->Def.iCalcModel) {
525 case HubbardGC:
526 break;
527 case HubbardNConserved:
528 case KondoNConserved:/*To be confirmed*/
529 case tJNConserved:/*To be confirmed*/
530 if (X->Def.SingleExcitationOperator[0][2] == 1) { //cis
531 X->Def.Ne = X->Def.NeMPI + 1;
532 }
533 else{
534 X->Def.Ne = X->Def.NeMPI - 1;
535 }
536 break;
537 case Hubbard:
538 case Kondo:
539 case KondoGC:
540 case tJ:
541 case tJGC:
542 if (X->Def.SingleExcitationOperator[0][2] == 1) { //cis
543 X->Def.Ne = X->Def.NeMPI + 1;
544 if (X->Def.SingleExcitationOperator[0][1] == 0) {//up
545 X->Def.Nup = X->Def.NupOrg + 1;
546 X->Def.Ndown=X->Def.NdownOrg;
547 } else {//down
548 X->Def.Nup=X->Def.NupOrg;
549 X->Def.Ndown = X->Def.NdownOrg + 1;
550 }
551 } else {//ajt
552 X->Def.Ne = X->Def.NeMPI - 1;
553 if (X->Def.SingleExcitationOperator[0][1] == 0) {//up
554 X->Def.Nup = X->Def.NupOrg - 1;
555 X->Def.Ndown=X->Def.NdownOrg;
556
557 } else {//down
558 X->Def.Nup=X->Def.NupOrg;
559 X->Def.Ndown = X->Def.NdownOrg - 1;
560 }
561 }
562 break;
563 case Spin:
564 case SpinGC:
565 return FALSE;
566 }
567 } else if (X->Def.NPairExcitationOperator > 0) {
568 X->Def.Ne=X->Def.NeMPI;
569 switch (X->Def.iCalcModel) {
570 case SpinGC:
571 case HubbardNConserved:
572 case HubbardGC:
573 case KondoNConserved:/*To be confirmed*/
574 case tJNConserved:/*To be confirmed*/
575 break;
576 case Hubbard:
577 case Kondo:
578 case KondoGC:
579 case tJ:
580 case tJGC:
581 if (X->Def.PairExcitationOperator[0][1] != X->Def.PairExcitationOperator[0][3]) {
582 if (X->Def.PairExcitationOperator[0][1] == 0) {//up
583 X->Def.Nup = X->Def.NupOrg + 1;
584 X->Def.Ndown = X->Def.NdownOrg - 1;
585 } else {//down
586 X->Def.Nup = X->Def.NupOrg - 1;
587 X->Def.Ndown = X->Def.NdownOrg + 1;
588 }
589 }
590 break;
591 case Spin:
592 if (X->Def.PairExcitationOperator[0][1] != X->Def.PairExcitationOperator[0][3]) {
593 if (X->Def.iFlgGeneralSpin == FALSE) {
594 if (X->Def.PairExcitationOperator[0][1] == 0) {//down
595 X->Def.Nup = X->Def.NupOrg - 1;
596 X->Def.Ndown = X->Def.NdownOrg + 1;
597 } else {//up
598 X->Def.Nup = X->Def.NupOrg + 1;
599 X->Def.Ndown = X->Def.NdownOrg - 1;
600 }
601 }
602 else{//for general spin
603 X->Def.Total2Sz = X->Def.Total2SzMPI+2*(X->Def.PairExcitationOperator[0][1]-X->Def.PairExcitationOperator[0][3]);
604 }
605 }
606 break;
607 }
608 } else {
609 return FALSE;
610 }
611 //Update Infomation
612 X->Def.Nsite=X->Def.NsiteMPI;
613
614 if (check(X) == MPIFALSE) {
615 FinalizeMPI();
616 return FALSE;
617 }
618 }
619
620 //set memory
621 if (setmem_large(X) != 0) {
622 fprintf(stdoutMPI, cErrLargeMem, iErrCodeMem);
623 exitMPI(-1);
624 }
625
626 if (sz(X, list_1, list_2_1, list_2_2) != 0) {
627 return FALSE;
628 }
629
630 if(X->Def.iCalcModel==HubbardNConserved){
631 X->Def.iCalcModel=Hubbard;
632 }
633
634#ifdef _DEBUG
635 if (*iFlgListModifed == TRUE) {
636 for(j=1; j<=X->Check.idim_maxOrg; j++){
637 fprintf(stdout, "Debug1: myrank=%d, list_1_org[ %ld] = %ld\n", myrank, j, list_1_org[j]+myrank*X->Def.OrgTpow[2*X->Def.NsiteMPI-1]);
638 }
639
640 for(j=1; j<=X->Check.idim_max; j++){
641 fprintf(stdout, "Debug2: myrank=%d, list_1[ %ld] = %ld\n", myrank, j, list_1[j]+myrank* 64);
642 }
643 }
644#endif
645
646 return TRUE;
647}
