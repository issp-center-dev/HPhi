61static int build_identity(const struct BindStruct *X, uint64_t *w)
62{
63 int invalid = X == NULL || w == NULL || sizeof(double) != 8 ||
64 sizeof(double complex) != 16 || DBL_MANT_DIG != 53 || DBL_MAX_EXP != 1024;
65 if (agree(invalid, "runtime validation")) return -1;
66 memset(w, 0, RESTART_V3_WORDS * sizeof(*w));
67 if (X->Def.iFlgSymmetryBasis) {
68 if (SymmetryCheckpointIdentityWords(X, w) != 0) return -1; /* words 0-29 as v1/v2 */
69 uint64_t order_xor = 0, order_sum = 0, tag = RestartHashTag("hphi-ckpt-order-v3");
70 for (unsigned long i = 1; i <= X->Sym->local_dim; ++i) {
71 const struct SymmetryBasisVector *e = SymmetryBasisLocalEntry(X->Sym, i);
72 if (e == NULL) { invalid = 1; break; }
73 uint64_t h = RestartHashWord(tag, X->Sym->local_offset + i);
74 h = RestartHashWord(h, (uint64_t)e->rep_state);
75 h = RestartHashWord(h, e->orbit_size);
76 h = RestartHashWord(h, e->stabilizer_size);
77 order_xor ^= h; order_sum += h;
78 }
79 if (agree(invalid, "basis order identity")) return -1;
80 w[RW_ORDER] = order_xor; w[RW_ORDER_SUM] = order_sum; /* reduced below */
81 w[RW_BASIS_KIND] = RESTART_BASIS_SECTOR;
82 } else {
83 const struct DefineList *d = &X->Def;
84 uint64_t flags = (uint64_t)d->iRestartFixedFlags; /* normalized in readdef (Q2) */
85 int sz_fixed = (flags & 2) != 0, ne_fixed = (flags & 1) != 0;
86 w[RW_MAGIC] = UINT64_C(0x0a31565349485048); w[RW_PHASE] = 0; w[RW_SCALAR] = 128;
87 w[RW_MODEL] = (uint64_t)d->iCalcModelInput; w[RW_NSITE] = d->NsiteMPI;
88 w[RW_NUP] = sz_fixed && !d->iFlgGeneralSpin ? d->NupMPI : 0;
89 w[RW_NDOWN] = sz_fixed && !d->iFlgGeneralSpin ? d->NdownMPI : 0;
90 w[RW_NE] = ne_fixed ? d->NeMPI : 0;
91 w[RW_DIM] = X->Check.idim_maxMPI; w[RW_LAYOUT] = 2;
92 w[RW_FIXED_FLAGS] = flags; w[RW_BASIS_KIND] = RESTART_BASIS_RAW;
93 w[RW_TOTAL2SZ] = sz_fixed ? (uint64_t)(int64_t)d->Total2SzMPI : 0;
94 w[RW_SITE_DIGEST] = raw_site_digest(d);
95 w[RW_HAMILTONIAN] = 0;
96 }
97 w[RW_VERSION] = 3; w[RW_RANKS] = (uint64_t)nproc;
98#ifdef MPI
99 if (X->Def.iFlgSymmetryBasis) {
100 uint64_t x = w[RW_ORDER], s = w[RW_ORDER_SUM];
101 if (MPI_Allreduce(&x, &w[RW_ORDER], 1, MPI_UINT64_T, MPI_BXOR, MPI_COMM_WORLD) != MPI_SUCCESS) exitMPI(-1);
102 if (MPI_Allreduce(&s, &w[RW_ORDER_SUM], 1, MPI_UINT64_T, MPI_SUM, MPI_COMM_WORLD) != MPI_SUCCESS) exitMPI(-1);
103 }
104#endif
105 return 0;
106}

116static int check_identity(const uint64_t *expected, const uint64_t *header, const char **bad)
117{
118 for (size_t i = 0; i < sizeof(identity_words)/sizeof(identity_words[0]); ++i) {
119 int w = identity_words[i];
120 if (header[w] != expected[w]) { *bad = word_names[w]; return -1; }
121 }
122 int layout_ok = header[RW_BASIS_KIND] == RESTART_BASIS_RAW ? header[RW_LAYOUT] == 2 : header[RW_LAYOUT] <= 1;
123 if (!layout_ok) { *bad = word_names[RW_LAYOUT]; return -1; }
124 if (header[RW_FLAGS] & ~UINT64_C(1)) { *bad = word_names[RW_FLAGS]; return -1; }
125 return 0;
126}

186static int write_part(struct BindStruct *X, const uint64_t *identity, const struct RestartVectorInfo *info,
187 int comp, uint64_t gen, uint64_t flags, const double complex *vector, char *final_name)
188{
189 uint64_t h[RESTART_V3_WORDS]; char part[D_FileNameMax];
190 memcpy(h, identity, sizeof(h));
191 h[RW_RANK] = (uint64_t)myrank; h[RW_OFFSET] = RestartLocalOffset(X); h[RW_LOCAL_DIM] = X->Check.idim_max;
192 h[RW_METHOD] = info->method; h[RW_STATE] = info->set; h[RW_STEP] = info->step; h[RW_TIME] = RestartDoubleWord(info->time);
193 h[RW_NSET] = info->nset; h[RW_COMPONENT] = (uint64_t)comp; h[RW_FLAGS] = flags;
194 h[RW_SCALAR_A] = RestartDoubleWord(info->scalar_a); h[RW_SCALAR_B] = RestartDoubleWord(info->scalar_b);
195 h[RW_SCALAR_C] = RestartDoubleWord(info->scalar_c);
196 uint64_t x = 0, s = 0; double norm = 0; int invalid = 0;
197 for (unsigned long i = 1; i <= X->Check.idim_max; ++i) {
198 double re = creal(vector[i]), im = cimag(vector[i]);
199 if (!isfinite(re) || !isfinite(im)) invalid = 1;
200 norm += re*re + im*im;
201 }
202 RestartDigestAccumulate(h[RW_OFFSET], vector, X->Check.idim_max, &x, &s);
203 norm = SumMPI_d(norm);
204 if (comp == RESTART_COMP_STATE && (!isfinite(norm) || fabs(norm - 1.0) > 1e-8)) invalid = 1;
205 allreduce_xor_sum(x, s, &h[RW_DATA_XOR], &h[RW_DATA_SUM]);
206 if (agree(invalid, "vector validation")) return -1;
207 int length = snprintf(final_name, D_FileNameMax, cFileNameRestartVector, X->Def.CDataFileHead,
208 (int)info->set, comp, (int)gen, myrank);
209 invalid = length < 0 || length >= D_FileNameMax;
210 length = snprintf(part, sizeof(part), "%s%s.part", cParentOutputFolder, final_name);
211 invalid |= length < 0 || length >= (int)sizeof(part);
212 if (agree(invalid, "path validation")) return -1;
213 FILE *fp = fopen(part, "wb");
214 invalid = fp == NULL;
215#ifdef HPHI_TEST_SYMMETRY_PROBE
216 if (!invalid && SymmetryFullDiagTestFail("restart_write")) invalid = 1;
217#endif
218 for (int i = 0; !invalid && i < RESTART_V3_WORDS; ++i) invalid = RestartPutWord(fp, h[i]) != 0;
219 for (unsigned long i = 1; !invalid && i <= X->Check.idim_max; ++i)
220 invalid = RestartPutWord(fp, RestartDoubleWord(creal(vector[i]))) != 0 ||
221 RestartPutWord(fp, RestartDoubleWord(cimag(vector[i]))) != 0;
222 if (fp != NULL) { int flushed = fflush(fp) != 0; int closed = fclose(fp) != 0; invalid |= flushed || closed; }
223 if (invalid && fp != NULL) remove(part);
224 return agree(invalid, "write"); /* every rank reaches this collective in the same order (Q1) */
225}

227int RestartWriteUnit(struct BindStruct *X, const struct RestartUnit *unit,
228 double complex *const *const *vectors, const struct RestartVectorInfo *infos, uint64_t flags)
229{
230 static int fresh_manifest_published = 0; /* ReStart=1: the first publication replaces the manifest (Q11) */
231 struct RestartManifest old = {0}, m = {0}; int exists, invalid = 0;
232 uint64_t identity[RESTART_V3_WORDS], gen = 0;
233 StartTimer(7100);
234 invalid = X == NULL || unit == NULL || vectors == NULL || infos == NULL;
235 if (!invalid) invalid = unit->nset <= 0 || unit->ncomp < 1 || unit->ncomp > 2 || unit->sets == NULL ||
236 unit->nset > INT_MAX / unit->ncomp ||
237 (size_t)unit->nset > SIZE_MAX / unit->ncomp / D_FileNameMax;
238 if (agree(invalid, "unit validation")) { StopTimer(7100); return -1; }
239 for (int i = 0; i < unit->nset; ++i) {
240 invalid |= vectors[i] == NULL || unit->sets[i] > INT_MAX || infos[i].set != unit->sets[i];
241 if (vectors[i] != NULL) for (int c = 0; c < unit->ncomp; ++c) invalid |= vectors[i][c] == NULL;
242 }
243 if (agree(invalid, "unit vectors / set indices")) { StopTimer(7100); return -1; }
244 char (*names)[D_FileNameMax] = malloc((size_t)unit->nset * unit->ncomp * D_FileNameMax);
245 if (RestartManifestLoad(X, &old, &exists) != 0) { free(names); StopTimer(7100); return -1; }
246 int fresh = X->Def.iReStart == RESTART_OUT && !fresh_manifest_published;
247 m.method = fresh || !exists ? X->Def.iCalcType : old.method;
248 m.basis = fresh || !exists ? basis_kind(X) : old.basis;
249 /* Keep old intact for collision checks and post-publication cleanup; rows are deep-copied. */
250 if (!fresh) for (size_t i = 0; i < old.nrows; ++i)
251 if (RestartManifestUpsert(&m, &old.rows[i]) != 0) invalid = 1;
252 invalid |= names == NULL;
253 if (agree(invalid, "publication allocation")) goto fail;
254 if (agree(RestartChooseGeneration(&old, unit->sets, unit->nset, &gen) != 0,
255 "no collision-free generation; use a different output directory or CDataFileHead")) goto fail;
256 if (agree(build_identity(X, identity) != 0, "identity")) goto fail;
257 for (int i = 0; i < unit->nset && !invalid; ++i)
258 for (int c = 0; c < unit->ncomp && !invalid; ++c)
259 invalid = write_part(X, identity, &infos[i], c, gen, flags, vectors[i][c], names[i * unit->ncomp + c]) != 0;
260 /* `invalid` is now identical on every rank (write_part agreed), so breaking here keeps the collective order. */
261 if (invalid) { remove_generation(X, unit, gen, 1 /* .part */); goto fail; }
262 for (int k = 0; k < unit->nset * unit->ncomp && !invalid; ++k) {
263 char part[D_FileNameMax], final[D_FileNameMax];
264 snprintf(part, sizeof(part), "%s%s.part", cParentOutputFolder, names[k]);
265 snprintf(final, sizeof(final), "%s%s", cParentOutputFolder, names[k]);
266#ifdef HPHI_TEST_SYMMETRY_PROBE
267 if (SymmetryFullDiagTestFail("restart_rename")) invalid = 1; else
268#endif
269 invalid = rename(part, final) != 0;
270 }
271 if (agree(invalid, "rename")) { remove_generation(X, unit, gen, 0); remove_generation(X, unit, gen, 1); goto fail; }
272 for (int i = 0; i < unit->nset; ++i) {
273 struct RestartManifestRow row = {unit->sets[i], (uint64_t)nproc, gen, infos[i].step, (uint64_t)unit->ncomp};
274 if (RestartManifestUpsert(&m, &row) != 0) invalid = 1;
275 }
276#ifdef HPHI_TEST_SYMMETRY_PROBE
277 if (SymmetryFullDiagTestFail("restart_manifest")) invalid = 1;
278#endif
279 if (agree(invalid, "manifest update")) { remove_generation(X, unit, gen, 0); goto fail; }
280 if (RestartManifestPublish(X, &m) != 0) { remove_generation(X, unit, gen, 0); goto fail; } /* collective on every rank */
281 if (fresh) fresh_manifest_published = 1;
282 remove_superseded_files(X, &old, &m); /* published now: delete only old files no longer referenced */
283 fprintf(stdoutMPI, "Restart checkpoint published: set %" PRIu64 " step %" PRIu64 " gen %" PRIu64 " flags %" PRIu64 "\n",
284 unit->sets[0], infos[0].step, gen, flags);
285#ifdef HPHI_TEST_SYMMETRY_PROBE
286 if (SumMPI_i(RestartTestAbortAfter(unit->sets[0], infos[0].step)) != 0) {
287 fprintf(stdoutMPI, "Test hook: stopping after checkpoint publication.\n");
288 RestartManifestFree(&m); RestartManifestFree(&old); free(names); StopTimer(7100); FinalizeMPI(); exit(0);
289 }
290#endif
291 RestartManifestFree(&m); RestartManifestFree(&old); free(names); StopTimer(7100); return 0;
292fail:
293 RestartManifestFree(&m); RestartManifestFree(&old); free(names); StopTimer(7100); return -1;
294}

308static int inspect_vector_header(struct BindStruct *X, const struct RestartManifestRow *row, int comp,
309 enum RestartReadPurpose purpose, const char *base, uint64_t *reference,
310 uint64_t **saved_offsets, uint64_t **saved_dims)
311{
312 if (agree(row == NULL, "missing manifest row")) return -1;
313 if (purpose == RESTART_READ_IMPORT &&
314 agree(comp != RESTART_COMP_STATE, "import requires state component")) return -1;
315 const uint64_t P = row->ranks;
316 uint64_t expected[RESTART_V3_WORDS];
317 uint64_t *offsets = NULL, *dims = NULL; int status = 0;
318 if (agree(row->set > INT_MAX || row->gen > 1 || row->ncomp < 1 || row->ncomp > 2 ||
319 comp < 0 || comp >= (int)row->ncomp || P > INT_MAX ||
320 P > SIZE_MAX / sizeof(uint64_t), "manifest row")) return -1;
321 /* A continuation manifest is self-describing. In particular, TPQ sets may
322 * be migrated from legacy raw files to v3 one at a time, so a source-rank
323 * hint for the remaining legacy sets must not reject an already migrated
324 * v3 set. Explicit state import keeps the optional equality check. */
325 if (purpose == RESTART_READ_IMPORT &&
326 agree(X->Def.RestartSourceRanks > 0 &&
327 (uint64_t)X->Def.RestartSourceRanks != P,
328 "RestartSourceRanks does not match the saved rank count")) return -1;
329 if (agree(build_identity(X, expected) != 0, "identity")) return -1;
330 offsets = calloc(P, sizeof(*offsets)); dims = calloc(P, sizeof(*dims));
331 if (agree(offsets == NULL || dims == NULL, "allocation")) goto fail;
332 if (myrank == 0) { /* header scan: every old rank, including empty ones */
333 uint64_t running = 0;
334 for (uint64_t r = 0; r < P && status == 0; ++r) {
335 uint64_t h[RESTART_V3_WORDS]; char name[D_FileNameMax];
336 if (vector_name(name, sizeof(name), X, row, comp, (int)r, base) != 0) { status = 1; break; }
337 FILE *fp = open_output(name, "rb");
338 if (fp == NULL) { status = 1; break; }
339 for (int i = 0; i < RESTART_V3_WORDS && status == 0; ++i) if (RestartGetWord(fp, &h[i]) != 0) status = 2;
340 if (status == 0 && (h[RW_MAGIC] != expected[RW_MAGIC] || h[RW_VERSION] != 3 || h[RW_RANKS] != P || h[RW_RANK] != r)) status = 3;
341 if (status == 0 && h[RW_OFFSET] != running) status = 4;
342 if (status == 0) {
343 long end;
344 if (fseek(fp, 0, SEEK_END) != 0 || (end = ftell(fp)) < 0 || h[RW_LOCAL_DIM] > (UINT64_MAX - RESTART_V3_HEADER_BYTES) / 16 ||
345 (uint64_t)end != RESTART_V3_HEADER_BYTES + 16 * h[RW_LOCAL_DIM]) status = 5;
346 }
347 if (status == 0) {
348 if (r == 0) memcpy(reference, h, sizeof(h));
349 else for (int i = 0; i < RESTART_V3_WORDS; ++i)
350 if (i != RW_RANK && i != RW_OFFSET && i != RW_LOCAL_DIM && h[i] != reference[i]) status = 6;
351 if (h[RW_LOCAL_DIM] > UINT64_MAX - running) status = 8;
352 else { offsets[r] = h[RW_OFFSET]; dims[r] = h[RW_LOCAL_DIM]; running += h[RW_LOCAL_DIM]; }
353 }
354 fclose(fp);
355 }
356 if (status == 0 && running != reference[RW_DIM]) status = 7;
357 }
358 status = BcastMPI_i(0, status);
359 if (status != 0) { fprintf(stdoutMPI, "Error: restart checkpoint header scan failed (code %d).\n", status); goto fail; }
360 bcast_words(reference, RESTART_V3_WORDS); bcast_words(offsets, P); bcast_words(dims, P);
361 const char *bad = NULL;
362 int invalid = check_identity(expected, reference, &bad) != 0 || reference[RW_COMPONENT] != (uint64_t)comp;
363 if (purpose == RESTART_READ_CONTINUATION) { /* Q3: method, unit size, step range, int range */
364 if (agree(reference[RW_STEP] < step_lower_bound(X->Def.iCalcType) ||
365 reference[RW_STEP] > (uint64_t)(INT_MAX - 1), "step range")) goto fail;
366 invalid |= reference[RW_STATE] != row->set || reference[RW_STEP] != row->step ||
367 reference[RW_METHOD] != (uint64_t)X->Def.iCalcType ||
368 (X->Def.iCalcType == CG && reference[RW_NSET] != (uint64_t)X->Def.k_exct);
369 }
370 if (purpose == RESTART_READ_CONTINUATION) {
371 invalid |= X->Def.iCalcType != CG && reference[RW_FLAGS] != 0;
372 if (X->Def.iCalcType == cTPQ) {
373 double expected_dt = 1.0 / LargeValue, saved_dt = RestartWordDouble(reference[RW_SCALAR_C]);
374 invalid |= RestartWordDouble(reference[RW_SCALAR_B]) < 0 || !isfinite(expected_dt) || expected_dt <= 0 ||
375 fabs(saved_dt - expected_dt) > 1e-12 * expected_dt;
376 }
377 }
378 if (!invalid) invalid = !isfinite(RestartWordDouble(reference[RW_TIME])) || !isfinite(RestartWordDouble(reference[RW_SCALAR_A])) ||
379 !isfinite(RestartWordDouble(reference[RW_SCALAR_B])) || !isfinite(RestartWordDouble(reference[RW_SCALAR_C]));
380 if (invalid && bad != NULL) fprintf(stderr, "Error: rank %d restart checkpoint %s differs.\n", myrank, bad);

382 *saved_offsets = offsets; *saved_dims = dims; return 0;
383fail:
384 free(offsets); free(dims); return -1;

420static int read_set_files(struct BindStruct *X, const struct RestartManifestRow *row, int comp,
421 enum RestartReadPurpose purpose, const char *base,
422 double complex *vector, struct RestartVectorInfo *info)
423{
424 uint64_t reference[RESTART_V3_WORDS], expected[RESTART_V3_WORDS];
425 uint64_t *offsets = NULL, *dims = NULL;
426 int invalid = 0;
427 if (inspect_vector_header(X, row, comp, purpose, base, reference, &offsets, &dims) != 0) return -1;
428 const uint64_t P = row->ranks;
429 /* expected is used only for the Hamiltonian provenance message below. */
430 if (agree(build_identity(X, expected) != 0, "identity")) goto fail;
431 /* intersection reads (S3): lo/hi per old file */
432 uint64_t a = RestartLocalOffset(X);
433 if (agree(a > reference[RW_DIM] || X->Check.idim_max > reference[RW_DIM] - a ||
434 X->Check.idim_max >= SIZE_MAX / sizeof(double complex), "local interval / allocation size")) goto fail;
435 uint64_t b = a + X->Check.idim_max;
436 double complex *tmp = calloc(X->Check.idim_max + 1, sizeof(*tmp));
437 unsigned long payload_files_opened = 0;
438 invalid = tmp == NULL;
439 for (uint64_t j = 0; !invalid && j < P; ++j) {
440 uint64_t lo, hi;
441 if (!RestartIntersect(a, b, offsets[j], dims[j], &lo, &hi)) continue;
442 char name[D_FileNameMax];
443 if (vector_name(name, sizeof(name), X, row, comp, (int)j, base) != 0) { invalid = 1; break; }
444 FILE *fp = open_output(name, "rb");
445 if (fp != NULL) ++payload_files_opened;
446 uint64_t byte = RESTART_V3_HEADER_BYTES + 16 * (lo - offsets[j]);
447 invalid = fp == NULL || byte > (uint64_t)LONG_MAX || fseek(fp, (long)byte, SEEK_SET) != 0;
448 for (uint64_t g = lo; !invalid && g < hi; ++g) {
449 uint64_t re, im;
450 invalid = RestartGetWord(fp, &re) != 0 || RestartGetWord(fp, &im) != 0;
451 if (!invalid) { double c[2] = {RestartWordDouble(re), RestartWordDouble(im)}; memcpy(&tmp[(g - a) + 1], c, sizeof(c)); }
452 }
453 if (!invalid && hi == offsets[j] + dims[j] && (fgetc(fp) != EOF || ferror(fp))) invalid = 1;
454 if (fp != NULL) fclose(fp);
455 }
456 if (agree(invalid, "payload read")) { free(tmp); goto fail; }
457 payload_files_opened = SumMPI_li(payload_files_opened);
458 fprintf(stdoutMPI, "Restart read: header files %" PRIu64
459 ", payload files opened %lu.\n", P, payload_files_opened);
460 uint64_t x = 0, s = 0, gx, gs; double norm = 0;
461 for (unsigned long i = 1; i <= X->Check.idim_max; ++i) {
462 double re = creal(tmp[i]), im = cimag(tmp[i]);
463 if (!isfinite(re) || !isfinite(im)) invalid = 1;
464 norm += re*re + im*im;
465 }
466 RestartDigestAccumulate(a, tmp, X->Check.idim_max, &x, &s);
467 norm = SumMPI_d(norm);
468 allreduce_xor_sum(x, s, &gx, &gs);
469 invalid |= gx != reference[RW_DATA_XOR] || gs != reference[RW_DATA_SUM];
470 if (comp == RESTART_COMP_STATE) invalid |= !isfinite(norm) || fabs(norm - 1.0) > 1e-8;
471 if (agree(invalid, "payload checksum / norm")) { free(tmp); goto fail; }
472 memcpy(vector, tmp, (X->Check.idim_max + 1) * sizeof(*vector)); free(tmp);
473 info->method = reference[RW_METHOD]; info->set = reference[RW_STATE]; info->step = reference[RW_STEP];
474 info->time = RestartWordDouble(reference[RW_TIME]); info->nset = reference[RW_NSET]; info->flags = reference[RW_FLAGS];
475 info->scalar_a = RestartWordDouble(reference[RW_SCALAR_A]); info->scalar_b = RestartWordDouble(reference[RW_SCALAR_B]);
476 info->scalar_c = RestartWordDouble(reference[RW_SCALAR_C]);
477 fprintf(stdoutMPI, "Restart checkpoint loaded: set=%" PRIu64 " comp=%d step=%" PRIu64 " flags=%" PRIu64 " time=%.17g hamiltonian_changed=%s\n",
478 info->set, comp, info->step, info->flags, info->time,
479 reference[RW_BASIS_KIND] == RESTART_BASIS_RAW ? "unknown" : (reference[RW_HAMILTONIAN] == expected[RW_HAMILTONIAN] ? "no" : "yes"));
480 if (purpose == RESTART_READ_IMPORT &&
481 record_import(X, info, reference[RW_HAMILTONIAN], info->method == FullDiag) != 0) goto fail;
482 free(offsets); free(dims); return 0;
483fail:
484 free(offsets); free(dims); return -1;
485}

533/* Read a sector v1/v2 state with a different rank count. The old DATA words
534 * hash each writer-rank partition, so first reproduce that checksum over every
535 * old file, then read only the intervals intersecting this run's ownership. */
536static int read_sector_cross_rank(struct BindStruct *X, const char *base,
537 int version, int source_ranks,
538 double complex *vector,
539 struct SymmetryCheckpointInfo *info)
540{
541 const int words = version == RESTART_FORMAT_SECTOR_V2 ? 30 : 28;
542 const uint64_t P = (uint64_t)source_ranks;
543 uint64_t expected[RESTART_V3_WORDS] = {0};
544 uint64_t reference[RESTART_V3_WORDS] = {0};
545 uint64_t *offsets = NULL, *dims = NULL;
546 int status = 0, invalid = 0;
547 if (agree(base == NULL || vector == NULL || info == NULL ||
548 (version != RESTART_FORMAT_SECTOR_V1 &&
549 version != RESTART_FORMAT_SECTOR_V2) || source_ranks < 1,
550 "sector cross-rank arguments")) return -1;
551 if (SymmetryCheckpointIdentityWords(X, expected) != 0) return -1;
552 offsets = calloc(P, sizeof(*offsets));
553 dims = calloc(P, sizeof(*dims));
554 if (agree(offsets == NULL || dims == NULL, "sector cross-rank allocation")) goto fail;
555
556 if (myrank == 0) {
557 uint64_t running = 0;
558 for (uint64_t rank = 0; rank < P && status == 0; ++rank) {
559 uint64_t header[RESTART_V3_WORDS] = {0};
560 char name[D_FileNameMax];
561 FILE *fp = NULL;
562 if (vector_name(name, sizeof(name), X, NULL, 0, (int)rank, base) != 0) {
563 status = 1;
564 break;
565 }
566 fp = open_output(name, "rb");
567 if (fp == NULL) {
568 status = 2;
569 break;
570 }
571 for (int i = 0; i < words && status == 0; ++i)
572 if (RestartGetWord(fp, &header[i]) != 0) status = 3;
573 if (status == 0 &&
574 (sector_legacy_runtime_identity(expected, header, words) != 0 ||
575 header[RW_RANKS] != P || header[RW_RANK] != rank ||
576 header[RW_LAYOUT] > 1 || header[RW_OFFSET] != running)) status = 4;
577 if (status == 0) {
578 long end = -1;
579 uint64_t bytes = (uint64_t)(8 * words);
580 if (header[RW_LOCAL_DIM] > (UINT64_MAX - bytes) / 16 ||
581 fseek(fp, 0, SEEK_END) != 0 || (end = ftell(fp)) < 0 ||
582 (uint64_t)end != bytes + 16 * header[RW_LOCAL_DIM]) status = 5;
583 }
584 if (status == 0) {
585 double time = RestartWordDouble(header[RW_TIME]);
586 invalid = !isfinite(time) ||
587 (header[RW_METHOD] != FullDiag && header[RW_METHOD] != CG &&
588 header[RW_METHOD] != TimeEvolution);
589 if (header[RW_METHOD] == FullDiag)
590 invalid |= header[RW_STEP] != 0 || header[RW_TIME] != 0 ||
591 header[RW_STATE] >= header[RW_DIM];
592 if (invalid) status = 6;
593 }
594 if (status == 0) {
595 if (rank == 0) memcpy(reference, header, sizeof(reference));
596 else if (sector_legacy_same_unit(reference, header, words) != 0) status = 7;
597 }
598 if (status == 0) {
599 if (header[RW_LOCAL_DIM] > UINT64_MAX - running) status = 8;
600 else {
601 offsets[rank] = header[RW_OFFSET];
602 dims[rank] = header[RW_LOCAL_DIM];
603 running += header[RW_LOCAL_DIM];
604 }
605 }
606 if (fclose(fp) != 0 && status == 0) status = 9;
607 }
608 if (status == 0 && running != reference[RW_DIM]) status = 10;
609 }
610 status = BcastMPI_i(0, status);
611 if (agree(status != 0, "sector cross-rank header scan")) goto fail;
612 bcast_words(reference, (uint64_t)words);
613 bcast_words(offsets, P);
614 bcast_words(dims, P);
615
616 uint64_t local_xor = 0, local_sum = 0, global_xor = 0, global_sum = 0;
617 double local_norm = 0;
618 for (uint64_t rank = (uint64_t)myrank; !invalid && rank < P; rank += (uint64_t)nproc) {
619 char name[D_FileNameMax];
620 FILE *fp = NULL;
621 uint64_t hash;
622 double norm;
623 invalid = vector_name(name, sizeof(name), X, NULL, 0, (int)rank, base) != 0;
624 if (!invalid) fp = open_output(name, "rb");
625 invalid |= fp == NULL || fseek(fp, 8L * words, SEEK_SET) != 0 ||
626 SymmetryCheckpointLegacyFileHash(fp, rank, offsets[rank],
627 dims[rank], &hash, &norm) != 0;
628 if (!invalid) {
629 local_xor ^= hash;
630 local_sum += hash;
631 local_norm += norm;
632 }
633 if (fp != NULL && fclose(fp) != 0) invalid = 1;
634 }
635 if (agree(invalid, "sector v1/v2 old-partition DATA read")) goto fail;
636 allreduce_xor_sum(local_xor, local_sum, &global_xor, &global_sum);
637 double norm = SumMPI_d(local_norm);
638 invalid = global_xor != reference[RW_DATA_XOR] ||
639 global_sum != reference[RW_DATA_SUM] || !isfinite(norm) ||
640 fabs(norm - 1.0) > 1e-8;
641 if (agree(invalid, "sector v1/v2 payload checksum / norm")) goto fail;
642
643 uint64_t a = RestartLocalOffset(X);
644 if (agree(a > reference[RW_DIM] || X->Check.idim_max > reference[RW_DIM] - a ||
645 X->Check.idim_max >= SIZE_MAX / sizeof(double complex),
646 "sector cross-rank local interval")) goto fail;
647 uint64_t b = a + X->Check.idim_max;
648 double complex *temporary = calloc(X->Check.idim_max + 1, sizeof(*temporary));
649 invalid = temporary == NULL;
650 for (uint64_t rank = 0; !invalid && rank < P; ++rank) {
651 uint64_t lo, hi;
652 if (!RestartIntersect(a, b, offsets[rank], dims[rank], &lo, &hi)) continue;
653 char name[D_FileNameMax];
654 FILE *fp = NULL;
655 uint64_t byte = (uint64_t)(8 * words) + 16 * (lo - offsets[rank]);
656 invalid = vector_name(name, sizeof(name), X, NULL, 0, (int)rank, base) != 0;
657 if (!invalid) fp = open_output(name, "rb");
658 invalid |= fp == NULL || byte > (uint64_t)LONG_MAX ||
659 fseek(fp, (long)byte, SEEK_SET) != 0;
660 for (uint64_t global = lo; !invalid && global < hi; ++global) {
661 uint64_t re_word, im_word;
662 invalid = RestartGetWord(fp, &re_word) != 0 ||
663 RestartGetWord(fp, &im_word) != 0;
664 if (!invalid) {
665 double components[2] = {RestartWordDouble(re_word),
666 RestartWordDouble(im_word)};
667 if (!isfinite(components[0]) || !isfinite(components[1])) invalid = 1;
668 else memcpy(&temporary[(global - a) + 1], components, sizeof(components));
669 }
670 }
671 if (fp != NULL && fclose(fp) != 0) invalid = 1;
672 }
673 if (agree(invalid, "sector cross-rank payload read")) {
674 free(temporary);
675 goto fail;
676 }
677 local_norm = 0;
678 for (unsigned long i = 1; i <= X->Check.idim_max; ++i) {
679 double re = creal(temporary[i]), im = cimag(temporary[i]);
680 local_norm += re*re + im*im;
681 }
682 norm = SumMPI_d(local_norm);
683 if (agree(!isfinite(norm) || fabs(norm - 1.0) > 1e-8,
684 "sector cross-rank redistributed norm")) {
685 free(temporary);
686 goto fail;
687 }
688 memcpy(vector, temporary,
689 (X->Check.idim_max + 1) * sizeof(*temporary));
690 free(temporary);
691 info->source_method = reference[RW_METHOD];
692 info->state_index = reference[RW_STATE];
693 info->step = reference[RW_STEP];
694 info->time = RestartWordDouble(reference[RW_TIME]);
695 info->hamiltonian_digest = reference[RW_HAMILTONIAN];
696 info->hamiltonian_digest_kind = reference[RW_METHOD] == FullDiag
697 ? SYMMETRY_HAMILTONIAN_MATRIX_V1 : SYMMETRY_HAMILTONIAN_PARSED;
698 /* The sector digest is an order-independent set digest. Equality of the
699 * set implies equality of global order because the basis builder sorts
700 * canonical representatives by their representative state. */
701 fprintf(stdoutMPI, "Symmetry checkpoint cross-rank v1/v2 read: payload checksum verified, "
702 "basis set verified by sector digest and ordered canonically by representative "
703 "(ORDER word not compared).\n");
704 free(offsets);
705 free(dims);
706 return 0;
707fail:
708 free(offsets);
709 free(dims);
710 return -1;
711}

737/* Inspect every declared writer file on rank zero. No file-name probing is
738 * used to infer P: legacy raw files do not carry a source-rank count, and an
739 * omitted empty trailing rank would make such inference ambiguous. */
740static int scan_legacy_raw(struct BindStruct *X, const char *pattern, int set,
741 int P, int minimum_step,
742 struct LegacyRawScan *scan)
743{
744 int status = 0, missing_rank = -1, present = 0, saved_step = 0;
745 uint64_t running = 0;
746 if (agree(X == NULL || pattern == NULL || scan == NULL || P < 1 ||
747 P > INT_MAX || (uint64_t)P > SIZE_MAX / sizeof(uint64_t),
748 "legacy raw arguments")) return -1;
749 memset(scan, 0, sizeof(*scan));
750 scan->offsets = calloc((size_t)P, sizeof(*scan->offsets));
751 scan->dims = calloc((size_t)P, sizeof(*scan->dims));
752 if (agree(scan->offsets == NULL || scan->dims == NULL,
753 "legacy raw allocation")) goto fail;
754
755 if (myrank == 0) {
756 for (int rank = 0; rank < P && status == 0; ++rank) {
757 char name[D_FileNameMax];
758 FILE *fp = NULL;
759 int file_step = 0;
760 unsigned long local_dim = 0;
761 if (legacy_raw_name(name, sizeof(name), pattern, set, rank) != 0) {
762 status = 1;
763 break;
764 }
765 errno = 0;
766 fp = open_output(name, "rb");
767 if (fp == NULL) {
768 if (errno == ENOENT) {
769 if (missing_rank < 0) missing_rank = rank;
770 continue;
771 }
772 status = 2;
773 break;
774 }
775 ++present;
776 if (fread(&file_step, sizeof(file_step), 1, fp) != 1 ||
777 fread(&local_dim, sizeof(local_dim), 1, fp) != 1) status = 3;
778 if (status == 0 &&
779 (file_step < minimum_step || file_step > INT_MAX - 1)) status = 4;
780 if (status == 0 && present == 1) saved_step = file_step;
781 if (status == 0 && file_step != saved_step) status = 5;
782 if (status == 0) {
783 const uint64_t header = sizeof(int) + sizeof(unsigned long);
784 const uint64_t dim = (uint64_t)local_dim;
785 long end = -1;
786 if (dim > (UINT64_MAX - header - 16) / 16 ||
787 fseek(fp, 0, SEEK_END) != 0 || (end = ftell(fp)) < 0 ||
788 (uint64_t)end != header + 16 * (dim + 1)) status = 6;
789 }
790 if (status == 0) {
791 uint64_t dim = (uint64_t)local_dim;
792 if (dim > UINT64_MAX - running) status = 7;
793 else {
794 scan->offsets[rank] = running;
795 scan->dims[rank] = dim;
796 running += dim;
797 }
798 }
799 if (fclose(fp) != 0 && status == 0) status = 8;
800 }
801 if (status == 0 && present != 0 && present != P) status = 9;
802 if (status == 0 && present == P && running != X->Check.idim_maxMPI)
803 status = 10;
804 }
805 status = BcastMPI_i(0, status);
806 present = BcastMPI_i(0, present);
807 missing_rank = BcastMPI_i(0, missing_rank);
808 saved_step = BcastMPI_i(0, saved_step);
809 if (status != 0) {
810 if (X->Def.RestartSourceRanks == 0 && present != 0 &&
811 (status == 9 || status == 10)) {
812 fprintf(stdoutMPI, "Error: RestartSourceRanks is required for rank-changing "
813 "legacy input; otherwise check missing or incompatible legacy files.\n");
814 } else if (status == 9 && missing_rank >= 0) {
815 fprintf(stdoutMPI, "Error: legacy restart file for rank_%d is missing "
816 "(RestartSourceRanks=%d).\n", missing_rank, P);
817 } else if (status == 10) {
818 fprintf(stdoutMPI, "Error: legacy restart dimensions do not sum to the "
819 "current global dimension (RestartSourceRanks=%d).\n", P);
820 } else {
821 fprintf(stdoutMPI, "Error: legacy recovery header / step / file length "
822 "validation failed (code=%d, RestartSourceRanks=%d).\n", status, P);
823 }
824 goto fail;
825 }
826 scan->present = present;
827 scan->step = saved_step;
828 if (present == 0) return 0;
829 bcast_words(scan->offsets, (uint64_t)P);
830 bcast_words(scan->dims, (uint64_t)P);
831 if (P == nproc) {
832 int incompatible = scan->dims[myrank] != (uint64_t)X->Check.idim_max;
833 if (agree(incompatible, "legacy raw local dimension")) {
834 if (X->Def.RestartSourceRanks == 0)
835 fprintf(stdoutMPI, "Error: RestartSourceRanks is required for rank-changing "
836 "legacy input; otherwise check missing or incompatible legacy files.\n");
837 goto fail;
838 }
839 }
840 return 0;
841fail:
842 free_legacy_scan(scan);
843 return -1;
844}

847 int source_ranks, double complex *vector, int *step)
848{
849 struct LegacyRawScan scan;
850 double complex *temporary = NULL;
851 int invalid = 0;
852 if (agree(vector == NULL || step == NULL || X == NULL ||
853 X->Def.iFlgSymmetryBasis, "legacy raw read arguments")) return -1;
854 if (scan_legacy_raw(X, pattern, set, source_ranks, 0, &scan) != 0)
855 return -1;
856 if (scan.present == 0) {
857 fprintf(stdoutMPI, "Error: legacy restart input is missing "
858 "(RestartSourceRanks=%d).\n", source_ranks);
859 free_legacy_scan(&scan);
860 return -1;
861 }
862 uint64_t a = RestartLocalOffset(X);
863 uint64_t global_dim = X->Check.idim_maxMPI;
864 if (agree(a > global_dim || X->Check.idim_max > global_dim - a ||
865 X->Check.idim_max >= SIZE_MAX / sizeof(*temporary),
866 "legacy raw local interval")) goto fail;
867 uint64_t b = a + X->Check.idim_max;
868 temporary = calloc(X->Check.idim_max + 1, sizeof(*temporary));
869 invalid = temporary == NULL;
870 for (int rank = 0; !invalid && rank < source_ranks; ++rank) {
871 uint64_t lo, hi;
872 if (!RestartIntersect(a, b, scan.offsets[rank], scan.dims[rank], &lo, &hi))
873 continue;
874 char name[D_FileNameMax];
875 FILE *fp = NULL;
876 const uint64_t header = sizeof(int) + sizeof(unsigned long);

975static int validate_set_boundary(struct BindStruct *X, const struct RestartManifest *m, uint64_t set)
976{

981 if (agree(bad, "recovery boundary manifest")) return -1;

984 if (inspect_vector_header(X, row, c, RESTART_READ_CONTINUATION, NULL, h, &offsets, &dims) != 0) return -1;

988 if (w != RW_COMPONENT && w != RW_DATA_XOR && w != RW_DATA_SUM && first[w] != h[w]) bad = 1;
989 }
990 if (agree(bad, "component scalars differ / recovery boundary")) return -1;
991 /* A saved CG block has one basis identity, including its recorded layout.
992 * Current-vs-saved layout compatibility is a separate reader decision. */
993 const struct RestartManifestRow *first_row = RestartManifestFind(m, 0);
994 if (X->Def.iCalcType == CG && set != 0 && first_row != NULL) {
995 uint64_t *offsets = NULL, *dims = NULL;
996 if (inspect_vector_header(X, first_row, RESTART_COMP_STATE, RESTART_READ_CONTINUATION, NULL,
997 h, &offsets, &dims) != 0) return -1;
998 free(offsets); free(dims);
999 for (size_t i = 0; i < sizeof(identity_words) / sizeof(identity_words[0]); ++i)
1000 if (first[identity_words[i]] != h[identity_words[i]]) bad = 1;
1001 bad |= first[RW_LAYOUT] != h[RW_LAYOUT];
1002 }
1003 return agree(bad, "CG state identities differ / recovery boundary");
1004}
