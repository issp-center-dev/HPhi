24static int agree(int invalid, const char *stage)
25{
26 if (SumMPI_i(invalid != 0) == 0) return 0;
27#ifdef HPHI_TEST_SYMMETRY_PROBE
28 /* Test-only evidence that every rank crossed the same agreement boundary
29 * before the solver's top-level MPI abort can terminate its peers. */
30 fprintf(stdout, "restart publication agreed failure stage=%s rank=%d\n", stage, myrank);
31 fflush(stdout);
32#endif
33 if (strcmp(stage, "manifest update") == 0)
34 fprintf(stdoutMPI, "Error: restart manifest update failed.\n");
35 else
36 fprintf(stdoutMPI, "Error: restart checkpoint %s failed.\n", stage);
37 return -1;
38}

60/* Identity words shared by every file of this run (RANK/OFFSET/LOCAL_DIM/COMPONENT/set words are set per file). */
61static int build_identity(const struct BindStruct *X, uint64_t *w)
62{
63 int invalid = X == NULL || w == NULL || sizeof(double) != 8 ||
64 sizeof(double complex) != 16 || DBL_MANT_DIG != 53 || DBL_MAX_EXP != 1024;
65 if (agree(invalid, "runtime validation")) return -1;
66 memset(w, 0, RESTART_V3_WORDS * sizeof(*w));
67 if (X->Def.iFlgSymmetryBasis) {
68 if (SymmetryCheckpointIdentityWords(X, w) != 0) return -1; /* words 0-29 as v1/v2 */
69 uint64_t order_xor = 0, order_sum = 0, tag = RestartHashTag("hphi-ckpt-order-v3");
70 for (unsigned long i = 1; i <= X->Sym->local_dim; ++i) {
71 const struct SymmetryBasisVector *e = SymmetryBasisLocalEntry(X->Sym, i);
72 if (e == NULL) { invalid = 1; break; }
73 uint64_t h = RestartHashWord(tag, X->Sym->local_offset + i);
74 h = RestartHashWord(h, (uint64_t)e->rep_state);
75 h = RestartHashWord(h, e->orbit_size);
76 h = RestartHashWord(h, e->stabilizer_size);
77 order_xor ^= h; order_sum += h;
78 }
79 if (agree(invalid, "basis order identity")) return -1;
80 w[RW_ORDER] = order_xor; w[RW_ORDER_SUM] = order_sum; /* reduced below */
81 w[RW_BASIS_KIND] = RESTART_BASIS_SECTOR;
82 } else {
83 const struct DefineList *d = &X->Def;
84 uint64_t flags = (uint64_t)d->iRestartFixedFlags; /* normalized in readdef (Q2) */
85 int sz_fixed = (flags & 2) != 0, ne_fixed = (flags & 1) != 0;
86 w[RW_MAGIC] = UINT64_C(0x0a31565349485048); w[RW_PHASE] = 0; w[RW_SCALAR] = 128;
87 w[RW_MODEL] = (uint64_t)d->iCalcModelInput; w[RW_NSITE] = d->NsiteMPI;
88 w[RW_NUP] = sz_fixed && !d->iFlgGeneralSpin ? d->NupMPI : 0;
89 w[RW_NDOWN] = sz_fixed && !d->iFlgGeneralSpin ? d->NdownMPI : 0;
90 w[RW_NE] = ne_fixed ? d->NeMPI : 0;
91 w[RW_DIM] = X->Check.idim_maxMPI; w[RW_LAYOUT] = 2;
92 w[RW_FIXED_FLAGS] = flags; w[RW_BASIS_KIND] = RESTART_BASIS_RAW;
93 w[RW_TOTAL2SZ] = sz_fixed ? (uint64_t)(int64_t)d->Total2SzMPI : 0;
94 w[RW_SITE_DIGEST] = raw_site_digest(d);
95 w[RW_HAMILTONIAN] = 0;
96 }
97 w[RW_VERSION] = 3; w[RW_RANKS] = (uint64_t)nproc;
98#ifdef MPI
99 if (X->Def.iFlgSymmetryBasis) {
100 uint64_t x = w[RW_ORDER], s = w[RW_ORDER_SUM];
101 if (MPI_Allreduce(&x, &w[RW_ORDER], 1, MPI_UINT64_T, MPI_BXOR, MPI_COMM_WORLD) != MPI_SUCCESS) exitMPI(-1);
102 if (MPI_Allreduce(&s, &w[RW_ORDER_SUM], 1, MPI_UINT64_T, MPI_SUM, MPI_COMM_WORLD) != MPI_SUCCESS) exitMPI(-1);
103 }
104#endif
105 return 0;
106}

113static const int identity_words[] = {RW_MAGIC, RW_VERSION, RW_PHASE, RW_SCALAR, RW_MODEL, RW_NSITE, RW_NUP, RW_NDOWN, RW_NE,

116static int check_identity(const uint64_t *expected, const uint64_t *header, const char **bad)
117{
118 for (size_t i = 0; i < sizeof(identity_words)/sizeof(identity_words[0]); ++i) {
119 int w = identity_words[i];
120 if (header[w] != expected[w]) { *bad = word_names[w]; return -1; }
121 }
122 int layout_ok = header[RW_BASIS_KIND] == RESTART_BASIS_RAW ? header[RW_LAYOUT] == 2 : header[RW_LAYOUT] <= 1;
123 if (!layout_ok) { *bad = word_names[RW_LAYOUT]; return -1; }
124 if (header[RW_FLAGS] & ~UINT64_C(1)) { *bad = word_names[RW_FLAGS]; return -1; }
125 return 0;
126}

132static void allreduce_xor_sum(uint64_t x, uint64_t s, uint64_t *gx, uint64_t *gs)
133{
134#ifdef MPI
135 if (MPI_Allreduce(&x, gx, 1, MPI_UINT64_T, MPI_BXOR, MPI_COMM_WORLD) != MPI_SUCCESS) exitMPI(-1);
136 if (MPI_Allreduce(&s, gs, 1, MPI_UINT64_T, MPI_SUM, MPI_COMM_WORLD) != MPI_SUCCESS) exitMPI(-1);
137#else
138 *gx = x; *gs = s;
139#endif
140}
141static void bcast_words(uint64_t *words, uint64_t count)
142{
143#ifdef MPI
144 if (count > INT_MAX || MPI_Bcast(words, (int)count, MPI_UINT64_T, 0, MPI_COMM_WORLD) != MPI_SUCCESS) exitMPI(-1);
145#else
146 (void)words; (void)count;
147#endif
148}

186static int write_part(struct BindStruct *X, const uint64_t *identity, const struct RestartVectorInfo *info,
187 int comp, uint64_t gen, uint64_t flags, const double complex *vector, char *final_name)
188{
189 uint64_t h[RESTART_V3_WORDS]; char part[D_FileNameMax];
190 memcpy(h, identity, sizeof(h));
191 h[RW_RANK] = (uint64_t)myrank; h[RW_OFFSET] = RestartLocalOffset(X); h[RW_LOCAL_DIM] = X->Check.idim_max;
192 h[RW_METHOD] = info->method; h[RW_STATE] = info->set; h[RW_STEP] = info->step; h[RW_TIME] = RestartDoubleWord(info->time);
193 h[RW_NSET] = info->nset; h[RW_COMPONENT] = (uint64_t)comp; h[RW_FLAGS] = flags;
194 h[RW_SCALAR_A] = RestartDoubleWord(info->scalar_a); h[RW_SCALAR_B] = RestartDoubleWord(info->scalar_b);
195 h[RW_SCALAR_C] = RestartDoubleWord(info->scalar_c);
196 uint64_t x = 0, s = 0; double norm = 0; int invalid = 0;
197 for (unsigned long i = 1; i <= X->Check.idim_max; ++i) {
198 double re = creal(vector[i]), im = cimag(vector[i]);
199 if (!isfinite(re) || !isfinite(im)) invalid = 1;
200 norm += re*re + im*im;
201 }
202 RestartDigestAccumulate(h[RW_OFFSET], vector, X->Check.idim_max, &x, &s);
203 norm = SumMPI_d(norm);
204 if (comp == RESTART_COMP_STATE && (!isfinite(norm) || fabs(norm - 1.0) > 1e-8)) invalid = 1;
205 allreduce_xor_sum(x, s, &h[RW_DATA_XOR], &h[RW_DATA_SUM]);
206 if (agree(invalid, "vector validation")) return -1;
207 int length = snprintf(final_name, D_FileNameMax, cFileNameRestartVector, X->Def.CDataFileHead,
208 (int)info->set, comp, (int)gen, myrank);
209 invalid = length < 0 || length >= D_FileNameMax;
210 length = snprintf(part, sizeof(part), "%s%s.part", cParentOutputFolder, final_name);
211 invalid |= length < 0 || length >= (int)sizeof(part);
212 if (agree(invalid, "path validation")) return -1;
213 FILE *fp = fopen(part, "wb");
214 invalid = fp == NULL;
215#ifdef HPHI_TEST_SYMMETRY_PROBE
216 if (!invalid && SymmetryFullDiagTestFail("restart_write")) invalid = 1;
217#endif
218 for (int i = 0; !invalid && i < RESTART_V3_WORDS; ++i) invalid = RestartPutWord(fp, h[i]) != 0;
219 for (unsigned long i = 1; !invalid && i <= X->Check.idim_max; ++i)
220 invalid = RestartPutWord(fp, RestartDoubleWord(creal(vector[i]))) != 0 ||
221 RestartPutWord(fp, RestartDoubleWord(cimag(vector[i]))) != 0;
222 if (fp != NULL) { int flushed = fflush(fp) != 0; int closed = fclose(fp) != 0; invalid |= flushed || closed; }
223 if (invalid && fp != NULL) remove(part);
224 return agree(invalid, "write"); /* every rank reaches this collective in the same order (Q1) */
225}

227int RestartWriteUnit(struct BindStruct *X, const struct RestartUnit *unit,
228 double complex *const *const *vectors, const struct RestartVectorInfo *infos, uint64_t flags)
229{
230 static int fresh_manifest_published = 0; /* ReStart=1: the first publication replaces the manifest (Q11) */
231 struct RestartManifest old = {0}, m = {0}; int exists, invalid = 0;
232 uint64_t identity[RESTART_V3_WORDS], gen = 0;
233 StartTimer(7100);
234 invalid = X == NULL || unit == NULL || vectors == NULL || infos == NULL;
235 if (!invalid) invalid = unit->nset <= 0 || unit->ncomp < 1 || unit->ncomp > 2 || unit->sets == NULL ||
236 unit->nset > INT_MAX / unit->ncomp ||
237 (size_t)unit->nset > SIZE_MAX / unit->ncomp / D_FileNameMax;
238 if (agree(invalid, "unit validation")) { StopTimer(7100); return -1; }
239 for (int i = 0; i < unit->nset; ++i) {
240 invalid |= vectors[i] == NULL || unit->sets[i] > INT_MAX || infos[i].set != unit->sets[i];
241 if (vectors[i] != NULL) for (int c = 0; c < unit->ncomp; ++c) invalid |= vectors[i][c] == NULL;
242 }
243 if (agree(invalid, "unit vectors / set indices")) { StopTimer(7100); return -1; }
244 char (*names)[D_FileNameMax] = malloc((size_t)unit->nset * unit->ncomp * D_FileNameMax);
245 if (RestartManifestLoad(X, &old, &exists) != 0) { free(names); StopTimer(7100); return -1; }
246 int fresh = X->Def.iReStart == RESTART_OUT && !fresh_manifest_published;
247 m.method = fresh || !exists ? X->Def.iCalcType : old.method;
248 m.basis = fresh || !exists ? basis_kind(X) : old.basis;
249 /* Keep old intact for collision checks and post-publication cleanup; rows are deep-copied. */
250 if (!fresh) for (size_t i = 0; i < old.nrows; ++i)
251 if (RestartManifestUpsert(&m, &old.rows[i]) != 0) invalid = 1;
252 invalid |= names == NULL;
253 if (agree(invalid, "publication allocation")) goto fail;
254 if (agree(RestartChooseGeneration(&old, unit->sets, unit->nset, &gen) != 0,
255 "no collision-free generation; use a different output directory or CDataFileHead")) goto fail;
256 if (agree(build_identity(X, identity) != 0, "identity")) goto fail;
257 for (int i = 0; i < unit->nset && !invalid; ++i)
258 for (int c = 0; c < unit->ncomp && !invalid; ++c)
259 invalid = write_part(X, identity, &infos[i], c, gen, flags, vectors[i][c], names[i * unit->ncomp + c]) != 0;
260 /* `invalid` is now identical on every rank (write_part agreed), so breaking here keeps the collective order. */
261 if (invalid) { remove_generation(X, unit, gen, 1 /* .part */); goto fail; }
262 for (int k = 0; k < unit->nset * unit->ncomp && !invalid; ++k) {
263 char part[D_FileNameMax], final[D_FileNameMax];
264 snprintf(part, sizeof(part), "%s%s.part", cParentOutputFolder, names[k]);
265 snprintf(final, sizeof(final), "%s%s", cParentOutputFolder, names[k]);
266#ifdef HPHI_TEST_SYMMETRY_PROBE
267 if (SymmetryFullDiagTestFail("restart_rename")) invalid = 1; else
268#endif
269 invalid = rename(part, final) != 0;
270 }
271 if (agree(invalid, "rename")) { remove_generation(X, unit, gen, 0); remove_generation(X, unit, gen, 1); goto fail; }
272 for (int i = 0; i < unit->nset; ++i) {
273 struct RestartManifestRow row = {unit->sets[i], (uint64_t)nproc, gen, infos[i].step, (uint64_t)unit->ncomp};
274 if (RestartManifestUpsert(&m, &row) != 0) invalid = 1;
275 }
276#ifdef HPHI_TEST_SYMMETRY_PROBE
277 if (SymmetryFullDiagTestFail("restart_manifest")) invalid = 1;
278#endif
279 if (agree(invalid, "manifest update")) { remove_generation(X, unit, gen, 0); goto fail; }
280 if (RestartManifestPublish(X, &m) != 0) { remove_generation(X, unit, gen, 0); goto fail; } /* collective on every rank */
281 if (fresh) fresh_manifest_published = 1;
282 remove_superseded_files(X, &old, &m); /* published now: delete only old files no longer referenced */
283 fprintf(stdoutMPI, "Restart checkpoint published: set %" PRIu64 " step %" PRIu64 " gen %" PRIu64 " flags %" PRIu64 "\n",
284 unit->sets[0], infos[0].step, gen, flags);
285#ifdef HPHI_TEST_SYMMETRY_PROBE
286 if (SumMPI_i(RestartTestAbortAfter(unit->sets[0], infos[0].step)) != 0) {
287 fprintf(stdoutMPI, "Test hook: stopping after checkpoint publication.\n");
288 RestartManifestFree(&m); RestartManifestFree(&old); free(names); StopTimer(7100); FinalizeMPI(); exit(0);
289 }
290#endif
291 RestartManifestFree(&m); RestartManifestFree(&old); free(names); StopTimer(7100); return 0;
292fail:
293 RestartManifestFree(&m); RestartManifestFree(&old); free(names); StopTimer(7100); return -1;
294}

308static int inspect_vector_header(struct BindStruct *X, const struct RestartManifestRow *row, int comp,
309 enum RestartReadPurpose purpose, const char *base, uint64_t *reference,
310 uint64_t **saved_offsets, uint64_t **saved_dims)
311{
312 if (agree(row == NULL, "missing manifest row")) return -1;
313 if (purpose == RESTART_READ_IMPORT &&
314 agree(comp != RESTART_COMP_STATE, "import requires state component")) return -1;
315 const uint64_t P = row->ranks;
316 uint64_t expected[RESTART_V3_WORDS];
317 uint64_t *offsets = NULL, *dims = NULL; int status = 0;
318 if (agree(row->set > INT_MAX || row->gen > 1 || row->ncomp < 1 || row->ncomp > 2 ||
319 comp < 0 || comp >= (int)row->ncomp || P > SIZE_MAX / sizeof(uint64_t), "manifest row")) return -1;
320 if (agree(P != (uint64_t)nproc, "rank count (rank redistribution arrives in PR-3)")) return -1;
321 if (agree(X->Def.RestartSourceRanks > 0 && (uint64_t)X->Def.RestartSourceRanks != P, "RestartSourceRanks does not match the saved rank count")) return -1;
322 if (agree(build_identity(X, expected) != 0, "identity")) return -1;
323 offsets = calloc(P, sizeof(*offsets)); dims = calloc(P, sizeof(*dims));
324 if (agree(offsets == NULL || dims == NULL, "allocation")) goto fail;
325 if (myrank == 0) { /* header scan: every old rank, including empty ones */
326 uint64_t running = 0;
327 for (uint64_t r = 0; r < P && status == 0; ++r) {
328 uint64_t h[RESTART_V3_WORDS]; char name[D_FileNameMax];
329 if (vector_name(name, sizeof(name), X, row, comp, (int)r, base) != 0) { status = 1; break; }
330 FILE *fp = open_output(name, "rb");
331 if (fp == NULL) { status = 1; break; }
332 for (int i = 0; i < RESTART_V3_WORDS && status == 0; ++i) if (RestartGetWord(fp, &h[i]) != 0) status = 2;
333 if (status == 0 && (h[RW_MAGIC] != expected[RW_MAGIC] || h[RW_VERSION] != 3 || h[RW_RANKS] != P || h[RW_RANK] != r)) status = 3;
334 if (status == 0 && h[RW_OFFSET] != running) status = 4;
335 if (status == 0) {
336 long end;
337 if (fseek(fp, 0, SEEK_END) != 0 || (end = ftell(fp)) < 0 || h[RW_LOCAL_DIM] > (UINT64_MAX - RESTART_V3_HEADER_BYTES) / 16 ||
338 (uint64_t)end != RESTART_V3_HEADER_BYTES + 16 * h[RW_LOCAL_DIM]) status = 5;
339 }
340 if (status == 0) {
341 if (r == 0) memcpy(reference, h, sizeof(h));
342 else for (int i = 0; i < RESTART_V3_WORDS; ++i)
343 if (i != RW_RANK && i != RW_OFFSET && i != RW_LOCAL_DIM && h[i] != reference[i]) status = 6;
344 if (h[RW_LOCAL_DIM] > UINT64_MAX - running) status = 8;
345 else { offsets[r] = h[RW_OFFSET]; dims[r] = h[RW_LOCAL_DIM]; running += h[RW_LOCAL_DIM]; }
346 }
347 fclose(fp);
348 }
349 if (status == 0 && running != reference[RW_DIM]) status = 7;
350 }
351 status = BcastMPI_i(0, status);
352 if (status != 0) { fprintf(stdoutMPI, "Error: restart checkpoint header scan failed (code %d).\n", status); goto fail; }
353 bcast_words(reference, RESTART_V3_WORDS); bcast_words(offsets, P); bcast_words(dims, P);
354 const char *bad = NULL;
355 int invalid = check_identity(expected, reference, &bad) != 0 || reference[RW_COMPONENT] != (uint64_t)comp;
356 if (purpose == RESTART_READ_CONTINUATION) { /* Q3: method, unit size, step range, int range */
357 if (agree(reference[RW_STEP] < step_lower_bound(X->Def.iCalcType) ||
358 reference[RW_STEP] > (uint64_t)(INT_MAX - 1), "step range")) goto fail;
359 invalid |= reference[RW_STATE] != row->set || reference[RW_STEP] != row->step ||
360 reference[RW_METHOD] != (uint64_t)X->Def.iCalcType ||
361 (X->Def.iCalcType == CG && reference[RW_NSET] != (uint64_t)X->Def.k_exct);
362 }
363 if (purpose == RESTART_READ_CONTINUATION) {
364 invalid |= X->Def.iCalcType != CG && reference[RW_FLAGS] != 0;
365 if (X->Def.iCalcType == cTPQ) {
366 double expected_dt = 1.0 / LargeValue, saved_dt = RestartWordDouble(reference[RW_SCALAR_C]);
367 invalid |= RestartWordDouble(reference[RW_SCALAR_B]) < 0 || !isfinite(expected_dt) || expected_dt <= 0 ||
368 fabs(saved_dt - expected_dt) > 1e-12 * expected_dt;
369 }
370 }
371 if (!invalid) invalid = !isfinite(RestartWordDouble(reference[RW_TIME])) || !isfinite(RestartWordDouble(reference[RW_SCALAR_A])) ||
372 !isfinite(RestartWordDouble(reference[RW_SCALAR_B])) || !isfinite(RestartWordDouble(reference[RW_SCALAR_C]));
373 if (invalid && bad != NULL) fprintf(stderr, "Error: rank %d restart checkpoint %s differs.\n", myrank, bad);
374 if (agree(invalid, "identity / unit validation")) goto fail;
375 *saved_offsets = offsets; *saved_dims = dims; return 0;
376fail:
377 free(offsets); free(dims); return -1;

413static int read_set_files(struct BindStruct *X, const struct RestartManifestRow *row, int comp,
414 enum RestartReadPurpose purpose, const char *base,
415 double complex *vector, struct RestartVectorInfo *info)
416{
417 uint64_t reference[RESTART_V3_WORDS], expected[RESTART_V3_WORDS];
418 uint64_t *offsets = NULL, *dims = NULL;
419 int invalid = 0;
420 if (inspect_vector_header(X, row, comp, purpose, base, reference, &offsets, &dims) != 0) return -1;
421 const uint64_t P = row->ranks;
422 /* expected is used only for the Hamiltonian provenance message below. */
423 if (agree(build_identity(X, expected) != 0, "identity")) goto fail;
424 /* intersection reads (S3): lo/hi per old file */
425 uint64_t a = RestartLocalOffset(X);
426 if (agree(a > reference[RW_DIM] || X->Check.idim_max > reference[RW_DIM] - a ||
427 X->Check.idim_max >= SIZE_MAX / sizeof(double complex), "local interval / allocation size")) goto fail;
428 uint64_t b = a + X->Check.idim_max;
429 double complex *tmp = calloc(X->Check.idim_max + 1, sizeof(*tmp));
430 invalid = tmp == NULL;
431 for (uint64_t j = 0; !invalid && j < P; ++j) {
432 uint64_t lo, hi;
433 if (!RestartIntersect(a, b, offsets[j], dims[j], &lo, &hi)) continue;
434 char name[D_FileNameMax];
435 if (vector_name(name, sizeof(name), X, row, comp, (int)j, base) != 0) { invalid = 1; break; }
436 FILE *fp = open_output(name, "rb");
437 uint64_t byte = RESTART_V3_HEADER_BYTES + 16 * (lo - offsets[j]);
438 invalid = fp == NULL || byte > (uint64_t)LONG_MAX || fseek(fp, (long)byte, SEEK_SET) != 0;
439 for (uint64_t g = lo; !invalid && g < hi; ++g) {
440 uint64_t re, im;
441 invalid = RestartGetWord(fp, &re) != 0 || RestartGetWord(fp, &im) != 0;
442 if (!invalid) { double c[2] = {RestartWordDouble(re), RestartWordDouble(im)}; memcpy(&tmp[(g - a) + 1], c, sizeof(c)); }
443 }
444 if (!invalid && hi == offsets[j] + dims[j] && (fgetc(fp) != EOF || ferror(fp))) invalid = 1;
445 if (fp != NULL) fclose(fp);
446 }
447 if (agree(invalid, "payload read")) { free(tmp); goto fail; }
448 uint64_t x = 0, s = 0, gx, gs; double norm = 0;
449 for (unsigned long i = 1; i <= X->Check.idim_max; ++i) {
450 double re = creal(tmp[i]), im = cimag(tmp[i]);
451 if (!isfinite(re) || !isfinite(im)) invalid = 1;
452 norm += re*re + im*im;
453 }
454 RestartDigestAccumulate(a, tmp, X->Check.idim_max, &x, &s);
455 norm = SumMPI_d(norm);
456 allreduce_xor_sum(x, s, &gx, &gs);
457 invalid |= gx != reference[RW_DATA_XOR] || gs != reference[RW_DATA_SUM];
458 if (comp == RESTART_COMP_STATE) invalid |= !isfinite(norm) || fabs(norm - 1.0) > 1e-8;
459 if (agree(invalid, "payload checksum / norm")) { free(tmp); goto fail; }
460 memcpy(vector, tmp, (X->Check.idim_max + 1) * sizeof(*vector)); free(tmp);
461 info->method = reference[RW_METHOD]; info->set = reference[RW_STATE]; info->step = reference[RW_STEP];
462 info->time = RestartWordDouble(reference[RW_TIME]); info->nset = reference[RW_NSET]; info->flags = reference[RW_FLAGS];
463 info->scalar_a = RestartWordDouble(reference[RW_SCALAR_A]); info->scalar_b = RestartWordDouble(reference[RW_SCALAR_B]);
464 info->scalar_c = RestartWordDouble(reference[RW_SCALAR_C]);
465 fprintf(stdoutMPI, "Restart checkpoint loaded: set=%" PRIu64 " comp=%d step=%" PRIu64 " flags=%" PRIu64 " time=%.17g hamiltonian_changed=%s\n",
466 info->set, comp, info->step, info->flags, info->time,
467 reference[RW_BASIS_KIND] == RESTART_BASIS_RAW ? "unknown" : (reference[RW_HAMILTONIAN] == expected[RW_HAMILTONIAN] ? "no" : "yes"));
468 if (purpose == RESTART_READ_IMPORT &&
469 record_import(X, info, reference[RW_HAMILTONIAN], info->method == FullDiag) != 0) goto fail;
470 free(offsets); free(dims); return 0;
471fail:
472 free(offsets); free(dims); return -1;
473}

497int RestartImportVector(struct BindStruct *X, const char *base, double complex *vector,
498 struct RestartVectorInfo *provenance, enum RestartFileFormat *format, int *source_ranks)
499{
500 int status = 0, kind = RESTART_FORMAT_LEGACY_RAW, ranks = nproc;
501 if (agree(base == NULL || vector == NULL || provenance == NULL || format == NULL ||
502 source_ranks == NULL, "import arguments")) return -1;
503 if (myrank == 0) {
504 char name[D_FileNameMax];
505 FILE *fp = NULL;
506 uint64_t magic, version, saved_ranks;
507 status = vector_name(name, sizeof(name), X, NULL, 0, 0, base) != 0;
508 if (!status) status = (fp = open_output(name, "rb")) == NULL;
509 if (!status) status = RestartGetWord(fp, &magic) != 0 || RestartGetWord(fp, &version) != 0;
510 if (!status && magic == UINT64_C(0x0a31565349485048)) {
511 if (version < 1 || version > 3) status = 1;
512 else {
513 kind = (int)version;
514 status = fseek(fp, 8 * RW_RANKS, SEEK_SET) != 0 || RestartGetWord(fp, &saved_ranks) != 0;
515 if (!status) {
516 status = saved_ranks == 0 || saved_ranks > INT_MAX;
517 if (status) fprintf(stdoutMPI, "Error: restart checkpoint MPI rank count differs.\n");
518 else ranks = (int)saved_ranks;
519 }
520 }
521 }
522 if (fp && fclose(fp) != 0) status = 1;
523 }
524 status = BcastMPI_i(0, status);
525 kind = BcastMPI_i(0, kind); ranks = BcastMPI_i(0, ranks);
526 if (import_probe_failure(X, status, "import format / version / source rank count")) return -1;
527 *format = (enum RestartFileFormat)kind; *source_ranks = ranks;
528 if (agree(X->Def.RestartSourceRanks > 0 && X->Def.RestartSourceRanks != ranks,
529 "RestartSourceRanks does not match the saved rank count")) return -1;
530 if (ranks != nproc) {
531 fprintf(stdoutMPI, "Error: restart checkpoint MPI rank count differs.\n");
532 if (import_probe_failure(X, 1, "rank count (rank redistribution arrives in PR-3)")) return -1;
533 }
534 memset(provenance, 0, sizeof(*provenance));
535 if (kind == RESTART_FORMAT_V3) {
536 const struct RestartManifestRow row = {0, (uint64_t)ranks, 0, 0, 1};
537 return read_set_files(X, &row, RESTART_COMP_STATE, RESTART_READ_IMPORT, base, vector, provenance);
538 }
539 if (kind == RESTART_FORMAT_LEGACY_RAW) {
540 if (import_probe_failure(X, X->Def.iFlgSymmetryBasis, "legacy raw import requires a raw basis")) return -1;
541 return -2;
542 }
543 if (agree(!X->Def.iFlgSymmetryBasis, "sector import requires a sector basis")) return -1;
544 char name[D_FileNameMax];
545 struct SymmetryCheckpointInfo info;
546 if (agree(vector_name(name, sizeof(name), X, NULL, 0, myrank, base) != 0, "import filename")) return -1;
547 if (ReadSymmetryCheckpoint(X, name, vector, &info) != 0) return -1;
548 provenance->method = info.source_method; provenance->set = info.state_index;
549 provenance->step = info.step; provenance->time = info.time;
550 return record_import(X, provenance, info.hamiltonian_digest,
551 info.hamiltonian_digest_kind == SYMMETRY_HAMILTONIAN_MATRIX_V1);
552}

554static int validate_set_boundary(struct BindStruct *X, const struct RestartManifest *m, uint64_t set)
555{
556 const struct RestartManifestRow *row = RestartManifestFind(m, set);
557 uint64_t first[RESTART_V3_WORDS], h[RESTART_V3_WORDS];
558 int bad = row == NULL || m->method != X->Def.iCalcType || m->basis != basis_kind(X);
559 if (!bad) bad = row->ncomp != (uint64_t)(X->Def.iCalcType == CG ? 2 : 1);
560 if (agree(bad, "recovery boundary manifest")) return -1;
561 for (int c = 0; c < (int)row->ncomp; ++c) {
562 uint64_t *offsets = NULL, *dims = NULL;
563 if (inspect_vector_header(X, row, c, RESTART_READ_CONTINUATION, NULL, h, &offsets, &dims) != 0) return -1;
564 free(offsets); free(dims);
565 if (c == 0) memcpy(first, h, sizeof(first));
566 else for (int w = 0; w < RESTART_V3_WORDS; ++w)
567 if (w != RW_COMPONENT && w != RW_DATA_XOR && w != RW_DATA_SUM && first[w] != h[w]) bad = 1;
568 }
569 if (agree(bad, "component scalars differ / recovery boundary")) return -1;
570 /* A saved CG block has one basis identity, including its recorded layout.
571 * Current-vs-saved layout compatibility is a separate reader decision. */
572 const struct RestartManifestRow *first_row = RestartManifestFind(m, 0);
573 if (X->Def.iCalcType == CG && set != 0 && first_row != NULL) {
574 uint64_t *offsets = NULL, *dims = NULL;
575 if (inspect_vector_header(X, first_row, RESTART_COMP_STATE, RESTART_READ_CONTINUATION, NULL,
576 h, &offsets, &dims) != 0) return -1;
577 free(offsets); free(dims);
578 for (size_t i = 0; i < sizeof(identity_words) / sizeof(identity_words[0]); ++i)
579 if (first[identity_words[i]] != h[identity_words[i]]) bad = 1;
580 bad |= first[RW_LAYOUT] != h[RW_LAYOUT];
581 }
582 return agree(bad, "CG state identities differ / recovery boundary");
583}

585int RestartClassifySet(struct BindStruct *X, const struct RestartManifest *manifest, uint64_t set,
586 const char *legacy_pattern, enum RestartSource *source, int *legacy_step)
587{
588 *source = RESTART_SOURCE_NONE; *legacy_step = 0;
589 if (agree(set > INT_MAX || (manifest != NULL &&
590 (manifest->method != X->Def.iCalcType || manifest->basis != basis_kind(X))),
591 "manifest method / basis / set")) return -1;
592 if (manifest != NULL && RestartManifestFind(manifest, set) != NULL) {
593 if (validate_set_boundary(X, manifest, set) != 0) return -1;
594 *source = RESTART_SOURCE_V3; return 0;
595 }
596 if (legacy_pattern == NULL) return 0;
597 if (X->Def.RestartSourceRanks > 0 && X->Def.RestartSourceRanks != nproc) {
598 fprintf(stdoutMPI, "Error: legacy restart files with a different rank count arrive in a later release (RestartSourceRanks).\n");
599 return -1; /* PR-3 lifts this */
600 }
601 char name[D_FileNameMax];
602 if (X->Def.iCalcType == TimeEvolution) snprintf(name, sizeof(name), legacy_pattern, myrank);
603 else snprintf(name, sizeof(name), legacy_pattern, (int)set, myrank);
604 FILE *fp = open_output(name, "rb"); int present = fp != NULL, step = 0, bad = fp == NULL && errno != ENOENT;
605 uint64_t dim = 0;
606 if (present) {
607 long end = -1;
608 bad |= fread(&step, sizeof(int), 1, fp) != 1 || fread(&dim, sizeof(dim), 1, fp) != 1;
609 bad |= step < (int)step_lower_bound(X->Def.iCalcType) || step > INT_MAX - 1;
610 bad |= dim != (uint64_t)X->Check.idim_max || dim > (UINT64_MAX - 28) / 16;
611 bad |= fseek(fp, 0, SEEK_END) != 0 || (end = ftell(fp)) < 0;
612 if (!bad) bad = (uint64_t)end != RESTART_LEGACY_HEADER_BYTES + 16 * (dim + 1);
613 bad |= fclose(fp) != 0;
614 }
615 if (agree(bad, "legacy recovery header / step / file length")) return -1;
616 int count = SumMPI_i(present);
617 if (count == 0) return 0;
618 if (count != nproc) { fprintf(stdoutMPI, "Error: legacy restart files for set %" PRIu64 " exist on %d of %d ranks.\n", set, count, nproc); return -1; }
619 int root_step = BcastMPI_i(0, step);
620 if (agree(bad || step != root_step, "legacy header step agreement")) return -1;
621 *source = RESTART_SOURCE_LEGACY; *legacy_step = step;
622 return 0;
623}
