18/* Read one line of any length. A line without a final newline is marked
19 * incomplete so callers can discard an EOF fragment after seeing EOF. */
20static int read_line(FILE *fp, char **buf, size_t *cap, size_t *len,
21 int *complete)
22{
23 *len = 0;
24 *complete = 0;
25 for (;;) {
26 int ch = fgetc(fp);
27 if (ch == EOF) {
28 if (ferror(fp)) return -1;
29 if (*len == 0) return 0;
30 break;
31 }
32 if (*len + 2 > *cap) {
33 size_t ncap = *cap ? 2 * *cap : 4096;
34 char *nbuf = realloc(*buf, ncap);
35 if (nbuf == NULL) return -1;
36 *buf = nbuf;
37 *cap = ncap;
38 }
39 (*buf)[(*len)++] = (char)ch;
40 if (ch == '\n') {
41 *complete = 1;
42 break;
43 }
44 }
45 if (*cap == 0) {
46 *buf = malloc(1);
47 if (*buf == NULL) return -1;
48 *cap = 1;
49 }
50 (*buf)[*len] = '\0';
51 return 1;
52}

54static int parse_u64(const char *token, size_t n, uint64_t *value)
55{
56 uint64_t parsed = 0;
57
58 if (n == 0) return -1;
59 for (size_t i = 0; i < n; ++i) {
60 unsigned char ch = (unsigned char)token[i];
61 uint64_t digit;
62 if (ch < '0' || ch > '9') return -1;
63 digit = (uint64_t)(ch - '0');
64 if (parsed > (UINT64_MAX - digit) / 10) return -1;
65 parsed = parsed * 10 + digit;
66 }
67 *value = parsed;
68 return 0;
69}

73static int classify_data_row(const char *line,
74 const struct RestartSeriesSpec *spec)
75{
76 const char *tokens[2] = {NULL, NULL};
77 size_t lengths[2] = {0, 0};
78 const char *last = NULL;
79 size_t last_len = 0;
80 int ntokens = 0;
81
82 for (const char *p = line; *p != '\0';) {
83 while (*p != '\0' && isspace((unsigned char)*p)) ++p;
84 if (*p == '\0') break;
85 const char *start = p;
86 while (*p != '\0' && !isspace((unsigned char)*p)) ++p;
87 if (ntokens == spec->set_column) {
88 tokens[0] = start;
89 lengths[0] = (size_t)(p - start);
90 }
91 if (ntokens == spec->step_column) {
92 tokens[1] = start;
93 lengths[1] = (size_t)(p - start);
94 }
95 last = start;
96 last_len = (size_t)(p - start);
97 ++ntokens;
98 }
99 if (ntokens == 0) return 2;
100 if (spec->step_column < 0) {
101 tokens[1] = last;
102 lengths[1] = last_len;
103 }
104 if (tokens[1] == NULL ||
105 (spec->set_column >= 0 && tokens[0] == NULL)) return -1;
106
107 uint64_t step, set = 0;
108 if (parse_u64(tokens[1], lengths[1], &step) != 0) return -1;
109 if (spec->set_column >= 0 &&
110 parse_u64(tokens[0], lengths[0], &set) != 0) return -1;
111
112 const struct RestartSeriesRule *rule = NULL;
113 if (spec->set_column < 0) {
114 if (spec->nrule > 0 && spec->rules != NULL) rule = &spec->rules[0];
115 } else {
116 for (int i = 0; i < spec->nrule; ++i) {
117 if (spec->rules != NULL && spec->rules[i].set == set) {
118 rule = &spec->rules[i];
119 break;
120 }
121 }
122 }
123 if (rule == NULL) return spec->drop_unlisted_sets ? 0 : 1;
124 return step < rule->keep_below_step ? 1 : 0;
125}
