100 double complex *tmp_v0,
101 double complex *tmp_v1
102) {
103#ifdef MPI
104 int mask1, mask2, state1, state2, ierr, origin, bitdiff, Fsgn;
105 unsigned long int idim_max_buf, j;
106 MPI_Status statusMPI;
107 double complex trans, dmv, dam_pr;
108
109 mask1 = (int)X->Def.Tpow[2 * org_isite1 + org_ispin1];
110 mask2 = (int)X->Def.Tpow[2 * org_isite2 + org_ispin2];
111 if (mask2 > mask1) bitdiff = mask2 - mask1 * 2;
112 else bitdiff = mask1 - mask2 * 2;
113 origin = myrank ^ (mask1 + mask2);
114
115 state1 = origin & mask1;
116 state2 = origin & mask2;
117
118 SgnBit((unsigned long int) (origin & bitdiff), &Fsgn); // Fermion sign
119
120 if (state1 == 0 && state2 == mask2) {
121 trans = -(double)Fsgn * tmp_trans;
122 }/*if (state1 == 0 && state2 == mask2)*/
123 else if (state1 == mask1 && state2 == 0) {
124 trans = -(double)Fsgn * conj(tmp_trans);
125 if (X->Large.mode == M_CORR || X->Large.mode == M_CALCSPEC) trans = 0.0;
126 }/*if (state1 == mask1 && state2 == 0)*/
127 else return 0.0;
128
129 ierr = MPI_Sendrecv(&X->Check.idim_max, 1, MPI_UNSIGNED_LONG, origin, 0,
130 &idim_max_buf, 1, MPI_UNSIGNED_LONG, origin, 0,
131 MPI_COMM_WORLD, &statusMPI);
132 if (ierr != 0) exitMPI(-1);
133 ierr = MPI_Sendrecv(tmp_v1, X->Check.idim_max + 1, MPI_DOUBLE_COMPLEX, origin, 0,
134 v1buf, idim_max_buf + 1, MPI_DOUBLE_COMPLEX, origin, 0,
135 MPI_COMM_WORLD, &statusMPI);
136 if (ierr != 0) exitMPI(-1);
137
138 dam_pr = 0.0;
139#pragma omp parallel default(none) reduction(+:dam_pr) private(j, dmv) \
140firstprivate(idim_max_buf, trans, X) shared(v1buf, tmp_v1, tmp_v0)
141 {
142 if (X->Large.mode == M_MLTPLY || X->Large.mode == M_CALCSPEC) {
143#pragma omp for
144 for (j = 1; j <= idim_max_buf; j++) {
145 dmv = trans * v1buf[j];
146 tmp_v0[j] += dmv;
147 dam_pr += conj(tmp_v1[j]) * dmv;
148 }/*for (j = 1; j <= idim_max_buf; j++)*/
149 }/*if (X->Large.mode == M_MLTPLY || X->Large.mode == M_CALCSPEC)*/

151#pragma omp for
152 for (j = 1; j <= idim_max_buf; j++) {
153 dmv = trans * v1buf[j];
154 dam_pr += conj(tmp_v1[j]) * dmv;
155 }/*for (j = 1; j <= idim_max_buf; j++)*/
156 }
157 }/*End of parallel region*/

290 Prepare index in the inter PE
291 */
292 mask2 = (int) X->Def.Tpow[2 * org_isite2 + org_ispin2];

338

341

346 }/*for (j = 0; j < idim_max_buf; j++)*/

360 dam_pr += conj(tmp_v1[ioff + 1]) * dmv;
361 }/*for (j = 0; j < idim_max_buf; j++)*/
362 }/*for (j = 0; j < idim_max_buf; j++)*/
363 }/*if (! (X->Large.mode == M_MLTPLY|| X->Large.mode == M_CALCSPEC))*/
