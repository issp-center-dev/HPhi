36static unsigned long long HashSymmetryBytes(unsigned long long hash,
37 const void *data,
38 size_t size)
39{
40 const unsigned char *bytes = (const unsigned char *)data;
41 size_t i;
42 for (i = 0U; i < size; i++) {
43 hash ^= (unsigned long long)bytes[i];
44 hash *= 1099511628211ULL;
45 }
46 return hash;
47}

49static unsigned long long SymmetryBasisDigest(
50 const struct SymmetryBasisRuntime *sym)
51{
52 unsigned long long hash = 14695981039346656037ULL;
53 unsigned long int index;
54 hash = HashSymmetryBytes(hash, &sym->dim, sizeof(sym->dim));
55 for (index = 1UL; index <= sym->dim; index++) {
56 const struct SymmetryBasisVector *entry = &sym->basis[index];
57 hash = HashSymmetryBytes(hash, &entry->rep_state,
58 sizeof(entry->rep_state));
59 hash = HashSymmetryBytes(hash, &entry->orbit_size,
60 sizeof(entry->orbit_size));
61 hash = HashSymmetryBytes(hash, &entry->stabilizer_size,
62 sizeof(entry->stabilizer_size));
63 hash = HashSymmetryBytes(hash, &entry->norm, sizeof(entry->norm));
64 hash = HashSymmetryBytes(hash, &entry->stabilizer_character_sum,
65 sizeof(entry->stabilizer_character_sum));
66 hash = HashSymmetryBytes(hash, &entry->diagonal,
67 sizeof(entry->diagonal));
68 }
69 return hash;
70}

72static void OutputSymmetryRankStats(const struct BindStruct *X)
73{
74 static const int timer_ids[] = {
75 1100, 1110, 1115, 1111, 1112, 1113, 1114,
76 1101, 1120, 1121, 1122, 4113
77 };
78 static const char *work_keys[] = {
79 "basis_raw_states",
80 "basis_representative_candidates",
81 "basis_compatible_survivors",
82 "basis_transform_calls",
83 "basis_orbit_metadata_calls",
84 "basis_thread_count",
85 "basis_thread_raw_states_max",
86 "basis_thread_representative_candidates_max",
87 "basis_thread_compatible_survivors_max",
88 "basis_thread_transform_calls_max",
89 "basis_gather_entries",
90 "basis_gather_bytes",
91 "plan_local_rows",
92 "plan_local_nnz",
93 "plan_row_nnz_max"
94 };
95 const size_t timer_count = sizeof(timer_ids) / sizeof(timer_ids[0]);
96 const size_t work_count = sizeof(work_keys) / sizeof(work_keys[0]);
97 const struct SymmetryMatvecPlan *plan;
98 double timer_local[sizeof(timer_ids) / sizeof(timer_ids[0])];
99 double timer_min[sizeof(timer_ids) / sizeof(timer_ids[0])];
100 double timer_max[sizeof(timer_ids) / sizeof(timer_ids[0])];
101 double timer_sum[sizeof(timer_ids) / sizeof(timer_ids[0])];
102 unsigned long long work_local[sizeof(work_keys) / sizeof(work_keys[0])];
103 unsigned long long work_min[sizeof(work_keys) / sizeof(work_keys[0])];
104 unsigned long long work_max[sizeof(work_keys) / sizeof(work_keys[0])];
105 unsigned long long work_sum[sizeof(work_keys) / sizeof(work_keys[0])];
106 double row_mean_local;
107 double row_mean_min;
108 double row_mean_max;
109 double row_mean_sum;
110 unsigned long long basis_digest_local;
111 unsigned long long basis_digest_min;
112 unsigned long long basis_digest_max;
113 char fileName[D_FileNameMax];
114 FILE *fp;
115 size_t i;
116
117 if (X == NULL || X->Def.iFlgSymmetryBasis == FALSE || X->Sym == NULL ||
118 X->Sym->enabled != TRUE) {
119 return;
120 }
121 plan = X->Sym->matvec_plan;
122 for (i = 0; i < timer_count; i++) timer_local[i] = Timer[timer_ids[i]];
123 work_local[0] = X->Sym->basis_raw_states;
124 work_local[1] = X->Sym->basis_representative_candidates;
125 work_local[2] = X->Sym->basis_compatible_survivors;
126 work_local[3] = X->Sym->basis_transform_calls;
127 work_local[4] = X->Sym->basis_orbit_metadata_calls;
128 work_local[5] = (unsigned long long)X->Sym->basis_thread_count;
129 work_local[6] = X->Sym->basis_thread_raw_states_max;
130 work_local[7] = X->Sym->basis_thread_representative_candidates_max;
131 work_local[8] = X->Sym->basis_thread_compatible_survivors_max;
132 work_local[9] = X->Sym->basis_thread_transform_calls_max;
133 work_local[10] = X->Sym->basis_gather_entries;
134 work_local[11] = X->Sym->basis_gather_bytes;
135 work_local[12] = plan != NULL ? (unsigned long long)plan->local_dim : 0ULL;
136 work_local[13] = plan != NULL ? (unsigned long long)plan->nnz : 0ULL;
137 work_local[14] = plan != NULL ? (unsigned long long)plan->row_nnz_max : 0ULL;
138 row_mean_local = plan != NULL && plan->local_dim > 0UL
139 ? (double)plan->nnz / (double)plan->local_dim
140 : 0.0;
141 basis_digest_local = SymmetryBasisDigest(X->Sym);
142
143 MPI_Allreduce(timer_local, timer_min, (int)timer_count, MPI_DOUBLE,
144 MPI_MIN, MPI_COMM_WORLD);
145 MPI_Allreduce(timer_local, timer_max, (int)timer_count, MPI_DOUBLE,
146 MPI_MAX, MPI_COMM_WORLD);
147 MPI_Allreduce(timer_local, timer_sum, (int)timer_count, MPI_DOUBLE,
148 MPI_SUM, MPI_COMM_WORLD);
149 MPI_Allreduce(work_local, work_min, (int)work_count, MPI_UNSIGNED_LONG_LONG,
150 MPI_MIN, MPI_COMM_WORLD);
151 MPI_Allreduce(work_local, work_max, (int)work_count, MPI_UNSIGNED_LONG_LONG,
152 MPI_MAX, MPI_COMM_WORLD);
153 MPI_Allreduce(work_local, work_sum, (int)work_count, MPI_UNSIGNED_LONG_LONG,
154 MPI_SUM, MPI_COMM_WORLD);
155 MPI_Allreduce(&row_mean_local, &row_mean_min, 1, MPI_DOUBLE,
156 MPI_MIN, MPI_COMM_WORLD);
157 MPI_Allreduce(&row_mean_local, &row_mean_max, 1, MPI_DOUBLE,
158 MPI_MAX, MPI_COMM_WORLD);
159 MPI_Allreduce(&row_mean_local, &row_mean_sum, 1, MPI_DOUBLE,
160 MPI_SUM, MPI_COMM_WORLD);
161 MPI_Allreduce(&basis_digest_local, &basis_digest_min, 1,
162 MPI_UNSIGNED_LONG_LONG, MPI_MIN, MPI_COMM_WORLD);
163 MPI_Allreduce(&basis_digest_local, &basis_digest_max, 1,
164 MPI_UNSIGNED_LONG_LONG, MPI_MAX, MPI_COMM_WORLD);
165
166 sprintf(fileName, "CalcTimerRankStats.dat");
167 if (childfopenMPI(fileName, "w", &fp) != 0) return;
168 fprintf(fp, "format=HPhiCalcTimerRankStats version=1 ranks=%d\n", nproc);
169 for (i = 0; i < timer_count; i++) {
170 fprintf(fp, "timer id=%d ranks=%d min=%.17g max=%.17g mean=%.17g\n",
171 timer_ids[i], nproc, timer_min[i], timer_max[i],
172 timer_sum[i] / (double)nproc);
173 }
174 for (i = 0; i < work_count; i++) {
175 fprintf(fp, "work key=%s ranks=%d min=%llu max=%llu mean=%.17g\n",
176 work_keys[i], nproc, work_min[i], work_max[i],
177 (double)work_sum[i] / (double)nproc);
178 }
179 fprintf(fp,
180 "work key=plan_row_nnz_mean ranks=%d min=%.17g max=%.17g mean=%.17g\n",
181 nproc, row_mean_min, row_mean_max, row_mean_sum / (double)nproc);
182 fprintf(fp,
183 "basis_digest algorithm=fnv1a64-fields ranks=%d min=%016llx max=%016llx\n",
184 nproc, basis_digest_min, basis_digest_max);
185 fclose(fp);
186}

209void InitTimer() {
210#ifdef MPI
211 int i;
212 int NTimer=10000;
213 Timer = (double*)malloc((NTimer)*sizeof(double));
214 TimerStart = (double*)malloc((NTimer)*sizeof(double));
215 for(i=0;i<NTimer;i++) Timer[i]=0.0;
216 for(i=0;i<NTimer;i++) TimerStart[i]=0.0;
217#endif
218 return;
219}
