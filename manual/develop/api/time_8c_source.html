53void InitTimer() {
54#ifdef MPI
55 int i;
56 int NTimer=10000;
57 Timer = (double*)malloc((NTimer)*sizeof(double));
58 TimerStart = (double*)malloc((NTimer)*sizeof(double));
59 for(i=0;i<NTimer;i++) Timer[i]=0.0;
60 for(i=0;i<NTimer;i++) TimerStart[i]=0.0;
61#endif
62 return;

83void StopTimer(int n) {
