29/* Share a local error across all ranks so every rank takes the same
30 branch before each collective ELPA call (design doc section 4). */
31static int SyncError(int ierr) {
32 int gerr = 0;
33 MPI_Allreduce(&ierr, &gerr, 1, MPI_INT, MPI_MIN, MPI_COMM_WORLD);
34 return gerr;
35}

37/* elpa_set for int values with mandatory error check. */
38static int SetElpaInt(elpa_t handle, const char *name, int value) {
39 int error = ELPA_OK;
40 elpa_set(handle, name, value, &error);
41 if (error != ELPA_OK) {
42 fprintf(stdout, " Error: elpa_set(\"%s\", %d) failed: %s\n",
43 name, value, elpa_strerr(error));
44 return -1;
45 }
46 return 0;
47}

55int diag_elpa_cmp(int xNsize, double complex *A_distr,
56 double complex *Z_distr, double *w,
57 int local_nrows, int local_ncols,
58 int myrow, int mycol, int nblk, int ngpu) {
59 elpa_t handle = NULL;
60 int error = ELPA_OK;
61 int ierr = 0;
62
63 if (elpa_init(20211125) != ELPA_OK) {
64 fprintf(stdout, " Error: the linked ELPA is older than API 20211125 (2021.11).\n");
65 ierr = -1;
66 }
67 if (SyncError(ierr) != 0) return -1;
68
69 handle = elpa_allocate(&error);
70 if (error != ELPA_OK) {
71 fprintf(stdout, " Error: elpa_allocate failed: %s\n", elpa_strerr(error));
72 ierr = -1;
73 }
74
75 /* Mandatory parameters: set BEFORE elpa_setup (ELPA manual sec. 2). */
76 if (ierr == 0) {
77 if (SetElpaInt(handle, "na", xNsize) != 0 ||
78 SetElpaInt(handle, "nev", xNsize) != 0 ||
79 SetElpaInt(handle, "local_nrows", local_nrows) != 0 ||
80 SetElpaInt(handle, "local_ncols", local_ncols) != 0 ||
81 SetElpaInt(handle, "nblk", nblk) != 0 ||
82 SetElpaInt(handle, "mpi_comm_parent",
83 (int)MPI_Comm_c2f(MPI_COMM_WORLD)) != 0 ||
84 SetElpaInt(handle, "process_row", myrow) != 0 ||
85 SetElpaInt(handle, "process_col", mycol) != 0) {
86 ierr = -1;
87 }
88 }
89 if (SyncError(ierr) != 0) goto cleanup_fail;
90
91 error = elpa_setup(handle);
92 if (error != ELPA_OK) {
93 fprintf(stdout, " Error: elpa_setup failed: %s\n", elpa_strerr(error));
94 ierr = -1;
95 }
96 if (SyncError(ierr) != 0) goto cleanup_fail;
97
98 /* Tunable runtime options: set AFTER elpa_setup (ELPA manual sec. 2).
99 GPU: 1stage is usually faster on GPU; 2stage on CPU. */
100 if (ngpu >= 1) {
101 if (SetElpaInt(handle, "solver", ELPA_SOLVER_1STAGE) != 0) ierr = -1;
102 if (ierr == 0 && SetElpaInt(handle, ELPA_GPU_OPTION, 1) != 0) {
103 fprintf(stdout,
104 " Error: the linked ELPA has no NVIDIA GPU support.\n"
105 " Set \"NGPU 0\" in calcmod.def for CPU execution.\n");
106 ierr = -1;
107 }
108 } else {
109 /* 2stage is preferred on CPU, but ELPA's 2stage solver returned
110 deterministically inaccurate eigenvectors (residual ~1e-6) for
111 capped block sizes (observed with nblk=24 on a 4x2 grid, ELPA
112 2025.06.001). Capping only happens for matrices small relative to
113 the process grid, where 1stage is safe and fast enough, so fall
114 back to 1stage whenever the block size was capped below ELPA_NBLK. */
115 int solver = (nblk < ELPA_NBLK) ? ELPA_SOLVER_1STAGE : ELPA_SOLVER_2STAGE;
116 if (SetElpaInt(handle, "solver", solver) != 0) ierr = -1;
117 }
118 if (SyncError(ierr) != 0) goto cleanup_fail;
119
120#ifdef _ELPA_GPU
121 if (ngpu >= 1) {
122 error = elpa_setup_gpu(handle);
123 if (error != ELPA_OK) {
124 fprintf(stdout,
125 " Error: elpa_setup_gpu failed: %s\n"
126 " Check the GPU environment, or set \"NGPU 0\" for CPU execution.\n",
127 elpa_strerr(error));
128 ierr = -1;
129 }
130 if (SyncError(ierr) != 0) goto cleanup_fail;
131 }
132#else
133 if (ngpu >= 1) {
134 /* readdef.c rejects this combination at startup; defense in depth. */
135 fprintf(stdout, " Error: this HPhi build has no ELPA GPU API (_ELPA_GPU).\n");
136 ierr = -1;
137 if (SyncError(ierr) != 0) goto cleanup_fail;
138 }
139#endif
140
141 /* Type-generic macro resolves to the double-complex solver. */
142 elpa_eigenvectors(handle, A_distr, w, Z_distr, &error);
143 if (error != ELPA_OK) {
144 fprintf(stdout, " Error: elpa_eigenvectors failed: %s\n", elpa_strerr(error));
145 ierr = -1;
146 }
147 if (SyncError(ierr) != 0) goto cleanup_fail;
148
149 elpa_deallocate(handle, &error);
150 elpa_uninit(&error);
151 return 0;
152
153cleanup_fail:
154 if (handle != NULL) {
155 elpa_deallocate(handle, &error);
156 }
157 elpa_uninit(&error);
158 return -1;
159}
160#endif /* _ELPA */
