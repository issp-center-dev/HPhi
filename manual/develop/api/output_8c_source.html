28
29 FILE *fp;
30 char sdt[D_FileNameMax];
31 long int i, i_max;
32 i_max = X->Check.idim_max;
33
34 if (X->Def.iCalcType == FullDiag) {
35 switch (X->Def.iCalcModel) {
36 case Spin:
37 case Hubbard:
38 case tJ:
39 case Kondo:
40 sprintf(sdt, cFileNamePhys_FullDiag, X->Def.CDataFileHead, X->Def.Nup, X->Def.Ndown);

42 case SpinGC:
43 case HubbardGC:
44 case KondoGC:
45 case tJGC:
46 sprintf(sdt, cFileNamePhys_FullDiag_GC, X->Def.CDataFileHead);
47 break;
48 default:
49 break;
50 }
51 if (childfopenMPI(sdt, "w", &fp) != 0) {
52 return -1;
53 }
54 fprintf(fp, " <H> <N> <Sz> <S2> <D> \n");
55 for (i = 0; i < i_max; i++) {
56 fprintf(fp, " %10lf %10lf %10lf %10lf %10lf\n", X->Phys.all_energy[i], X->Phys.all_num_up[i]+X->Phys.all_num_down[i], X->Phys.all_sz[i],
57 X->Phys.all_s2[i], X->Phys.all_doublon[i]);
58 }
59 fclose(fp);
60 }
61 else{
62 fprintf(stdoutMPI, "Error: output function is used only for FullDiag mode.");

64 }

75int outputHam(struct BindStruct *X){
76 long int i=0;
77 long int j=0;
78 long int imax = X->Check.idim_max;
79 long int ihermite=0;
80 char cHeader[256];
81 FILE *fp;
82 char sdt[D_FileNameMax];
83
84#pragma omp parallel for default(none) reduction(+:ihermite) firstprivate(imax) private(i, j) shared(Ham)
85 for (i=1; i<=imax; i++){
86 for (j=1; j<=i; j++){
87 if(cabs(Ham[i][j])>1.0e-13){
88 ihermite += 1;
89 }
90 }
91 }
92
93 strcpy(cHeader, "%%MatrixMarket matrix coordinate complex hermitian\n");
94 sprintf(sdt,cFileNamePhys_FullDiag_Ham, X->Def.CDataFileHead);
95 if(childfopenMPI(sdt,"w",&fp)!=0){
96 return -1;
97 }
98 fprintf(fp, "%s", cHeader);
99 fprintf(fp, "%ld %ld %ld \n", imax, imax, ihermite);
100 for (i=1; i<=imax; i++){
101 for (j=1; j<=i; j++){
102 if(cabs(Ham[i][j])>1.0e-13){
103 fprintf(fp, "%ld %ld %lf %lf\n",i,j,creal(Ham[i][j]),cimag(Ham[i][j]));
104 }
105 }
106 }
107 fclose(fp);
108 return 0;
109}
