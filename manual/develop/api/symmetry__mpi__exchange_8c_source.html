699 next_result.counts, 1, MPI_UINT64_T,
700 MPI_COMM_WORLD) != MPI_SUCCESS) {
701 goto fail;
702 }
703 }
704#endif
705 else {
706 next_result.counts[0] = send_layout->counts[0];
707 next_result.displacements[0] = 0U;
708 }
709
710 for (peer = 0; peer < nrank; peer++) {
711 if (known_receive_layout != NULL &&
712 next_result.displacements[peer] != recv_total) {
713 local_error = 1;
714 break;
715 }
716 if (known_receive_layout == NULL) {
717 next_result.displacements[peer] = recv_total;
718 }
719 if (SymmetryCheckedU64Add(recv_total, next_result.counts[peer],
720 &recv_total) != 0) {
721 local_error = 1;
722 break;
723 }
724 }
725 if (known_receive_layout != NULL &&
726 recv_total != known_receive_layout->count) {
727 local_error = 1;
728 }

730 recv_total > (uint64_t)(SIZE_MAX / descriptor->extent) ||
731 SymmetryCheckedSizeMul((size_t)recv_total,
732 descriptor != NULL ? descriptor->extent : 0U,
733 &payload_bytes) != 0) {
734 local_error = 1;
735 }
736 next_result.count = recv_total;
737 next_result.nrank = nrank;
738
739#ifdef MPI
740 if (mpi_active != FALSE && nrank > 1) {
741 use_chunked =
742 local_error == 0
743 ? layout_requires_chunking(send_layout, &next_result, chunk_limit,
744 force_chunked)
745 : TRUE;
746 if (MPI_Allreduce(&use_chunked, &global_use_chunked, 1, MPI_INT, MPI_MAX,
747 MPI_COMM_WORLD) != MPI_SUCCESS) {
748 goto fail;
749 }
750 use_chunked = global_use_chunked;
751 }
752#endif
753 if (SymmetryCheckedSizeAdd(result_bytes, payload_bytes,
754 &result_bytes) != 0 ||
755 exchange_workspace_bytes(nrank, mpi_active, use_chunked,
756 &workspace_bytes) != 0 ||
757 checked_budget_peak(budget, result_bytes, workspace_bytes) != 0) {
758 local_error = 1;
759 }
760 if (SymmetryMpiAgreeError(mpi_active, local_error) != 0) goto fail;
761 if (payload_bytes > 0U) {
762 next_result.entries = malloc(payload_bytes);
763 if (next_result.entries == NULL) local_error = 1;
764 }
765 if (SymmetryMpiAgreeError(mpi_active, local_error) != 0) goto fail;
766
767 next_stats.send_entries = send_layout->count;
768 next_stats.recv_entries = recv_total;
769 next_stats.message_entry_limit = chunk_limit;
770 next_stats.message_byte_limit = message_byte_limit;
771
772#ifdef MPI
773 if (mpi_active != FALSE && nrank > 1) {
774 {
775 uint64_t max_message_entries = 0U;
776 for (peer = 0; peer < nrank; peer++) {
777 uint64_t send_count;
778 uint64_t recv_count;
779 if (peer == rank) continue;
780 send_count = send_layout->counts[peer];
781 recv_count = next_result.counts[peer];
782 if (use_chunked != FALSE) {
783 if (send_count > chunk_limit) send_count = chunk_limit;
784 if (recv_count > chunk_limit) recv_count = chunk_limit;
785 }
786 if (send_count > max_message_entries) {
787 max_message_entries = send_count;
788 }
789 if (recv_count > max_message_entries) {
790 max_message_entries = recv_count;
791 }
792 }
793 next_stats.max_message_entries = max_message_entries;
794 next_stats.max_message_bytes =
795 max_message_entries * (uint64_t)descriptor->extent;
796 }
797 if (descriptor->create_type != NULL) {
798 if (descriptor->create_type(&payload_type) != 0) {
799 local_error = 1;
800 } else {
801 owns_payload_type = TRUE;
802 }
803 } else {
804 payload_type = descriptor->builtin_type;
805 if (payload_type == MPI_DATATYPE_NULL) local_error = 1;
806 }
807 if (local_error == 0 &&
808 (MPI_Type_get_extent(payload_type, &type_lower_bound, &type_extent) !=
809 MPI_SUCCESS ||
810 type_lower_bound != 0 ||
811 type_extent != (MPI_Aint)descriptor->extent)) {
812 local_error = 1;
813 }
814 if (SymmetryMpiAgreeError(mpi_active, local_error) != 0) goto fail;
815 if (use_chunked != FALSE) {
816 if (exchange_chunked(send_entries, send_layout, rank, chunk_limit,
817 &next_result, &next_stats, descriptor,
818 payload_type) != 0) {
819 goto fail;
820 }
821 next_stats.used_chunked = TRUE;
822 } else {
823 if (exchange_alltoallv(send_entries, send_layout, &next_result,
824 payload_type) != 0) {
825 goto fail;
826 }
827 }
828 if (owns_payload_type != FALSE) {
829 MPI_Type_free(&payload_type);
830 owns_payload_type = FALSE;
831 }
832 } else
833#endif
834 {
835 if (send_layout->count != recv_total) goto fail;
836 if (payload_bytes > 0U) {
837 memcpy(next_result.entries, send_entries, payload_bytes);
838 }
839 }
840
841 *result = next_result;
842 if (stats != NULL) *stats = next_stats;
843 return 0;
844
845fail:
846#ifdef MPI
847 if (owns_payload_type != FALSE &&
848 payload_type != MPI_DATATYPE_NULL) {
849 MPI_Type_free(&payload_type);
850 }
851#endif
852 free_raw_result(&next_result);
853 return -1;
854}
