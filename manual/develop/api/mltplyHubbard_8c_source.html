169 struct BindStruct *X,

171 double complex *tmp_v1

173 long unsigned int i;
174 long unsigned int isite1, isite2, sigma1, sigma2;
175 long unsigned int isite3, isite4, sigma3, sigma4;
176 long unsigned int ibitsite1, ibitsite2, ibitsite3, ibitsite4;
177
178 double complex dam_pr;
179 double complex tmp_trans;
180 /*[s] For InterAll */
181 double complex tmp_V;
182 /*[e] For InterAll */
183
184 int ihermite=0;
185 int idx=0;
186
187 StartTimer(300);
188
189#ifdef MPI
190 // Disable batching for TimeEvolution with step-dependent interactions.
191 // Same guard as mltplyHubbardGC: batched groups are initialized once
192 // and cannot track terms added by MakeTEDTransfer/MakeTEDInterAll.
193 int use_batching_H = MPIBatchingEnabled() &&
194 !(X->Def.iCalcType == TimeEvolution &&
195 (X->Def.NTEInterAllMax > 0 || X->Def.NTETransferMax > 0));
196#endif

201 StartTimer(310);
202
203#ifdef MPI
204 if (use_batching_H) {
205 // MPIdouble transfers - use batched communication
206 StartTimer(311);
207 if (!batched_double_Hubbard_initialized) {
208 if (InitializeMPIBatchedDoubleTransfers_Hubbard(X, &batched_double_Hubbard) != 0) {
209 fprintf(stderr, "Error: Failed to initialize batched MPI double transfers for Hubbard\n");
210 return -1;
211 }
212 batched_double_Hubbard_initialized = 1;
213 }
214
215 for (int g = 0; g < batched_double_Hubbard.num_groups; g++) {
216 dam_pr = X_child_general_hopp_MPIdouble_batched(
217 &batched_double_Hubbard.groups[g], X, tmp_v0, tmp_v1);
218 X->Large.prdct += dam_pr;
219 }
220 StopTimer(311);
221
222 // MPIsingle transfers - use batched communication
223 StartTimer(312);
224 if (!batched_transfers_Hubbard_initialized) {
225 if (InitializeMPIBatchedTransfers_Hubbard(X, &batched_transfers_Hubbard) != 0) {
226 fprintf(stderr, "Error: Failed to initialize batched MPI transfers for Hubbard\n");
227 return -1;
228 }
229 batched_transfers_Hubbard_initialized = 1;
230 }
231
232 for (int g = 0; g < batched_transfers_Hubbard.num_groups; g++) {
233 dam_pr = X_child_general_hopp_MPIsingle_batched(
234 &batched_transfers_Hubbard.groups[g], X, tmp_v0, tmp_v1);
235 X->Large.prdct += dam_pr;
236 }
237 StopTimer(312);
238 }
239#endif

241 for (i = 0; i < X->Def.EDNTransfer; i+=2) {
242#ifdef MPI
243 if (!use_batching_H) {
244 // Non-batched per-term MPI path (original develop behavior)
245 if (X->Def.EDGeneralTransfer[i][0] + 1 > X->Def.Nsite &&
246 X->Def.EDGeneralTransfer[i][2] + 1 > X->Def.Nsite) {
247 StartTimer(311);
248 general_hopp_MPIdouble(i, X, tmp_v0, tmp_v1);
249 StopTimer(311);
250 continue;
251 }
252 else if (X->Def.EDGeneralTransfer[i][2] + 1 > X->Def.Nsite) {
253 StartTimer(312);
254 general_hopp_MPIsingle(i, X, tmp_v0, tmp_v1);
255 StopTimer(312);
256 continue;
257 }
258 else if (X->Def.EDGeneralTransfer[i][0] + 1 > X->Def.Nsite) {
259 StartTimer(312);
260 general_hopp_MPIsingle(i + 1, X, tmp_v0, tmp_v1);
261 StopTimer(312);
262 continue;

265 X->Def.EDGeneralTransfer[i][2] + 1 > X->Def.Nsite) {
266 continue; // inter-PE handled by batching above

268#endif

270 StartTimer(313);
271 for (ihermite = 0; ihermite<2; ihermite++) {
272 idx = i + ihermite;
273 isite1 = X->Def.EDGeneralTransfer[idx][0] + 1;
274 isite2 = X->Def.EDGeneralTransfer[idx][2] + 1;
275 sigma1 = X->Def.EDGeneralTransfer[idx][1];
276 sigma2 = X->Def.EDGeneralTransfer[idx][3];
277 if (general_hopp_GetInfo(X, isite1, isite2, sigma1, sigma2) != 0) {
278 return -1;
279 }
280 tmp_trans = -X->Def.EDParaGeneralTransfer[idx];
281 X->Large.tmp_trans = tmp_trans;
282 dam_pr = general_hopp(tmp_v0, tmp_v1, X, tmp_trans);
283 X->Large.prdct += dam_pr;
284 }
285 StopTimer(313);
286 }
287 StopTimer(310);
291 StartTimer(320);
292 for (i = 0; i < X->Def.NInterAll_OffDiagonal; i+=2) {
293
294 isite1 = X->Def.InterAll_OffDiagonal[i][0] + 1;
295 isite2 = X->Def.InterAll_OffDiagonal[i][2] + 1;
296 isite3 = X->Def.InterAll_OffDiagonal[i][4] + 1;
297 isite4 = X->Def.InterAll_OffDiagonal[i][6] + 1;
298 sigma1 = X->Def.InterAll_OffDiagonal[i][1];
299 sigma2 = X->Def.InterAll_OffDiagonal[i][3];
300 sigma3 = X->Def.InterAll_OffDiagonal[i][5];
301 sigma4 = X->Def.InterAll_OffDiagonal[i][7];
302 tmp_V = X->Def.ParaInterAll_OffDiagonal[i];
303
304 dam_pr = 0.0;
305 if (CheckPE(isite1 - 1, X) == TRUE || CheckPE(isite2 - 1, X) == TRUE ||
306 CheckPE(isite3 - 1, X) == TRUE || CheckPE(isite4 - 1, X) == TRUE) {
307 StartTimer(321);
308 ibitsite1 = X->Def.OrgTpow[2*isite1-2+sigma1] ;
309 ibitsite2 = X->Def.OrgTpow[2 * isite2 - 2 + sigma2];
310 ibitsite3 = X->Def.OrgTpow[2 * isite3 - 2 + sigma3];
311 ibitsite4 = X->Def.OrgTpow[2 * isite4 - 2 + sigma4];
312 if (ibitsite1 == ibitsite2 && ibitsite3 == ibitsite4) {
313 dam_pr += child_CisAisCjtAjt_Hubbard_MPI(isite1 - 1, sigma1,
314 isite3 - 1, sigma3,
315 tmp_V, X, tmp_v0, tmp_v1);
316 }
317 else if (ibitsite1 == ibitsite2 && ibitsite3 != ibitsite4) {
318 dam_pr += child_CisAisCjtAku_Hubbard_MPI(isite1 - 1, sigma1,
319 isite3 - 1, sigma3, isite4 - 1, sigma4,
320 tmp_V, X, tmp_v0, tmp_v1);
321 }
322 else if (ibitsite1 != ibitsite2 && ibitsite3 == ibitsite4) {
323 dam_pr += child_CisAjtCkuAku_Hubbard_MPI(isite1 - 1, sigma1, isite2 - 1, sigma2,
324 isite3 - 1, sigma3, tmp_V, X, tmp_v0, tmp_v1);
325 }
326 else if (ibitsite1 != ibitsite2 && ibitsite3 != ibitsite4) {
327 dam_pr += child_CisAjtCkuAlv_Hubbard_MPI(isite1 - 1, sigma1, isite2 - 1, sigma2,
328 isite3 - 1, sigma3, isite4 - 1, sigma4, tmp_V, X, tmp_v0, tmp_v1);
329 }
330 StopTimer(321);
331 }
332 else {
333 StartTimer(322);
334 for (ihermite = 0; ihermite < 2; ihermite++) {
335 idx = i + ihermite;
336 isite1 = X->Def.InterAll_OffDiagonal[idx][0] + 1;
337 isite2 = X->Def.InterAll_OffDiagonal[idx][2] + 1;
338 isite3 = X->Def.InterAll_OffDiagonal[idx][4] + 1;
339 isite4 = X->Def.InterAll_OffDiagonal[idx][6] + 1;
340 sigma1 = X->Def.InterAll_OffDiagonal[idx][1];
341 sigma2 = X->Def.InterAll_OffDiagonal[idx][3];
342 sigma3 = X->Def.InterAll_OffDiagonal[idx][5];
343 sigma4 = X->Def.InterAll_OffDiagonal[idx][7];
344 tmp_V = X->Def.ParaInterAll_OffDiagonal[idx];
345
346 general_int_GetInfo(i, X, isite1, isite2, isite3, isite4,
347 sigma1, sigma2, sigma3, sigma4, tmp_V);
348
349 dam_pr += general_int(tmp_v0, tmp_v1, X);
350 }/*for (ihermite = 0; ihermite < 2; ihermite++)*/
351 StopTimer(322);
352 }
353 X->Large.prdct += dam_pr;
354 }/*for (i = 0; i < X->Def.NInterAll_OffDiagonal; i+=2)*/
355 StopTimer(320);
356
357 if (X->Def.iCalcModel == Hubbard && X->Def.NNBodyInterAll_OffDiagonal > 0) {
358 if (MultiplyNBodyInterAllHubbard(X, tmp_v0, tmp_v1) != 0) {
359 return -1;
360 }
361 }
365 StartTimer(330);
366 for (i = 0; i < X->Def.NPairHopping; i +=2) {
367 sigma1=0;
368 sigma2=1;
369 dam_pr = 0.0;
370
371 if ( X->Def.PairHopping[i][0] + 1 > X->Def.Nsite
372 || X->Def.PairHopping[i][1] + 1 > X->Def.Nsite)
373 {
374 StartTimer(331);
375 dam_pr = child_CisAjtCkuAlv_Hubbard_MPI(
376 X->Def.PairHopping[i][0], sigma1, X->Def.PairHopping[i][1], sigma1,
377 X->Def.PairHopping[i][0], sigma2, X->Def.PairHopping[i][1], sigma2,
378 X->Def.ParaPairHopping[i], X, tmp_v0, tmp_v1);
379 StopTimer(331);
380 }
381 else {
382 StartTimer(332);
383 for (ihermite = 0; ihermite<2; ihermite++) {
384 idx = i + ihermite;
385 pairhopp_GetInfo(idx, X);
386 dam_pr += pairhopp(tmp_v0, tmp_v1, X);
387 }/*for (ihermite = 0; ihermite<2; ihermite++)*/
388 StopTimer(332);
389 }
390 X->Large.prdct += dam_pr;
391 }/*for (i = 0; i < X->Def.NPairHopping; i += 2)*/
392 StopTimer(330);
396 StartTimer(340);
397 for (i = 0; i < X->Def.NExchangeCoupling; i ++) {
398 sigma1 = 0;
399 sigma2 = 1;
400 dam_pr=0.0;
401 if (X->Def.ExchangeCoupling[i][0] + 1 > X->Def.Nsite ||
402 X->Def.ExchangeCoupling[i][1] + 1 > X->Def.Nsite)
403 {
404 StartTimer(341);
405 dam_pr = child_CisAjtCkuAlv_Hubbard_MPI(
406 X->Def.ExchangeCoupling[i][0], sigma1, X->Def.ExchangeCoupling[i][1], sigma1,
407 X->Def.ExchangeCoupling[i][1], sigma2, X->Def.ExchangeCoupling[i][0], sigma2,
408 X->Def.ParaExchangeCoupling[i], X, tmp_v0, tmp_v1);
409 StopTimer(341);
410 }
411 else {
412 StartTimer(342);
413 exchange_GetInfo(i, X);
414 dam_pr = exchange(tmp_v0, tmp_v1, X);
415 StopTimer(342);
416 }
417 X->Large.prdct += dam_pr;
418 }/*for (i = 0; i < X->Def.NExchangeCoupling; i ++)*/
419 StopTimer(340);
420
421 StopTimer(300);
422 return 0;

435 long unsigned int i;

440 double complex dam_pr = 0.0;
441 double complex tmp_trans;
442 /*[s] For InterAll */
443 double complex tmp_V;
444 /*[e] For InterAll */
445
446 int ihermite=0;
447 int idx=0;
448
449 StartTimer(200);
450
451#ifdef MPI
452 // Disable batching for TimeEvolution with step-dependent interactions,
453 // because batched groups are initialized once and cannot track terms

455 // Peierls substitution (NLaser > 0) is safe since it only modifies
456 // existing coefficients, which the batched apply re-reads.
457 int use_batching = MPIBatchingEnabled() &&
458 !(X->Def.iCalcType == TimeEvolution &&
459 (X->Def.NTEInterAllMax > 0 || X->Def.NTETransferMax > 0));
460#endif
461
465 StartTimer(210);
466
467#ifdef MPI
468 if (use_batching) {
469 // MPIdouble transfers - use batched communication
470 StartTimer(211);
471 if (!batched_double_HubbardGC_initialized) {
472 if (InitializeMPIBatchedDoubleTransfers_HubbardGC(X, &batched_double_HubbardGC) != 0) {
473 fprintf(stderr, "Error: Failed to initialize batched MPI double transfers for HubbardGC\n");
474 return -1;
475 }
476 batched_double_HubbardGC_initialized = 1;
477 }
478
479 for (int g = 0; g < batched_double_HubbardGC.num_groups; g++) {
480 dam_pr = X_child_GC_general_hopp_MPIdouble_batched(
481 &batched_double_HubbardGC.groups[g], X, tmp_v0, tmp_v1);
482 X->Large.prdct += dam_pr;
483 }
484 StopTimer(211);
485
486 // MPIsingle transfers - use batched communication
487 StartTimer(212);
488 if (!batched_transfers_HubbardGC_initialized) {
489 if (InitializeMPIBatchedTransfers_HubbardGC(X, &batched_transfers_HubbardGC) != 0) {
490 fprintf(stderr, "Error: Failed to initialize batched MPI transfers for HubbardGC\n");
491 return -1;
492 }
493 batched_transfers_HubbardGC_initialized = 1;
494 }
495
496 for (int g = 0; g < batched_transfers_HubbardGC.num_groups; g++) {
497 dam_pr = X_child_GC_general_hopp_MPIsingle_batched(
498 &batched_transfers_HubbardGC.groups[g], X, tmp_v0, tmp_v1);
499 X->Large.prdct += dam_pr;
500 }
501 StopTimer(212);
502 }
503#endif
504
505 for (i = 0; i < X->Def.EDNTransfer; i += 2) {
506#ifdef MPI
507 if (!use_batching) {
508 // Non-batched per-term MPI path (original develop behavior)
509 if (X->Def.EDGeneralTransfer[i][0] + 1 > X->Def.Nsite &&
510 X->Def.EDGeneralTransfer[i][2] + 1 > X->Def.Nsite) {
511 StartTimer(211);
512 GC_general_hopp_MPIdouble(i, X, tmp_v0, tmp_v1);
513 StopTimer(211);
514 continue;
515 }
516 else if (X->Def.EDGeneralTransfer[i][2] + 1 > X->Def.Nsite) {
517 StartTimer(212);
518 GC_general_hopp_MPIsingle(i, X, tmp_v0, tmp_v1);
519 StopTimer(212);
520 continue;
521 }
522 else if (X->Def.EDGeneralTransfer[i][0] + 1 > X->Def.Nsite) {
523 StartTimer(212);
524 GC_general_hopp_MPIsingle(i + 1, X, tmp_v0, tmp_v1);
525 StopTimer(212);
526 continue;
527 }
528 } else if (X->Def.EDGeneralTransfer[i][0] + 1 > X->Def.Nsite ||
529 X->Def.EDGeneralTransfer[i][2] + 1 > X->Def.Nsite) {
530 continue; // inter-PE handled by batching above
531 }
532#endif
533 // Local transfers (both sites intra-process)
534 StartTimer(213);
535 for (ihermite = 0; ihermite<2; ihermite++) {
536 idx = i + ihermite;
537 isite1 = X->Def.EDGeneralTransfer[idx][0] + 1;
538 isite2 = X->Def.EDGeneralTransfer[idx][2] + 1;
539 sigma1 = X->Def.EDGeneralTransfer[idx][1];
540 sigma2 = X->Def.EDGeneralTransfer[idx][3];
541 if (general_hopp_GetInfo(X, isite1, isite2, sigma1, sigma2) != 0) {
542 return -1;
543 }
544 tmp_trans = -X->Def.EDParaGeneralTransfer[idx];
545 dam_pr = GC_general_hopp(tmp_v0, tmp_v1, X, tmp_trans);
546 X->Large.prdct += dam_pr;
547 }
548 StopTimer(213);
549 }
550 StopTimer(210);
554 StartTimer(220);
555
556#ifdef MPI
557 if (use_batching) {
558 // Initialize batched InterAll if not already done
559 StartTimer(221);
560 if (!batched_interall_HubbardGC_initialized) {
561 if (InitializeMPIBatchedInterAll_HubbardGC(X, &batched_interall_HubbardGC) != 0) {
562 fprintf(stderr, "Error: Failed to initialize batched MPI InterAll for HubbardGC\n");
563 return -1;
564 }
565 batched_interall_HubbardGC_initialized = 1;
566 }

569 for (int g = 0; g < batched_interall_HubbardGC.num_groups; g++) {
570 dam_pr = X_child_GC_InterAll_Hubbard_MPI_batched(
571 &batched_interall_HubbardGC.groups[g], X, tmp_v0, tmp_v1);
572 X->Large.prdct += dam_pr;
573 }
574 StopTimer(221);
575 }
576#endif

578 for (i = 0; i < X->Def.NInterAll_OffDiagonal; i+=2) {
579 isite1 = X->Def.InterAll_OffDiagonal[i][0] + 1;
580 isite2 = X->Def.InterAll_OffDiagonal[i][2] + 1;
581 isite3 = X->Def.InterAll_OffDiagonal[i][4] + 1;
582 isite4 = X->Def.InterAll_OffDiagonal[i][6] + 1;
583 sigma1 = X->Def.InterAll_OffDiagonal[i][1];
584 sigma2 = X->Def.InterAll_OffDiagonal[i][3];
585 sigma3 = X->Def.InterAll_OffDiagonal[i][5];
586 sigma4 = X->Def.InterAll_OffDiagonal[i][7];
587 tmp_V = X->Def.ParaInterAll_OffDiagonal[i];
588
589 if ( CheckPE(isite1 - 1, X) == TRUE || CheckPE(isite2 - 1, X) == TRUE
590 || CheckPE(isite3 - 1, X) == TRUE || CheckPE(isite4 - 1, X) == TRUE)
591 {
592#ifdef MPI
593 if (use_batching) {
594 // Inter-PE terms are normally handled by the batched groups above.
595 // But the batched initializer drops terms whose communication partner
596 // is the local rank (origin==myrank) -- e.g. an off-diagonal InterAll
597 // whose inter-process factor is a number operator, which leaves the
598 // rank bit unchanged. Those terms still contribute and would otherwise
599 // be lost (skipped both here and in batching), silently corrupting H.
600 // Recompute the partner with the same routine the batched initializer
601 // uses, and only skip terms the batched groups actually cover
602 // (origin != myrank). Self-partner / non-contributing terms fall
603 // through to the per-term path below, whose child_GC_*_Hubbard_MPI
604 // routines apply the myrank==origin branch locally.
605 unsigned long int origin_chk;
606 int is_hermite_chk;
607 int ret_chk = ComputeInterAllOrigin(
608 isite1 - 1, sigma1, isite2 - 1, sigma2,
609 isite3 - 1, sigma3, isite4 - 1, sigma4,
610 X, &origin_chk, &is_hermite_chk);
611 if (ret_chk == 0 && (int)origin_chk != myrank) {
612 continue;
613 }
614 }
615 // Non-batched per-term MPI path (original develop behavior)
616 StartTimer(221);
617 ibitsite1 = X->Def.OrgTpow[2 * isite1 - 2 + sigma1];
618 ibitsite2 = X->Def.OrgTpow[2 * isite2 - 2 + sigma2];
619 ibitsite3 = X->Def.OrgTpow[2 * isite3 - 2 + sigma3];
620 ibitsite4 = X->Def.OrgTpow[2 * isite4 - 2 + sigma4];
621 if (ibitsite1 == ibitsite2 && ibitsite3 == ibitsite4)
622 dam_pr = child_GC_CisAisCjtAjt_Hubbard_MPI(
623 isite1 - 1, sigma1, isite3 - 1, sigma3, tmp_V, X, tmp_v0, tmp_v1);
624 else if (ibitsite1 == ibitsite2 && ibitsite3 != ibitsite4)
625 dam_pr = child_GC_CisAisCjtAku_Hubbard_MPI(
626 isite1 - 1, sigma1, isite3 - 1, sigma3, isite4 - 1, sigma4, tmp_V, X, tmp_v0, tmp_v1);
627 else if (ibitsite1 != ibitsite2 && ibitsite3 == ibitsite4)
628 dam_pr = child_GC_CisAjtCkuAku_Hubbard_MPI(
629 isite1 - 1, sigma1, isite2 - 1, sigma2, isite3 - 1, sigma3, tmp_V, X, tmp_v0, tmp_v1);
630 else if (ibitsite1 != ibitsite2 && ibitsite3 != ibitsite4)
631 dam_pr = child_GC_CisAjtCkuAlv_Hubbard_MPI(
632 isite1 - 1, sigma1, isite2 - 1, sigma2, isite3 - 1, sigma3, isite4 - 1, sigma4, tmp_V, X, tmp_v0, tmp_v1);
633 StopTimer(221);
634#endif
635 }
636 else{
637 StartTimer(222);
638 dam_pr = 0.0;
639 for(ihermite=0; ihermite<2; ihermite++){
640 idx=i+ihermite;
641 isite1 = X->Def.InterAll_OffDiagonal[idx][0] + 1;
642 isite2 = X->Def.InterAll_OffDiagonal[idx][2] + 1;
643 isite3 = X->Def.InterAll_OffDiagonal[idx][4] + 1;
644 isite4 = X->Def.InterAll_OffDiagonal[idx][6] + 1;
645 sigma1 = X->Def.InterAll_OffDiagonal[idx][1];
646 sigma2 = X->Def.InterAll_OffDiagonal[idx][3];
647 sigma3 = X->Def.InterAll_OffDiagonal[idx][5];
648 sigma4 = X->Def.InterAll_OffDiagonal[idx][7];
649 tmp_V = X->Def.ParaInterAll_OffDiagonal[idx];
650
651 general_int_GetInfo(i, X, isite1, isite2, isite3, isite4,
652 sigma1, sigma2, sigma3, sigma4, tmp_V);
653 dam_pr += GC_general_int(tmp_v0, tmp_v1, X);
654 }/*for(ihermite=0; ihermite<2; ihermite++)*/
655 StopTimer(222);
656 }
657 X->Large.prdct += dam_pr;
658 }/*for (i = 0; i < X->Def.NInterAll_OffDiagonal; i+=2)*/
659 StopTimer(220);
660
661 if (X->Def.NNBodyInterAll_OffDiagonal > 0) {
662 if (MultiplyNBodyInterAllHubbardGC(X, tmp_v0, tmp_v1) != 0) {
663 return -1;
664 }
665 }
666
670 StartTimer(230);
671 for (i = 0; i < X->Def.NPairHopping; i +=2) {
672 sigma1 = 0;
673 sigma2 = 1;
674 dam_pr = 0.0;
675 if ( X->Def.PairHopping[i][0] + 1 > X->Def.Nsite
676 || X->Def.PairHopping[i][1] + 1 > X->Def.Nsite)
677 {
678 StartTimer(231);
