90int expec_cisajs(struct BindStruct *X,double complex *vec){
91
92 FILE *fp;
93 char sdt[D_FileNameMax];
94
95 long unsigned int irght,ilft,ihfbit;
96 long int i_max;
97 //For TPQ
98 int step=0;
99 int rand_i=0;
100
101 if(X->Def.NCisAjt <1) return 0;
102
103 i_max = X->Check.idim_max;
104 if(GetSplitBitByModel(X->Def.Nsite, X->Def.iCalcModel, &irght, &ilft, &ihfbit)!=0){
105 return -1;
106 }
107 X->Large.i_max = i_max;
108 X->Large.irght = irght;
109 X->Large.ilft = ilft;
110 X->Large.ihfbit = ihfbit;
111 X->Large.mode = M_CORR;
112
113 switch(X->Def.iCalcType){
114 case Lanczos:
115 if(X->Def.St==0){
116 sprintf(sdt, cFileName1BGreen_Lanczos, X->Def.CDataFileHead);
117 fprintf(stdoutMPI, "%s", cLogLanczosExpecOneBodyGStart);
118 TimeKeeper(X, cFileNameTimeKeep, cLanczosExpecOneBodyGStart, "a");
119 }else if(X->Def.St==1){
120 sprintf(sdt, cFileName1BGreen_CG, X->Def.CDataFileHead);
121 TimeKeeper(X, cFileNameTimeKeep, cCGExpecOneBodyGStart, "a");
122 fprintf(stdoutMPI, "%s", cLogCGExpecOneBodyGStart);
123 }
124 //vec=v0;
125 break;
126 case TPQCalc:
127 case cTPQ:
128 step=X->Def.istep;
129 rand_i=X->Def.irand;
130 TimeKeeperWithRandAndStep(X, cFileNameTimeKeep, cTPQExpecOneBodyGStart, "a", rand_i, step);
131 sprintf(sdt, cFileName1BGreen_TPQ, X->Def.CDataFileHead, rand_i, step);
132 //vec=v0;
133 break;
134 case TimeEvolution:
135 step=X->Def.istep;
136 TimeKeeperWithStep(X, cFileNameTimeKeep, cTEExpecOneBodyGStart, "a", step);
137 sprintf(sdt, cFileName1BGreen_TE, X->Def.CDataFileHead, step);
138 break;
139
140 case FullDiag:
141 case CG:
142 sprintf(sdt, cFileName1BGreen_FullDiag, X->Def.CDataFileHead, X->Phys.eigen_num);
143 //vec=v0;
144 break;
145 }
146
147 if(childfopenMPI(sdt, "w", &fp)!=0){
148 return -1;
149 }
150 switch(X->Def.iCalcModel){
151 case HubbardGC:
152 if(expec_cisajs_HubbardGC(X, vec, &fp)!=0){
153 return -1;
154 }
155 break;
156
157 case Hubbard:
158 case tJ:
159 case tJGC:
160 case Kondo:
161 case KondoGC:
162 if(expec_cisajs_Hubbard(X, vec, &fp)!=0){
163 return -1;
164 }
165 break;
166
167 case Spin: // for the Sz-conserved spin system
168 if(expec_cisajs_Spin(X, vec, &fp)!=0){
169 return -1;
170 }
171 break;
172
173 case SpinGC:
174 if(expec_cisajs_SpinGC(X, vec, &fp)!=0){
175 return -1;
176 }
177 break;
178
179 case SpinlessFermion:
180 case SpinlessFermionGC:
181 // For spinless fermions, calculate one-body Green's function
182 {
183 long unsigned int i_sp, j_sp;
184 for(i_sp = 0; i_sp < X->Def.NCisAjt; i_sp++){
185 long unsigned int org_isite1_sp = X->Def.CisAjt[i_sp][0]+1;
186 long unsigned int org_isite2_sp = X->Def.CisAjt[i_sp][2]+1;
187 long unsigned int org_sigma1_sp = X->Def.CisAjt[i_sp][1];
188 long unsigned int org_sigma2_sp = X->Def.CisAjt[i_sp][3];
189 int site1_is_interPE = (org_isite1_sp > X->Def.Nsite) ? 1 : 0;
190 int site2_is_interPE = (org_isite2_sp > X->Def.Nsite) ? 1 : 0;
191 double complex dam_pr_sp = 0;
192
193 if((site1_is_interPE || site2_is_interPE) &&
194 (org_sigma1_sp == org_sigma2_sp)){
195#ifdef MPI
196 if(X->Def.iCalcModel == SpinlessFermionGC){
197 dam_pr_sp = X_GC_CisAjt_SpinlessFermion_MPI(
198 org_isite1_sp - 1, org_isite2_sp - 1, X, vec);
199 } else {
200 dam_pr_sp = X_CisAjt_SpinlessFermion_MPI(
201 org_isite1_sp - 1, org_isite2_sp - 1, X, vec);
202 }
203#else
204 dam_pr_sp = 0;
205#endif
206 } else if(org_isite1_sp == org_isite2_sp && org_sigma1_sp == org_sigma2_sp){
207 // Diagonal case: <n_i>
208 long unsigned int is_sp = X->Def.Tpow[org_isite1_sp - 1];
209 if(X->Def.iCalcModel == SpinlessFermionGC){
210 // Grand canonical - use (j_sp-1)
211#pragma omp parallel for default(none) reduction(+:dam_pr_sp) shared(vec) \
212 firstprivate(i_max, is_sp) private(j_sp)
213 for(j_sp = 1; j_sp <= i_max; j_sp++){
214 long unsigned int ibit_sp = ((j_sp-1) & is_sp) / is_sp;
215 dam_pr_sp += ibit_sp * conj(vec[j_sp]) * vec[j_sp];
216 }
217 } else {
218 // Canonical - use list_1[j_sp]
219#pragma omp parallel for default(none) reduction(+:dam_pr_sp) shared(vec, list_1) \
220 firstprivate(i_max, is_sp) private(j_sp)
221 for(j_sp = 1; j_sp <= i_max; j_sp++){
222 long unsigned int ibit_sp = (list_1[j_sp] & is_sp) / is_sp;
223 dam_pr_sp += ibit_sp * conj(vec[j_sp]) * vec[j_sp];
224 }
225 }
226 } else {
227 // Local off-diagonal case: <c^+_i c_j>
228 long unsigned int is1_sp = X->Def.Tpow[org_isite1_sp - 1];
229 long unsigned int is2_sp = X->Def.Tpow[org_isite2_sp - 1];
230
231 if(X->Def.iCalcModel == SpinlessFermionGC){
232#pragma omp parallel for default(none) reduction(+:dam_pr_sp) shared(vec) \
233 firstprivate(i_max, is1_sp, is2_sp) private(j_sp)
234 for(j_sp = 1; j_sp <= i_max; j_sp++){
235 long unsigned int org_bit = j_sp - 1;
236 long unsigned int tmp_bit, off_bit;
237 unsigned long int mask, bit;
238 int sgn = 1, tmp_sgn;
239
240 if((org_bit & is2_sp) == 0) continue;
241 tmp_bit = org_bit ^ is2_sp;
242 mask = is2_sp - 1;
243 bit = tmp_bit & mask;
244 SgnBit(bit, &tmp_sgn);
245 sgn *= tmp_sgn;
246
247 if((tmp_bit & is1_sp) != 0) continue;
248 off_bit = tmp_bit ^ is1_sp;
249 mask = is1_sp - 1;
250 bit = off_bit & mask;
251 SgnBit(bit, &tmp_sgn);
252 sgn *= tmp_sgn;
253
254 dam_pr_sp += (double)sgn * conj(vec[off_bit + 1]) * vec[j_sp];
255 }
256 } else {
257#pragma omp parallel for default(none) reduction(+:dam_pr_sp) shared(vec, list_1, list_2_1, list_2_2) \
258 firstprivate(i_max, is1_sp, is2_sp, X) private(j_sp)
259 for(j_sp = 1; j_sp <= i_max; j_sp++){
260 long unsigned int org_bit = list_1[j_sp];
261 long unsigned int tmp_bit, off_bit, off_idx;
262 unsigned long int mask, bit;
263 int sgn = 1, tmp_sgn;
264
265 if((org_bit & is2_sp) == 0) continue;
266 tmp_bit = org_bit ^ is2_sp;
267 mask = is2_sp - 1;
268 bit = tmp_bit & mask;
269 SgnBit(bit, &tmp_sgn);
270 sgn *= tmp_sgn;
271
272 if((tmp_bit & is1_sp) != 0) continue;
273 off_bit = tmp_bit ^ is1_sp;
274 mask = is1_sp - 1;
275 bit = off_bit & mask;
276 SgnBit(bit, &tmp_sgn);
277 sgn *= tmp_sgn;
278
279 if(GetOffComp(list_2_1, list_2_2, off_bit,
280 X->Large.irght, X->Large.ilft, X->Large.ihfbit, &off_idx) != TRUE){
281 continue;
282 }
283 dam_pr_sp += (double)sgn * conj(vec[off_idx]) * vec[j_sp];
284 }
285 }
286 }
287 dam_pr_sp = SumMPI_dc(dam_pr_sp);
288 fprintf(fp, " %4lu %4lu %4lu %4lu %.10lf %.10lf\n",
289 org_isite1_sp-1, org_sigma1_sp, org_isite2_sp-1, org_sigma2_sp,
290 creal(dam_pr_sp), cimag(dam_pr_sp));
291 }
292 }
293 break;
294
295 default:
296 return -1;
297 }
298
299 fclose(fp);
300 if(X->Def.St==0){
301 if(X->Def.iCalcType==Lanczos){
302 TimeKeeper(X, cFileNameTimeKeep, cLanczosExpecOneBodyGFinish, "a");
303 fprintf(stdoutMPI, "%s", cLogLanczosExpecOneBodyGEnd);
304 TimeKeeper(X, cFileNameTimeKeep, cLanczosExpecOneBodyGFinish, "a");
305 }
306 else if(X->Def.iCalcType==TPQCalc){
307 TimeKeeperWithRandAndStep(X, cFileNameTimeKeep, cTPQExpecOneBodyGFinish, "a", rand_i, step);
308 }
309 else if(X->Def.iCalcType==TimeEvolution){
310 TimeKeeperWithStep(X, cFileNameTimeKeep, cTEExpecOneBodyGFinish, "a", step);
311 }
312 }else if(X->Def.St==1){
313 TimeKeeper(X, cFileNameTimeKeep, cCGExpecOneBodyGFinish, "a");
314 fprintf(stdoutMPI, "%s", cLogCGExpecOneBodyGEnd);
315 }
316 return 0;
317}

328int expec_cisajs_HubbardGC(struct BindStruct *X, double complex *vec, FILE **_fp){
329 long unsigned int i,j;
330 long unsigned int org_isite1,org_isite2,org_sigma1,org_sigma2;
331 double complex dam_pr=0;
332 long int i_max;
333 long int ibit;
334 long unsigned int is;
335 double complex tmp_OneGreen=1.0;
336
337 i_max = X->Check.idim_max;
338
339 for(i=0;i<X->Def.NCisAjt;i++){
340 org_isite1 = X->Def.CisAjt[i][0]+1;
341 org_isite2 = X->Def.CisAjt[i][2]+1;
342 org_sigma1 = X->Def.CisAjt[i][1];
343 org_sigma2 = X->Def.CisAjt[i][3];
344 dam_pr=0;
345 if (org_isite1 > X->Def.Nsite &&
346 org_isite2 > X->Def.Nsite) {
347 if(org_isite1==org_isite2 && org_sigma1==org_sigma2){
348 if(org_sigma1==0){
349 is = X->Def.Tpow[2 * org_isite1 - 2];
350 }
351 else{
352 is = X->Def.Tpow[2 * org_isite1 - 1];
353 }
354 ibit = (unsigned long int)myrank & is;
355 if (ibit == is) {
356#pragma omp parallel for default(none) reduction(+:dam_pr) shared(vec) \
357 firstprivate(i_max) private(j)
358 for (j = 1; j <= i_max; j++) dam_pr += vec[j]*conj(vec[j]);
359 }
360 }
361 else{
362 dam_pr =child_GC_general_hopp_MPIdouble(org_isite1-1, org_sigma1, org_isite2-1, org_sigma2, -tmp_OneGreen, X, vec, vec);
363 }
364 }
365 else if (org_isite2 > X->Def.Nsite || org_isite1 > X->Def.Nsite){
366 if(org_isite1<org_isite2){
367 dam_pr =child_GC_general_hopp_MPIsingle(org_isite1-1, org_sigma1, org_isite2-1, org_sigma2, -tmp_OneGreen, X, vec, vec);
368 }
369 else{
370 dam_pr =child_GC_general_hopp_MPIsingle(org_isite2-1, org_sigma2, org_isite1-1, org_sigma1, -tmp_OneGreen, X, vec, vec);
371 dam_pr = conj(dam_pr);
372 }
373 }
374 else{
375 if(general_hopp_GetInfo( X,org_isite1,org_isite2,org_sigma1,org_sigma2)!=0){
376 return -1;
377 }
378 dam_pr = GC_general_hopp(vec,vec,X,tmp_OneGreen);
379 }
380
381 dam_pr= SumMPI_dc(dam_pr);
382 fprintf(*_fp," %4lu %4lu %4lu %4lu %.10lf %.10lf\n",org_isite1-1,org_sigma1,org_isite2-1,org_sigma2,creal(dam_pr),cimag(dam_pr));
383 }
384
385 return 0;
386}

397int expec_cisajs_Hubbard(struct BindStruct *X, double complex *vec, FILE **_fp) {
398 long unsigned int i,j;
399 long unsigned int org_isite1,org_isite2,org_sigma1,org_sigma2;
400 double complex dam_pr=0;
401 long int i_max;
402 int num1;
403 long int ibit;
404 long unsigned int is;
405 double complex tmp_OneGreen=1.0;
406
407 i_max = X->Check.idim_max;
408 for(i=0;i<X->Def.NCisAjt;i++){
409 org_isite1 = X->Def.CisAjt[i][0]+1;
410 org_isite2 = X->Def.CisAjt[i][2]+1;
411 org_sigma1 = X->Def.CisAjt[i][1];
412 org_sigma2 = X->Def.CisAjt[i][3];
413 dam_pr=0.0;
414
415 if(X->Def.iFlgSzConserved ==TRUE){
416 if(org_sigma1 != org_sigma2){
417 dam_pr =0.0;
418 fprintf(*_fp," %4lu %4lu %4lu %4lu %.10lf %.10lf\n",org_isite1-1,org_sigma1,org_isite2-1,org_sigma2,creal(dam_pr),cimag(dam_pr));
419 continue;
420 }
421 }
422
423 if(X->Def.iCalcModel==Kondo || X->Def.iCalcModel==KondoGC) {
424 if( (X->Def.LocSpn[org_isite1 - 1] == 1 && X->Def.LocSpn[org_isite2 - 1] == 0) ||
425 (X->Def.LocSpn[org_isite1 - 1] == 0 && X->Def.LocSpn[org_isite2 - 1] == 1)
426 )
427 {
428 dam_pr =0.0;
429 fprintf(*_fp," %4lu %4lu %4lu %4lu %.10lf %.10lf\n",org_isite1-1,org_sigma1,org_isite2-1,org_sigma2,creal(dam_pr),cimag(dam_pr));
430 continue;
431 }
432 }
433
434 if (org_isite1 > X->Def.Nsite &&
435 org_isite2 > X->Def.Nsite) {
436 if(org_isite1==org_isite2 && org_sigma1==org_sigma2){//diagonal
437
438 is = X->Def.Tpow[2 * org_isite1 - 2+org_sigma1];
439 ibit = (unsigned long int)myrank & is;
440 if (ibit == is) {
441#pragma omp parallel for default(none) reduction(+:dam_pr) shared(vec) \
442 firstprivate(i_max) private(j)
443 for (j = 1; j <= i_max; j++) dam_pr += vec[j]*conj(vec[j]);
444 }
445
446 }
447 else{
448 dam_pr =child_general_hopp_MPIdouble(org_isite1-1, org_sigma1, org_isite2-1, org_sigma2, -tmp_OneGreen, X, vec, vec);
449 }
450 }
451 else if (org_isite2 > X->Def.Nsite || org_isite1 > X->Def.Nsite){
452 if(org_isite1 < org_isite2){
453 dam_pr =child_general_hopp_MPIsingle(org_isite1-1, org_sigma1,org_isite2-1, org_sigma2, -tmp_OneGreen, X, vec, vec);
454 }
455 else{
456 dam_pr = child_general_hopp_MPIsingle(org_isite2-1, org_sigma2, org_isite1-1, org_sigma1, -tmp_OneGreen, X, vec, vec);
457 dam_pr = conj(dam_pr);
458 }
459 }
460 else{
461 if(general_hopp_GetInfo( X,org_isite1,org_isite2,org_sigma1,org_sigma2)!=0){
462 return -1;
463 }
464 if(org_isite1==org_isite2 && org_sigma1==org_sigma2){
465 //fprintf(stdoutMPI,"DEBUG1-3-1\n");
466 is = X->Def.Tpow[2 * org_isite1 - 2 + org_sigma1];
467
468#pragma omp parallel for default(none) shared(list_1, vec) reduction(+:dam_pr) firstprivate(i_max, is) private(num1, ibit)
469 for(j = 1;j <= i_max;j++){
470 ibit = list_1[j]&is;
471 num1 = ibit/is;
472 dam_pr += num1*conj(vec[j])*vec[j];
473 }
474 }
475 else{
476 dam_pr = general_hopp(vec,vec,X,tmp_OneGreen);
477 }
478 }
479 dam_pr= SumMPI_dc(dam_pr);
480 //fprintf(stdoutMPI, "rank=%d, dam_pr=%lf\n", myrank, creal(dam_pr));
481 fprintf(*_fp," %4lu %4lu %4lu %4lu %.10lf %.10lf\n",org_isite1-1,org_sigma1,org_isite2-1,org_sigma2,creal(dam_pr),cimag(dam_pr));
482 }
483 return 0;
484}

553 dam_pr=0.0;
554 }
555 }else{

558 }

561 }

574int expec_cisajs_SpinGeneral(struct BindStruct *X, double complex *vec, FILE **_fp) {
575 long unsigned int i,j;
576 long unsigned int org_isite1,org_isite2,org_sigma1,org_sigma2;
577 double complex dam_pr=0;
578 long int i_max;
579 int num1;
580 i_max = X->Check.idim_max;
581
582 for(i=0;i<X->Def.NCisAjt;i++){
583 org_isite1 = X->Def.CisAjt[i][0]+1;
584 org_isite2 = X->Def.CisAjt[i][2]+1;
585 org_sigma1 = X->Def.CisAjt[i][1];
586 org_sigma2 = X->Def.CisAjt[i][3];
587
588 if(org_isite1 == org_isite2){
589 if(org_isite1 >X->Def.Nsite){
590 if(org_sigma1==org_sigma2){
591 // longitudinal magnetic field
592 num1 = BitCheckGeneral((unsigned long int)myrank,
593 org_isite1, org_sigma1, X->Def.SiteToBit, X->Def.Tpow);
594 dam_pr=0.0;
595 if (num1 != 0) {
596#pragma omp parallel for default(none) reduction(+:dam_pr) private(j) firstprivate(i_max, org_isite1, org_sigma1, X) shared(vec)
597 for(j=1;j<=i_max;j++){
598 dam_pr+=conj(vec[j])*vec[j];
599 }
600 }
601 }else{
602 dam_pr=0.0;
603 }
604 }
605 else {//org_isite1 <= X->Def.Nsite
606 if(org_sigma1==org_sigma2){
607 // longitudinal magnetic field
608 dam_pr=0.0;
609#pragma omp parallel for default(none) reduction(+:dam_pr) private(j, num1) firstprivate(i_max, org_isite1, org_sigma1, X) shared(vec, list_1)
610 for(j=1;j<=i_max;j++){
611 num1 = BitCheckGeneral(list_1[j], org_isite1, org_sigma1, X->Def.SiteToBit, X->Def.Tpow);
612 dam_pr+=conj(vec[j])*vec[j]*num1;
613 }
614 }else{
615 dam_pr=0.0;
616 }
617 }
618 }else{
619 // hopping is not allowed in localized spin system
620 dam_pr=0.0;
621 }//org_isite1 != org_isite2
622
623 dam_pr = SumMPI_dc(dam_pr);
624 fprintf(*_fp," %4lu %4lu %4lu %4lu %.10lf %.10lf\n",org_isite1-1, org_sigma1, org_isite2-1, org_sigma2,creal(dam_pr),cimag(dam_pr));
625 }
626
627 return 0;
628}

639int expec_cisajs_SpinGC(struct BindStruct *X, double complex *vec, FILE **_fp) {
