67static int expec_cisajs_Symmetry(struct BindStruct *X, double complex *vec,
68 FILE **_fp)
69{
70 unsigned int i, n = 0;
71 int *index = NULL;
72 long *map = NULL;
73 struct SymmetryCorrelationOperator *ops = NULL;
74 double complex *values = NULL;
75 int status = -1, local_error;
76 index = (int *)malloc((4U * (size_t)X->Def.NCisAjt + 4U) * sizeof(*index));
77 map = (long *)malloc(((size_t)X->Def.NCisAjt + 1U) * sizeof(*map));
78 ops = (struct SymmetryCorrelationOperator *)calloc(
79 (size_t)X->Def.NCisAjt + 1U, sizeof(*ops));
80 values = (double complex *)calloc(
81 (size_t)X->Def.NCisAjt + 1U, sizeof(*values));
82 local_error = index == NULL || map == NULL || ops == NULL || values == NULL;
83 if (SymmetryMpiAgreeError(SymmetryMpiCollectivesActive(), local_error) != 0)
84 goto done;
85 for (i = 0; i < X->Def.NCisAjt; ++i) {
86 const int *row = X->Def.CisAjt[i];
87 if (X->Def.iCalcModel == Spin && row[0] != row[2]) {
88 map[i] = -1L;
89 continue;
90 }
91 memcpy(index + 4U * n, row, 4U * sizeof(*index));
92 ops[n].factors = 1U;
93 ops[n].index = index + 4U * n;
94 map[i] = (long)n++;
95 }
96 if (n > 0U && SymmetryCorrelationExpectation(X, vec, ops, n, values) != 0)
97 goto done;
98 for (i = 0; i < X->Def.NCisAjt; ++i) {
99 double complex value = map[i] >= 0L ? values[map[i]] : 0.0;
100 GreenOutputWriteIndexPrefix(*_fp, X);
101 fprintf(*_fp, GREEN_ONEBODY_ROW_FORMAT,
102 (unsigned long)X->Def.CisAjt[i][0],
103 (unsigned long)X->Def.CisAjt[i][1],
104 (unsigned long)X->Def.CisAjt[i][2],
105 (unsigned long)X->Def.CisAjt[i][3],
106 creal(value), cimag(value));
107 }
108 status = 0;
109done:
110 free(index);
111 free(map);
112 free(ops);
113 free(values);
114 return status;
115}

144int expec_cisajs(struct BindStruct *X,double complex *vec){
145
146 FILE *fp = NULL;
147 char sdt[D_FileNameMax];
148
149 long unsigned int irght,ilft,ihfbit;
150 long int i_max;
151 //For TPQ
152 int step=0;
153 int rand_i=0;
154 int sector_status = 0;
155
156 if(X->Def.NCisAjt <1) return 0;
157
158 i_max = X->Check.idim_max;
159 if(GetSplitBitByModel(X->Def.Nsite, X->Def.iCalcModel, &irght, &ilft, &ihfbit)!=0){
160 return -1;
161 }
162 X->Large.i_max = i_max;
163 X->Large.irght = irght;
164 X->Large.ilft = ilft;
165 X->Large.ihfbit = ihfbit;
166 X->Large.mode = M_CORR;
167
168 switch(X->Def.iCalcType){
169 case Lanczos:
170 if(X->Def.St==0){
171 sprintf(sdt, cFileName1BGreen_Lanczos, X->Def.CDataFileHead);
172 fprintf(stdoutMPI, "%s", cLogLanczosExpecOneBodyGStart);
173 TimeKeeper(X, cFileNameTimeKeep, cLanczosExpecOneBodyGStart, "a");
174 }else if(X->Def.St==1){
175 sprintf(sdt, cFileName1BGreen_CG, X->Def.CDataFileHead);
176 TimeKeeper(X, cFileNameTimeKeep, cCGExpecOneBodyGStart, "a");
177 fprintf(stdoutMPI, "%s", cLogCGExpecOneBodyGStart);
178 }
179 //vec=v0;
180 break;
181 case TPQCalc:
182 case cTPQ:
183 step=X->Def.istep;
184 rand_i=X->Def.irand;
185 TimeKeeperWithRandAndStep(X, cFileNameTimeKeep, cTPQExpecOneBodyGStart, "a", rand_i, step);
186 sprintf(sdt, cFileName1BGreen_TPQ, X->Def.CDataFileHead, rand_i, step);
187 //vec=v0;
188 break;
189 case TimeEvolution:
190 step=X->Def.istep;
191 TimeKeeperWithStep(X, cFileNameTimeKeep, cTEExpecOneBodyGStart, "a", step);
192 sprintf(sdt, cFileName1BGreen_TE, X->Def.CDataFileHead, step);
193 break;
194
195 case FullDiag:
196 case CG:
197 sprintf(sdt, cFileName1BGreen_FullDiag, X->Def.CDataFileHead, X->Phys.eigen_num);
198 //vec=v0;
199 break;
200 }
201 {
202 int open_failed = GreenOutputKindUsesAggregate(X, GreenOutputOneBody)
203 ? GreenOutputOpenAggregate(X, GreenOutputOneBody, &fp) != 0
204 : childfopenMPI(sdt, "w", &fp) != 0;
205 /* Only rank 0 opens the real file; agree before the sector collective. */
206 if (GreenOutputAgreeSectorError(X, open_failed) != 0) {
207 if (fp != NULL) fclose(fp);
208 if (X->Def.iFlgSymmetryBasis == TRUE)
209 GreenOutputReportSectorFailure(X, GreenOutputOneBody, sdt, "open");
210 return -1;
211 }
212 }
213 if (X->Def.iFlgSymmetryBasis == TRUE) {
214 sector_status = expec_cisajs_Symmetry(X, vec, &fp);
215 } else switch(X->Def.iCalcModel){
216 case HubbardGC:
217 if(expec_cisajs_HubbardGC(X, vec, &fp)!=0){
218 return -1;
219 }
220 break;
221
222 case Hubbard:
223 case tJ:
224 case tJGC:
225 case Kondo:
226 case KondoGC:
227 if(expec_cisajs_Hubbard(X, vec, &fp)!=0){
228 return -1;
229 }
230 break;
231
232 case Spin: // for the Sz-conserved spin system
233 if(expec_cisajs_Spin(X, vec, &fp)!=0){
234 return -1;
235 }
236 break;
237
238 case SpinGC:
239 if(expec_cisajs_SpinGC(X, vec, &fp)!=0){
240 return -1;
241 }
242 break;
243
244 case SpinlessFermion:
245 case SpinlessFermionGC:
246 // For spinless fermions, calculate one-body Green's function
247 {
248 long unsigned int i_sp, j_sp;
249 for(i_sp = 0; i_sp < X->Def.NCisAjt; i_sp++){
250 long unsigned int org_isite1_sp = X->Def.CisAjt[i_sp][0]+1;
251 long unsigned int org_isite2_sp = X->Def.CisAjt[i_sp][2]+1;
252 long unsigned int org_sigma1_sp = X->Def.CisAjt[i_sp][1];
253 long unsigned int org_sigma2_sp = X->Def.CisAjt[i_sp][3];
254 int site1_is_interPE = (org_isite1_sp > X->Def.Nsite) ? 1 : 0;
255 int site2_is_interPE = (org_isite2_sp > X->Def.Nsite) ? 1 : 0;
256 double complex dam_pr_sp = 0;
257
258 if((site1_is_interPE || site2_is_interPE) &&
259 (org_sigma1_sp == org_sigma2_sp)){
260#ifdef MPI
261 if(X->Def.iCalcModel == SpinlessFermionGC){
262 dam_pr_sp = X_GC_CisAjt_SpinlessFermion_MPI(
263 org_isite1_sp - 1, org_isite2_sp - 1, X, vec);
264 } else {
265 dam_pr_sp = X_CisAjt_SpinlessFermion_MPI(
266 org_isite1_sp - 1, org_isite2_sp - 1, X, vec);
267 }
268#else
269 dam_pr_sp = 0;
270#endif
271 } else if(org_isite1_sp == org_isite2_sp && org_sigma1_sp == org_sigma2_sp){
272 // Diagonal case: <n_i>
273 long unsigned int is_sp = X->Def.Tpow[org_isite1_sp - 1];
274 if(X->Def.iCalcModel == SpinlessFermionGC){
275 // Grand canonical - use (j_sp-1)
276#pragma omp parallel for default(none) reduction(+:dam_pr_sp) shared(vec) \
277 firstprivate(i_max, is_sp) private(j_sp)
278 for(j_sp = 1; j_sp <= i_max; j_sp++){
279 long unsigned int ibit_sp = ((j_sp-1) & is_sp) / is_sp;
280 dam_pr_sp += ibit_sp * conj(vec[j_sp]) * vec[j_sp];
281 }
282 } else {
283 // Canonical - use list_1[j_sp]
284#pragma omp parallel for default(none) reduction(+:dam_pr_sp) shared(vec, list_1) \
285 firstprivate(i_max, is_sp) private(j_sp)
286 for(j_sp = 1; j_sp <= i_max; j_sp++){
287 long unsigned int ibit_sp = (list_1[j_sp] & is_sp) / is_sp;
288 dam_pr_sp += ibit_sp * conj(vec[j_sp]) * vec[j_sp];
289 }
290 }
291 } else {
292 // Local off-diagonal case: <c^+_i c_j>
293 long unsigned int is1_sp = X->Def.Tpow[org_isite1_sp - 1];
294 long unsigned int is2_sp = X->Def.Tpow[org_isite2_sp - 1];
295
296 if(X->Def.iCalcModel == SpinlessFermionGC){
297#pragma omp parallel for default(none) reduction(+:dam_pr_sp) shared(vec) \
298 firstprivate(i_max, is1_sp, is2_sp) private(j_sp)
299 for(j_sp = 1; j_sp <= i_max; j_sp++){
300 long unsigned int org_bit = j_sp - 1;
301 long unsigned int tmp_bit, off_bit;
302 unsigned long int mask, bit;
303 int sgn = 1, tmp_sgn;
304
305 if((org_bit & is2_sp) == 0) continue;
306 tmp_bit = org_bit ^ is2_sp;
307 mask = is2_sp - 1;
308 bit = tmp_bit & mask;
309 SgnBit(bit, &tmp_sgn);
310 sgn *= tmp_sgn;
311
312 if((tmp_bit & is1_sp) != 0) continue;
313 off_bit = tmp_bit ^ is1_sp;
314 mask = is1_sp - 1;
315 bit = off_bit & mask;
316 SgnBit(bit, &tmp_sgn);
317 sgn *= tmp_sgn;
318
319 dam_pr_sp += (double)sgn * conj(vec[off_bit + 1]) * vec[j_sp];
320 }
321 } else {
322#pragma omp parallel for default(none) reduction(+:dam_pr_sp) shared(vec, list_1, list_2_1, list_2_2) \
323 firstprivate(i_max, is1_sp, is2_sp, X) private(j_sp)
324 for(j_sp = 1; j_sp <= i_max; j_sp++){
325 long unsigned int org_bit = list_1[j_sp];
326 long unsigned int tmp_bit, off_bit, off_idx;
327 unsigned long int mask, bit;
328 int sgn = 1, tmp_sgn;
329
330 if((org_bit & is2_sp) == 0) continue;
331 tmp_bit = org_bit ^ is2_sp;
332 mask = is2_sp - 1;
333 bit = tmp_bit & mask;
334 SgnBit(bit, &tmp_sgn);
335 sgn *= tmp_sgn;
336
337 if((tmp_bit & is1_sp) != 0) continue;
338 off_bit = tmp_bit ^ is1_sp;
339 mask = is1_sp - 1;
340 bit = off_bit & mask;
341 SgnBit(bit, &tmp_sgn);
342 sgn *= tmp_sgn;
343
344 if(GetOffComp(list_2_1, list_2_2, off_bit,
345 X->Large.irght, X->Large.ilft, X->Large.ihfbit, &off_idx) != TRUE){
346 continue;
347 }
348 dam_pr_sp += (double)sgn * conj(vec[off_idx]) * vec[j_sp];
349 }
350 }
351 }
352 dam_pr_sp = SumMPI_dc(dam_pr_sp);
353 GreenOutputWriteIndexPrefix(fp, X);
354 fprintf(fp, GREEN_ONEBODY_ROW_FORMAT,
355 org_isite1_sp-1, org_sigma1_sp, org_isite2_sp-1, org_sigma2_sp,
356 creal(dam_pr_sp), cimag(dam_pr_sp));
357 }
358 }
359 break;
360
361 default:
362 return -1;
363 }
364
365 {
366 int close_failed = GreenOutputKindUsesAggregate(X, GreenOutputOneBody)
367 ? GreenOutputCloseAggregate(GreenOutputOneBody, fp) != 0
368 : GreenOutputCloseChecked(fp) != 0;
369 if (X->Def.iFlgSymmetryBasis == TRUE &&
370 GreenOutputAgreeSectorError(X, sector_status != 0 || close_failed) != 0) {
371 if (close_failed) GreenOutputReportSectorFailure(X, GreenOutputOneBody, sdt, "close");
372 return -1;
373 }
374 }
375 if(X->Def.St==0){
376 if(X->Def.iCalcType==Lanczos){
377 TimeKeeper(X, cFileNameTimeKeep, cLanczosExpecOneBodyGFinish, "a");
378 fprintf(stdoutMPI, "%s", cLogLanczosExpecOneBodyGEnd);
379 TimeKeeper(X, cFileNameTimeKeep, cLanczosExpecOneBodyGFinish, "a");
380 }
381 else if(X->Def.iCalcType==TPQCalc){
382 TimeKeeperWithRandAndStep(X, cFileNameTimeKeep, cTPQExpecOneBodyGFinish, "a", rand_i, step);
383 }
384 else if(X->Def.iCalcType==TimeEvolution){
385 TimeKeeperWithStep(X, cFileNameTimeKeep, cTEExpecOneBodyGFinish, "a", step);
386 }
387 }else if(X->Def.St==1){
388 TimeKeeper(X, cFileNameTimeKeep, cCGExpecOneBodyGFinish, "a");
389 fprintf(stdoutMPI, "%s", cLogCGExpecOneBodyGEnd);
390 }
391 if (sector_status != 0) return -1;
392 return 0;
393}

404int expec_cisajs_HubbardGC(struct BindStruct *X, double complex *vec, FILE **_fp){
405 long unsigned int i,j;
406 long unsigned int org_isite1,org_isite2,org_sigma1,org_sigma2;
407 double complex dam_pr=0;
408 long int i_max;
409 long int ibit;
410 long unsigned int is;
411 double complex tmp_OneGreen=1.0;
412
413 i_max = X->Check.idim_max;
414
415 for(i=0;i<X->Def.NCisAjt;i++){
416 org_isite1 = X->Def.CisAjt[i][0]+1;
417 org_isite2 = X->Def.CisAjt[i][2]+1;
418 org_sigma1 = X->Def.CisAjt[i][1];
419 org_sigma2 = X->Def.CisAjt[i][3];
420 dam_pr=0;
421 if (org_isite1 > X->Def.Nsite &&
422 org_isite2 > X->Def.Nsite) {
423 if(org_isite1==org_isite2 && org_sigma1==org_sigma2){
424 if(org_sigma1==0){
425 is = X->Def.Tpow[2 * org_isite1 - 2];
426 }
427 else{
428 is = X->Def.Tpow[2 * org_isite1 - 1];
429 }
430 ibit = (unsigned long int)myrank & is;
431 if (ibit == is) {
432#pragma omp parallel for default(none) reduction(+:dam_pr) shared(vec) \
433 firstprivate(i_max) private(j)
434 for (j = 1; j <= i_max; j++) dam_pr += vec[j]*conj(vec[j]);
435 }
436 }
437 else{
438 dam_pr =child_GC_general_hopp_MPIdouble(org_isite1-1, org_sigma1, org_isite2-1, org_sigma2, -tmp_OneGreen, X, vec, vec);
439 }
440 }
441 else if (org_isite2 > X->Def.Nsite || org_isite1 > X->Def.Nsite){
442 if(org_isite1<org_isite2){
443 dam_pr =child_GC_general_hopp_MPIsingle(org_isite1-1, org_sigma1, org_isite2-1, org_sigma2, -tmp_OneGreen, X, vec, vec);
444 }
445 else{
446 dam_pr =child_GC_general_hopp_MPIsingle(org_isite2-1, org_sigma2, org_isite1-1, org_sigma1, -tmp_OneGreen, X, vec, vec);
447 dam_pr = conj(dam_pr);
448 }
449 }
450 else{
451 if(general_hopp_GetInfo( X,org_isite1,org_isite2,org_sigma1,org_sigma2)!=0){
452 return -1;
453 }
454 dam_pr = GC_general_hopp(vec,vec,X,tmp_OneGreen);
455 }
456
457 dam_pr= SumMPI_dc(dam_pr);
458 GreenOutputWriteIndexPrefix(*_fp, X);
459 fprintf(*_fp,GREEN_ONEBODY_ROW_FORMAT,org_isite1-1,org_sigma1,org_isite2-1,org_sigma2,creal(dam_pr),cimag(dam_pr));
460 }
461
462 return 0;
463}

474int expec_cisajs_Hubbard(struct BindStruct *X, double complex *vec, FILE **_fp) {
475 long unsigned int i,j;
476 long unsigned int org_isite1,org_isite2,org_sigma1,org_sigma2;
477 double complex dam_pr=0;
478 long int i_max;
479 int num1;
480 long int ibit;
481 long unsigned int is;
482 double complex tmp_OneGreen=1.0;
483
484 i_max = X->Check.idim_max;
485 for(i=0;i<X->Def.NCisAjt;i++){
486 org_isite1 = X->Def.CisAjt[i][0]+1;
487 org_isite2 = X->Def.CisAjt[i][2]+1;
488 org_sigma1 = X->Def.CisAjt[i][1];
489 org_sigma2 = X->Def.CisAjt[i][3];
490 dam_pr=0.0;
491
492 if(X->Def.iFlgSzConserved ==TRUE){
493 if(org_sigma1 != org_sigma2){
494 dam_pr =0.0;
495 GreenOutputWriteIndexPrefix(*_fp, X);
496 fprintf(*_fp,GREEN_ONEBODY_ROW_FORMAT,org_isite1-1,org_sigma1,org_isite2-1,org_sigma2,creal(dam_pr),cimag(dam_pr));
497 continue;
498 }
499 }
500
501 if(X->Def.iCalcModel==Kondo || X->Def.iCalcModel==KondoGC) {
502 if( (X->Def.LocSpn[org_isite1 - 1] == 1 && X->Def.LocSpn[org_isite2 - 1] == 0) ||
503 (X->Def.LocSpn[org_isite1 - 1] == 0 && X->Def.LocSpn[org_isite2 - 1] == 1)
504 )
505 {
506 dam_pr =0.0;
507 GreenOutputWriteIndexPrefix(*_fp, X);
508 fprintf(*_fp,GREEN_ONEBODY_ROW_FORMAT,org_isite1-1,org_sigma1,org_isite2-1,org_sigma2,creal(dam_pr),cimag(dam_pr));
509 continue;
510 }
511 }
512
513 if (org_isite1 > X->Def.Nsite &&
514 org_isite2 > X->Def.Nsite) {
515 if(org_isite1==org_isite2 && org_sigma1==org_sigma2){//diagonal
516
517 is = X->Def.Tpow[2 * org_isite1 - 2+org_sigma1];
518 ibit = (unsigned long int)myrank & is;
519 if (ibit == is) {
520#pragma omp parallel for default(none) reduction(+:dam_pr) shared(vec) \
521 firstprivate(i_max) private(j)
522 for (j = 1; j <= i_max; j++) dam_pr += vec[j]*conj(vec[j]);
523 }
524
525 }
526 else{
527 dam_pr =child_general_hopp_MPIdouble(org_isite1-1, org_sigma1, org_isite2-1, org_sigma2, -tmp_OneGreen, X, vec, vec);
528 }
529 }
530 else if (org_isite2 > X->Def.Nsite || org_isite1 > X->Def.Nsite){
531 if(org_isite1 < org_isite2){
532 dam_pr =child_general_hopp_MPIsingle(org_isite1-1, org_sigma1,org_isite2-1, org_sigma2, -tmp_OneGreen, X, vec, vec);
533 }
534 else{
535 dam_pr = child_general_hopp_MPIsingle(org_isite2-1, org_sigma2, org_isite1-1, org_sigma1, -tmp_OneGreen, X, vec, vec);
536 dam_pr = conj(dam_pr);
537 }
538 }
539 else{
540 if(general_hopp_GetInfo( X,org_isite1,org_isite2,org_sigma1,org_sigma2)!=0){
541 return -1;
542 }
543 if(org_isite1==org_isite2 && org_sigma1==org_sigma2){
544 //fprintf(stdoutMPI,"DEBUG1-3-1\n");
545 is = X->Def.Tpow[2 * org_isite1 - 2 + org_sigma1];
546
547#pragma omp parallel for default(none) shared(list_1, vec) reduction(+:dam_pr) firstprivate(i_max, is) private(num1, ibit)
548 for(j = 1;j <= i_max;j++){
549 ibit = list_1[j]&is;
550 num1 = ibit/is;
551 dam_pr += num1*conj(vec[j])*vec[j];
552 }
553 }
554 else{
555 dam_pr = general_hopp(vec,vec,X,tmp_OneGreen);
556 }
557 }
558 dam_pr= SumMPI_dc(dam_pr);
559 //fprintf(stdoutMPI, "rank=%d, dam_pr=%lf\n", myrank, creal(dam_pr));
560 GreenOutputWriteIndexPrefix(*_fp, X);
561 fprintf(*_fp,GREEN_ONEBODY_ROW_FORMAT,org_isite1-1,org_sigma1,org_isite2-1,org_sigma2,creal(dam_pr),cimag(dam_pr));
562 }
563 return 0;
564}

594int expec_cisajs_SpinHalf(struct BindStruct *X, double complex *vec, FILE **_fp) {
595 long unsigned int i,j;
596 long unsigned int isite1;
597 long unsigned int org_isite1,org_isite2,org_sigma1,org_sigma2;
598 double complex dam_pr=0;
599 long int i_max;
600 long int ibit1;
601 long unsigned int is1_up;
602
603 i_max = X->Check.idim_max;
604
605 for(i=0;i<X->Def.NCisAjt;i++){
606 org_isite1 = X->Def.CisAjt[i][0]+1;
607 org_isite2 = X->Def.CisAjt[i][2]+1;
608 org_sigma1 = X->Def.CisAjt[i][1];
609 org_sigma2 = X->Def.CisAjt[i][3];
610
611 if(org_sigma1 == org_sigma2){
612 if(org_isite1==org_isite2){
613 if(org_isite1 > X->Def.Nsite){
614 is1_up = X->Def.Tpow[org_isite1 - 1];
615 ibit1 = child_SpinGC_CisAis((unsigned long int)myrank + 1, X, is1_up, org_sigma1);
616 dam_pr=0;
617 if(ibit1 !=0){
618#pragma omp parallel for reduction(+:dam_pr)default(none) shared(vec) \
619 firstprivate(i_max) private(j)
620 for (j = 1; j <= i_max; j++) dam_pr += conj(vec[j])*vec[j];
621 }
622 }// org_isite1 > X->Def.Nsite
