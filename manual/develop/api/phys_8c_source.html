73int FullDiagOutputEigenvector(const struct BindStruct *X,
74 unsigned long int state,
75 const double complex *eigenvector) {
76 char sdt[D_FileNameMax];
77 FILE *fp = NULL;
78 int iteration = 0;
79 unsigned long int dimension;
80 double complex unused = 0.0;
81 int nfn;
82 int failed = 0;
83
84 if (X == NULL || eigenvector == NULL || state > (unsigned long int)INT_MAX) {
85 fprintf(stderr, "Error: invalid FullDiag eigenvector output request.\n");
86 return -1;
87 }
88 dimension = X->Check.idim_max;
89 nfn = snprintf(sdt, sizeof(sdt), cFileNameOutputEigen,
90 X->Def.CDataFileHead, (int)state, 0);
91 if (nfn < 0 || nfn >= (int)sizeof(sdt)) {
92 fprintf(stderr,
93 "Error: FullDiag eigenvector file name is too long (state=%lu).\n",
94 state);
95 return -1;
96 }
97 if (childfopenALL(sdt, "wb", &fp) != 0) return -1;
98
99 if (fwrite(&iteration, sizeof(iteration), 1, fp) != 1 ||
100 fwrite(&dimension, sizeof(dimension), 1, fp) != 1 ||
101 fwrite(&unused, sizeof(unused), 1, fp) != 1 ||
102 fwrite(eigenvector, sizeof(*eigenvector), dimension, fp) != dimension) {
103 fprintf(stderr,
104 "Error: failed to write FullDiag eigenvector file (%s).\n", sdt);
105 failed = 1;
106 }
107 if (fclose(fp) != 0) {
108 fprintf(stderr,
109 "Error: failed to close FullDiag eigenvector file (%s).\n", sdt);
110 failed = 1;
111 }
112 return failed ? -1 : 0;
113}

143 i_max = X->Check.idim_max;
144#ifdef _SCALAPACK
145 double complex *vec_tmp;
146 int rank;
147 if (use_scalapack && X->Def.iExpecMode != EXPECMODE_SERIAL) {
148 int phys_rc = phys_stateparallel(X, neig);
149 FreeDistributedEigenvectors(&Z_vec, descZ_vec, &use_scalapack);
150 if (phys_rc != 0) exitMPI(-1);
151 assert(!ExpecLocalActive());
152 return;
153 }
154 if(use_scalapack){
155 /* S2/Sz are now computed on rank 0 in the state loop below (ExpecLocal-

157 vec_tmp = malloc(i_max*sizeof(double complex));
158 }
159#endif
160 if (GreenOutputInitializeAggregateFiles(X) != 0) {
161 fprintf(stderr, "Error: initialize aggregate Green output files.\n");
162 exitMPI(-1);
163 }
164 for (i = 0; i < neig; i++) {
165#ifdef _SCALAPACK
166 for (j = 0; j < i_max; j++) {
167 v0[j + 1] = 0.0;
168 }
169 if(use_scalapack){
170 MPI_Comm_rank(MPI_COMM_WORLD, &rank);
171#ifdef _ELPA
172 if (X->Def.iSolver == SOLVER_ELPA) {
173 GetEigenVectorBlock(i, i_max, Z_vec, descZ_vec, vec_tmp);
174 } else {
175 GetEigenVector(i, i_max, Z_vec, descZ_vec, vec_tmp);
176 }
177#else
178 GetEigenVector(i, i_max, Z_vec, descZ_vec, vec_tmp);
179#endif
180 if(rank == 0) {
181 for (j = 0; j < i_max; j++) {
182 v0[j + 1] = vec_tmp[j];
183 }
184 }
185 else{
186 for (j = 0; j < i_max; j++) {
187 v0[j + 1] = 0.0;
188 }
189 }
190 } else {
191 if (X->Def.iCalcType == FullDiag) {
192 if (myrank == 0) {
193 for (j = 0; j < i_max; j++) {
194 v0[j + 1] = L_vec[i][j];
195 }
196 }
197 } else {
198 for (j = 0; j < i_max; j++) {
199 v0[j + 1] = L_vec[i][j];
200 }
201
202 }
203 }
204#else
205 for (j = 0; j < i_max; j++) {
206 v0[j + 1] = L_vec[i][j];
207 }
208#endif
209
210 if (X->Def.iCalcType == FullDiag &&
211 X->Def.iOutputEigenVec == TRUE && myrank == 0 &&
212 !eigen_output_failed) {
213#ifdef _SCALAPACK
214 const double complex *eigenvector =
215 use_scalapack ? vec_tmp : L_vec[i];
216#else
217 const double complex *eigenvector = L_vec[i];
218#endif
219 if (FullDiagOutputEigenvector(X, i, eigenvector) != 0)
220 eigen_output_failed = 1;
221 }
222
223 X->Phys.eigen_num = i;
224 if (expec_energy_flct(X) != 0) {
225 fprintf(stderr, "Error: calc expec_energy.\n");
226 exitMPI(-1);

233 fprintf(stderr, "Error: calc TwoBodyG.\n");
234 exitMPI(-1);
235 }
236 if (expec_nbodyg(X, v1) != 0) {
237 fprintf(stderr, "Error: calc NBodyG.\n");
238 exitMPI(-1);
239 }
240 if (expec_anomalousg(X, v1) != 0) {
241 fprintf(stderr, "Error: calc AnomalousG.\n");
242 exitMPI(-1);
243 }
244
245#ifdef _SCALAPACK
246 if(use_scalapack){
247 if (X->Def.iCalcType == FullDiag) {
248 /* Mode 0 (distributed) S2/Sz unification: rank 0 holds the gathered
249 eigenvector in v1 (moved there from v0 by expec_energy_flct), so
250 compute totalspin locally under an ExpecLocal wrapper (reductions
251 become no-communication pass-throughs). Other ranks hold a zeroed
252 vector; their values are not used (display/all_* take rank 0's). */
253 ExpecLocalEnter();
254 if (myrank == 0) {
255 if (expec_totalspin(X, v1) != 0) { ExpecLocalLeave(); exitMPI(-1); }
256 } else {
257 X->Phys.s2 = 0.0;
258 X->Phys.Sz = 0.0;
259 }
260 ExpecLocalLeave();
261 }
262 }else{
263 if (X->Def.iCalcType == FullDiag) {
264 if (expec_totalspin(X, v1) != 0) {
265 fprintf(stderr, "Error: calc TotalSpin.\n");
266 exitMPI(-1);
267 }
268 }
269 }
270#else
271 if (X->Def.iCalcType == FullDiag) {
272 if (expec_totalspin(X, v1) != 0) {
273 fprintf(stderr, "Error: calc TotalSpin.\n");
274 exitMPI(-1);
275 }
276 }
277#endif
278
279 if (X->Def.iCalcModel == Spin || X->Def.iCalcModel == SpinGC) {
280 tmp_N = X->Def.NsiteMPI;
281 } else {
282 tmp_N = X->Phys.num_up + X->Phys.num_down;
283 }
284
285 if (X->Def.iCalcType == FullDiag){
286 /* Unified serial format (S2 column) for both distributed and
287 non-distributed paths: distributed Mode 0 now computes S2/Sz on
288 rank 0 above rather than zero-filling. */
289 fprintf(stdoutMPI, "i=%5ld Energy=%10lf N=%10lf Sz=%10lf S2=%10lf Doublon=%10lf \n", i, X->Phys.energy, tmp_N,
290 X->Phys.Sz, X->Phys.s2, X->Phys.doublon);
291 }
292 else if (X->Def.iCalcType == CG)
293 fprintf(stdoutMPI, "i=%5ld Energy=%10lf N=%10lf Sz=%10lf Doublon=%10lf \n", i, X->Phys.energy, tmp_N,
294 X->Phys.Sz, X->Phys.doublon);
295 X->Phys.all_energy[i] = X->Phys.energy;
296 X->Phys.all_doublon[i] = X->Phys.doublon;
297 X->Phys.all_sz[i] = X->Phys.Sz;
298 X->Phys.all_s2[i] = X->Phys.s2;
299 X->Phys.all_num_up[i] = X->Phys.num_up;
300 X->Phys.all_num_down[i] = X->Phys.num_down;
301 }
302#ifdef MPI
303 if (X->Def.iCalcType == FullDiag && X->Def.iOutputEigenVec == TRUE) {
304 int any_eigen_output_failed = 0;
305 MPI_Allreduce(&eigen_output_failed, &any_eigen_output_failed, 1, MPI_INT,
306 MPI_MAX, MPI_COMM_WORLD);
307 eigen_output_failed = any_eigen_output_failed;
308 }
309#endif
310#ifdef _SCALAPACK
311 if(use_scalapack) {
312 free(vec_tmp);
313 FreeDistributedEigenvectors(&Z_vec, descZ_vec, &use_scalapack);
314 }
315#endif
316 if (eigen_output_failed) exitMPI(-1);
317 assert(!ExpecLocalActive());
318}
