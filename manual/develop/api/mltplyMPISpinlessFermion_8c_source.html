64double complex X_child_general_hopp_Spinless_MPIdouble(
65 int org_isite1,
66 int org_isite2,
67 double complex tmp_trans,
68 struct BindStruct *X,
69 double complex *tmp_v0,
70 double complex *tmp_v1) {
71#ifdef MPI
72 int mask1, mask2, state1, state2, ierr, origin, bitdiff, Fsgn;
73 unsigned long int idim_max_buf, j, ioff;
74 MPI_Status statusMPI;
75 double complex trans, dmv, dam_pr;
76
77 mask1 = (int) X->Def.Tpow[org_isite1];
78 mask2 = (int) X->Def.Tpow[org_isite2];
79
80 if (mask2 > mask1)
81 bitdiff = mask2 - mask1 * 2;
82 else
83 bitdiff = mask1 - mask2 * 2;
84 origin = myrank ^ (mask1 + mask2);
85
86 state1 = origin & mask1;
87 state2 = origin & mask2;
88
89 SgnBit((unsigned long int) (origin & bitdiff), &Fsgn);
90
91 if (state1 == 0 && state2 == mask2) {
92 trans = -(double) Fsgn * tmp_trans;
93 } else if (state1 == mask1 && state2 == 0) {
94 trans = -(double) Fsgn * conj(tmp_trans);
95 if (X->Large.mode == M_CORR) {
96 trans = 0;
97 }
98 } else
99 return 0.0;
100
101 ierr = MPI_Sendrecv(&X->Check.idim_max, 1, MPI_UNSIGNED_LONG, origin, 0,
102 &idim_max_buf, 1, MPI_UNSIGNED_LONG, origin, 0, MPI_COMM_WORLD, &statusMPI);
103 if (ierr != 0)
104 exitMPI(-1);
105 ierr = MPI_Sendrecv(list_1, X->Check.idim_max + 1, MPI_UNSIGNED_LONG, origin, 0,
106 list_1buf, idim_max_buf + 1, MPI_UNSIGNED_LONG, origin, 0, MPI_COMM_WORLD, &statusMPI);
107 if (ierr != 0)
108 exitMPI(-1);
109 ierr = MPI_Sendrecv(tmp_v1, X->Check.idim_max + 1, MPI_DOUBLE_COMPLEX, origin, 0,
110 v1buf, idim_max_buf + 1, MPI_DOUBLE_COMPLEX, origin, 0, MPI_COMM_WORLD, &statusMPI);
111 if (ierr != 0)
112 exitMPI(-1);
113
114 dam_pr = 0.0;
115 if (X->Large.mode == M_MLTPLY) {
116#pragma omp parallel for default(none) reduction(+:dam_pr) private(j, dmv, Fsgn, ioff) \
117 firstprivate(idim_max_buf, trans, X) shared(list_2_1, list_2_2, list_1buf, v1buf, tmp_v1, tmp_v0)
118 for (j = 1; j <= idim_max_buf; j++) {
119 if (GetOffComp(list_2_1, list_2_2, list_1buf[j],
120 X->Large.irght, X->Large.ilft, X->Large.ihfbit, &ioff) == FALSE) {
121 ioff = 0;
122 continue;
123 }
124 dmv = trans * v1buf[j];
125 tmp_v0[ioff] += dmv;
126 dam_pr += conj(tmp_v1[ioff]) * dmv;
127 }
128 } else {
129#pragma omp parallel for default(none) reduction(+:dam_pr) private(j, dmv, Fsgn, ioff) \
130 firstprivate(idim_max_buf, trans, X) shared(list_2_1, list_2_2, list_1buf, v1buf, tmp_v1, tmp_v0)
131 for (j = 1; j <= idim_max_buf; j++) {
132 if (GetOffComp(list_2_1, list_2_2, list_1buf[j],
133 X->Large.irght, X->Large.ilft, X->Large.ihfbit, &ioff) == FALSE) {
134 ioff = 0;
135 continue;
136 }
137 dmv = trans * v1buf[j];
138 dam_pr += conj(tmp_v1[ioff]) * dmv;
139 }
140 }
141 return (dam_pr);
142#else
143 return 0.0;
144#endif
145}

180double complex X_child_general_hopp_Spinless_MPIsingle(
181 int org_isite1,
182 int org_isite2,
183 double complex tmp_trans,
184 struct BindStruct *X,
185 double complex *tmp_v0,
186 double complex *tmp_v1) {
187#ifdef MPI
188 unsigned long int mask1, state1;
189 int mask2, state2, ierr, origin, bit2diff, Fsgn;
190 unsigned long int idim_max_buf, j, state1check, bit1diff, ioff, jreal;
191 MPI_Status statusMPI;
192 double complex trans, dmv, dam_pr;
193
194 mask2 = (int) X->Def.Tpow[org_isite2];
195 bit2diff = mask2 - 1;
196 origin = myrank ^ mask2;
197
198 state2 = origin & mask2;
199
200 SgnBit((unsigned long int) (origin & bit2diff), &Fsgn);
201
202 ierr = MPI_Sendrecv(&X->Check.idim_max, 1, MPI_UNSIGNED_LONG, origin, 0,
203 &idim_max_buf, 1, MPI_UNSIGNED_LONG, origin, 0, MPI_COMM_WORLD, &statusMPI);
204 if (ierr != 0)
205 exitMPI(-1);
206 ierr = MPI_Sendrecv(list_1, X->Check.idim_max + 1, MPI_UNSIGNED_LONG, origin, 0,
207 list_1buf, idim_max_buf + 1, MPI_UNSIGNED_LONG, origin, 0, MPI_COMM_WORLD, &statusMPI);
208 if (ierr != 0)
209 exitMPI(-1);
210 ierr = MPI_Sendrecv(tmp_v1, X->Check.idim_max + 1, MPI_DOUBLE_COMPLEX, origin, 0,
211 v1buf, idim_max_buf + 1, MPI_DOUBLE_COMPLEX, origin, 0, MPI_COMM_WORLD, &statusMPI);
212 if (ierr != 0)
213 exitMPI(-1);
214
215 mask1 = X->Def.Tpow[org_isite1];
216 if (state2 == mask2) {
217 trans = -(double) Fsgn * tmp_trans;
218 state1check = 0;
219 } else if (state2 == 0) {
220 state1check = mask1;
221 trans = -(double) Fsgn * conj(tmp_trans);
222 if (X->Large.mode == M_CORR) {
223 trans = 0;
224 }
225 } else
226 return 0.0;
227
228 bit1diff = X->Def.Tpow[X->Def.Nsite - 1] * 2 - mask1 * 2;
229
230 dam_pr = 0.0;
231 if (X->Large.mode == M_MLTPLY) {
232#pragma omp parallel for default(none) reduction(+:dam_pr) private(j, dmv, Fsgn, ioff, jreal, state1) \
233 firstprivate(idim_max_buf, trans, X, mask1, state1check, bit1diff, myrank) shared(list_2_1, list_2_2, list_1buf, v1buf, tmp_v1, tmp_v0, list_1)
234 for (j = 1; j <= idim_max_buf; j++) {
235 Fsgn = 1;
236 jreal = list_1buf[j];
237 state1 = jreal & mask1;
238 if (state1 == state1check) {
239 SgnBit((unsigned long int) (jreal & bit1diff), &Fsgn);
240 if (GetOffComp(list_2_1, list_2_2, jreal ^ mask1,
241 X->Large.irght, X->Large.ilft, X->Large.ihfbit, &ioff) == FALSE) {
242 ioff = 0;
243 continue;
244 }
245 dmv = (double) Fsgn * trans * v1buf[j];
246 tmp_v0[ioff] += dmv;
247 dam_pr += conj(tmp_v1[ioff]) * dmv;
248 }
249 }
250 } else {
251#pragma omp parallel for default(none) reduction(+:dam_pr) private(j, dmv, Fsgn, ioff, jreal, state1) \
252 firstprivate(idim_max_buf, trans, X, mask1, state1check, bit1diff) shared(list_2_1, list_2_2, list_1buf, v1buf, tmp_v1, tmp_v0)
253 for (j = 1; j <= idim_max_buf; j++) {
254 jreal = list_1buf[j];
255 state1 = jreal & mask1;
256 if (state1 == state1check) {
257 SgnBit(jreal & bit1diff, &Fsgn);
258 if (GetOffComp(list_2_1, list_2_2, jreal ^ mask1,
259 X->Large.irght, X->Large.ilft, X->Large.ihfbit, &ioff) == FALSE) {
260 ioff = 0;
261 continue;
262 }
263 dmv = (double) Fsgn * trans * v1buf[j];
264 dam_pr += conj(tmp_v1[ioff]) * dmv;
265 }
266 }
267 }
268 return (dam_pr);
269#else
270 return 0.0;
271#endif
272}

497double complex X_child_GC_general_hopp_SpinlessFermion_MPIsingle(
498 int org_isite1,
499 int org_isite2,
500 double complex tmp_trans,
501 struct BindStruct *X,
502 double complex *tmp_v0,
503 double complex *tmp_v1) {
504#ifdef MPI
505 unsigned long int mask1, state1;
506 int mask2, state2, ierr, origin, bit2diff, Fsgn;
507 unsigned long int idim_max_buf, j, state1check, bit1diff, ioff, jreal;
508 MPI_Status statusMPI;
509 double complex trans, dmv, dam_pr;
510
511 mask2 = (int) X->Def.Tpow[org_isite2];
512 bit2diff = mask2 - 1;
513 origin = myrank ^ mask2;
514
515 state2 = origin & mask2;
516
517 SgnBit((unsigned long int) (origin & bit2diff), &Fsgn);
518
519 // For GC mode, we only need to exchange vector (no list_1)
520 ierr = MPI_Sendrecv(&X->Check.idim_max, 1, MPI_UNSIGNED_LONG, origin, 0,
521 &idim_max_buf, 1, MPI_UNSIGNED_LONG, origin, 0, MPI_COMM_WORLD, &statusMPI);
522 if (ierr != 0)
523 exitMPI(-1);
524 ierr = MPI_Sendrecv(tmp_v1, X->Check.idim_max + 1, MPI_DOUBLE_COMPLEX, origin, 0,
525 v1buf, idim_max_buf + 1, MPI_DOUBLE_COMPLEX, origin, 0, MPI_COMM_WORLD, &statusMPI);
526 if (ierr != 0)
527 exitMPI(-1);
528
529 mask1 = X->Def.Tpow[org_isite1];
530 if (state2 == mask2) {
531 trans = -(double) Fsgn * tmp_trans;
532 state1check = 0;
533 } else if (state2 == 0) {
534 state1check = mask1;
535 trans = -(double) Fsgn * conj(tmp_trans);
536 if (X->Large.mode == M_CORR || X->Large.mode == M_CALCSPEC) {
537 trans = 0;
538 }
539 } else
540 return 0.0;
541
542 bit1diff = X->Def.Tpow[X->Def.Nsite - 1] * 2 - mask1 * 2;
543
544 dam_pr = 0.0;
545 if (X->Large.mode == M_MLTPLY || X->Large.mode == M_CALCSPEC) {
546#pragma omp parallel for default(none) reduction(+:dam_pr) private(j, dmv, Fsgn, ioff, jreal, state1) \
547 firstprivate(idim_max_buf, trans, X, mask1, state1check, bit1diff, myrank) shared(v1buf, tmp_v1, tmp_v0)
548 for (j = 1; j <= idim_max_buf; j++) {
549 Fsgn = 1;
550 // For GC mode, jreal = j - 1 (direct bit representation)
551 jreal = j - 1;
552 state1 = jreal & mask1;
553 if (state1 == state1check) {
554 SgnBit((unsigned long int) (jreal & bit1diff), &Fsgn);
555 // For GC mode, ioff is calculated directly
556 ioff = (jreal ^ mask1) + 1;
557 dmv = (double) Fsgn * trans * v1buf[j];

563#pragma omp parallel for default(none) reduction(+:dam_pr) private(j, dmv, Fsgn, ioff, jreal, state1) \
564 firstprivate(idim_max_buf, trans, X, mask1, state1check, bit1diff) shared(v1buf, tmp_v1, tmp_v0)
565 for (j = 1; j <= idim_max_buf; j++) {
566 Fsgn = 1;
567 jreal = j - 1;
568 state1 = jreal & mask1;
569 if (state1 == state1check) {
570 SgnBit(jreal & bit1diff, &Fsgn);
571 ioff = (jreal ^ mask1) + 1;
572 dmv = (double) Fsgn * trans * v1buf[j];

574 }

580#endif

589void child_GC_general_hopp_SpinlessFermion_MPIsingle_per_site(
590 unsigned long int org_isite2,
591 struct BindStruct *X,
592 double complex *tmp_v0,
593 double complex *tmp_v1) {
594#ifdef MPI
595 // For SpinlessFermionGC, iterate over all transfer terms for this site
596 // Note: X_child_GC_...MPIsingle internally handles both directions
597 // (forward and hermitian conjugate), so we only need to call it once
598 // per transfer with the inter-process site as org_isite2
599 unsigned long int i;
600 double complex dam_pr;
601
602 for (i = 0; i < X->Def.EDNTransfer; i += 2) {
603 // Case 1: destination site (creation) is the inter-process site
604 if (X->Def.EDGeneralTransfer[i][2] == (int)org_isite2 &&
605 X->Def.EDGeneralTransfer[i][0] + 1 <= X->Def.Nsite) {
606 dam_pr = X_child_GC_general_hopp_SpinlessFermion_MPIsingle(
607 X->Def.EDGeneralTransfer[i][0],
608 X->Def.EDGeneralTransfer[i][2],
609 X->Def.EDParaGeneralTransfer[i],
610 X,
611 tmp_v0,
612 tmp_v1);
613 X->Large.prdct += dam_pr;
614 }
615 // Case 2: source site (annihilation) is the inter-process site
616 // Use entry i+1 which has the sites swapped (hermitian conjugate pair)
617 else if (X->Def.EDGeneralTransfer[i][0] == (int)org_isite2 &&
618 X->Def.EDGeneralTransfer[i][2] + 1 <= X->Def.Nsite) {
619 // Entry i+1 has sites swapped: [i+1][0] = [i][2], [i+1][2] = [i][0]
620 // So [i+1] has the inter-process site as destination
621 dam_pr = X_child_GC_general_hopp_SpinlessFermion_MPIsingle(
622 X->Def.EDGeneralTransfer[i + 1][0],
623 X->Def.EDGeneralTransfer[i + 1][2],
624 X->Def.EDParaGeneralTransfer[i + 1],
625 X,
626 tmp_v0,
627 tmp_v1);
628 X->Large.prdct += dam_pr;
629 }
630 }
631#endif
632}

634#ifdef MPI
639static int ApplyCisAjtGlobalSpinless(
640 unsigned long int org_global_bit,
641 unsigned long int global_is1,
642 unsigned long int global_is2,
643 int *sgn,
644 unsigned long int *off_global_bit) {
645 unsigned long int tmp_bit;
646 unsigned long int mask;
647 int tmp_sgn;
648 int tmp_sign;
649
650 tmp_bit = org_global_bit;
651 tmp_sign = 1;
652
653 // c_j
654 if ((tmp_bit & global_is2) == 0) return FALSE;
655 tmp_bit ^= global_is2;
656 mask = global_is2 - 1;
657 SgnBit(tmp_bit & mask, &tmp_sgn);
658 tmp_sign *= tmp_sgn;
659
660 // c^\dagger_i
661 if ((tmp_bit & global_is1) != 0) return FALSE;
662 tmp_bit ^= global_is1;
663 mask = global_is1 - 1;
664 SgnBit(tmp_bit & mask, &tmp_sgn);
665 tmp_sign *= tmp_sgn;
666
667 *sgn = tmp_sign;
668 *off_global_bit = tmp_bit;
669 return TRUE;
670}

678double complex X_CisAjt_SpinlessFermion_MPI(
679 int org_isite1,
680 int org_isite2,
681 struct BindStruct *X,
682 double complex *vec) {
683#ifdef MPI
684 double complex dam_pr;
685 unsigned long int i_max;
686 unsigned long int idim_max_buf;
687 unsigned long int rank_flip_mask;
688 unsigned long int global_is1, global_is2;
689 unsigned long int j, src_local_bit, src_global_bit, off_global_bit, off_local_bit, ioff;
690 int origin, off_rank, ierr, sgn;
691 MPI_Status statusMPI;
692
693 dam_pr = 0.0;
694 i_max = X->Check.idim_max;
695 idim_max_buf = 0;
696
697 rank_flip_mask = 0;
698 if (org_isite1 >= X->Def.Nsite) rank_flip_mask ^= X->Def.Tpow[org_isite1];
699 if (org_isite2 >= X->Def.Nsite) rank_flip_mask ^= X->Def.Tpow[org_isite2];
700 origin = myrank ^ (int)rank_flip_mask;
701
702 global_is1 = (1UL << org_isite1);
703 global_is2 = (1UL << org_isite2);
