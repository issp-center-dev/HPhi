113static inline const struct SymmetryBasisVector *SymmetryBasisLocalEntry(
114 const struct SymmetryBasisRuntime *sym,
115 unsigned long int local_index)
116{
117 unsigned long int global_index;
118 if (sym == NULL || sym->enabled != TRUE ||
119 local_index == 0UL || local_index > sym->local_dim ||
120 sym->local_offset > sym->dim ||
121 sym->local_dim > sym->dim - sym->local_offset) {
122 return NULL;
123 }
124 if (sym->basis_layout == SYMMETRY_BASIS_DISTRIBUTED) {
125 if (sym->basis != NULL || sym->local_basis == NULL ||
126 sym->local_capacity <= local_index) {
127 return NULL;
128 }
129 return &sym->local_basis[local_index];
130 }
131 if (sym->basis_layout != SYMMETRY_BASIS_REPLICATED ||
132 sym->basis == NULL) {
133 return NULL;
134 }
135 global_index = sym->local_offset + local_index;
136 if (global_index > sym->capacity) return NULL;
137 return &sym->basis[global_index];
138}
