93int phys_stateparallel(struct BindStruct *X, unsigned long int neig) {
94 long int NN = (long int)neig;
95 int P = nproc;
96 long int NC = (NN + P - 1) / P;
97 long int jb = (long int)myrank * NC + 1;
98 long int je = ((long int)myrank + 1) * NC;
99 long int ncols;
100 int rc_local = 0;
101 double complex *panel = NULL;
102 int ok, gok, g;
103 long int r;
104
105 if (je > NN) je = NN;
106 ncols = (je >= jb) ? (je - jb + 1) : 0;
107
108 /* --- allocate the state panel; sync the malloc verdict with a MIN-Allreduce
109 BEFORE any collective touches it (same pattern as lapack_diag_elpa). --- */
110 panel = malloc((((NN * ncols) > 0) ? (size_t)(NN * ncols) : 1)
111 * sizeof(double complex));
112 ok = (panel != NULL) ? 0 : -1;
113 if (ok != 0)
114 fprintf(stdout, " Error: malloc failed for state panel (rank %d).\n", myrank);
115 MPI_Allreduce(&ok, &gok, 1, MPI_INT, MPI_MIN, MPI_COMM_WORLD);
116 if (gok != 0) {
117 free(panel);
118 return -1;
119 }
120
121 /* Redistribute the 2D block-cyclic eigenvectors into this rank's state
122 column panel, then release Z_vec (peak memory drops back to O(N^2/P)). */
123 if (RedistBlockCyclicToStatePanel(NN, Z_vec, descZ_vec, jb, ncols, NN, panel)
124 != 0) {
125 free(panel);
126 return -1;
127 }
128 free(Z_vec);
129 Z_vec = NULL;
130
131 /* --- ExpecMode 2 plan: build once, identically on every rank (the
132 capability table gates Hubbard/HubbardGC/half-Spin/half-SpinGC as of
133 Task 5; every other model, and any runtime-demoted quantity of those
134 four, gets the all-fallback plan -- see src/expec_trace.c). ---- */
135 {
136 TraceExecutionPlan tplan;
137 /* overflow-free ceiling division (NN>=0, nproc>0 are the caller's
138 precondition) -- distinct from NC above, which the plan API commits
139 to calling nc_uniform to make explicit it is NOT a rank-local ncols. */
140 long int nc_uniform = NN / (long int)nproc + ((NN % (long int)nproc) != 0);
141 uint64_t gbuf_max = 0;
142 size_t gbuf_max_bytes;
143 /* Energy trace-kernel CSR context (phase 3c Task 5): collected once per
144 run (below), passed BY POINTER to the local loop, freed after it. Zeroed
145 here so TraceHamFree() is a safe no-op on every path that never collects
146 (ExpecMode 0/1, InputHam static demotion, or an unsupported model whose
147 energy slot the finalize step demotes). */
148 TraceHamCsr ham_csr;
149 double t_energy_map = 0.0; /* collect (map) seconds, rank-local */
150 memset(&ham_csr, 0, sizeof(ham_csr));
151 /* Final whole-branch review fix: only ExpecMode 2 runs parse (and can
152 warn about) HPHI_TRACE_BUF_MAX_MB. ExpecMode 0/1 leave gbuf_max at 0
153 and skip the Bcast too -- TraceBuildPlan() never reads this argument
154 on the ExpecMode!=TRACE path (see its own early return), so 0 is
155 never divided by or otherwise used there. */
156 if (X->Def.iExpecMode == EXPECMODE_TRACE) {
157 if (myrank == 0) gbuf_max = (uint64_t)TraceGbufMaxBytesFromEnv(); /* rank 0 only */
158 MPI_Bcast(&gbuf_max, 1, MPI_UINT64_T, 0, MPI_COMM_WORLD);
159 }
160 /* checked narrowing in case size_t is narrower than 64 bits here -- the
161 SAME capped cap feeds both TraceBuildPlan() and TraceHamCollect(). */
162 gbuf_max_bytes =
163 (gbuf_max > (uint64_t)SIZE_MAX) ? (size_t)SIZE_MAX : (size_t)gbuf_max;
164 TraceBuildPlan(X, nc_uniform, gbuf_max_bytes,
165 &tplan); /* ExpecMode!=2 => all-fallback plan */
166
167 /* Energy build->finalize lifecycle (design spec 3c): when TraceBuildPlan()
168 left the energy slot PROVISIONALLY kernel=1 (ExpecMode 2, not InputHam),
169 run the CSR collector on this rank, then reconcile every rank's outcome
170 with TraceFinalizeEnergyPlan() -- a single MPI_Allreduce(MIN) that
171 demotes energy on ALL ranks if ANY rank failed, or aborts on a
172 cross-rank nnz mismatch. Done BEFORE TraceReportPlan() so the INFO line
173 reflects the final verdict. Skipped when the slot is already 0
174 (InputHam static demotion, or ExpecMode 0/1). */
175 if (tplan.kernel[TRACE_Q_ENERGY]) {
176 int local_ok;
177 long int nnz_merged;
178 struct timespec ts0, ts1;
179 clock_gettime(CLOCK_MONOTONIC, &ts0);
180 local_ok = TraceHamCollect(X, gbuf_max_bytes, -1, &ham_csr);
181 clock_gettime(CLOCK_MONOTONIC, &ts1);
182 t_energy_map = (double)(ts1.tv_sec - ts0.tv_sec) +
183 (double)(ts1.tv_nsec - ts0.tv_nsec) * 1e-9;
184 /* On demotion TraceHamCollect() already freed+zeroed ham_csr, so
185 nnz_merged has no meaning -- pass 0 (the finalize contract ignores it
186 when local_ok==0). ham_csr.nnz is the MERGED count (rowptr[n]). */
187 nnz_merged = local_ok ? ham_csr.nnz : 0;
188 TraceFinalizeEnergyPlan(&tplan, local_ok, nnz_merged);
189 /* If finalize demoted energy (this or another rank failed) but THIS rank
190 had collected a valid CSR, free it now -- it will not reach the loop. */
191 if (!tplan.kernel[TRACE_Q_ENERGY])
192 TraceHamFree(&ham_csr);
193 }
194
195 if (X->Def.iExpecMode == EXPECMODE_TRACE && myrank == 0)
196 TraceReportPlan(&tplan, stdoutMPI);
197
198 /* --- MPI-free per-rank observable session (partial green_output
199 session). Single-exit invariant: once ExpecLocalEnter() has run,
200 GreenOutputClearPartialSuffix() and ExpecLocalLeave() below MUST run
201 exactly once before this block ends, regardless of whether the
202 kernel dispatch or the fallback loop reports failure -- do NOT add an
203 early return between Enter() and Leave(). --- */
204 ExpecLocalEnter(); /* also clears the sticky ExpecLocal error */
205 GreenOutputSetPartialSuffix(myrank);
206 rc_local = expec_trace_owned_states(X, &tplan, panel, jb, je, NN);
207 if (rc_local == 0)
208 rc_local = phys_stateparallel_local_loop(
209 X, panel, jb, je, NN, &tplan,
210 tplan.kernel[TRACE_Q_ENERGY] ? &ham_csr : NULL);
211 GreenOutputClearPartialSuffix();
212 ExpecLocalLeave();
213
214 /* Task 8 benchmark-breakdown line (final whole-branch review fix):
215 rank 0 only, after Leave (so ExpecLocal's guard window is already
216 closed -- this print is not itself scanned/restricted the way the
217 per-rank observable session above is), and only when this rank's
218 local trace-kernel dispatch actually completed
219 (rc_local==0 -- a failed call may have left the accumulators
220 mid-phase). One line per quantity this run selected the trace kernel
221 for; a quantity that fell back is not printed (nothing was timed for
222 it -- see TraceGetTimings()'s doc comment). */
223 if (X->Def.iExpecMode == EXPECMODE_TRACE && myrank == 0 && rc_local == 0) {
224 double timings[TRACE_Q_NQUANT][3];
225 TraceGetTimings(timings);
226 if (tplan.kernel[TRACE_Q_ONEBODY])
227 fprintf(stdoutMPI,
228 " ExpecMode 2 timing (rank 0): one-body map=%.3fs stream=%.3fs output=%.3fs\n",
229 timings[TRACE_Q_ONEBODY][0], timings[TRACE_Q_ONEBODY][1],
230 timings[TRACE_Q_ONEBODY][2]);
231 if (tplan.kernel[TRACE_Q_TWOBODY])
232 fprintf(stdoutMPI,
233 " ExpecMode 2 timing (rank 0): two-body map=%.3fs stream=%.3fs output=%.3fs\n",
234 timings[TRACE_Q_TWOBODY][0], timings[TRACE_Q_TWOBODY][1],
235 timings[TRACE_Q_TWOBODY][2]);
236 /* Energy family (phase 3c): map = the CSR collector (t_energy_map),
237 stream = the per-state SpMV/field kernel accumulated in the local loop
238 (phys_stateparallel_energy_stream_seconds()). Phys-field stores are
239 folded into the streaming step, so the output field is reported as 0 --
240 the three-field format is kept for a stable, greppable line. */
241 if (tplan.kernel[TRACE_Q_ENERGY])
242 fprintf(stdoutMPI,
243 " ExpecMode 2 timing (rank 0): energy map=%.3fs stream=%.3fs output=%.3fs\n",
244 t_energy_map, phys_stateparallel_energy_stream_seconds(), 0.0);
245 }
246
247 /* Free the energy CSR context (design spec 3c: orchestrator-owned, freed
248 after the loop). Safe no-op when it was never collected or already freed
249 on a finalize demotion -- TraceHamFree() tolerates a zeroed struct. */
250 TraceHamFree(&ham_csr);
251 }
252 free(panel);
253
254 /* --- single rendezvous: share the failure verdict across all ranks. --- */
255 MPI_Allreduce(&rc_local, &g, 1, MPI_INT, MPI_MIN, MPI_COMM_WORLD);
256 if (g != 0)
257 return -1;
258
259 /* --- Gatherv the all_* summary arrays to rank 0. Each rank owns the
260 block-contiguous segment [jb-1, je-1]; recvcounts/displs are derived from
261 the deterministic block ownership. All all_* arrays are double, so the
262 same recvcounts/displs are reused for every array. --- */
263 {
264 int *recvcounts = NULL, *displs = NULL;
265 double *tmp = NULL;
266 int sendcount;
267
268 /* long->int width check: the largest int-valued quantity fed to
269 MPI_Gatherv is a displ/count bounded by NN. */
270 ok = 0;
271 if (NN > (long int)INT_MAX) {
272 ok = -1;
273 if (myrank == 0)
274 fprintf(stdout,
275 " Error: N=%ld exceeds INT_MAX; cannot Gatherv all_* arrays.\n",
276 NN);
277 }
278 /* rank-0 receive buffers; sync malloc success before the collective. */
279 if (ok == 0 && myrank == 0) {
280 recvcounts = malloc((size_t)P * sizeof(int));
281 displs = malloc((size_t)P * sizeof(int));
282 tmp = malloc((size_t)NN * sizeof(double));
283 if (recvcounts == NULL || displs == NULL || tmp == NULL) {
284 ok = -1;
285 fprintf(stdout,
286 " Error: malloc failed for all_* Gatherv buffers (rank 0).\n");
287 }
288 }
289 MPI_Allreduce(&ok, &gok, 1, MPI_INT, MPI_MIN, MPI_COMM_WORLD);
290 if (gok != 0) {
291 free(recvcounts);
292 free(displs);
293 free(tmp);
294 return -1;
295 }
296
297 if (myrank == 0) {
298 for (r = 0; r < P; r++) {
299 long int jb_r = r * NC + 1;
300 long int je_r = (r + 1) * NC;
301 long int nc_r;
302 if (je_r > NN) je_r = NN;
303 nc_r = (je_r >= jb_r) ? (je_r - jb_r + 1) : 0;
304 recvcounts[r] = (int)nc_r;
305 displs[r] = (int)(r * NC); /* 0-based start; unused when nc_r==0 */
306 }
307 }
308 sendcount = (int)ncols;
309
310 /* rank 0 receives into a separate temp buffer, then copies into
311 X->Phys.all_* (avoids MPI_IN_PLACE aliasing with its own segment).
312 Senders pass &all_xxx[jb-1] with sendcount=ncols (0 for zero-owner). */
313#define PHYS_GATHER_ONE(arr) \
314 MPI_Gatherv((ncols > 0) ? &X->Phys.arr[jb - 1] : X->Phys.arr, sendcount, \
315 MPI_DOUBLE, tmp, recvcounts, displs, MPI_DOUBLE, 0, \
316 MPI_COMM_WORLD); \
317 if (myrank == 0) { \
318 long int k; \
319 for (k = 0; k < NN; k++) X->Phys.arr[k] = tmp[k]; \
320 }
321 PHYS_GATHER_ONE(all_energy)
322 PHYS_GATHER_ONE(all_doublon)
323 PHYS_GATHER_ONE(all_sz)
324 PHYS_GATHER_ONE(all_s2)
325 PHYS_GATHER_ONE(all_num_up)
326 PHYS_GATHER_ONE(all_num_down)
327#undef PHYS_GATHER_ONE
328
329 free(recvcounts);
330 free(displs);
331 free(tmp);
332 }
333
334 /* --- rank 0 re-renders the progress lines in state order, in the serial
335 format (S2 column included). Mode 1 is only ever entered for FullDiag. --- */
336 if (myrank == 0 && X->Def.iCalcType == FullDiag) {
337 long int i;
338 double tmp_N;
339 for (i = 0; i < NN; i++) {
340 if (X->Def.iCalcModel == Spin || X->Def.iCalcModel == SpinGC) {
341 tmp_N = X->Def.NsiteMPI;
342 } else {
343 tmp_N = X->Phys.all_num_up[i] + X->Phys.all_num_down[i];
344 }
345 fprintf(stdoutMPI,
346 "i=%5ld Energy=%10lf N=%10lf Sz=%10lf S2=%10lf Doublon=%10lf \n",
347 i, X->Phys.all_energy[i], tmp_N, X->Phys.all_sz[i],
348 X->Phys.all_s2[i], X->Phys.all_doublon[i]);
349 }
350 }
351
352 /* --- publish the aggregate green_output files: rank 0 concatenates the
353 rank-ordered (= state-ordered) partials. Collective; identical rc on all
354 ranks; publishes/deletes only on global success. --- */
355 if (GreenOutputMergePartials(X) != 0)
356 return -1;
357
358 return 0;
359}
