56void TraceFinalizeEnergyPlan(TraceExecutionPlan *plan, int local_ok,
57 long int nnz_merged) {
58 long long buf[3];
59
60 if (local_ok && nnz_merged >= 0 && (long long)nnz_merged != LLONG_MAX) {
61 /* success rank with a representable merged nnz distinct from the failure
62 sentinel (long int always fits long long; nnz_merged>=0 makes the
63 negation safe). */
64 buf[0] = 1;
65 buf[1] = (long long)nnz_merged;
66 buf[2] = -(long long)nnz_merged;
67 } else {
68 /* failed rank (gate exceeded, any allocation failure), or an nnz that
69 cannot be encoded distinctly from the sentinel -> treat self as failed. */
70 buf[0] = 0;
71 buf[1] = LLONG_MAX;
72 buf[2] = LLONG_MAX;
73 }
74
75#ifdef MPI
76 MPI_Allreduce(MPI_IN_PLACE, buf, 3, MPI_LONG_LONG, MPI_MIN, MPI_COMM_WORLD);
77#endif
78
79 /* Verdict order (identical reduced buffer on every rank -> identical branch),
80 design spec 3c: demote-before-consistency so failed-rank sentinels never
81 pollute the nnz agreement check. */
82 if (buf[0] == 0) {
83 plan->kernel[TRACE_Q_ENERGY] = 0;
84 plan->demoted_memory[TRACE_Q_ENERGY] = 1;
85 } else if (buf[1] != -buf[2]) {
86 fprintf(stderr,
87 " Error: ExpecMode 2 energy trace kernel: the collected "
88 "Hamiltonian nnz differs across MPI ranks (min nnz=%lld, "
89 "max nnz=%lld) -- a nondeterministic makeHam enumeration, which is "
90 "a correctness error, not a fallback case. Aborting.\n",
91 buf[1], -buf[2]);
92 exitMPI(-1);
93 }
94 /* else: every rank succeeded with an agreeing nnz -> energy stays
95 kernel[TRACE_Q_ENERGY]=1 (final). */
96}
