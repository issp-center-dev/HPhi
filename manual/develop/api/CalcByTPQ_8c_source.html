81static int tpq_output_open(const char *name, const char *mode, FILE **fp)
82{
83 int failed = childfopenMPI(name, mode, fp) != 0 || *fp == NULL;
84#ifdef HPHI_TEST_SYMMETRY_PROBE
85 if (SymmetryFullDiagTestFail("tpq_output_open")) failed = 1;
86#endif
87 if (SumMPI_i(failed) == 0) return 0;
88 if (*fp != NULL) {
89 fclose(*fp);
90 *fp = NULL;
91 }
92#ifdef HPHI_TEST_SYMMETRY_PROBE
93 tpq_report_agreed_failure("output_open");
94#endif
95 fprintf(stdoutMPI,
96 "Error: mTPQ output open failed on one or more MPI ranks.\n");
97 return -1;
98}

100static int tpq_output_close(FILE **fp)
101{
102 int failed = *fp == NULL;
103 if (*fp != NULL) {
104 if (ferror(*fp) != 0) failed = 1;
105 if (fclose(*fp) != 0) failed = 1;
106 *fp = NULL;
107 }
108#ifdef HPHI_TEST_SYMMETRY_PROBE
109 if (SymmetryFullDiagTestFail("tpq_output_close")) failed = 1;
110#endif
111 if (SumMPI_i(failed) == 0) return 0;
112#ifdef HPHI_TEST_SYMMETRY_PROBE
113 tpq_report_agreed_failure("output_close");
114#endif
115 fprintf(stdoutMPI,
116 "Error: mTPQ output close failed on one or more MPI ranks.\n");
117 return -1;
118}
