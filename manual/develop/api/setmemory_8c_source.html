162int***i_3d_allocate(const long unsigned int N, const long unsigned int M, const long unsigned int L){
163 long unsigned int int_i, int_j;
164 int*** A;
165 A = (int***)calloc(nonzero_count(N),sizeof(int**));
166 A[0] = (int**)calloc(nonzero_count(M*N),sizeof(int*));
167 A[0][0] = (int*)calloc(nonzero_count(L*M*N),sizeof(int));
168 for(int_i=0;int_i<N; int_i++) {
169 A[int_i] = A[0] + int_i*M;
170 for(int_j = 0; int_j<M; int_j++){
171 A[int_i][int_j]= A[0][0] + int_i*M*L + int_j*L;
172 }
173 }
174 return A;
175}
