395double complex exchange_spin_element(
396 long unsigned int j,
397 double complex *tmp_v0,
398 double complex *tmp_v1,
399 struct BindStruct *X,
400 long unsigned int *tmp_off
401) {
402 long unsigned int off;
403 double complex dmv;
404 long unsigned int iexchg;
405 long unsigned int is_up = X->Large.isA_spin;
406 long unsigned int irght = X->Large.irght;
407 long unsigned int ilft = X->Large.ilft;
408 long unsigned int ihfbit = X->Large.ihfbit;
409 double complex tmp_J = X->Large.tmp_J;
410 int mode = X->Large.mode;
411 double complex dam_pr = 0;
412 long unsigned int ibit_tmp;
413
414 ibit_tmp = (list_1[j] & is_up);
415 if (ibit_tmp == 0 || ibit_tmp == is_up) {
416 return dam_pr;
417 }
418 else {
419 iexchg = list_1[j] ^ is_up;
420 GetOffComp(list_2_1, list_2_2, iexchg, irght, ilft, ihfbit, &off);
421 *tmp_off = off;
422 dmv = tmp_J * tmp_v1[j];
423 if (mode == M_MLTPLY) {
424 tmp_v0[off] += dmv;
425 }
426 dam_pr += dmv * conj(tmp_v1[off]);
427 return dam_pr;
428 }
