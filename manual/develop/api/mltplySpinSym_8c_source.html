25static unsigned long int mask_between_sites(unsigned int site0, unsigned int site1)
26{
27 unsigned int lo = site0 < site1 ? site0 : site1;
28 unsigned int hi = site0 < site1 ? site1 : site0;
29 if (hi <= lo + 1U) return 0UL;
30 return (1UL << hi) - (1UL << (lo + 1U));
31}

33static int apply_spinless_hopping_hermite(unsigned long int state,
34 unsigned int site1,
35 unsigned int site2,
36 double complex trans,
37 unsigned long int *out_state,
38 double complex *hval)
39{
40 unsigned long int mask1 = 1UL << site1;
41 unsigned long int mask2 = 1UL << site2;
42 unsigned long int occupied1 = state & mask1;
43 unsigned long int occupied2 = state & mask2;
44 int sgn = 1;
45 if (site1 == site2) return FALSE;
46 if ((occupied1 == 0UL && occupied2 == 0UL) ||
47 (occupied1 != 0UL && occupied2 != 0UL)) {
48 return FALSE;
49 }
50 SgnBit(state & mask_between_sites(site1, site2), &sgn);
51 *out_state = state ^ mask1 ^ mask2;
52 if (occupied1 != 0UL && occupied2 == 0UL) {
53 *hval = (double)sgn * conj(trans);

56 }
