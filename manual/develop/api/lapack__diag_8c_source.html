39static int lapack_diag_elpa(struct BindStruct *X, long int xMsize) {
40 int i_negone = -1, i_zero = 0;
41 int size;
42 int nprow, npcol, myrow, mycol;
43 int ictxt;
44 long int mb, mp, nq, i, j;
45 int lld, dims[2] = {0, 0};
46 int iam, nprocs, info;
47 double complex *A_distr;
48 double *w;
49 int descA[9];
50 int ierr;
51
52 fprintf(stdoutMPI, "Using ELPA (%s)\n\n",
53 X->Def.iNGPU >= 1 ? "GPU" : "CPU");
54
55 MPI_Comm_size(MPI_COMM_WORLD, &size);
56 MPI_Dims_create(size, 2, dims);
57 nprow = dims[0]; npcol = dims[1];
58
59 if (xMsize < ((nprow > npcol) ? nprow : npcol)) {
60 fprintf(stdoutMPI,
61 "Error: matrix dimension (%ld) is smaller than the process grid (%d x %d).\n"
62 " Reduce the number of MPI ranks for this problem size.\n",
63 xMsize, nprow, npcol);
64 /* Pre-existing early return, predating the descinit-verdict block
65 below (PR #276 review round 2): this runs before any BLACS grid
66 exists, but iHamPanelActive/Ham_local are set by the caller
67 (xsetmem.c) well before lapack_diag_elpa() is entered, so the
68 distributed-panel mode leak fixed in the descinit-failure branch
69 applies here too. Same one-line cleanup idiom, minus
70 blacs_gridexit_ (no grid to tear down yet). */
71 if (iHamPanelActive) {
72 free(Ham_local);
73 Ham_local = NULL;
74 iHamPanelActive = 0;
75 }
76 return -1;
77 }
78 mb = ElpaBlockSize(xMsize, nprow, npcol);
79
80#ifdef _ELPA_GPU
81 /* Startup consistency warning (design doc sec. 2): ranks per node
82 should be a multiple of NGPU (ideally equal: 1 rank per GPU). */
83 if (X->Def.iNGPU >= 1) {
84 MPI_Comm comm_node;
85 int nrank_node;
86 MPI_Comm_split_type(MPI_COMM_WORLD, MPI_COMM_TYPE_SHARED, 0,
87 MPI_INFO_NULL, &comm_node);
88 MPI_Comm_size(comm_node, &nrank_node);
89 MPI_Comm_free(&comm_node);
90 if (nrank_node % X->Def.iNGPU != 0) {
91 fprintf(stdoutMPI,
92 "Warning: ranks per node (%d) is not a multiple of NGPU (%d):\n"
93 " GPUs may idle or be shared unevenly. Recommended: 1 rank per GPU.\n",
94 nrank_node, X->Def.iNGPU);
95 }
96 }
97#endif
98
99 blacs_pinfo_(&iam, &nprocs);
100 blacs_get_(&i_negone, &i_zero, &ictxt);
101 blacs_gridinit_(&ictxt, "R", &nprow, &npcol);
102 blacs_gridinfo_(&ictxt, &nprow, &npcol, &myrow, &mycol);
103
104 mp = numroc_(&xMsize, &mb, &myrow, &i_zero, &nprow);
105 nq = numroc_(&xMsize, &mb, &mycol, &i_zero, &npcol);
106 lld = (mp > 0) ? mp : 1;
107
108 descinit_(descA, &xMsize, &xMsize, &mb, &mb, &i_zero, &i_zero, &ictxt, &lld, &info);
109 {
110 /* descinit_ overwrites info per call: fold both verdicts, then
111 synchronize so no rank proceeds into the collectives below with an
112 invalid descriptor while others abort (same pattern as
113 RedistBlockCyclicToStatePanel). Capture descA's info before the
114 second descinit_ call overwrites it, so a descA-only failure is
115 not misreported as info=0 (PR #276 review round 2). */
116 int info_a = info;
117 int ok = (info_a == 0) ? 0 : -1, gok;
118 descinit_(descZ_vec, &xMsize, &xMsize, &mb, &mb, &i_zero, &i_zero, &ictxt, &lld, &info);
119 if (info != 0) ok = -1;
120 if (ok != 0) {
121 fprintf(stdout,
122 " Error: descinit_ failed (info_A=%d, info_Z=%d) for the ELPA\n"
123 " descriptors on rank %d; aborting the ELPA diagonalization.\n",
124 info_a, info, myrank);
125 }
126 MPI_Allreduce(&ok, &gok, 1, MPI_INT, MPI_MIN, MPI_COMM_WORLD);
127 if (gok != 0) {
128 /* Distributed-panel mode (phase 2): this return happens before the
129 panel-mode branch below consumes Ham_local, so free it here too
130 (same idiom as the panel-mode allocation-failure paths further
131 down in this function; PR #276 review round 2). */
132 if (iHamPanelActive) {
133 free(Ham_local);
134 Ham_local = NULL;
135 iHamPanelActive = 0;
136 }
137 blacs_gridexit_(&ictxt);
138 return -1;
139 }
140 }
141
142 if (iHamPanelActive) {
143 /* Distributed generation (phase 2): allocate only A_distr + w first,
144 redistribute the 1D panel into A_distr, then free the panel and
145 only THEN allocate Z_vec. This keeps the redistribution-time peak
146 at 2 coexisting matrices (design doc sec. 3), not 3. Every
147 rank-local malloc (including failure) is synced with
148 MPI_Allreduce(MIN) before any collective touches the buffer
149 (design doc sec. 4), same pattern as SyncError in
150 matrixlapack_elpa.c / the ownership guard in
151 RedistPanelToBlockCyclic. */
152 int rerr, ok, gok;
153
154 A_distr = malloc(((mp * nq > 0) ? mp * nq : 1) * sizeof(double complex));
155 w = malloc(xMsize * sizeof(double));
156 ok = (A_distr != NULL && w != NULL) ? 0 : -1;
157 if (ok != 0) {
158 fprintf(stdout, " Error: malloc failed for A_distr/w (rank %d).\n", myrank);
159 }
160 MPI_Allreduce(&ok, &gok, 1, MPI_INT, MPI_MIN, MPI_COMM_WORLD);
161 if (gok != 0) {
162 free(A_distr);
163 free(w);
164 free(Ham_local);
165 Ham_local = NULL;
166 iHamPanelActive = 0;
167 blacs_gridexit_(&ictxt);
168 return -1;
169 }
170
171 rerr = RedistPanelToBlockCyclic(xMsize, HamColBegin,
172 (HamColEnd >= HamColBegin)
173 ? (HamColEnd - HamColBegin + 1) : 0,
174 HamPanelLd, Ham_local, A_distr, descA);
175 free(Ham_local);
176 Ham_local = NULL;
177 iHamPanelActive = 0; /* panel consumed; phys.c uses Z_vec only */
178 if (rerr != 0) {
179 free(A_distr);
180 free(w);
181 blacs_gridexit_(&ictxt);
182 return -1;
183 }
184
185 Z_vec = malloc(((mp * nq > 0) ? mp * nq : 1) * sizeof(double complex));
186 ok = (Z_vec != NULL) ? 0 : -1;
187 if (ok != 0) {
188 fprintf(stdout, " Error: malloc failed for Z_vec (rank %d).\n", myrank);
189 }
190 MPI_Allreduce(&ok, &gok, 1, MPI_INT, MPI_MIN, MPI_COMM_WORLD);
191 if (gok != 0) {
192 free(A_distr);
193 free(Z_vec);
194 Z_vec = NULL;
195 free(w);
196 blacs_gridexit_(&ictxt);
197 return -1;
198 }
199 } else {
200 int ok, gok;
201
202 A_distr = malloc(((mp * nq > 0) ? mp * nq : 1) * sizeof(double complex));
203 Z_vec = malloc(((mp * nq > 0) ? mp * nq : 1) * sizeof(double complex));
204 w = malloc(xMsize * sizeof(double));
205 ok = (A_distr != NULL && Z_vec != NULL && w != NULL) ? 0 : -1;
206 if (ok != 0) {
207 fprintf(stdout, " Error: malloc failed for A_distr/Z_vec/w (rank %d).\n", myrank);
208 }
209 MPI_Allreduce(&ok, &gok, 1, MPI_INT, MPI_MIN, MPI_COMM_WORLD);
210 if (gok != 0) {
211 free(A_distr);
212 free(Z_vec);
213 Z_vec = NULL;
214 free(w);
215 blacs_gridexit_(&ictxt);
216 return -1;
217 }
218
219 for (i = 0; i < xMsize; i++) {
220 for (j = 0; j < xMsize; j++) {
221 DivMat(i, j, Ham[i][j], A_distr, descA);
222 }
223 }
224 }
225
226 ierr = diag_elpa_cmp((int)xMsize, A_distr, Z_vec, w,
227 (int)mp, (int)nq, (int)myrow, (int)mycol,
228 (int)mb, X->Def.iNGPU);
229 free(A_distr);
230 if (ierr != 0) {
231 /* Failed diagonalization must not leave a dangling non-NULL Z_vec
232 (PR #276 review): downstream code treats Z_vec != NULL as "valid
233 distributed eigenvectors exist". */
234 free(Z_vec);
235 Z_vec = NULL;
236 free(w);
237 blacs_gridexit_(&ictxt);
238 return -1;
239 }
240
241 for (i = 0; i < xMsize; i++) {
242 v0[i] = w[i];
243 }
244 free(w);
245 use_scalapack = 1;
246 return 0;
247}
