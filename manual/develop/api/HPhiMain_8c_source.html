649 int mode=0;

653 if(JudgeDefType(argc, argv, &mode)!=0){
654 exitMPI(-1);

659 nproc = 1;
660 stdoutMPI = stdout;
661 splash();

665 //Timer
666 InitTimer();
667 if (mode != STANDARD_DRY_MODE) StartTimer(0);
668
669 //MakeDirectory for output
670 struct stat tmpst;
671 if (myrank == 0) {
672 if (stat(cParentOutputFolder, &tmpst) != 0) {
673 if (mkdir(cParentOutputFolder, 0777) != 0) {
674 fprintf(stdoutMPI, "%s", cErrOutput);
675 exitMPI(-1);
676 }
677 }
678 }/*if (myrank == 0)*/
679
680 strcpy(cFileListName, argv[2]);
681
682 if(mode==STANDARD_MODE || mode == STANDARD_DRY_MODE){
683 if (myrank == 0) StdFace_main(argv[2]);
684 strcpy(cFileListName, "namelist.def");
685 if (mode == STANDARD_DRY_MODE){
686 fprintf(stdout, "Dry run is Finished. \n\n");
687 return 0;
688 }
689 }
690
691 setmem_HEAD(&X.Bind);
692 if(ReadDefFileNInt(cFileListName, &(X.Bind.Def), &(X.Bind.Boost))!=0){
693 fprintf(stdoutMPI, "%s", cErrDefFile);
694 exitMPI(-1);
695 }
696
697 if (X.Bind.Def.nvec < X.Bind.Def.k_exct){

700 exitMPI(-1);
701 }
702 fprintf(stdoutMPI, "%s", cProFinishDefFiles);
703
704 /*ALLOCATE-------------------------------------------*/
705 setmem_def(&X.Bind, &X.Bind.Boost);
706 /*-----------------------------------------------------*/
707
708 /*Read Def files.*/
709 TimeKeeper(&(X.Bind), cFileNameTimeKeep, cReadDefStart, "w");
710 if(ReadDefFileIdxPara(&(X.Bind.Def), &(X.Bind.Boost))!=0){
711 fprintf(stdoutMPI, "%s", cErrIndices);
712 exitMPI(-1);
713 }
714 TimeKeeper(&(X.Bind), cFileNameTimeKeep, cReadDefFinish, "a");
715 fprintf(stdoutMPI, "%s", cProFinishDefCheck);
716
717 /*Set convergence Factor*/
718 SetConvergenceFactor(&(X.Bind.Def));
719
720 if (X.Bind.Def.iCalcType == FullDiag
721 && X.Bind.Def.iFlgScaLAPACK ==0
722 && nproc != 1) {
723 fprintf(stdoutMPI, "Error: Full Diagonalization by LAPACK is only allowed for one process.\n");
724 FinalizeMPI();
725 return(-1);
726 }
727
728 /*---------------------------*/
729 if(HPhiTrans(&(X.Bind))!=0) {
730 exitMPI(-1);
731 }
732
733 //Start Calculation
734 if(X.Bind.Def.iFlgCalcSpec == CALCSPEC_NOT) {
735
736 if(check(&(X.Bind))==MPIFALSE){
737 exitMPI(-1);
738 }
739
740 /*LARGE VECTORS ARE ALLOCATED*/
741 if (setmem_large(&X.Bind) != 0) {
742 fprintf(stdoutMPI, cErrLargeMem, iErrCodeMem);
743 exitMPI(-1);
744 }
745
746 StartTimer(1000);
747 if(sz(&(X.Bind), list_1, list_2_1, list_2_2)!=0){
748 exitMPI(-1);
749 }
750
751 StopTimer(1000);
752 if(X.Bind.Def.WRITE==1){
753 output_list(&(X.Bind));
754 exitMPI(-2);
755 }
756 StartTimer(2000);
757 diagonalcalc(&(X.Bind));
758 StopTimer(2000);
759
760 switch (X.Bind.Def.iCalcType) {
761 case Lanczos:
762 StartTimer(4000);
763 if (CalcByLanczos(&X) != TRUE) {
764 StopTimer(4000);
765 exitMPI(-3);
766 }
767 StopTimer(4000);
768 break;
769
770 case CG:
771 if (CalcByLOBPCG(&X) != TRUE) {
772 exitMPI(-3);
773 }
774 break;
775
776 case FullDiag:
777 StartTimer(5000);
778 if (CalcByFullDiag(&X) != TRUE) {
779 FinalizeMPI();
780 }
781 StopTimer(5000);
782 break;
783
784 case TPQCalc:
785 StartTimer(3000);
786 if (CalcByTPQ(NumAve, X.Bind.Def.Param.ExpecInterval, &X) != TRUE) {
787 StopTimer(3000);
788 exitMPI(-3);
789 }
790 StopTimer(3000);
791 break;
792
793 case cTPQ:
794 StartTimer(3000);
795 if (CalcByCanonicalTPQ(NumAve, X.Bind.Def.Param.ExpecInterval, &X) != TRUE) {
796 StopTimer(3000);
797 exitMPI(-3);
798 }
799 StopTimer(3000);
800 break;
801
802
803 case TimeEvolution:
804 if(CalcByTEM(X.Bind.Def.Param.ExpecInterval, &X)!=0){
805 exitMPI(-3);
806 }
807 break;
808
809 default:
810 StopTimer(0);
811 exitMPI(-3);
812 }
813 }
814 else{
815 StartTimer(6000);
816 if (CalcSpectrum(&X) != TRUE) {
817 StopTimer(6000);
818 exitMPI(-3);
819 }
820 StopTimer(6000);
821 }
822
823 StopTimer(0);
824 OutputTimer(&(X.Bind));
825 FinalizeMPI();
826 return 0;
827}
