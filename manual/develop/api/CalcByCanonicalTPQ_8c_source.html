467int count_file_lines(const char *file_name) {
468 FILE *file;
469 int lines = 0;
470 int ch;
471 int prev = '\n'; // assume file starts with a new line
472
473 file = fopen(file_name, "r");
474 if (file == NULL) {
475 fprintf(stderr, "could not open file: %s\n", file_name);
476 return -1; // distinguishable error code
477 }
478
479 while ((ch = fgetc(file)) != EOF) {
480 if (ch == '\n') {
481 lines++;
482 }
483 prev = ch;
484 }
485
486 fclose(file);
487
488 if (prev != '\n') {
489 lines++;
490 }
491
492 return lines;
493}

495/* * @brief Read inverse temperature data from a file
496 *
497 * @param read_invtemp Pointer to an array for storing inverse temperatures
498 * @param read_nmax Pointer to an array for storing nmax values
499 * @param read_physcal Pointer to an array for storing physical calculation flags
500 * @param read_eigen Pointer to an array for storing eigenvalue flags
501 * @param file_name Name of the file to read from
502 * @param max_lines Maximum number of lines to read from the file
503 * @return Number of lines read, or -1 if the file could not be opened, or -2 if the number of lines exceeds max_lines
504 * @author Takahiro Misawa (ISSP)
505 */
506int func_read_invtemp(double *read_invtemp, int *read_nmax, int *read_physcal, int *read_eigen, const char *file_name, int max_lines) {
507
508 FILE *file = fopen(file_name, "r");
509 char line[1024];
510 if (file == NULL) {
511 fprintf(stderr, "could not open file: %s\n", file_name);
512 return -1;
513 }
514
515 int i = 0;
516 int line_no = 0;
517 while (fgets(line, sizeof(line), file) != NULL) {
518 double invtemp_tmp;
519 int nmax_tmp, physcal_tmp, eigen_tmp;
520 int nread = 0;
521 char *cursor = line;
522 line_no++;
523
524 while (isspace((unsigned char)*cursor)) cursor++;
525 if (*cursor == '\0') continue;
526
527 if (i >= max_lines) {
528 fprintf(stderr, "Error: number of lines in file exceeds expected %d\n", max_lines);
529 fclose(file);
530 return -2;
531 }
532
533 if (sscanf(cursor, "%lf %d %d %d %n",
534 &invtemp_tmp, &nmax_tmp, &physcal_tmp, &eigen_tmp, &nread) != 4) {
535 fprintf(stderr, "Error: invalid InvTemp row %d in file: %s\n", line_no, file_name);
536 fclose(file);
537 return -3;
538 }
539 cursor += nread;
540 while (isspace((unsigned char)*cursor)) cursor++;
541 if (*cursor != '\0') {
542 fprintf(stderr, "Error: invalid InvTemp row %d in file: %s\n", line_no, file_name);
543 fclose(file);
544 return -3;
545 }
546
547 read_invtemp[i] = invtemp_tmp;
548 read_nmax[i] = nmax_tmp;
549 read_physcal[i] = physcal_tmp;
550 read_eigen[i] = eigen_tmp;
551 i++;
552 }
553
554 fclose(file);
555 return i;
556}
