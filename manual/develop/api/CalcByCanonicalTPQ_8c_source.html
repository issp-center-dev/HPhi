512int count_file_lines(const char *file_name) {
513 FILE *file;
514 int lines = 0;
515 int ch;
516 int prev = '\n'; // assume file starts with a new line
517
518 file = fopen(file_name, "r");
519 if (file == NULL) {
520 fprintf(stderr, "could not open file: %s\n", file_name);
521 return -1; // distinguishable error code
522 }
523
524 while ((ch = fgetc(file)) != EOF) {
525 if (ch == '\n') {
526 lines++;
527 }
528 prev = ch;
529 }
530
531 fclose(file);
532
533 if (prev != '\n') {
534 lines++;
535 }
536
537 return lines;
538}

540/* * @brief Read inverse temperature data from a file
541 *
542 * @param read_invtemp Pointer to an array for storing inverse temperatures
543 * @param read_nmax Pointer to an array for storing nmax values
544 * @param read_physcal Pointer to an array for storing physical calculation flags
545 * @param read_eigen Pointer to an array for storing eigenvalue flags
546 * @param file_name Name of the file to read from
547 * @param max_lines Maximum number of lines to read from the file
548 * @return Number of lines read, or -1 if the file could not be opened, or -2 if the number of lines exceeds max_lines
549 * @author Takahiro Misawa (ISSP)
550 */
551int func_read_invtemp(double *read_invtemp, int *read_nmax, int *read_physcal, int *read_eigen, const char *file_name, int max_lines) {
552
553 FILE *file = fopen(file_name, "r");
554 char line[1024];
555 if (file == NULL) {
556 fprintf(stderr, "could not open file: %s\n", file_name);
557 return -1;
558 }
559
560 int i = 0;
561 int line_no = 0;
562 while (fgets(line, sizeof(line), file) != NULL) {
563 double invtemp_tmp;
564 int nmax_tmp, physcal_tmp, eigen_tmp;
565 int nread = 0;
566 char *cursor = line;
567 line_no++;
568
569 while (isspace((unsigned char)*cursor)) cursor++;
570 if (*cursor == '\0') continue;
571
572 if (i >= max_lines) {
573 fprintf(stderr, "Error: number of lines in file exceeds expected %d\n", max_lines);
574 fclose(file);
575 return -2;
576 }
577
578 if (sscanf(cursor, "%lf %d %d %d %n",
579 &invtemp_tmp, &nmax_tmp, &physcal_tmp, &eigen_tmp, &nread) != 4) {
580 fprintf(stderr, "Error: invalid InvTemp row %d in file: %s\n", line_no, file_name);
581 fclose(file);
582 return -3;
583 }
584 cursor += nread;
585 while (isspace((unsigned char)*cursor)) cursor++;
586 if (*cursor != '\0') {
587 fprintf(stderr, "Error: invalid InvTemp row %d in file: %s\n", line_no, file_name);
588 fclose(file);
589 return -3;
590 }
591
592 read_invtemp[i] = invtemp_tmp;
593 read_nmax[i] = nmax_tmp;
594 read_physcal[i] = physcal_tmp;
595 read_eigen[i] = eigen_tmp;
596 i++;
597 }
598
599 fclose(file);
600 return i;
