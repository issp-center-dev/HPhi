158int***i_3d_allocate(const long unsigned int N, const long unsigned int M, const long unsigned int L){
159 long unsigned int int_i, int_j;
160 int*** A;
161 A = (int***)calloc((N),sizeof(int**));
162 A[0] = (int**)calloc((M*N),sizeof(int*));
163 A[0][0] = (int*)calloc((L*M*N),sizeof(int));
164 for(int_i=0;int_i<N; int_i++) {
165 A[int_i] = A[0] + int_i*M;
166 for(int_j = 0; int_j<M; int_j++){
167 A[int_i][int_j]= A[0][0] + int_i*M*L + int_j*L;
168 }
169 }
170 return A;
171}
