35SectorShift GetExcitationSectorShift(int iCalcModel, int isGeneralSpin, int isPair, const int *op) {
36 SectorShift s = {0, 0, 0, 0, TRUE, OFFDIAG_SHIFT_OK};
37
38 if (isPair == FALSE) {
39 const int spin = op[1]; /* 0 = up, 1 = down */
40 const int isCreation = (op[2] == 1); /* type==1 -> cis (creation), else ajt (annihilation) */
41 switch (iCalcModel) {
42 case Hubbard: /* v1 allow-list (CalcSpectrum.c:537-562; spin 0 == up) */
43 if (isCreation) {
44 s.dNe = +1;
45 if (spin == 0) s.dNup = +1; else s.dNdown = +1;
46 } else {
47 s.dNe = -1;
48 if (spin == 0) s.dNup = -1; else s.dNdown = -1;
49 }
50 break;
51 case Spin:
52 case SpinGC: /* single excitation N/A for spin (CalcSpectrum.c:563-565, SingleEx.c:54) */
53 s.valid = FALSE;
54 break;
55 case HubbardGC:
56 /* Grand canonical: the full Fock space is one Hilbert space, so a single c/c^dag
57 excitation stays in it -- no sector shift (the doc's "single: HubbardGC -> none").
58 s keeps the default {0,0,0,0, valid=TRUE}. This lets the bra/ket off-diagonal path
59 (SpectrumNumBra) project ket and bra -- of either spin -- in the shared GC space. */
60 break;
61 case HubbardNConserved:
62 /* Ne conserved, 2Sz free: c/c^dag shifts the sector by +-1 in Ne ONLY (the excited list
63 is Ne+-1 with all 2Sz). Track dNe alone and leave the Sz components zero, so the
64 bra/ket and cross-operator sector checks match same- AND cross-spin operators (both
65 spins land in the same Ne+-1 space) -- the spin-orbit-capable off-diagonal route. */
66 if (isCreation) s.dNe = +1;
67 else s.dNe = -1;
68 break;
69 case Kondo:
70 case KondoGC:
71 case tJ:
72 case tJGC:
73 case tJNConserved:
74 case KondoNConserved:
75 /* deferred: not in v1 allow-list (rejected upstream by the allow-list guard) */
76 s.valid = FALSE;
77 break;
78 default:
79 s.valid = FALSE;
80 break;
81 }
82 } else {
83 const int spin1 = op[1];
84 const int spin2 = op[3];
85 switch (iCalcModel) {
86 case Hubbard: /* v1 allow-list (CalcSpectrum.c:576-590; spin 0 == up) */
87 if (spin1 != spin2) {
88 if (spin1 == 0) { s.dNup = +1; s.dNdown = -1; }
89 else { s.dNup = -1; s.dNdown = +1; }
90 }
91 break;
92 case SpinGC: /* v1 allow-list: full grand-canonical space, no shift */
93 break;
94 case Spin: /* v1 allow-list; NOTE: spin convention is opposite to Hubbard (CalcSpectrum.c:591-606) */
95 if (spin1 != spin2) {
96 if (isGeneralSpin == FALSE) {
97 if (spin1 == 0) { s.dNup = -1; s.dNdown = +1; } /* CalcSpectrum.c:594 //down */
98 else { s.dNup = +1; s.dNdown = -1; } /* CalcSpectrum.c:597 //up */
99 } else {
100 s.dTotal2Sz = 2 * (spin1 - spin2); /* CalcSpectrum.c:603 */
101 }
102 }
103 break;
104 case Kondo:
105 case KondoGC:
106 case tJ:
107 case tJGC:
108 case HubbardGC:
109 case HubbardNConserved:
110 case tJNConserved:
111 case KondoNConserved:
112 /* deferred: not in v1 allow-list (rejected upstream by the allow-list guard) */
113 s.valid = FALSE;
114 break;
115 default:
116 s.valid = FALSE;
117 break;
118 }
119 }
120 /* A single operator row can only be invalid because the model/operator is
121 not in the allow-list (set-internal inconsistency needs >= 2 rows). */
122 s.reason = (s.valid == TRUE) ? OFFDIAG_SHIFT_OK : OFFDIAG_SHIFT_MODEL_NOT_ALLOWED;
123 return s;
124}

126SectorShift GetExcitationOperatorSetShift(int iCalcModel, int isGeneralSpin, int isPair,
127 int **op, int nOp) {
128 SectorShift acc = {0, 0, 0, 0, FALSE, OFFDIAG_SHIFT_MODEL_NOT_ALLOWED};
129 int i;
130
131 if (nOp <= 0) return acc; /* empty set: valid=FALSE */
132
133 for (i = 0; i < nOp; i++) {
134 SectorShift cur = GetExcitationSectorShift(iCalcModel, isGeneralSpin, isPair, op[i]);
135 if (cur.valid == FALSE) {
136 acc.valid = FALSE;
137 acc.reason = cur.reason; /* model/operator not allowed */
138 return acc;
139 }
140 if (i == 0) {
141 acc = cur;
142 } else if (acc.dNe != cur.dNe || acc.dNup != cur.dNup ||
143 acc.dNdown != cur.dNdown || acc.dTotal2Sz != cur.dTotal2Sz) {
144 acc.valid = FALSE; /* rows disagree on sector shift */
145 acc.reason = OFFDIAG_SHIFT_SET_INCONSISTENT;
146 return acc;
147 }
148 }
149 return acc; /* valid=TRUE; shift is common to all rows */
150}
