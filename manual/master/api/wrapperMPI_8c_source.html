66void InitializeMPI(int argc, char *argv[]){
67 int ierr;
68
69#ifdef MPI
70 ierr = MPI_Init(&argc, &argv);
71 ierr = MPI_Comm_size(MPI_COMM_WORLD, &nproc);
72 ierr = MPI_Comm_rank(MPI_COMM_WORLD, &myrank);
73 if(ierr != 0) exitMPI(ierr);
74 /* Parse HPHI_MPI_NOBATCH on rank 0 and broadcast so every rank agrees
75 (a per-rank divergence would deadlock the collective MPI_Sendrecv pattern). */
76 if (myrank == 0) {
77 const char *e = getenv("HPHI_MPI_NOBATCH");
78 iFlgMPIBatch = (e != NULL && atoi(e) != 0) ? 0 : 1;
79 }
80 MPI_Bcast(&iFlgMPIBatch, 1, MPI_INT, 0, MPI_COMM_WORLD);
81#else
82 nproc = 1;
83 myrank = 0;
84 {
85 const char *e = getenv("HPHI_MPI_NOBATCH");
86 iFlgMPIBatch = (e != NULL && atoi(e) != 0) ? 0 : 1;
87 }
88#endif
89 if (myrank == 0) stdoutMPI = stdout;
90 else stdoutMPI = fopen("/dev/null", "w");
91 splash();
92
93#pragma omp parallel default(none) shared(nthreads)
94#pragma omp master
95#ifdef _OPENMP
96 nthreads = omp_get_num_threads();
97#else
98 nthreads=1;
99#endif
100 fprintf(stdoutMPI, "\n\n##### Parallelization Info. #####\n\n");
101 fprintf(stdoutMPI, " OpenMP threads : %d\n", nthreads);
102 fprintf(stdoutMPI, " MPI PEs : %d \n\n", nproc);
103 fprintf(stdoutMPI, " MPI batching : %s\n\n", iFlgMPIBatch ? "ON" : "OFF");
104}/*void InitializeMPI(int argc, char *argv[])*/

159 int maxcount,
160 FILE* fp
161){
162 int inull;
163 char *ctmp;
164
165 ctmp = InputString;
166 inull = 0;
167 if (myrank == 0) {
168 ctmp = fgets(InputString, maxcount, fp);
169 if (ctmp == NULL){
170 inull = 1;
171 }
172
173 while(*InputString == '\n' || strncmp(InputString, "#", 1)==0){
174 ctmp = fgets(InputString, maxcount, fp);
175 if (ctmp == NULL){
176 inull=1;
177 break;
178 }
179 }
180 }
181#ifdef MPI
182 MPI_Bcast(InputString, maxcount, MPI_CHAR, 0, MPI_COMM_WORLD);
183 MPI_Bcast(&inull, 1, MPI_INT, 0, MPI_COMM_WORLD);
184#endif
185 if (myrank != 0 && inull == 1) {
186 ctmp = NULL;
187 }
188

324double NormMPI_dc(
325 unsigned long int idim,
326 double complex *_v1
327){
328 double complex cdnorm=0;
329 double dnorm =0;
330 unsigned long int i;
331 //DEBUG
332#pragma omp parallel for default(none) private(i) firstprivate(myrank) shared(_v1, idim) reduction(+: cdnorm)
333 for(i=1;i<=idim;i++){
334 cdnorm += conj(_v1[i])*_v1[i];
335 }
336#ifdef MPI
337 cdnorm = SumMPI_dc(cdnorm);
338#endif
339 dnorm=creal(cdnorm);
340 dnorm=sqrt(dnorm);
341
342 return dnorm;
343}/*double NormMPI_dc*/
